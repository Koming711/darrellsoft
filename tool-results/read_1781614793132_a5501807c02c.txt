    66→function getTodayStr(): string {
    67→  const d = new Date()
    68→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`

    75→// --- Parse dataJson for document info ---
    76→function parseDocInfo(entry: HistoryEntry) {
    77→  try {
    78→    const parsed = JSON.parse(entry.dataJson)
    79→    const items = parsed.items || []
    80→    const firstItem = items[0]
    81→    const namaBarang = firstItem?.deskripsi || ''
    82→    const hargaSatuan = firstItem?.harga || 0
    83→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0)
    84→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0)
    85→    const ppn = parsed.ppn || 0
    86→    const dpPercent = parsed.dp || 0
    87→    const totalHarga = subtotal + (subtotal * ppn / 100)
    88→    // Always derive dpAmount from originalTotal
    89→    const originalTotal = parsed.originalTotal !== undefined ? parsed.originalTotal : totalHarga
    90→    const dpAmount = originalTotal * (dpPercent / 100)
    91→    const sisa = totalHarga - dpAmount
    92→    const lunas = parsed.lunas === true
    93→    const tanggalJatuhTempo = parsed.tanggalJatuhTempo || ''
    94→    const tanggalPelunasan = parsed.tanggalPelunasan || ''
    95→    const referensi = parsed.referensi || ''
    96→    const referensiInvoiceNomor = parsed.referensiInvoiceNomor || ''
    97→    const isPelunasan = entry.docType === 'invoice-pelunasan'
    98→    const uangCapek = parsed.uangCapek || 0
    99→    return { namaBarang, hargaSatuan, totalQty, totalHarga, dpPercent, dp: dpAmount, sisa, lunas, tanggalJatuhTempo, tanggalPelunasan, referensi, referensiInvoiceNomor, isPelunasan, originalTotal, uangCapek }
   100→  } catch {
   101→    return { namaBarang: '', hargaSatuan: 0, totalQty: 0, totalHarga: 0, dpPercent: 0, dp: 0, sisa: 0, lunas: false, tanggalJatuhTempo: '', tanggalPelunasan: '', referensi: '', referensiInvoiceNomor: '', isPelunasan: false, originalTotal: 0, uangCapek: 0 }
   102→  }
   103→}

   105→// --- Parse dataJson for InvoiceData ---
   106→function parseInvoiceData(entry: HistoryEntry): InvoiceData {
   107→  try {
   108→    const parsed = JSON.parse(entry.dataJson)
   109→    const company: CompanyInfo = {
   110→      nama: parsed.company?.nama || DEFAULT_COMPANY.nama,
   111→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
   112→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
   113→      email: parsed.company?.email || DEFAULT_COMPANY.email,
   114→      npwp: parsed.company?.npwp || '',
   115→      website: parsed.company?.website || '',
   116→      ppn: parsed.company?.ppn ?? parsed.ppn ?? 11,
   117→      logo: parsed.company?.logo || '',
   118→      bankName: parsed.company?.bankName || '',
   119→      bankAccount: parsed.company?.bankAccount || '',
   120→      bankHolder: parsed.company?.bankHolder || '',
   121→      bankName2: parsed.company?.bankName2 || '',
   122→      bankAccount2: parsed.company?.bankAccount2 || '',
   123→      bankHolder2: parsed.company?.bankHolder2 || '',
   124→    }
   125→    const client = parsed.client || { nama: '', kontak: '', alamat: '' }
   126→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number }, i: number) => ({
   127→      id: it.id || `item-${i}`,
   128→      deskripsi: it.deskripsi || '',
   129→      qty: it.qty || 0,
   130→      satuan: it.satuan || '',
   131→      harga: it.harga || 0,
   132→    }))
   133→    const docType = (entry.docType === 'invoice-pelunasan' ? 'invoice-pelunasan' : 'invoice') as 'invoice' | 'invoice-pelunasan'
   134→    return {
   135→      type: docType,
   136→      company,
   137→      nomor: parsed.nomor || entry.nomor || '',
   138→      tanggal: parsed.tanggal || entry.tanggal || '',
   139→      referensi: parsed.referensi || '',
   140→      client,
   141→      items,
   142→      ppn: parsed.ppn ?? 11,
   143→      dp: parsed.dp || 0,
   144→      dpAmount: parsed.dpAmount,
   145→      catatan: parsed.catatan || '',
   146→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   147→      caraPembayaran: parsed.caraPembayaran || '',
   148→      tanggalGiro: parsed.tanggalGiro || '',
   149→      lunas: parsed.lunas === true,
   150→      tanggalPelunasan: parsed.tanggalPelunasan || '',
   151→      referensiInvoiceId: parsed.referensiInvoiceId || '',
   152→      referensiInvoiceNomor: parsed.referensiInvoiceNomor || '',
   153→      originalTotal: parsed.originalTotal,
   154→      uangCapek: parsed.uangCapek || 0,
   155→    }
   156→  } catch {
   157→    const docType = (entry.docType === 'invoice-pelunasan' ? 'invoice-pelunasan' : 'invoice') as 'invoice' | 'invoice-pelunasan'
   158→    return {
   159→      type: docType,
   160→      company: { ...DEFAULT_COMPANY },
   161→      nomor: entry.nomor || '',
   162→      tanggal: entry.tanggal || '',
   163→      referensi: '',
   164→      client: { nama: '', kontak: '', alamat: '' },
   165→      items: [],
   166→      ppn: 11,
   167→      dp: 0,
   168→      catatan: '',
   169→      tanggalJatuhTempo: '',
   170→      caraPembayaran: '',
   171→      tanggalGiro: '',
   172→      lunas: false,
   173→      tanggalPelunasan: '',
   174→      referensiInvoiceId: '',
   175→      referensiInvoiceNomor: '',
   176→      uangCapek: 0,
   177→    }
   178→  }
   179→}
