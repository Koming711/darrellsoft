    81→// ===== Helpers =====
    82→function formatRupiah(n: number): string {
    83→  return new Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR', minimumFractionDigits: 0 }).format(n || 0)
    84→}
    85→
    86→function formatDateID(iso: string): string {
    87→  if (!iso) return '-'
    88→  const d = new Date(iso + (iso.length === 10 ? 'T00:00:00' : ''))
    89→  if (isNaN(d.getTime())) return iso
    90→  return d.toLocaleDateString('id-ID', { day: '2-digit', month: 'short', year: 'numeric' })
    91→}
    92→
    93→function fmtISO(d: Date): string {
    94→  return new Date(d.getTime() - d.getTimezoneOffset() * 60000).toISOString().slice(0, 10)
    95→}
    96→
    97→function getFilterDates(period: PeriodType, customStart: string, customEnd: string): { start: string; end: string } {
    98→  const now = new Date()
    99→  const today = fmtISO(now)
   100→  if (period === 'today') return { start: today, end: today }
   101→  if (period === 'week') {
   102→    const day = now.getDay()
   103→    const monday = new Date(now)
   104→    monday.setDate(now.getDate() - (day === 0 ? 6 : day - 1))
   105→    const sunday = new Date(monday)
   106→    sunday.setDate(monday.getDate() + 6)
   107→    return { start: fmtISO(monday), end: fmtISO(sunday) }
   108→  }
   109→  if (period === 'month') {
   110→    return { start: fmtISO(new Date(now.getFullYear(), now.getMonth(), 1)), end: fmtISO(new Date(now.getFullYear(), now.getMonth() + 1, 0)) }
   111→  }
   112→  if (period === 'year') return { start: `${now.getFullYear()}-01-01`, end: `${now.getFullYear()}-12-31` }
   113→  return { start: customStart || '2000-01-01', end: customEnd || today }
   114→}
   115→
   116→function periodLabel(period: PeriodType, customStart: string, customEnd: string): string {
   117→  if (period === 'today') return 'Hari Ini'
   118→  if (period === 'week') return 'Minggu Ini'
   119→  if (period === 'month') return 'Bulan Ini'
   120→  if (period === 'year') return 'Tahun Ini'
   121→  return `${formatDateID(customStart)} s/d ${formatDateID(customEnd)}`
   122→}
