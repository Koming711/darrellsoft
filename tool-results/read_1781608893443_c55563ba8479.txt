    12→// === MODULE-LEVEL SESSION CACHE ===
    13→// Persists across navigations so DashboardLayout doesn't need to re-verify on every mount
    14→interface CachedSession {
    15→  user: any
    16→  userProfile: { createdAt: string | null; validUntil: string | null } | null
    17→  autoLogoutMin: number
    18→  logoutWarningSec: number
    19→  sessionWarning: string | null
    20→  forceLogoutAvailable: boolean
    21→  accountExpired: boolean
    22→  cachedAt: number
    23→}
    24→let _sessionCache: CachedSession | null = null
    25→const SESSION_CACHE_TTL = 120_000 // 2 minutes
    26→
    27→function getSessionCache(): CachedSession | null {
    28→  // Check module-level cache first
    29→  if (_sessionCache && Date.now() - _sessionCache.cachedAt <= SESSION_CACHE_TTL) {
    30→    return _sessionCache
    31→  }
    32→  // Fallback to sessionStorage
    33→  if (typeof window !== 'undefined') {
    34→    try {
    35→      const stored = sessionStorage.getItem('__dashboard_session_cache')
    36→      if (stored) {
    37→        const parsed = JSON.parse(stored) as CachedSession
    38→        if (Date.now() - parsed.cachedAt <= SESSION_CACHE_TTL) {
    39→          _sessionCache = parsed
    40→          return parsed
    41→        }
    42→        sessionStorage.removeItem('__dashboard_session_cache')
    43→      }
    44→    } catch {}
    45→  }
    46→  _sessionCache = null
    47→  return null
    48→}
    49→
    50→function setSessionCache(data: Omit<CachedSession, 'cachedAt'>) {
    51→  const entry = { ...data, cachedAt: Date.now() }
    52→  _sessionCache = entry
    53→  if (typeof window !== 'undefined') {
    54→    try {
    55→      sessionStorage.setItem('__dashboard_session_cache', JSON.stringify(entry))
    56→    } catch {}
    57→  }
    58→}
    59→
    60→function invalidateSessionCache() {
    61→  _sessionCache = null
    62→  if (typeof window !== 'undefined') {
    63→    try {
    64→      sessionStorage.removeItem('__dashboard_session_cache')
    65→    } catch {}
    66→  }
    67→}

    69→/** Subtle overlay shown in main content area during route navigation */
    70→function NavigatingOverlay() {
    71→  const [show, setShow] = useState(false)
    72→  const pathname = usePathname()
    73→
    74→  // Listen for navigation start event
    75→  useEffect(() => {
    76→    const onStart = () => setShow(true)
    77→    window.addEventListener('navigation-start', onStart)
    78→    return () => window.removeEventListener('navigation-start', onStart)
    79→  }, [])
    80→
    81→  // When pathname changes, navigation is complete — hide overlay after brief delay
    82→  useEffect(() => {
    83→    const t = setTimeout(() => setShow(false), 100)
    84→    return () => clearTimeout(t)
    85→  }, [pathname])
    86→
    87→  if (!show) return null
    88→
    89→  return (
    90→    <div className="absolute inset-0 z-10 flex items-center justify-center pointer-events-none">
    91→      <div className="flex items-center gap-2 px-4 py-2 rounded-lg bg-white/80 dark:bg-gray-900/80 backdrop-blur-sm shadow-sm">
    92→        <div className="w-4 h-4 border-2 border-blue-500 border-t-transparent rounded-full animate-spin" />
    93→        <span className="text-sm text-muted-foreground">Memuat...</span>
    94→      </div>
    95→    </div>
    96→  )
    97→}

   115→  const [isReclaiming, setIsReclaiming] = useState(false)

   122→  const [autoLogoutMin, setAutoLogoutMin] = useState(0)
   123→  const [logoutWarningSec, setLogoutWarningSec] = useState(0)
   124→  const [showCountdown, setShowCountdown] = useState(false)
   125→  const [countdown, setCountdown] = useState(0)

   127→
   128→  // === NO ACCESS STATE ===
   129→  const [noAccess, setNoAccess] = useState(false)
   130→
   131→  // === INIT GUARD (prevent double init in StrictMode) ===
   132→  const initDoneRef = useRef(false)
   133→
   134→  // === USER PROFILE (for header dates) ===

   157→    autoLogoutIntervalRef.current = null

   162→  const handleStayLoggedIn = useCallback(() => {
   163→    lastActivityRef.current = Date.now()
   164→    countdownActiveRef.current = false
   165→    setShowCountdown(false)
   166→    setCountdown(0)
   167→  }, [])
   168→
   169→  // === INITIAL AUTH + SETTINGS LOAD (runs once) ===
   170→  useEffect(() => {
   171→    if (initDoneRef.current) return
   172→    initDoneRef.current = true
   173→
   174→    // If we have a valid cache, restore state immediately — then do a background refresh

   179→      setSessionWarning(existingCache.sessionWarning ?? null)
   180→      setForceLogoutAvailable(existingCache.forceLogoutAvailable ?? false)

   199→            setLogoutWarningSec(data.securitySettings.logout_warning_sec || 0)

   537→                className="mt-5 px-5 py-2.5 bg-primary text-primary-foreground text-sm font-medium rounded-lg hover:bg-primary/90 transition-colors no-underline inline-block cursor-pointer"
   538→              >
   539→                {backLabel}
   540→              </a>
   541→            </div>
   542→          </main>
   543→        </div>
   544→      </div>
   545→    )
   546→  }
   547→
   548→  // === RENDER ===
   549→  return (
   550→    <div className="min-h-screen" style={{ backgroundColor: 'var(--app-content-bg)' }}>
   551→      <Sidebar
   552→        username={user?.username || 'User'}
   553→        role={user?.role}
   554→        onLogout={handleLogout}
   555→        isOpen={sidebarOpen}
   556→        onToggle={() => setSidebarOpen(!sidebarOpen)}
   557→        permVersion={permVersion}
   558→      />
   559→
   560→      <div className="lg:ml-16 transition-all duration-300">
   561→        <MobileHeader
   562→          username={user?.username}
   563→          title={title}
   564→          subtitle={subtitle}
   565→          userProfile={userProfile}
   566→        />
   567→
   568→        {/* Main Content — extra bottom padding on mobile for bottom nav + safe area */}
   569→        <main className="p-4 pb-20 lg:p-8 lg:pb-8 relative">
   570→          {/* Navigation loading overlay */}
   571→          <NavigatingOverlay />
   572→          {children}
   573→        </main>
   574→      </div>
   575→
   576→      {/* ===== Mobile Bottom Navigation ===== */}
   577→      <MobileBottomNav role={user?.role} onMoreClick={() => setSidebarOpen(true)} username={user?.username} onLogout={handleLogout} />
   578→
   579→      {/* ===== MODAL: ACCOUNT EXPIRED ===== */}
   580→      {accountExpired && sessionWarning && (
   581→        <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/50 backdrop-blur-sm px-4">
   582→          <div className="rounded-2xl shadow-2xl border border-red-200 dark:border-red-800 max-w-sm w-full p-6 animate-in fade-in zoom-in duration-200" style={{ backgroundColor: 'var(--card)' }}>
   583→            <div className="flex items-start gap-3 mb-4">
   584→              <div className="w-10 h-10 rounded-xl bg-red-100 dark:bg-red-900/30 flex items-center justify-center flex-shrink-0">
   585→                <TimerOff className="w-5 h-5 text-red-600 dark:text-red-400" />
   586→              </div>
   587→              <div>
   588→                <h3 className="text-lg font-bold text-foreground">Akun Kadaluarsa</h3>
   589→                <p className="text-xs text-muted-foreground mt-0.5">Masa berlaku akun telah habis</p>
   590→              </div>
   591→            </div>
   592→            <div className="bg-red-50 dark:bg-red-950 border border-red-200 dark:border-red-800 rounded-xl p-4 mb-5">
   593→              <p className="text-sm text-red-700 dark:text-red-400 leading-relaxed">{sessionWarning}</p>
   594→            </div>
   595→            <button
   596→              onClick={handleLogout}
   597→              className="w-full bg-red-600 hover:bg-red-700 text-white font-semibold py-2.5 rounded-xl transition-colors"
   598→              autoFocus
   599→            >
   600→              <span className="flex items-center justify-center gap-2">
   601→                <LogOut className="w-4 h-4" /> OK, Mengerti
   602→              </span>
   603→            </button>
   604→          </div>
   605→        </div>
   606→      )}
   607→
   608→      {/* ===== MODAL: SESSION WARNING (Multi-Device) ===== */}
   609→      {sessionWarning && !accountExpired && (
   610→        <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/50 backdrop-blur-sm px-4">
   611→          <div className="rounded-2xl shadow-2xl border border-red-200 dark:border-red-800 max-w-sm w-full p-6 animate-in fade-in zoom-in duration-200" style={{ backgroundColor: 'var(--card)' }}>
   612→            <div className="flex items-start gap-3 mb-4">
   613→              <div className="w-10 h-10 rounded-xl bg-red-100 dark:bg-red-900/30 flex items-center justify-center flex-shrink-0">
   614→                {forceLogoutAvailable ? <ShieldAlert className="w-5 h-5 text-red-600 dark:text-red-400" /> : <Smartphone className="w-5 h-5 text-red-600 dark:text-red-400" />}
   615→              </div>
   616→              <div>
   617→                <h3 className="text-lg font-bold text-foreground">Peringatan Keamanan</h3>
   618→                <p className="text-xs text-muted-foreground mt-0.5">Akun digunakan di perangkat lain</p>
   619→              </div>
   620→            </div>
   621→            <div className="bg-red-50 dark:bg-red-950 border border-red-200 dark:border-red-800 rounded-xl p-4 mb-5">
   622→              <p className="text-sm text-red-700 dark:text-red-400 leading-relaxed">{sessionWarning}</p>
   623→            </div>
   624→            {forceLogoutAvailable ? (
   625→              <div className="space-y-2.5">
   626→                <button
   627→                  onClick={async () => {
   628→                    setIsReclaiming(true)
   629→                    try {
   630→                      const authUser = getAuthUser()
   631→                      if (!authUser?.sessionId) {
   632→                        toast.error('Data sesi tidak ditemukan, silakan login ulang')
   633→                        setIsReclaiming(false)
   634→                        handleLogout()
   635→                        return
   636→                      }
   637→                      const res = await authFetch('/api/auth/reclaim-session', {
   638→                        method: 'POST',
   639→                        headers: { 'Content-Type': 'application/json' },
   640→                        body: JSON.stringify({ username: authUser.username, sessionId: authUser.sessionId, userId: authUser.id }),
   641→                      })
   642→                      const data = await res.json()
   643→                      if (res.ok) {
   644→                        setSessionWarning(null)
   645→                        setForceLogoutAvailable(false)
   646→                        toast.success('Perangkat lain berhasil di-logout')
   647→                      } else {
   648→                        toast.error(data.error || 'Gagal mengklaim sesi. Silakan coba lagi.')
   649→                      }
   650→                    } catch {
   651→                      toast.error('Terjadi kesalahan jaringan')
   652→                    }
   653→                    setIsReclaiming(false)
   654→                  }}
   655→                  disabled={isReclaiming}
   656→                  className="w-full flex items-center justify-center gap-2 bg-amber-500 hover:bg-amber-600 disabled:bg-amber-400 text-white font-semibold py-2.5 rounded-xl transition-colors"
   657→                  autoFocus
   658→                >
   659→                  {isReclaiming ? (
   660→                    <><div className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin" /> Memproses...</>
   661→                  ) : (
   662→                    <><ShieldAlert className="w-4 h-4" /> Paksa Logout Perangkat Lain</>
   663→                  )}
   664→                </button>
   665→                <button
   666→                  onClick={handleLogout}
   667→                  className="w-full flex items-center justify-center gap-2 border-2 border-red-200 dark:border-red-800 text-red-600 dark:text-red-400 hover:bg-red-50 dark:hover:bg-red-950 font-semibold py-2.5 rounded-xl transition-colors"
   668→                >
   669→                  <LogOut className="w-4 h-4" /> Logout dari Sini
   670→                </button>
   671→              </div>
   672→            ) : (
   673→              <button
   674→                onClick={handleLogout}
   675→                className="w-full bg-red-600 hover:bg-red-700 text-white font-semibold py-2.5 rounded-xl transition-colors"
   676→                autoFocus
   677→              >
   678→                OK, Logout
   679→              </button>
   680→            )}
   681→          </div>
   682→        </div>
   683→      )}
   684→
   685→      {/* ===== MODAL: AUTO-LOGOUT COUNTDOWN ===== */}
   686→      {showCountdown && (
   687→        <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/50 backdrop-blur-sm px-4">
   688→          <div className="rounded-2xl shadow-2xl border border-amber-200 dark:border-amber-800 max-w-sm w-full p-6 animate-in fade-in zoom-in duration-200" style={{ backgroundColor: 'var(--card)' }}>
   689→            <div className="flex items-start gap-3 mb-4">
   690→              <div className="w-10 h-10 rounded-xl bg-amber-100 dark:bg-amber-900/30 flex items-center justify-center flex-shrink-0">
   691→                <AlertTriangle className="w-5 h-5 text-amber-600 dark:text-amber-400" />
   692→              </div>
   693→              <div>
   694→                <h3 className="text-lg font-bold text-foreground">Sesi Akan Berakhir</h3>
   695→                <p className="text-xs text-muted-foreground mt-0.5">Anda tidak aktif, akan otomatis logout</p>
   696→              </div>
   697→            </div>
   698→            <div className="text-center py-5">
   699→              <div className="text-6xl font-bold text-amber-600 dark:text-amber-400 tabular-nums">{countdown}</div>
   700→              <p className="text-sm text-muted-foreground mt-2">detik tersisa</p>
   701→            </div>
   702→            <button
   703→              onClick={handleStayLoggedIn}
   704→              className="w-full bg-amber-500 hover:bg-amber-600 text-white font-semibold py-2.5 rounded-xl transition-colors"
   705→              autoFocus
   706→            >
   707→              Tetap Login
   708→            </button>
   709→          </div>
   710→        </div>
   711→      )}
   712→    </div>
   713→  )
   714→}
