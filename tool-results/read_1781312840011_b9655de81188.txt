    65→function getTodayStr(): string {
    66→  const d = new Date()
    67→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
    68→}

    70→function formatRupiahShort(n: number): string {
    71→  return new Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR', minimumFractionDigits: 0 }).format(n)
    72→}

    74→// --- Parse dataJson for document info ---
    75→function parseDocInfo(entry: HistoryEntry) {
    76→  try {
    77→    const parsed = JSON.parse(entry.dataJson)
    78→    const items = parsed.items || []
    79→    const firstItem = items[0]
    80→    const namaBarang = firstItem?.deskripsi || ''
    81→    const hargaSatuan = firstItem?.harga || 0
    82→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0)
    83→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0)
    84→    const ppn = parsed.ppn || 0
    85→    const dpPercent = parsed.dp || 0
    86→    const totalHarga = subtotal + (subtotal * ppn / 100)
    87→    // Use saved dpAmount if available, otherwise calculate from percentage
    88→    const dpAmount = parsed.dpAmount !== undefined ? parsed.dpAmount : totalHarga * (dpPercent / 100)
    89→    const sisa = totalHarga - dpAmount
    90→    const lunas = parsed.lunas === true
    91→    const tanggalJatuhTempo = parsed.tanggalJatuhTempo || ''
    92→    const tanggalPelunasan = parsed.tanggalPelunasan || ''
    93→    const referensi = parsed.referensi || ''
    94→    return { namaBarang, hargaSatuan, totalQty, totalHarga, dpPercent, dp: dpAmount, sisa, lunas, tanggalJatuhTempo, tanggalPelunasan, referensi }
    95→  } catch {
    96→    return { namaBarang: '', hargaSatuan: 0, totalQty: 0, totalHarga: 0, dpPercent: 0, dp: 0, sisa: 0, lunas: false, tanggalJatuhTempo: '', tanggalPelunasan: '', referensi: '' }
    97→  }
    98→}

   100→// --- Parse dataJson for InvoiceData ---
   101→function parseInvoiceData(entry: HistoryEntry): InvoiceData {
   102→  try {
   103→    const parsed = JSON.parse(entry.dataJson)
   104→    const company: CompanyInfo = {
   105→      nama: parsed.company?.nama || DEFAULT_COMPANY.nama,
   106→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
   107→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
   108→      email: parsed.company?.email || DEFAULT_COMPANY.email,
   109→      npwp: parsed.company?.npwp || '',
   110→      website: parsed.company?.website || '',
   111→      ppn: parsed.company?.ppn ?? parsed.ppn ?? 11,
   112→      logo: parsed.company?.logo || '',
   113→      bankName: parsed.company?.bankName || '',
   114→      bankAccount: parsed.company?.bankAccount || '',
   115→      bankHolder: parsed.company?.bankHolder || '',
   116→      bankName2: parsed.company?.bankName2 || '',
   117→      bankAccount2: parsed.company?.bankAccount2 || '',
   118→      bankHolder2: parsed.company?.bankHolder2 || '',
   119→    }
   120→    const client = parsed.client || { nama: '', kontak: '', alamat: '' }
   121→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number }, i: number) => ({
   122→      id: it.id || `item-${i}`,
   123→      deskripsi: it.deskripsi || '',
   124→      qty: it.qty || 0,
   125→      satuan: it.satuan || '',
   126→      harga: it.harga || 0,
   127→    }))
   128→    return {
   129→      type: 'invoice',
   130→      company,
   131→      nomor: parsed.nomor || entry.nomor || '',
   132→      tanggal: parsed.tanggal || entry.tanggal || '',
   133→      referensi: parsed.referensi || '',
   134→      client,
   135→      items,
   136→      ppn: parsed.ppn ?? 11,
   137→      dp: parsed.dp || 0,
   138→      dpAmount: parsed.dpAmount,
   139→      catatan: parsed.catatan || '',
   140→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   141→      caraPembayaran: parsed.caraPembayaran || '',
   142→      tanggalGiro: parsed.tanggalGiro || '',
   143→      lunas: parsed.lunas === true,
   144→      tanggalPelunasan: parsed.tanggalPelunasan || '',
   145→    }
   146→  } catch {
   147→    return {
   148→      type: 'invoice',
   149→      company: { ...DEFAULT_COMPANY },
   150→      nomor: entry.nomor || '',
   151→      tanggal: entry.tanggal || '',
   152→      referensi: '',
   153→      client: { nama: '', kontak: '', alamat: '' },
   154→      items: [],
   155→      ppn: 11,
   156→      dp: 0,
   157→      catatan: '',
   158→      tanggalJatuhTempo: '',
   159→      caraPembayaran: '',
   160→      tanggalGiro: '',
   161→      lunas: false,
   162→      tanggalPelunasan: '',
   163→    }
   164→  }
   165→}

   329→          previewEl.style.transformOrigin = origTransformOrigin
   330→          if (scaleWrapper) scaleWrapper.style.transform = scaleWrapperOrigTransform

   333→        toast.error('Preview tidak ditemukan')
   334→      }
   335→    } catch (err) {
   336→      console.error(err)

   339→      setSendingPdf(false)

   354→      if (blob.size === 0) { toast.error('Backup kosong'); return }

   425→            </div>
   426→          </div>

   429→        {/* Table / Cards */}
   430→        {loading ? (
   431→          <div className="px-4 py-6 text-center"><Loader2 className="w-6 h-6 mx-auto text-blue-500 animate-spin" /><p className="text-xs text-slate-400 mt-2">Memuat riwayat...</p></div>
   432→        ) : filteredHistory.length > 0 ? (
   433→          <>
   434→            {/* Mobile */}

   452→                    <div className="flex items-center justify-between gap-2">
