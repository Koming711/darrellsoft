    65→    65→export function InvoiceEditor({ dpDisabled = false, onSaved }: { dpDisabled?: boolean; onSaved?: (id: string) => void }) {
    66→    66→  const invoice = useDokuproStore((s) => s.invoice);
    67→    67→  const setInvoice = useDokuproStore((s) => s.setInvoice);
    68→    68→  const resetDocument = useDokuproStore((s) => s.resetDocument);
    69→    69→  const loadCompanyFromAPI = useDokuproStore((s) => s.loadCompanyFromAPI);
    70→    70→  const invoiceEditingId = useDokuproStore((s) => s.invoiceEditingId);
    71→    71→  const setInvoiceEditingId = useDokuproStore((s) => s.setInvoiceEditingId);
    72→    72→  const searchParams = useSearchParams();
    73→    73→  const riwayatIdFromUrl = searchParams.get('riwayatId');
    74→    74→  const autoSelectDoneRef = useRef<string | null>(null); // track which riwayatId was auto-selected
    75→    75→
    76→    76→  // Daftar barang milik customer terpilih (Master Barang per pelanggan)
    77→    77→  const [barangList, setBarangList] = useState<BarangOption[]>([]);
    78→    78→  // Nama customer sebelumnya — untuk mendeteksi perubahan customer
    79→    79→  const prevClientNameRef = useRef<string>(invoice.client.nama);
    80→    80→  // Nama yang diisi otomatis oleh pemilihan referensi (hitung cetakan) —
    81→    81→  // perubahan nama dari alur ini TIDAK boleh mengosongkan item yang baru diisi.
    82→    82→  const autoFillNameRef = useRef<string>('');
    83→    83→
    84→    84→  // Riwayat cetakan dropdown state
    85→    85→  const [riwayatList, setRiwayatList] = useState<RiwayatCetakanItem[]>([]);
    86→    86→  const [referensiInput, setReferensiInput] = useState(invoice.referensi);
    87→    87→  const [dropdownOpen, setDropdownOpen] = useState(false);
    88→    88→
    89→    89→  // Customer dropdown state
    90→    90→  const [customerList, setCustomerList] = useState<CustomerItem[]>([]);
    91→    91→  const [clientInput, setClientInput] = useState(invoice.client.nama);
    92→    92→  const [clientDropdownOpen, setClientDropdownOpen] = useState(false);
    93→    93→  const [clientTyping, setClientTyping] = useState(false);
    94→    94→
    95→    95→  const fetchRiwayatCetakan = useCallback(async () => {
    96→    96→    try {
    97→    97→      const res = await fetch('/api/riwayat-cetakan', { headers: getAuthHeaders() });
    98→    98→      if (res.ok) {
    99→    99→        const data = await res.json();
   100→   100→        setRiwayatList(data);
   101→   101→        return data;
   102→   102→      }
   103→   103→    } catch {
   104→   104→      // silently fail
   105→   105→    }
   106→   106→    return [];
   107→   107→  }, []);
   108→   108→
   109→   109→  // Fetch customer list
   110→   110→  const fetchCustomers = useCallback(async () => {
   111→   111→    try {
   112→   112→      const res = await fetch('/api/customers', { headers: getAuthHeaders() });
   113→   113→      if (res.ok) {
   114→   114→        const data = await res.json();
   115→   115→        setCustomerList(Array.isArray(data) ? data : []);
   116→   116→      }
   117→   117→    } catch {
   118→   118→      // silently fail
   119→   119→    }
   120→   120→  }, []);
   121→   121→
   122→   122→  useEffect(() => { loadCompanyFromAPI() }, [loadCompanyFromAPI]);
   123→   123→  useEffect(() => { fetchRiwayatCetakan() }, [fetchRiwayatCetakan]);
   124→   124→  useEffect(() => { fetchCustomers() }, [fetchCustomers]);
   125→   125→
   126→   126→  // Muat daftar barang milik customer terpilih (Master Barang per pelanggan).
   127→   127→  // Jika customer tidak dikenal / belum memilih, daftar barang = kosong.
   128→   128→  useEffect(() => {
   129→   129→    const nama = invoice.client.nama.trim().toLowerCase();
   130→   130→    const found = nama
   131→   131→      ? customerList.find((c) => c.name.trim().toLowerCase() === nama)
   132→   132→      : undefined;
   133→   133→    if (!found) {
   134→   134→      setBarangList([]);
   135→   135→      return;
   136→   136→    }
   137→   137→    let cancelled = false;
   138→   138→    fetch(`/api/items?customerId=${encodeURIComponent(found.id)}&active=1`, { headers: getAuthHeaders() })
   139→   139→      .then((res) => (res.ok ? res.json() : { items: [] }))
   140→   140→      .then((data) => {
   141→   141→        if (cancelled) return;
   142→   142→        const rows = Array.isArray(data?.items) ? data.items : [];
   143→   143→        setBarangList(rows.map((r: { id: string; name: string; unit: string; standardPrice: number; hpp: number | null }) => ({
   144→   144→          id: r.id,
   145→   145→          name: r.name,
   146→   146→          unit: r.unit || 'pcs',
   147→   147→          standardPrice: r.standardPrice || 0,
   148→   148→          hpp: r.hpp ?? null,
   149→   149→        })));
   150→   150→      })
   151→   151→      .catch(() => { if (!cancelled) setBarangList([]); });
   152→   152→    return () => { cancelled = true; };
   153→   153→  }, [invoice.client.nama, customerList]);
   154→   154→
   155→   155→  // Ganti customer → otomatis KOSONGKAN nama barang di kotak item
   156→   156→  // (barang milik tiap pelanggan tidak boleh tercampur).
   157→   157→  // Pengecualian: nama yang diisi otomatis oleh pemilihan referensi cetakan.
   158→   158→  useEffect(() => {
   159→   159→    const nama = invoice.client.nama.trim();
   160→   160→    const prev = prevClientNameRef.current.trim();
   161→   161→    if (nama !== prev) {
   162→   162→      prevClientNameRef.current = invoice.client.nama;
   163→   163→      if (autoFillNameRef.current && nama === autoFillNameRef.current.trim()) {
   164→   164→        // Diisi otomatis oleh applyReferensi — jangan kosongkan
   165→   165→        autoFillNameRef.current = '';
   166→   166→        return;
   167→   167→      }
   168→   168→      autoFillNameRef.current = '';
   169→   169→      setInvoice((prevInv) => ({
   170→   170→        ...prevInv,
   171→   171→        items: prevInv.items.map((it, i) => (
   172→   172→          i === 0
   173→   173→            ? { ...it, deskripsi: '', qty: 1, harga: 0, modal: 0 }
   174→   174→            : null
   175→   175→        )).filter((it): it is typeof prevInv.items[number] => it !== null),
   176→   176→      }));
   177→   177→    }
   178→   178→  }, [invoice.client.nama, setInvoice]);
   179→   179→
   180→   180→  // Fetch next Invoice number from server (only when creating new, not editing)
   181→   181→  const fetchNextNumber = useCallback(() => {
   182→   182→    // Read current editingId from store to avoid stale closure
   183→   183→    const currentEditingId = useDokuproStore.getState().invoiceEditingId;
   184→   184→    if (currentEditingId) return; // Don't overwrite nomor when editing
   185→   185→    fetch('/api/history?preview=next-number&docType=invoice', { headers: getAuthHeaders() })
   186→   186→      .then(res => res.ok ? res.json() : null)
   187→   187→      .then(data => { if (data?.nextNumber) setInvoice((prev) => ({ ...prev, nomor: data.nextNumber })) })
   188→   188→      .catch(() => {});
   189→   189→  }, [setInvoice]);
   190→   190→
   191→   191→  useEffect(() => { fetchNextNumber() }, [fetchNextNumber]);
   192→   192→
   193→   193→  // Re-fetch next number after a document is saved
   194→   194→  useEffect(() => {
   195→   195→    const handler = () => { fetchNextNumber() };
   196→   196→    window.addEventListener('dokupro:history-updated', handler);
   197→   197→    return () => { window.removeEventListener('dokupro:history-updated', handler) };
   198→   198→  }, [fetchNextNumber]);
   199→   199→
   200→   200→  // Mode Regular (dpDisabled): paksa DP = 0 dan sembunyikan input DP
   201→   201→  useEffect(() => {
   202→   202→    if (dpDisabled) {
   203→   203→      setInvoice((prev) => (prev.dp === 0 ? prev : ({ ...prev, dp: 0 })));
   204→   204→    }
   205→   205→  }, [dpDisabled, setInvoice]);
   206→   206→
   207→   207→  // Sync referensiInput when invoice.referensi changes externally
   208→   208→  useEffect(() => { setReferensiInput(invoice.referensi) }, [invoice.referensi]);
   209→   209→  // Sync clientInput when invoice.client.nama changes externally
   210→   210→  useEffect(() => { setClientInput(invoice.client.nama) }, [invoice.client.nama]);
   211→   211→
   212→   212→  // Filter riwayat list — when dropdown is open show all, otherwise not used
   213→   213→  const filteredRiwayatList = riwayatList;
   214→   214→
   215→   215→  const handleReferensiInputChange = (value: string) => {
   216→   216→    setReferensiInput(value);
   217→   217→    setInvoice((prev) => ({ ...prev, referensi: value }));
   218→   218→    setDropdownOpen(true);
   219→   219→  };
   220→   220→
   221→   221→  // Apply referensi data using functional setInvoice to avoid stale closure
   222→   222→  const applyReferensi = (item: RiwayatCetakanItem) => {
   223→   223→    const ref = item.nomorUrut || item.printName || '';
   224→   224→    setReferensiInput(ref);
   225→   225→
   226→   226→    // Build single item with multi-line description
   227→   227→    const descLines: string[] = [];
   228→   228→    if (item.printName) descLines.push(item.printName);
   229→   229→
   230→   230→    const bahanParts: string[] = [];
   231→   231→    if (item.paperName) bahanParts.push(item.paperName);
   232→   232→    if (item.paperGrammage) bahanParts.push(item.paperGrammage + 'g');
   233→   233→    if (bahanParts.length > 0) descLines.push(bahanParts.join(' '));
   234→   234→
   235→   235→    if (item.finishingNames) {
   236→   236→      const finishings = item.finishingNames.split(',').map(f => f.trim()).filter(Boolean);
   237→   237→      descLines.push('Finishing:');
   238→   238→      finishings.forEach(f => descLines.push(f));
   239→   239→    }
   240→   240→
   241→   241→    const deskripsi = descLines.join('\n');
   242→   242→    const jumlahPesanan = parseInt(item.jumlahPesanan) || 1;
   243→   243→    const hargaPerPcs = jumlahPesanan > 0 ? Math.round(item.grandTotal / jumlahPesanan) : 0;
   244→   244→
   245→   245→    const newItems: InvoiceData['items'] = [{
   246→   246→      id: 'riwayat-0',
   247→   247→      deskripsi: deskripsi || ref,
   248→   248→      qty: jumlahPesanan,
   249→   249→      satuan: 'pcs',
   250→   250→      harga: hargaPerPcs,
   251→   251→    }];
   252→   252→
   253→   253→    // Tandai bahwa perubahan nama customer berikutnya berasal dari alur
   254→   254→    // referensi cetakan — jangan kosongkan item yang baru saja diisi.
   255→   255→    if (item.customerName) autoFillNameRef.current = item.customerName;
   256→   256→
   257→   257→    // Use functional form to always get the LATEST state
   258→   258→    setInvoice((prev) => ({
   259→   259→      ...prev,
   260→   260→      referensi: ref,
   261→   261→      client: {
   262→   262→        ...prev.client,
   263→   263→        nama: item.customerName || prev.client.nama,
   264→   264→      },
   265→   265→      items: newItems,
   266→   266→      catatan: '',
   267→   267→      uangCapek: item.profitAmount || 0,
   268→   268→    }));
   269→   269→    setDropdownOpen(false);
   270→   270→  };
   271→   271→
   272→   272→  const handleReferensiSelect = (item: RiwayatCetakanItem) => {
   273→   273→    applyReferensi(item);
   274→   274→  };
   275→   275→
   276→   276→  // Auto-select referensi when coming from hitung cetakan with riwayatId
   277→   277→  // Fetches the specific riwayat directly by ID for reliability (no race condition with list loading)
   278→   278→  useEffect(() => {
   279→   279→    if (!riwayatIdFromUrl) return;
   280→   280→    if (autoSelectDoneRef.current === riwayatIdFromUrl) return;
   281→   281→
   282→   282→    let cancelled = false;
   283→   283→
   284→   284→    const fetchAndApply = async () => {
   285→   285→      // First, try to find it in the already-loaded list (fast path)
   286→   286→      const foundInList = riwayatList.find((r) => r.id === riwayatIdFromUrl);
   287→   287→      if (foundInList) {
   288→   288→        if (!cancelled) {
   289→   289→          autoSelectDoneRef.current = riwayatIdFromUrl;
   290→   290→          applyReferensi(foundInList);
   291→   291→        }
   292→   292→        return;
   293→   293→      }
   294→   294→
   295→   295→      // Not in list yet — fetch directly by ID from the API (reliable path)
   296→   296→      try {
   297→   297→        const res = await fetch(`/api/riwayat-cetakan?id=${riwayatIdFromUrl}`, { headers: getAuthHeaders() });
   298→   298→        if (res.ok && !cancelled) {
   299→   299→          const data = await res.json();
   300→   300→          if (data && !data.error) {
   301→   301→            autoSelectDoneRef.current = riwayatIdFromUrl;
   302→   302→            applyReferensi(data as RiwayatCetakanItem);
   303→   303→          }
   304→   304→        }
   305→   305→      } catch {
   306→   306→        // Fallback: retry once after a short delay
   307→   307→        if (cancelled) return;
   308→   308→        await new Promise(r => setTimeout(r, 800));
   309→   309→        if (cancelled) return;
   310→   310→        try {
   311→   311→          const res = await fetch(`/api/riwayat-cetakan?id=${riwayatIdFromUrl}`, { headers: getAuthHeaders() });
   312→   312→          if (res.ok) {
   313→   313→            const data = await res.json();
   314→   314→            if (data && !data.error) {
   315→   315→              autoSelectDoneRef.current = riwayatIdFromUrl;
   316→   316→              applyReferensi(data as RiwayatCetakanItem);
   317→   317→            }
   318→   318→          }
   319→   319→        } catch {
   320→   320→          // Give up silently
   321→   321→        }
   322→   322→      }
   323→   323→    };
   324→   324→
   325→   325→    fetchAndApply();
   326→   326→
   327→   327→    return () => { cancelled = true; };
   328→   328→  }, [riwayatIdFromUrl]);
   329→   329→
   330→   330→  // Filter customer list (only when user is actively typing)
   331→   331→  const filteredCustomerList = useMemo(() =>
   332→   332→    clientTyping
   333→   333→      ? customerList.filter((c) => {
   334→   334→          const search = clientInput.toLowerCase().trim();
   335→   335→          if (!search) return true;
   336→   336→          return c.name.toLowerCase().includes(search) ||
   337→   337→                 (c.companyName || '').toLowerCase().includes(search) ||
   338→   338→                 (c.phone || '').toLowerCase().includes(search) ||
   339→   339→                 (c.address || '').toLowerCase().includes(search);
   340→   340→        })
   341→   341→      : customerList
   342→   342→  , [clientTyping, customerList, clientInput]);
   343→   343→
   344→   344→  const handleClientInputChange = (value: string) => {
   345→   345→    setClientInput(value);
   346→   346→    setClientTyping(true);
   347→   347→    updateClient('nama', value);
   348→   348→    setClientDropdownOpen(true);
   349→   349→  };
   350→   350→
   351→   351→  const handleClientSelect = (item: CustomerItem) => {
   352→   352→    setClientInput(item.name);
   353→   353→    setClientTyping(false);
   354→   354→    setInvoice((prev) => ({
   355→   355→      ...prev,
   356→   356→      client: {
   357→   357→        nama: item.name,
   358→   358→        kontak: item.phone || item.email || '',
   359→   359→        alamat: item.address || '',
   360→   360→      },
   361→   361→    }));
   362→   362→    setClientDropdownOpen(false);
   363→   363→  };
   364→   364→
   365→   365→  // Pilih barang dari dropdown Master Barang customer → isi deskripsi +
   366→   366→  // harga satuan + harga modal otomatis (semuanya tetap bisa diedit manual).
   367→   367→  const handlePickBarang = (itemIndex: number, barang: BarangOption) => {
   368→   368→    setInvoice((prev) => ({
   369→   369→      ...prev,
   370→   370→      items: prev.items.map((it, i) => (
   371→   371→        i === itemIndex
   372→   372→          ? {
   373→   373→              ...it,
   374→   374→              deskripsi: barang.name,
   375→   375→              satuan: barang.unit || it.satuan || 'pcs',
   376→   376→              harga: barang.standardPrice || 0,
   377→   377→              modal: barang.hpp ?? 0,
   378→   378→            }
   379→   379→          : it
   380→   380→      )),
   381→   381→    }));
   382→   382→  };
   383→   383→
   384→   384→  const updateClient = (field: string, value: string) => {
   385→   385→    setInvoice((prev) => ({
   386→   386→      ...prev,
   387→   387→      client: { ...prev.client, [field]: value },
   388→   388→    }));
   389→   389→  };
   390→   390→
   391→   391→  const subtotal = invoice.items.reduce((sum, item) => sum + item.qty * item.harga, 0);
   392→   392→  const ppnAmount = subtotal * (invoice.ppn / 100);
   393→   393→  const total = subtotal + ppnAmount;
   394→   394→  const dpPercent = invoice.dp || 0;
   395→   395→  const dpAmount = total * (dpPercent / 100);
   396→   396→  const sisa = total - dpAmount;
   397→   397→
   398→   398→  // Pratinjau langsung muncul begitu ada data inti (referensi dipilih /
   399→   399→  // customer terisi / ada barang) — tanpa tombol "Lihat Pratinjau A5".
   400→   400→  const hasPreviewData = Boolean(
   401→   401→    referensiInput.trim() ||
   402→   402→    invoice.client.nama.trim() ||
   403→   403→    invoice.items.some((it) => it.deskripsi.trim() !== '')
   404→   404→  );
   405→   405→
   406→   406→  return (
   407→   407→    <>
   408→   408→      <DocumentEditorLayout
   409→   409→        title="Invoice"
   410→   410→        previewMode="inline-bottom"
   411→   411→        formColumns={2}
   412→   412→        previewMaxWidth={560}
   413→   413→        previewContent={
   414→   414→          hasPreviewData ? (
   415→   415→            <InvoicePreview data={invoice} />
   416→   416→          ) : (
   417→   417→            <div className="flex min-h-[320px] w-full items-center justify-center rounded-lg border-2 border-dashed border-slate-200 bg-slate-50/60 p-6 text-center">
   418→   418→              <p className="max-w-[280px] text-sm text-slate-400">
   419→   419→                Pratinjau invoice akan muncul di sini setelah Anda memilih referensi (No. HC) atau customer.
   420→   420→              </p>
   421→   421→            </div>
   422→   422→          )
   423→   423→        }
   424→   424→        actions={
   425→   425→          <DocumentActionButtons
   426→   426→            docType="invoice"
   427→   427→            documentLabel="Invoice"
   428→   428→            currentData={invoice}
   429→   429→            onReset={() => resetDocument('invoice')}
   430→   430→            editingId={invoiceEditingId}
   431→   431→            onUpdateSuccess={() => setInvoiceEditingId(null)}
   432→   432→            showPrintActions={false}
   433→   433→            onSaved={onSaved}
   434→   434→          />
   435→   435→        }
   436→   436→      >
   437→   437→        {/* FORM 2 KOLOM (desktop) — KIRI: Detail Dokumen + Informasi Pembayaran +
   438→   438→            Kepada Yth, KANAN: Item + Informasi Tambahan. Kotak Data Perusahaan
