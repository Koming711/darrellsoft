    74→// Helper: create a Block from calculation results
    75→function makeBlock(
    76→  name: string, x: number, y: number, containerW: number, containerH: number,
    77→  result: { pieces: number; horizontal: number; vertical: number; rotated: boolean },
    78→  cw: number, ch: number
    79→): Block {
    80→  const pw = result.rotated ? ch : cw
    81→  const ph = result.rotated ? cw : ch
    82→  const usedW = result.horizontal * pw
    83→  const usedH = result.vertical * ph
    84→  return {
    85→    name, x, y,
    86→    width: usedW,
    87→    height: usedH,
    88→    horizontal: result.horizontal,
    89→    vertical: result.vertical,
    90→    pieces: result.pieces,
    91→    pieceWidth: pw,
    92→    pieceHeight: ph,
    93→    rotated: result.rotated,
    94→    usedWidth: usedW,
    95→    usedHeight: usedH,
    96→    wasteWidth: containerW - usedW,
    97→    wasteHeight: containerH - usedH,
    98→  }
    99→}

   101→// Calculate pieces in a rectangle with optional rotation
   102→export function calculatePiecesInRect(
   103→  rectWidth: number,
   104→  rectHeight: number,
   105→  pieceWidth: number,
   106→  pieceHeight: number
   107→) {
   108→  // Normal orientation
   109→  const hNormal = Math.floor(rectWidth / pieceWidth)
   110→  const vNormal = Math.floor(rectHeight / pieceHeight)
   111→  const totalNormal = hNormal * vNormal
   112→
   113→  // Rotated orientation
   114→  const hRotated = Math.floor(rectWidth / pieceHeight)
   115→  const vRotated = Math.floor(rectHeight / pieceWidth)
   116→  const totalRotated = hRotated * vRotated
   117→
   118→  if (totalRotated > totalNormal) {
   119→    return { pieces: totalRotated, horizontal: hRotated, vertical: vRotated, rotated: true }
   120→  }
   121→  return { pieces: totalNormal, horizontal: hNormal, vertical: vNormal, rotated: false }
   122→}

   124→// Normal grid layout (with best rotation)
   125→export function normalGridLayout(pw: number, ph: number, cw: number, ch: number): Scenario {
   126→  const { pieces, horizontal, vertical, rotated } = calculatePiecesInRect(pw, ph, cw, ch)
   127→
   128→  return {
   129→    strategy: 'Grid Normal',
   130→    type: 'normal',
   131→    blocks: [makeBlock('Blok A', 0, 0, pw, ph, { pieces, horizontal, vertical, rotated }, cw, ch)],
   132→    total: pieces,
   133→    steps: [`Langkah 1: Potong ${horizontal}×${vertical} grid ${rotated ? '(rotasi 90°)' : ''}`]
   134→  }
   135→}

   137→// Guillotine cutting - single cut
   138→export function guillotineSingleCut(
   139→  pw: number,
   140→  ph: number,
   141→  cw: number,
   142→  ch: number,
   143→  optimizationMode: 'fast' | 'maximal'
   144→): Scenario[] {
   145→  const scenarios: Scenario[] = []
   146→  const step = optimizationMode === 'fast' ? 2 : 0.5
   147→  // FIX: Use Math.min(cw, ch) because calculatePiecesInRect already tries both orientations
   148→  const minSize = Math.min(cw, ch)
   149→
   150→  // Vertical cuts
   151→  for (let x = minSize; x <= pw - minSize; x += step) {
   152→    const leftWidth = x
   153→    const rightWidth = pw - x
   154→
   155→    const leftResult = calculatePiecesInRect(leftWidth, ph, cw, ch)
   156→    const rightResult = calculatePiecesInRect(rightWidth, ph, cw, ch)
   157→
   158→    const total = leftResult.pieces + rightResult.pieces
   159→
   160→    if (total > 0) {
   161→      scenarios.push({
   162→        strategy: `Guillotine Vertikal (potong di ${x.toFixed(1)}cm)`,
   163→        type: 'block-cut',
   164→        cutPosition: x,
   165→        cutPositionY: undefined,
   166→        blocks: [
   167→          ...(leftResult.pieces > 0 ? [makeBlock('Blok A', 0, 0, leftWidth, ph, leftResult, cw, ch)] : []),
   168→          ...(rightResult.pieces > 0 ? [makeBlock('Blok B', x, 0, rightWidth, ph, rightResult, cw, ch)] : []),
   169→        ],
   170→        total,
   171→        steps: [
   172→          `Langkah 1: Potong vertikal pada posisi ${x.toFixed(1)}cm dari kiri`,
   173→          `Langkah 2: Bagian kiri (${leftWidth.toFixed(1)}×${ph}cm) → ${leftResult.pieces} potongan ${leftResult.rotated ? '(rotasi 90°)' : ''}`,
   174→          `Langkah 3: Bagian kanan (${rightWidth.toFixed(1)}×${ph}cm) → ${rightResult.pieces} potongan ${rightResult.rotated ? '(rotasi 90°)' : ''}`
   175→        ]
   176→      })
   177→    }
   178→  }
   179→
   180→  // Horizontal cuts
   181→  for (let y = minSize; y <= ph - minSize; y += step) {
   182→    const bottomHeight = y
   183→    const topHeight = ph - y
   184→
   185→    const bottomResult = calculatePiecesInRect(pw, bottomHeight, cw, ch)
   186→    const topResult = calculatePiecesInRect(pw, topHeight, cw, ch)
   187→
   188→    const total = bottomResult.pieces + topResult.pieces
   189→
   190→    if (total > 0) {
   191→      scenarios.push({
   192→        strategy: `Guillotine Horizontal (potong di ${y.toFixed(1)}cm)`,
   193→        type: 'block-cut',
   194→        cutPosition: undefined,
   195→        cutPositionY: y,
   196→        blocks: [
   197→          ...(bottomResult.pieces > 0 ? [makeBlock('Blok A', 0, 0, pw, bottomHeight, bottomResult, cw, ch)] : []),
   198→          ...(topResult.pieces > 0 ? [makeBlock('Blok B', 0, y, pw, topHeight, topResult, cw, ch)] : []),
   199→        ],
   200→        total,
   201→        steps: [
   202→          `Langkah 1: Potong horizontal pada posisi ${y.toFixed(1)}cm dari bawah`,
   203→          `Langkah 2: Bagian bawah (${pw}×${bottomHeight.toFixed(1)}cm) → ${bottomResult.pieces} potongan ${bottomResult.rotated ? '(rotasi 90°)' : ''}`,
   204→          `Langkah 3: Bagian atas (${pw}×${topHeight.toFixed(1)}cm) → ${topResult.pieces} potongan ${topResult.rotated ? '(rotasi 90°)' : ''}`
   205→        ]
   206→      })
   207→    }
   208→  }
   209→
   210→  return scenarios
   211→}

   213→// Two-stage guillotine cutting (FIXED: now actually uses both dimensions)
   214→export function twoStageGuillotine(
   215→  pw: number,
   216→  ph: number,
   217→  cw: number,
   218→  ch: number,
   219→  optimizationMode: 'fast' | 'maximal'
   220→): Scenario[] {
   221→  const scenarios: Scenario[] = []
   222→  const step = optimizationMode === 'fast' ? 5 : 1
   223→  const minSize = Math.min(cw, ch)
   224→
   225→  // V→H: First cut vertical, then horizontal on one or both sections
   226→  for (let x = minSize; x <= pw - minSize; x += step) {
   227→    const leftW = x
   228→    const rightW = pw - x
   229→
   230→    for (let y = minSize; y <= ph - minSize; y += step) {
   231→      // Split left section horizontally: bottom-left + top-left
   232→      const blResult = calculatePiecesInRect(leftW, y, cw, ch)
   233→      const tlResult = calculatePiecesInRect(leftW, ph - y, cw, ch)
   234→      // Split right section horizontally: bottom-right + top-right
   235→      const brResult = calculatePiecesInRect(rightW, y, cw, ch)
   236→      const trResult = calculatePiecesInRect(rightW, ph - y, cw, ch)
   237→
   238→      // Option A: Cut left vertically, then both halves horizontally (4 blocks)
   239→      const totalA = blResult.pieces + tlResult.pieces + brResult.pieces + trResult.pieces
   240→      if (totalA > 0) {
   241→        const blocks: Block[] = []
   242→        if (blResult.pieces > 0) blocks.push(makeBlock('Blok A', 0, 0, leftW, y, blResult, cw, ch))
   243→        if (tlResult.pieces > 0) blocks.push(makeBlock('Blok B', 0, y, leftW, ph - y, tlResult, cw, ch))
   244→        if (brResult.pieces > 0) blocks.push(makeBlock('Blok C', x, 0, rightW, y, brResult, cw, ch))
   245→        if (trResult.pieces > 0) blocks.push(makeBlock('Blok D', x, y, rightW, ph - y, trResult, cw, ch))
   246→        scenarios.push({
   247→          strategy: `2-Stage V-H (${x.toFixed(1)}cm, ${y.toFixed(1)}cm)`,
   248→          type: 'block-cut',
   249→          cutPosition: x,
   250→          cutPositionY: y,
   251→          blocks,
   252→          total: totalA,
   253→          steps: [
   254→            `Langkah 1: Potong vertikal pada posisi ${x.toFixed(1)}cm`,
   255→            `Langkah 2: Potong horizontal pada posisi ${y.toFixed(1)}cm`,
   256→            ...blocks.map(b => `→ ${b.name}: ${b.pieces} potongan ${b.rotated ? '(rotasi 90°)' : ''}`)
   257→          ]
   258→        })
   259→      }
   260→
   261→      // Option B: Cut vertical first, then ONLY horizontal on left side (3 blocks)
   262→      const rightFull = calculatePiecesInRect(rightW, ph, cw, ch)
   263→      const totalB = blResult.pieces + tlResult.pieces + rightFull.pieces
   264→      if (totalB > 0 && totalB !== totalA) {
   265→        const blocks: Block[] = []
   266→        if (blResult.pieces > 0) blocks.push(makeBlock('Blok A', 0, 0, leftW, y, blResult, cw, ch))
   267→        if (tlResult.pieces > 0) blocks.push(makeBlock('Blok B', 0, y, leftW, ph - y, tlResult, cw, ch))
   268→        if (rightFull.pieces > 0) blocks.push(makeBlock('Blok C', x, 0, rightW, ph, rightFull, cw, ch))
   269→        scenarios.push({
   270→          strategy: `2-Stage V-H 3-Block (${x.toFixed(1)}cm, ${y.toFixed(1)}cm)`,
   271→          type: 'block-cut',
   272→          cutPosition: x,
   273→          cutPositionY: y,
   274→          blocks,
   275→          total: totalB,
   276→          steps: [
   277→            `Langkah 1: Potong vertikal pada posisi ${x.toFixed(1)}cm`,
   278→            `Langkah 2: Potong bagian kiri horizontal pada posisi ${y.toFixed(1)}cm`,
   279→            ...blocks.map(b => `→ ${b.name}: ${b.pieces} potongan ${b.rotated ? '(rotasi 90°)' : ''}`)
   280→          ]
   281→        })
   282→      }
   283→
   284→      // Option C: Cut vertical first, then ONLY horizontal on right side (3 blocks)
   285→      const leftFull = calculatePiecesInRect(leftW, ph, cw, ch)
   286→      const totalC = leftFull.pieces + brResult.pieces + trResult.pieces
   287→      if (totalC > 0 && totalC !== totalA) {
   288→        const blocks: Block[] = []
   289→        if (leftFull.pieces > 0) blocks.push(makeBlock('Blok A', 0, 0, leftW, ph, leftFull, cw, ch))
   290→        if (brResult.pieces > 0) blocks.push(makeBlock('Blok B', x, 0, rightW, y, brResult, cw, ch))
   291→        if (trResult.pieces > 0) blocks.push(makeBlock('Blok C', x, y, rightW, ph - y, trResult, cw, ch))
   292→        scenarios.push({
   293→          strategy: `2-Stage V-H 3-Block Alt (${x.toFixed(1)}cm, ${y.toFixed(1)}cm)`,
   294→          type: 'block-cut',
   295→          cutPosition: x,
   296→          cutPositionY: y,
   297→          blocks,
   298→          total: totalC,
   299→          steps: [
   300→            `Langkah 1: Potong vertikal pada posisi ${x.toFixed(1)}cm`,
   301→            `Langkah 2: Potong bagian kanan horizontal pada posisi ${y.toFixed(1)}cm`,
   302→            ...blocks.map(b => `→ ${b.name}: ${b.pieces} potongan ${b.rotated ? '(rotasi 90°)' : ''}`)
   303→          ]
   304→        })
   305→      }
   306→    }
   307→  }
   308→
   309→  return scenarios
   310→}

   312→// Hybrid layout - grid + rotated pieces in waste
   313→export function hybridLayout(pw: number, ph: number, cw: number, ch: number): Scenario[] {
   314→  const scenarios: Scenario[] = []
   315→
   316→  // Normal grid first
   317→  const normalResult = normalGridLayout(pw, ph, cw, ch)
   318→  const mainBlock = normalResult.blocks[0]
   319→
   320→  // Check if there's waste area that can fit rotated pieces
   321→  const wasteWidth = pw - mainBlock.usedWidth
   322→  const wasteHeight = ph - mainBlock.usedHeight
   323→
   324→  if (wasteWidth > 0 || wasteHeight > 0) {
   325→    // Try to fit rotated pieces in waste areas
   326→    if (wasteWidth >= Math.min(cw, ch)) {
   327→      const rotatedInWaste = calculatePiecesInRect(wasteWidth, ph, cw, ch)
   328→      if (rotatedInWaste.pieces > 0 && rotatedInWaste.rotated !== mainBlock.rotated) {
   329→        scenarios.push({
   330→          strategy: 'Hybrid (Grid + Sisa Rotasi)',
   331→          type: 'block-cut',
   332→          cutPosition: mainBlock.usedWidth,
   333→          cutPositionY: undefined,
   334→          blocks: [
   335→            mainBlock,
   336→            makeBlock('Blok B', mainBlock.usedWidth, 0, wasteWidth, ph, rotatedInWaste, cw, ch)
   337→          ],
   338→          total: mainBlock.pieces + rotatedInWaste.pieces,
   339→          steps: [
   340→            `Langkah 1: Potong grid normal ${mainBlock.horizontal}×${mainBlock.vertical}`,
   341→            `Langkah 2: Gunakan sisa (${wasteWidth.toFixed(1)}×${ph}cm) untuk ${rotatedInWaste.pieces} potongan rotasi`
   342→          ]
   343→        })
   344→      }
   345→    }
   346→
   347→    if (wasteHeight >= Math.min(cw, ch)) {
   348→      const rotatedInWaste = calculatePiecesInRect(pw, wasteHeight, cw, ch)
   349→      if (rotatedInWaste.pieces > 0 && rotatedInWaste.rotated !== mainBlock.rotated) {
   350→        scenarios.push({
   351→          strategy: 'Hybrid (Grid + Sisa Rotasi Vertikal)',
   352→          type: 'block-cut',
   353→          cutPosition: undefined,
   354→          cutPositionY: mainBlock.usedHeight,
   355→          blocks: [
   356→            mainBlock,
   357→            makeBlock('Blok B', 0, mainBlock.usedHeight, pw, wasteHeight, rotatedInWaste, cw, ch)
   358→          ],
   359→          total: mainBlock.pieces + rotatedInWaste.pieces,
   360→          steps: [
   361→            `Langkah 1: Potong grid normal ${mainBlock.horizontal}×${mainBlock.vertical}`,
   362→            `Langkah 2: Gunakan sisa (${pw}×${wasteHeight.toFixed(1)}cm) untuk ${rotatedInWaste.pieces} potongan rotasi`
   363→          ]
   364→        })
   365→      }
   366→    }
   367→  }
   368→
   369→  return scenarios
   370→}

   372→// NEW: Belah 2 - Split paper in half first, then cut each half into pieces
   373→// This is a common printing industry technique
   374→export function splitInHalf(pw: number, ph: number, cw: number, ch: number): Scenario[] {
   375→  const scenarios: Scenario[] = []
   376→
   377→  // --- Belah 2 Vertikal: Split width in half ---
   378→  const halfW = pw / 2
   379→  if (halfW >= Math.min(cw, ch)) {
   380→    const leftResult = calculatePiecesInRect(halfW, ph, cw, ch)
   381→    const rightResult = calculatePiecesInRect(halfW, ph, cw, ch)
   382→    const total = leftResult.pieces + rightResult.pieces
   383→
   384→    if (total > 0) {
   385→      const blocks: Block[] = []
   386→      if (leftResult.pieces > 0) blocks.push(makeBlock('Blok A', 0, 0, halfW, ph, leftResult, cw, ch))
   387→      if (rightResult.pieces > 0) blocks.push(makeBlock('Blok B', halfW, 0, halfW, ph, rightResult, cw, ch))
   388→
   389→      scenarios.push({
   390→        strategy: 'Belah 2 Vertikal',
   391→        type: 'split-half',
   392→        cutPosition: halfW,
   393→        cutPositionY: undefined,
   394→        blocks,
   395→        total,
   396→        steps: [
   397→          `Langkah 1: Belah kertas menjadi 2 bagian sama besar vertikal (masing-masing ${halfW.toFixed(1)}×${ph}cm)`,
   398→          `Langkah 2: Setiap bagian dipotong ${leftResult.horizontal}×${leftResult.vertical} = ${leftResult.pieces} potongan ${leftResult.rotated ? '(rotasi 90°)' : ''}`,
   399→          `Langkah 3: Total = ${total} potongan per lembar`
   400→        ]
   401→      })
   402→    }
   403→  }
   404→
   405→  // --- Belah 2 Horizontal: Split height in half ---
   406→  const halfH = ph / 2
   407→  if (halfH >= Math.min(cw, ch)) {
   408→    const bottomResult = calculatePiecesInRect(pw, halfH, cw, ch)
   409→    const topResult = calculatePiecesInRect(pw, halfH, cw, ch)
   410→    const total = bottomResult.pieces + topResult.pieces
   411→
   412→    if (total > 0) {
   413→      const blocks: Block[] = []
   414→      if (bottomResult.pieces > 0) blocks.push(makeBlock('Blok A', 0, 0, pw, halfH, bottomResult, cw, ch))
   415→      if (topResult.pieces > 0) blocks.push(makeBlock('Blok B', 0, halfH, pw, halfH, topResult, cw, ch))
   416→
   417→      scenarios.push({
   418→        strategy: 'Belah 2 Horizontal',
   419→        type: 'split-half',
   420→        cutPosition: undefined,
   421→        cutPositionY: halfH,
   422→        blocks,
   423→        total,
   424→        steps: [
   425→          `Langkah 1: Belah kertas menjadi 2 bagian sama besar horizontal (masing-masing ${pw}×${halfH.toFixed(1)}cm)`,
   426→          `Langkah 2: Setiap bagian dipotong ${bottomResult.horizontal}×${bottomResult.vertical} = ${bottomResult.pieces} potongan ${bottomResult.rotated ? '(rotasi 90°)' : ''}`,
   427→          `Langkah 3: Total = ${total} potongan per lembar`
   428→        ]
   429→      })
   430→    }
   431→  }
   432→
   433→  // --- Belah 2 Vertikal dengan rotasi berbeda per bagian ---
   434→  // Left half: try normal; Right half: try forced rotation (and vice versa)
   435→  if (halfW >= Math.min(cw, ch)) {
   436→    // Left normal, right rotated
   437→    const leftNormal = calculatePiecesInRect(halfW, ph, cw, ch)
   438→    // Force rotated for right half
   439→    const rightForcedNormal = Math.floor(halfW / cw) * Math.floor(ph / ch)
   440→    const rightForcedRotated = Math.floor(halfW / ch) * Math.floor(ph / cw)
   441→    const rightResult = rightForcedNormal >= rightForcedRotated
   442→      ? { pieces: rightForcedNormal, horizontal: Math.floor(halfW / cw), vertical: Math.floor(ph / ch), rotated: false }
   443→      : { pieces: rightForcedRotated, horizontal: Math.floor(halfW / ch), vertical: Math.floor(ph / cw), rotated: true }
   444→    
   445→    if (leftNormal.pieces > 0 && rightResult.pieces > 0) {
   446→      const total = leftNormal.pieces + rightResult.pieces
   447→      // Only add if different from the symmetric case
   448→      if (leftNormal.pieces !== rightResult.pieces || leftNormal.rotated !== rightResult.rotated) {
   449→        const blocks: Block[] = []
   450→        blocks.push(makeBlock('Blok A', 0, 0, halfW, ph, leftNormal, cw, ch))
   451→        blocks.push(makeBlock('Blok B', halfW, 0, halfW, ph, rightResult, cw, ch))
   452→        scenarios.push({
   453→          strategy: 'Belah 2 Vertikal (Campuran Rotasi)',
   454→          type: 'split-half',
   455→          cutPosition: halfW,
   456→          cutPositionY: undefined,
   457→          blocks,
   458→          total,
   459→          steps: [
   460→            `Langkah 1: Belah kertas menjadi 2 bagian vertikal (masing-masing ${halfW.toFixed(1)}×${ph}cm)`,
   461→            `Langkah 2: Bagian kiri: ${leftNormal.horizontal}×${leftNormal.vertical} = ${leftNormal.pieces} potongan ${leftNormal.rotated ? '(rotasi 90°)' : ''}`,
   462→            `Langkah 3: Bagian kanan: ${rightResult.horizontal}×${rightResult.vertical} = ${rightResult.pieces} potongan ${rightResult.rotated ? '(rotasi 90°)' : ''}`,
   463→            `Langkah 4: Total = ${total} potongan per lembar`
   464→          ]
   465→        })
   466→      }
   467→    }
   468→  }
   469→
   470→  if (halfH >= Math.min(cw, ch)) {
   471→    const topNormal = calculatePiecesInRect(pw, halfH, cw, ch)
   472→    const bottomForcedNormal = Math.floor(pw / cw) * Math.floor(halfH / ch)
   473→    const bottomForcedRotated = Math.floor(pw / ch) * Math.floor(halfH / cw)
   474→    const bottomResult = bottomForcedNormal >= bottomForcedRotated
   475→      ? { pieces: bottomForcedNormal, horizontal: Math.floor(pw / cw), vertical: Math.floor(halfH / ch), rotated: false }
   476→      : { pieces: bottomForcedRotated, horizontal: Math.floor(pw / ch), vertical: Math.floor(halfH / cw), rotated: true }
   477→
   478→    if (topNormal.pieces > 0 && bottomResult.pieces > 0) {
   479→      const total = topNormal.pieces + bottomResult.pieces
   480→      if (topNormal.pieces !== bottomResult.pieces || topNormal.rotated !== bottomResult.rotated) {
   481→        const blocks: Block[] = []
   482→        blocks.push(makeBlock('Blok A', 0, 0, pw, halfH, topNormal, cw, ch))
   483→        blocks.push(makeBlock('Blok B', 0, halfH, pw, halfH, bottomResult, cw, ch))
   484→        scenarios.push({
   485→          strategy: 'Belah 2 Horizontal (Campuran Rotasi)',
   486→          type: 'split-half',
   487→          cutPosition: undefined,
   488→          cutPositionY: halfH,
   489→          blocks,
   490→          total,
   491→          steps: [
   492→            `Langkah 1: Belah kertas menjadi 2 bagian horizontal (masing-masing ${pw}×${halfH.toFixed(1)}cm)`,
   493→            `Langkah 2: Bagian bawah: ${topNormal.horizontal}×${topNormal.vertical} = ${topNormal.pieces} potongan ${topNormal.rotated ? '(rotasi 90°)' : ''}`,
   494→            `Langkah 3: Bagian atas: ${bottomResult.horizontal}×${bottomResult.vertical} = ${bottomResult.pieces} potongan ${bottomResult.rotated ? '(rotasi 90°)' : ''}`,
   495→            `Langkah 4: Total = ${total} potongan per lembar`
   496→          ]
   497→        })
   498→      }
   499→    }
   500→  }
   501→
   502→  return scenarios
   503→}

   505→// NEW: Belah 4 - Split paper into 4 quadrants (2×2), then cut each quadrant
   506→export function splitInFour(pw: number, ph: number, cw: number, ch: number): Scenario[] {
   507→  const scenarios: Scenario[] = []
   508→
   509→  const halfW = pw / 2
   510→  const halfH = ph / 2
   511→
   512→  if (halfW < Math.min(cw, ch) || halfH < Math.min(cw, ch)) return scenarios
   513→
   514→  // Each quadrant: halfW × halfH
   515→  const qResult = calculatePiecesInRect(halfW, halfH, cw, ch)
   516→  const total = qResult.pieces * 4
   517→
   518→  if (total > 0) {
   519→    scenarios.push({
   520→      strategy: 'Belah 4 (2×2)',
   521→      type: 'split-four',
   522→      cutPosition: halfW,
   523→      cutPositionY: halfH,
   524→      blocks: [
   525→        makeBlock('Blok A', 0, 0, halfW, halfH, qResult, cw, ch),
   526→        makeBlock('Blok B', halfW, 0, halfW, halfH, qResult, cw, ch),
   527→        makeBlock('Blok C', 0, halfH, halfW, halfH, qResult, cw, ch),
   528→        makeBlock('Blok D', halfW, halfH, halfW, halfH, qResult, cw, ch),
   529→      ],
   530→      total,
   531→      steps: [
   532→        `Langkah 1: Belah kertas jadi 2 bagian vertikal (masing-masing ${halfW.toFixed(1)}×${ph}cm)`,
   533→        `Langkah 2: Masing-masing bagian dibelah horizontal jadi 2 (masing-masing ${halfW.toFixed(1)}×${halfH.toFixed(1)}cm)`,
   534→        `Langkah 3: Setiap kuadran dipotong ${qResult.horizontal}×${qResult.vertical} = ${qResult.pieces} potongan ${qResult.rotated ? '(rotasi 90°)' : ''}`,
   535→        `Langkah 4: Total = 4 × ${qResult.pieces} = ${total} potongan per lembar`
   536→      ]
   537→    })
   538→  }
   539→
   540→  // Belah 4 dengan rotasi berbeda per baris
   541→  // Bottom row: normal, Top row: try different orientation
   542→  const bottomQ = calculatePiecesInRect(halfW, halfH, cw, ch)
   543→  const topQNormal = Math.floor(halfW / cw) * Math.floor(halfH / ch)
   544→  const topQRotated = Math.floor(halfW / ch) * Math.floor(halfH / cw)
   545→  const topQ = topQNormal >= topQRotated
   546→    ? { pieces: topQNormal, horizontal: Math.floor(halfW / cw), vertical: Math.floor(halfH / ch), rotated: false }
   547→    : { pieces: topQRotated, horizontal: Math.floor(halfW / ch), vertical: Math.floor(halfH / cw), rotated: true }
   548→
   549→  if (bottomQ.pieces > 0 && topQ.pieces > 0) {
   550→    const totalMixed = (bottomQ.pieces + topQ.pieces) * 2
   551→    if (totalMixed > total && bottomQ.rotated !== topQ.rotated) {
   552→      scenarios.push({
   553→        strategy: 'Belah 4 (2×2 Campuran Rotasi)',
   554→        type: 'split-four',
   555→        cutPosition: halfW,
   556→        cutPositionY: halfH,
   557→        blocks: [
   558→          makeBlock('Blok A', 0, 0, halfW, halfH, bottomQ, cw, ch),
   559→          makeBlock('Blok B', halfW, 0, halfW, halfH, bottomQ, cw, ch),
   560→          makeBlock('Blok C', 0, halfH, halfW, halfH, topQ, cw, ch),
   561→          makeBlock('Blok D', halfW, halfH, halfW, halfH, topQ, cw, ch),
   562→        ],
   563→        total: totalMixed,
   564→        steps: [
   565→          `Langkah 1: Belah kertas jadi 2 bagian vertikal (masing-masing ${halfW.toFixed(1)}×${ph}cm)`,
   566→          `Langkah 2: Belah horizontal jadi 4 kuadran (masing-masing ${halfW.toFixed(1)}×${halfH.toFixed(1)}cm)`,
   567→          `Langkah 3: Baris bawah: ${bottomQ.pieces} potongan ${bottomQ.rotated ? '(rotasi 90°)' : ''} per kuadran`,
   568→          `Langkah 4: Baris atas: ${topQ.pieces} potongan ${topQ.rotated ? '(rotasi 90°)' : ''} per kuadran`,
   569→          `Langkah 5: Total = ${(bottomQ.pieces + topQ.pieces) * 2} potongan per lembar`
   570→        ]
   571→      })
   572→    }
   573→  }
   574→
   575→  return scenarios
   576→}

   580→  paperHeight: number
   581→  cutWidth: number
   582→  cutHeight: number
   583→  quantity: number
   584→  pricePerSheet: number
   585→  optimizationMode: 'fast' | 'maximal'
   586→  customerName: string
   587→  paperMaterial: string
   588→  grammage: number
   589→}

   591→// Main calculation function - standalone, no React dependencies
   592→export function calculateCuts(params: CalculateCutsParams): CuttingResult {
   593→  const { paperWidth: pw, paperHeight: ph, cutWidth: cw, cutHeight: ch, quantity: qty, pricePerSheet: price, optimizationMode, customerName, paperMaterial, grammage } = params
   594→
   595→  const scenarios: Scenario[] = []
   596→
   597→  // 1. Normal grid layout (always try)
   598→  scenarios.push(normalGridLayout(pw, ph, cw, ch))
   599→
   600→  // 2. Belah 2 - Split in half first (always try - fast and common technique)
   601→  scenarios.push(...splitInHalf(pw, ph, cw, ch))
   602→
   603→  // 3. Belah 4 - Split into 4 quadrants (always try)
   604→  scenarios.push(...splitInFour(pw, ph, cw, ch))
   605→
   606→  // 4. Guillotine single cut
   607→  if (optimizationMode === 'maximal') {
   608→    scenarios.push(...guillotineSingleCut(pw, ph, cw, ch, optimizationMode))
   609→  } else {
   610→    const fastScenarios = guillotineSingleCut(pw, ph, cw, ch, optimizationMode)
   611→    scenarios.push(...fastScenarios.slice(0, Math.min(20, fastScenarios.length)))
   612→  }
   613→
   614→  // 5. Hybrid layout (always try - can improve results significantly)
   615→  scenarios.push(...hybridLayout(pw, ph, cw, ch))
   616→
   617→  // 6. Two-stage guillotine (expensive - only in maximal mode, or limited in fast mode)
   618→  if (optimizationMode === 'maximal') {
   619→    scenarios.push(...twoStageGuillotine(pw, ph, cw, ch, optimizationMode))
   620→  } else {
   621→    const fastScenarios = twoStageGuillotine(pw, ph, cw, ch, optimizationMode)
   622→    scenarios.push(...fastScenarios.slice(0, Math.min(15, fastScenarios.length)))
   623→  }
   624→
   625→  // Find best scenario (prioritize: 1. max pieces, 2. min waste)
   626→  let bestScenario = scenarios[0]
   627→  for (const scenario of scenarios) {
   628→    if (scenario.total > bestScenario.total) {
   629→      bestScenario = scenario
   630→    } else if (scenario.total === bestScenario.total) {
   631→      // Calculate waste for tie-breaking
   632→      const currentWaste = scenario.blocks.reduce((sum, b) =>
   633→        sum + (b.wasteWidth * b.height + b.width * b.wasteHeight - b.wasteWidth * b.wasteHeight), 0)
   634→      const bestWaste = bestScenario.blocks.reduce((sum, b) =>
   635→        sum + (b.wasteWidth * b.height + b.width * b.wasteHeight - b.wasteWidth * b.wasteHeight), 0)
   636→
   637→      if (currentWaste < bestWaste) {
   638→        bestScenario = scenario
   639→      }
   640→    }
   641→  }
   642→
   643→  // Calculate total waste and efficiency
   644→  const totalUsedArea = bestScenario.blocks.reduce((sum, b) =>
   645→    sum + (b.usedWidth * b.usedHeight), 0)
   646→  const totalPaperArea = pw * ph
   647→  const totalWasteArea = totalPaperArea - totalUsedArea
   648→  const efficiency = ((totalUsedArea / totalPaperArea) * 100)
   649→
   650→  // Calculate sheets needed and price
   651→  const sheetsNeeded = qty > 0 ? Math.ceil(qty / bestScenario.total) : 1
   652→  const totalPrice = sheetsNeeded * price
   653→
   654→  return {
   655→    totalPieces: bestScenario.total,
   656→    quantity: qty,
   657→    sheetsNeeded,
   658→    totalPrice,
   659→    pricePerSheet: price,
   660→    cutPosition: bestScenario.cutPosition,
   661→    cutPositionY: bestScenario.cutPositionY,
   662→    blocks: bestScenario.blocks,
   663→    paperWidth: pw,
   664→    paperHeight: ph,
   665→    cutWidth: cw,
   666→    cutHeight: ch,
   667→    customerName,
   668→    paperMaterial,
   669→    grammage,
   670→    scenarioType: bestScenario.type,
   671→    totalWasteArea,
   672→    efficiency,
   673→    steps: bestScenario.steps,
   674→    strategy: bestScenario.strategy
   675→  }
   676→}
