   379→  const getFinishingCost = (finishing: Finishing): { cost: number; isMin: boolean; breakdown: string } => {
   380→    const qty = parseInt(formData.quantity) || 0
   381→    const cw = parseFloat(formData.cutWidth) || 0
   382→    const ch = parseFloat(formData.cutHeight) || 0
   383→
   384→    const minSheets = finishing.minimumSheets
   385→    const minPrice = finishing.minimumPrice
   386→    const hargaLebih = finishing.additionalPrice
   387→    const hargaPerCm = finishing.pricePerCm
   388→    const isPond = finishing.name.toLowerCase().includes('pond')
   389→
   390→    if (qty <= 0) {
   391→      return { cost: minPrice, isMin: true, breakdown: `Harga minimum: Rp ${Math.round(minPrice).toLocaleString('id-ID')}` }
   392→    }
   393→
   394→    if (isPond) {
   395→      if (qty <= minSheets) {
   396→        return {
   397→          cost: minPrice,
   398→          isMin: true,
   399→          breakdown: `Qty ${qty} ≤ minim ${minSheets} → Harga minimum: Rp ${Math.round(minPrice).toLocaleString('id-ID')}`
   400→        }
   401→      }
   402→      const selisih = qty - minSheets
   403→      const tambahan = selisih * hargaLebih
   404→      const total = minPrice + tambahan
   405→      return {
   406→        cost: total,
   407→        isMin: false,
   408→        breakdown: `Harga minimum Rp ${Math.round(minPrice).toLocaleString('id-ID')} + ((${qty} - ${minSheets}) × Rp ${Math.round(hargaLebih).toLocaleString('id-ID')}) = Rp ${Math.round(total).toLocaleString('id-ID')}`
   409→      }
   410→    }
   411→
   412→    let part1 = 0
   413→    let part1Text = ''
   414→    if (qty > minSheets && hargaLebih > 0) {
   415→      const selisih = qty - minSheets
   416→      part1 = selisih * hargaLebih
   417→      part1Text = `(${qty} - ${minSheets}) × Rp ${Math.round(hargaLebih).toLocaleString('id-ID')} = Rp ${Math.round(part1).toLocaleString('id-ID')}`
   418→    }
   419→
   420→    let part2 = 0
   421→    let part2Text = ''
   422→    if (cw > 0 && ch > 0 && hargaPerCm > 0) {
   423→      const areaCost = cw * ch * hargaPerCm
   424→      part2 = areaCost * qty
   425→      part2Text = `(${cw} × ${ch}) × Rp ${Math.round(hargaPerCm).toLocaleString('id-ID')} × ${qty} = Rp ${Math.round(part2).toLocaleString('id-ID')}`
   426→    }
   427→
   428→    const total = part1 + part2
   429→
   430→    if (total <= minPrice) {
   431→      const parts = [part1Text, part2Text].filter(Boolean)
   432→      const calcText = parts.length > 0 ? `${parts.join(' + ')} = Rp ${Math.round(total).toLocaleString('id-ID')} ≤ Rp ${Math.round(minPrice).toLocaleString('id-ID')}` : ''
   433→      return {
   434→        cost: minPrice,
   435→        isMin: true,
   436→        breakdown: calcText ? `${calcText} → Harga minimum: Rp ${Math.round(minPrice).toLocaleString('id-ID')}` : `Harga minimum: Rp ${Math.round(minPrice).toLocaleString('id-ID')}`
   437→      }
   438→    }
   439→
   440→    const parts = [part1Text, part2Text].filter(Boolean)
   441→    const breakdown = parts.join(' + ') + ` = Rp ${Math.round(total).toLocaleString('id-ID')}`
   442→
   443→    return { cost: total, isMin: false, breakdown }
   444→  }
