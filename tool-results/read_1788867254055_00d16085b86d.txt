    93→// --- Parse dataJson for document info ---
    94→function parseDocInfo(entry: HistoryEntry) {
    95→  try {
    96→    const parsed = JSON.parse(entry.dataJson)
    97→    const items = parsed.items || []
    98→    const itemCount = items.length
    99→    const firstItem = items[0]
   100→    const namaBarang = firstItem?.deskripsi || ''
   101→    const hargaSatuan = firstItem?.harga || 0
   102→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0)
   103→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0)
   104→    const ppn = parsed.ppn || 0
   105→    const dpPercent = parsed.dp || 0
   106→    const totalHarga = subtotal + (subtotal * ppn / 100)
   107→    // Always derive dpAmount from originalTotal
   108→    const originalTotal = parsed.originalTotal !== undefined ? parsed.originalTotal : totalHarga
   109→    const dpAmount = originalTotal * (dpPercent / 100)
   110→    const sisa = totalHarga - dpAmount
   111→    const lunas = parsed.lunas === true
   112→    const tanggalJatuhTempo = parsed.tanggalJatuhTempo || ''
   113→    const tanggalPelunasan = parsed.tanggalPelunasan || ''
   114→    const referensi = parsed.referensi || ''
   115→    const referensiInvoiceNomor = parsed.referensiInvoiceNomor || ''
   116→    const isPelunasan = entry.docType === 'invoice-pelunasan'
   117→    const uangCapek = parsed.uangCapek || 0
   118→    return { namaBarang, hargaSatuan, totalQty, totalHarga, dpPercent, dp: dpAmount, sisa, lunas, tanggalJatuhTempo, tanggalPelunasan, referensi, referensiInvoiceNomor, isPelunasan, originalTotal, uangCapek, itemCount }
   119→  } catch {
   120→    return { namaBarang: '', hargaSatuan: 0, totalQty: 0, totalHarga: 0, dpPercent: 0, dp: 0, sisa: 0, lunas: false, tanggalJatuhTempo: '', tanggalPelunasan: '', referensi: '', referensiInvoiceNomor: '', isPelunasan: false, originalTotal: 0, uangCapek: 0, itemCount: 0 }
   121→  }
   122→}

   124→// --- Parse dataJson for InvoiceData ---
   125→function parseInvoiceData(entry: HistoryEntry): InvoiceData {
   126→  try {
   127→    const parsed = JSON.parse(entry.dataJson)
   128→    const company: CompanyInfo = {
   129→      nama: parsed.company?.nama || DEFAULT_COMPANY.nama,
   130→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
   131→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
   132→      email: parsed.company?.email || DEFAULT_COMPANY.email,
   133→      npwp: parsed.company?.npwp || '',
   134→      website: parsed.company?.website || '',
   135→      ppn: parsed.company?.ppn ?? parsed.ppn ?? 11,
   136→      logo: parsed.company?.logo || '',
   137→      bankName: parsed.company?.bankName || '',
   138→      bankAccount: parsed.company?.bankAccount || '',
   139→      bankHolder: parsed.company?.bankHolder || '',
   140→      bankName2: parsed.company?.bankName2 || '',
   141→      bankAccount2: parsed.company?.bankAccount2 || '',
   142→      bankHolder2: parsed.company?.bankHolder2 || '',
   143→    }
   144→    const client = parsed.client || { nama: '', kontak: '', alamat: '' }
   145→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number }, i: number) => ({
   146→      id: it.id || `item-${i}`,
   147→      deskripsi: it.deskripsi || '',
   148→      qty: it.qty || 0,
   149→      satuan: it.satuan || '',
   150→      harga: it.harga || 0,
   151→    }))
   152→    const docType = (entry.docType === 'invoice-pelunasan' ? 'invoice-pelunasan' : 'invoice') as 'invoice' | 'invoice-pelunasan'
   153→    return {
   154→      type: docType,
   155→      company,
   156→      nomor: parsed.nomor || entry.nomor || '',
   157→      tanggal: parsed.tanggal || entry.tanggal || '',
   158→      referensi: parsed.referensi || '',
   159→      client,
   160→      items,
   161→      ppn: parsed.ppn ?? 11,
   162→      dp: parsed.dp || 0,
   163→      dpAmount: parsed.dpAmount,
   164→      catatan: parsed.catatan || '',
   165→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   166→      caraPembayaran: parsed.caraPembayaran || '',
   167→      tanggalGiro: parsed.tanggalGiro || '',
   168→      lunas: parsed.lunas === true,
   169→      tanggalPelunasan: parsed.tanggalPelunasan || '',
   170→      referensiInvoiceId: parsed.referensiInvoiceId || '',
   171→      referensiInvoiceNomor: parsed.referensiInvoiceNomor || '',
   172→      originalTotal: parsed.originalTotal,
   173→      uangCapek: parsed.uangCapek || 0,
   174→    }
   175→  } catch {
   176→    const docType = (entry.docType === 'invoice-pelunasan' ? 'invoice-pelunasan' : 'invoice') as 'invoice' | 'invoice-pelunasan'
   177→    return {
   178→      type: docType,
   179→      company: { ...DEFAULT_COMPANY },
   180→      nomor: entry.nomor || '',
   181→      tanggal: entry.tanggal || '',
   182→      referensi: '',
   183→      client: { nama: '', kontak: '', alamat: '' },
   184→      items: [],
   185→      ppn: 11,
   186→      dp: 0,
   187→      catatan: '',
   188→      tanggalJatuhTempo: '',
   189→      caraPembayaran: '',
   190→      tanggalGiro: '',
   191→      lunas: false,
   192→      tanggalPelunasan: '',
   193→      referensiInvoiceId: '',
   194→      referensiInvoiceNomor: '',
   195→      uangCapek: 0,
   196→    }
   197→  }
   198→}
