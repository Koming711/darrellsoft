    35→// --- Types ---

    47→// --- Helper ---
    48→function getTodayStr(): string {
    49→  const d = new Date();
    50→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
    51→}
    52→
    53→// --- Parse dataJson for document info ---
    54→function parseDocInfo(entry: HistoryEntry) {
    55→  try {
    56→    const parsed = JSON.parse(entry.dataJson);
    57→    const items = parsed.items || [];
    58→    const firstItem = items[0];
    59→    const namaBarang = firstItem?.deskripsi || '';
    60→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0);
    61→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0);
    62→    const ppn = parsed.ppn || 0;
    63→    const dpPercent = parsed.dp || 0;
    64→    const totalHarga = subtotal + (subtotal * ppn / 100);
    65→    // Use saved dpAmount if available, otherwise calculate from percentage
    66→    const dpAmount = parsed.dpAmount !== undefined ? parsed.dpAmount : totalHarga * (dpPercent / 100);
    67→    const sisa = totalHarga - dpAmount;
    68→    const lunas = parsed.lunas === true;
    69→    const tanggalJatuhTempo = parsed.tanggalJatuhTempo || '';
    70→    const tanggalPelunasan = parsed.tanggalPelunasan || '';
    71→    return { namaBarang, totalQty, totalHarga, dpPercent, dp: dpAmount, sisa, lunas, tanggalJatuhTempo, tanggalPelunasan };
    72→  } catch {
    73→    return { namaBarang: '', totalQty: 0, totalHarga: 0, dpPercent: 0, dp: 0, sisa: 0, lunas: false, tanggalJatuhTempo: '', tanggalPelunasan: '' };
    74→  }
    75→}
    76→
    77→// --- Parse dataJson for InvoiceData ---
    78→function parseInvoiceData(entry: HistoryEntry): InvoiceData {
    79→  try {
    80→    const parsed = JSON.parse(entry.dataJson);
    81→    const company: CompanyInfo = {
    82→      nama: parsed.company?.nama || DEFAULT_COMPANY.nama,
    83→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
    84→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
    85→      email: parsed.company?.email || DEFAULT_COMPANY.email,
    86→      npwp: parsed.company?.npwp || '',
    87→      website: parsed.company?.website || '',
    88→      ppn: parsed.company?.ppn ?? parsed.ppn ?? 11,
    89→      logo: parsed.company?.logo || '',
    90→      bankName: parsed.company?.bankName || '',
    91→      bankAccount: parsed.company?.bankAccount || '',
    92→      bankHolder: parsed.company?.bankHolder || '',
    93→      bankName2: parsed.company?.bankName2 || '',
    94→      bankAccount2: parsed.company?.bankAccount2 || '',
    95→      bankHolder2: parsed.company?.bankHolder2 || '',
    96→    };
    97→    const client = parsed.client || { nama: '', kontak: '', alamat: '' };
    98→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number }, i: number) => ({
    99→      id: it.id || `item-${i}`,
   100→      deskripsi: it.deskripsi || '',
   101→      qty: it.qty || 0,
   102→      satuan: it.satuan || '',
   103→      harga: it.harga || 0,
   104→    }));
   105→    return {
   106→      type: 'invoice',
   107→      company,
   108→      nomor: parsed.nomor || entry.nomor || '',
   109→      tanggal: parsed.tanggal || entry.tanggal || '',
   110→      referensi: parsed.referensi || '',
   111→      client,
   112→      items,
   113→      ppn: parsed.ppn ?? 11,
   114→      dp: parsed.dp || 0,
   115→      dpAmount: parsed.dpAmount,
   116→      catatan: parsed.catatan || '',
   117→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   118→      caraPembayaran: parsed.caraPembayaran || '',
   119→      tanggalGiro: parsed.tanggalGiro || '',
   120→      lunas: parsed.lunas === true,
   121→      tanggalPelunasan: parsed.tanggalPelunasan || '',
   122→    };
   123→  } catch {
   124→    return {
   125→      type: 'invoice',
   126→      company: { ...DEFAULT_COMPANY },
   127→      nomor: entry.nomor || '',
   128→      tanggal: entry.tanggal || '',
   129→      referensi: '',
   130→      client: { nama: '', kontak: '', alamat: '' },
   131→      items: [],
   132→      ppn: 11,
   133→      dp: 0,
   134→      catatan: '',
   135→      tanggalJatuhTempo: '',
   136→      caraPembayaran: '',
   137→      tanggalGiro: '',
   138→      lunas: false,
   139→      tanggalPelunasan: '',
   140→    };
   141→  }
   142→}

   144→export function InvoicePelunasanEditor() {
   145→  // Local invoice state (not using global store to avoid conflicts with regular editor)
   146→  const [invoiceData, setInvoiceData] = useState<InvoiceData | null>(null);
   147→  const [selectedEntry, setSelectedEntry] = useState<HistoryEntry | null>(null);
   148→  const [invoiceHistory, setInvoiceHistory] = useState<HistoryEntry[]>([]);
   149→  const [loading, setLoading] = useState(true);
   150→  const [searchQuery, setSearchQuery] = useState('');
   151→  const [dropdownOpen, setDropdownOpen] = useState(false);
   152→  const [pelunasanSaving, setPelunasanSaving] = useState(false);
   153→  const [jpgGenerating, setJpgGenerating] = useState(false);
   154→
   155→  // Pelunasan fields
   156→  const [lunasToggle, setLunasToggle] = useState(false);
   157→  const [tanggalPelunasan, setTanggalPelunasan] = useState('');
   158→  const [tanggalJatuhTempo, setTanggalJatuhTempo] = useState('');
   159→  const [caraPembayaran, setCaraPembayaran] = useState('');
   160→  const [tanggalGiro, setTanggalGiro] = useState('');
   161→
   162→  // Original DP amount — fixed when invoice is first selected, does NOT change when items are added
   163→  const [originalDpAmount, setOriginalDpAmount] = useState(0);
   164→
   165→  const fetchHistory = useCallback(async () => {
   166→    try {
   167→      setLoading(true);
   168→      const headers = getAuthHeaders();
   169→      const res = await fetch('/api/history?docType=invoice', { headers });
   170→      if (res.ok) {
   171→        const json = await res.json();
   172→        setInvoiceHistory(json.data || []);
   173→      }
   174→    } catch (err) {
   175→      console.error('Failed to fetch invoice history:', err);
   176→    } finally {
   177→      setLoading(false);
   178→    }
   179→  }, []);
   180→
   181→  useEffect(() => {
   182→    fetchHistory();
   183→  }, [fetchHistory]);
   184→
   185→  useEffect(() => {
   186→    const handler = () => fetchHistory();
   187→    window.addEventListener('dokupro:history-updated', handler);
   188→    return () => window.removeEventListener('dokupro:history-updated', handler);
   189→  }, [fetchHistory]);
   190→
   191→  // Filter pending invoices (with DP and not yet lunas)
   192→  // Business rule: having DP means belum lunas, only explicit lunas flag marks it as paid
   193→  const pendingInvoices = useMemo(() => {
   194→    return invoiceHistory.filter(entry => {
   195→      const info = parseDocInfo(entry);
   196→      return (info.dpPercent > 0 || info.dp > 0) && !info.lunas;
   197→    });
   198→  }, [invoiceHistory]);
   199→
   200→  // Filter by search
   201→  const filteredPending = useMemo(() => {
   202→    if (!searchQuery.trim()) return pendingInvoices;
   203→    const q = searchQuery.toLowerCase().trim();
   204→    return pendingInvoices.filter(entry => {
   205→      const info = parseDocInfo(entry);
   206→      return entry.nomor?.toLowerCase().includes(q) || entry.pihakKedua?.toLowerCase().includes(q) || info.namaBarang?.toLowerCase().includes(q);
   207→    });
   208→  }, [pendingInvoices, searchQuery]);
   209→
   210→  // Select an invoice
   211→  const selectInvoice = (entry: HistoryEntry) => {
   212→    const parsed = parseInvoiceData(entry);
   213→    const info = parseDocInfo(entry);
   214→    setInvoiceData(parsed);
   215→    setSelectedEntry(entry);
   216→    setLunasToggle(info.lunas);
   217→    setTanggalPelunasan(info.tanggalPelunasan || getTodayStr());
   218→    setTanggalJatuhTempo(info.tanggalJatuhTempo || '');
   219→    setCaraPembayaran(parsed.caraPembayaran || '');
   220→    setTanggalGiro(parsed.tanggalGiro || '');
   221→    // Fix: store original DP amount so it doesn't change when items are added
   222→    setOriginalDpAmount(info.dp);
   223→    setDropdownOpen(false);
   224→  };
   225→
   226→  // Clear selection
   227→  const clearSelection = () => {
   228→    setInvoiceData(null);
   229→    setSelectedEntry(null);
   230→    setLunasToggle(false);
   231→    setTanggalPelunasan('');
   232→    setTanggalJatuhTempo('');
   233→    setCaraPembayaran('');
   234→    setTanggalGiro('');
   235→    setOriginalDpAmount(0);
   236→  };
   237→
   238→  // Update invoice data locally
   239→  const updateInvoice = (updates: Partial<InvoiceData>) => {
   240→    setInvoiceData(prev => prev ? { ...prev, ...updates } : null);
   241→  };
   242→
   243→  const updateCompany = (company: CompanyInfo) => {
   244→    setInvoiceData(prev => prev ? { ...prev, company } : null);
   245→  };
   246→
   247→  const updateClient = (field: string, value: string) => {
   248→    setInvoiceData(prev => prev ? { ...prev, client: { ...prev.client, [field]: value } } : null);
   249→  };
   250→
   251→  // Build preview data with pelunasan overrides
   252→  const previewData: InvoiceData | null = useMemo(() => {
   253→    if (!invoiceData) {
   254→      // Default preview when no invoice is selected
   255→      return {
   256→        type: 'invoice',
   257→        company: { ...DEFAULT_COMPANY },
   258→        nomor: '-',
   259→        tanggal: getTodayStr(),
   260→        referensi: '',
   261→        client: { nama: '', kontak: '', alamat: '' },
   262→        items: [],
   263→        ppn: 11,
   264→        dp: 0,
   265→        catatan: '',
   266→        tanggalJatuhTempo: '',
   267→        caraPembayaran: '',
   268→        tanggalGiro: '',
   269→        lunas: false,
   270→        tanggalPelunasan: '',
   271→      };
   272→    }
   273→    return {
   274→      ...invoiceData,
   275→      tanggalJatuhTempo,
   276→      caraPembayaran: caraPembayaran as InvoiceData['caraPembayaran'],
   277→      tanggalGiro,
   278→      lunas: lunasToggle,
   279→      tanggalPelunasan: lunasToggle ? tanggalPelunasan : '',
   280→    };
   281→  }, [invoiceData, tanggalJatuhTempo, caraPembayaran, tanggalGiro, lunasToggle, tanggalPelunasan]);
   282→
   283→  // Calculate amounts — DP is FIXED at original amount, only subtotal and sisa change
   284→  const subtotal = invoiceData?.items.reduce((sum, item) => sum + item.qty * item.harga, 0) || 0;
   285→  const ppnAmount = subtotal * ((invoiceData?.ppn || 0) / 100);
   286→  const total = subtotal + ppnAmount;
   287→  const dpPercent = invoiceData?.dp || 0;
   288→  // DP amount stays fixed at original value, doesn't recalculate when items change
   289→  const dpAmount = originalDpAmount;
   290→  const sisa = total - dpAmount;
   291→
   292→  // Generate JPG from preview and send to WhatsApp
   293→  const handleGenerateJpg = async () => {
   294→    const previewEl = document.querySelector('[data-document-preview]') as HTMLElement;
   295→    if (!previewEl) {
   296→      toast.error('Preview tidak ditemukan');
   297→      return;
   298→    }
   299→    setJpgGenerating(true);
   300→    try {
   301→      const dataUrl = await toJpeg(previewEl, {
   302→        quality: 0.95,
   303→        pixelRatio: 2,
   304→        backgroundColor: '#ffffff',
   305→      });
   306→
   307→      // Convert data URL to Blob for sharing
   308→      const res = await fetch(dataUrl);
   309→      const blob = await res.blob();
   310→      const fileName = `Invoice-PEL-${invoiceData?.nomor || 'draft'}.jpg`;
   311→      const file = new File([blob], fileName, { type: 'image/jpeg' });
   312→
   313→      // Try Web Share API (mobile) — can share file directly to WhatsApp
   314→      if (navigator.share && navigator.canShare?.({ files: [file] })) {
   315→        await navigator.share({
   316→          files: [file],
   317→          title: `Invoice ${invoiceData?.nomor || ''}`,
   318→          text: `Berikut invoice ${invoiceData?.nomor || ''} dari ${invoiceData?.company?.nama || ''}`,
   319→        });
   320→        toast.success('JPG berhasil dikirim ke WhatsApp');
   321→      } else {
   322→        // Fallback (desktop): download JPG + open WhatsApp link
   323→        const link = document.createElement('a');
   324→        link.download = fileName;
   325→        link.href = dataUrl;
   326→        link.click();
   327→
   328→        const phone = invoiceData?.client?.kontak?.replace(/\D/g, '') || '';
   329→        const message = encodeURIComponent(
   330→          `Berikut invoice ${invoiceData?.nomor || ''} dari ${invoiceData?.company?.nama || ''}\n\nJPG invoice sudah didownload, silakan lampirkan ke chat ini.`
   331→        );
   332→        const waUrl = phone
   333→          ? `https://wa.me/${phone}?text=${message}`
   334→          : `https://wa.me/?text=${message}`;
   335→        window.open(waUrl, '_blank');
   336→        toast.success('JPG didownload & WhatsApp terbuka');
   337→      }
   338→    } catch (err) {
   339→      console.error('Failed to generate JPG:', err);
   340→      toast.error('Gagal membuat JPG');
   341→    } finally {
   342→      setJpgGenerating(false);
   343→    }
   344→  };
   345→
   346→  // Save pelunasan — update the existing history entry
   347→  const handleSavePelunasan = async () => {
   348→    if (!selectedEntry || !invoiceData) return;
   349→    setPelunasanSaving(true);
   350→    try {
   351→      const parsed = JSON.parse(selectedEntry.dataJson);
   352→      // Update pelunasan fields
   353→      parsed.tanggalJatuhTempo = tanggalJatuhTempo;
   354→      parsed.caraPembayaran = caraPembayaran;
   355→      parsed.tanggalGiro = tanggalGiro;
   356→      if (lunasToggle) {
   357→        parsed.lunas = true;
   358→        parsed.tanggalPelunasan = tanggalPelunasan || getTodayStr();
   359→      } else {
   360→        parsed.lunas = false;
   361→        parsed.tanggalPelunasan = '';
   362→      }
   363→      // Also update any edited fields from the form
   364→      parsed.company = invoiceData.company;
   365→      parsed.client = invoiceData.client;
   366→      parsed.items = invoiceData.items;
   367→      parsed.ppn = invoiceData.ppn;
   368→      parsed.dp = invoiceData.dp;
   369→      // Save fixed DP amount so it doesn't change on restore
   370→      parsed.dpAmount = originalDpAmount;
   371→      parsed.catatan = invoiceData.catatan;
   372→      parsed.nomor = invoiceData.nomor;
   373→      parsed.tanggal = invoiceData.tanggal;
   374→      parsed.referensi = invoiceData.referensi;
   375→      delete parsed.statusPembayaran;
   376→
   377→      const newDataJson = JSON.stringify(parsed);
   378→      const res = await fetcher(`/api/history/${selectedEntry.id}`, {
   379→        method: 'PUT',
   380→        headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   381→        body: JSON.stringify({ dataJson: newDataJson }),
   382→      });
   383→      if (res.ok) {
   384→        toast.success(lunasToggle ? 'Pelunasan berhasil dicatat!' : 'Invoice berhasil diperbarui');
   385→        clearSelection();
   386→        fetchHistory();
   387→        notifyDataChange('invoice');
   388→      } else {
   389→        toast.error('Gagal menyimpan perubahan');
   390→      }
   391→    } catch {
   392→      toast.error('Gagal menyimpan perubahan');
   393→    } finally {
   394→      setPelunasanSaving(false);
   395→    }
   396→  };
   397→
   398→  return (
   399→    <DocumentEditorLayout
   400→      title="Invoice Pelunasan"
   401→      previewContent={<InvoicePreview data={previewData!} showPelunasanLabel dpAmountOverride={originalDpAmount} />}
   402→      actions={
   403→        invoiceData ? (
   404→          <div className="flex items-center gap-2 flex-wrap">
   405→            <Button
   406→              variant="outline"
   407→              size="sm"
   408→              onClick={clearSelection}
   409→              className="border-gray-200 text-gray-600 hover:bg-gray-50"
   410→            >
   411→              <X className="mr-1.5 h-3.5 w-3.5" />
   412→              Batal
   413→            </Button>
   414→            <Button
   415→              size="sm"
   416→              onClick={() => {
   417→                toast.dismiss();
   418→                const origTitle = document.title;
   419→                document.title = ' ';
   420→                setTimeout(() => {
   421→                  window.print();
   422→                  document.title = origTitle;
   423→                }, 100);
   424→              }}
   425→              className="bg-emerald-600 hover:bg-emerald-700"
   426→            >
   427→              Cetak
   428→            </Button>
   429→            <Button
   430→              size="sm"
   431→              onClick={handleGenerateJpg}
   432→              disabled={jpgGenerating}
   433→              className="bg-violet-600 hover:bg-violet-700 text-white"
   434→            >
   435→              {jpgGenerating ? (
   436→                <><Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" /> Membuat...</>
   437→              ) : (
   438→                <><ImageIcon className="mr-1.5 h-3.5 w-3.5" /> JPG</>
   439→              )}
   440→            </Button>
   441→            <Button
   442→              size="sm"
   443→              onClick={handleSavePelunasan}
   444→              disabled={pelunasanSaving}
   445→              className={cn(
   446→                'text-white',
   447→                lunasToggle
   448→                  ? 'bg-green-600 hover:bg-green-700'
   449→                  : 'bg-blue-600 hover:bg-blue-700'
   450→              )}
   451→            >
   452→              {pelunasanSaving ? (
   453→                <><Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" /> Menyimpan...</>
   454→              ) : (
   455→                <><CheckCircle2 className="mr-1.5 h-3.5 w-3.5" /> {lunasToggle ? 'Simpan Pelunasan' : 'Simpan Perubahan'}</>
   456→              )}
   457→            </Button>
   458→          </div>
   459→        ) : undefined
   460→      }
   461→    >
   462→      {/* Invoice Selector */}
   463→      <div className="rounded-lg border bg-card p-3 sm:p-4 shadow-sm">
   464→        <h3 className="mb-3 text-xs font-semibold uppercase tracking-wider text-muted-foreground flex items-center gap-1.5">
   465→          <Wallet className="w-3.5 h-3.5 text-amber-600" />
   466→          Pilih Invoice Belum Lunas
   467→        </h3>
   468→        <Popover open={dropdownOpen} onOpenChange={setDropdownOpen}>
   469→          <PopoverAnchor asChild>
   470→            <div className="relative">
   471→              <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 w-3.5 h-3.5 text-slate-400" />
   472→              <input
   473→                type="text"
   474→                placeholder={selectedEntry ? `${selectedEntry.nomor} — ${selectedEntry.pihakKedua}` : "Cari no. invoice, customer..."}
   475→                value={selectedEntry ? `${selectedEntry.nomor} — ${selectedEntry.pihakKedua}` : searchQuery}
   476→                onChange={(e) => { setSearchQuery(e.target.value); if (selectedEntry) { clearSelection(); } }}
   477→                onFocus={() => { if (selectedEntry) { clearSelection(); } setDropdownOpen(true); }}
   478→                className="flex h-9 w-full rounded-md border border-input bg-transparent pl-8 pr-8 py-2 text-sm shadow-xs transition-[color,box-shadow] outline-none focus-visible:border-ring focus-visible:ring-ring/50 focus-visible:ring-[3px]"
   479→              />
   480→              {selectedEntry && (
   481→                <button
   482→                  type="button"
   483→                  onClick={(e) => { e.preventDefault(); clearSelection(); }}
   484→                  className="absolute right-2 top-1/2 -translate-y-1/2 text-slate-400 hover:text-slate-600"
   485→                >
   486→                  <X className="w-4 h-4" />
   487→                </button>
   488→              )}
   489→            </div>
   490→          </PopoverAnchor>
   491→          <PopoverContent
   492→            align="start"
   493→            className="p-0 w-[var(--radix-popover-trigger-width)] max-h-64 overflow-y-auto"
   494→            onOpenAutoFocus={(e) => e.preventDefault()}
   495→          >
   496→            {loading ? (
   497→              <div className="px-3 py-4 text-center"><Loader2 className="w-5 h-5 mx-auto text-blue-500 animate-spin" /><p className="text-xs text-slate-400 mt-1">Memuat...</p></div>

   499→              <div>
   500→                <div className="px-3 py-1.5 text-[10px] font-semibold text-slate-400 uppercase bg-amber-50 border-b border-amber-100 sticky top-0">
   501→                  Menunggu Pelunasan ({filteredPending.length})
   502→                </div>
   503→                {filteredPending.map((entry) => {
