    79→   668→    }

   112→   701→    }

   123→   712→        window.dispatchEvent(new CustomEvent('dokupro:history-updated'))

   125→   714→      } else {
   126→   715→        toast.error('Gagal menghapus permanen')
   127→   716→      }
   128→   717→    } catch {
   129→   718→      toast.error('Gagal menghapus permanen')
   130→   719→    } finally {
   131→   720→      setSampahActionId(null)
   132→   721→    }

   139→   728→      const res = await authFetch('/api/database/backup-master?table=invoice_history')

   141→   730→        let errMsg = 'Gagal backup data riwayat invoice'
   142→   731→        try { const errData = await res.json(); errMsg = errData?.error || errMsg } catch {}
   143→   732→        toast.error(errMsg)
   144→   733→        return

   146→   735→      const blob = await res.blob()

   150→   739→      a.href = url

   153→   742→      a.download = match ? match[1] : `backup-invoice-history-${Date.now()}.xlsx`
   154→   743→      document.body.appendChild(a); a.click(); document.body.removeChild(a)

   176→   765→      } catch { toast.error('File backup tidak valid') }

   178→   767→    }

   181→   770→

   186→   775→    if (docType === 'purchase-order') return 'Purchase Order'
   187→   776→    return docType
   188→   777→  }
   189→   778→
   190→   779→  const hasResults = dpInvoices.length > 0 || pelunasanInvoices.length > 0
   191→   780→  const countLabel = loading ? 'Memuat data…' : `${allHistory.length} invoice`
   192→   781→
   193→   782→  return (
   194→   783→    <>
   195→   784→      <div className="space-y-5">
   196→   785→        {/* Header — gaya Master Customer */}
   197→   786→        <div className="flex flex-col gap-3 md:flex-row md:items-end md:justify-between">
   198→   787→          <div>
   199→   788→            <h1 className="text-xl md:text-2xl font-bold tracking-tight">Riwayat Invoice</h1>
   200→   789→            <p className="text-sm text-muted-foreground mt-1">{countLabel}</p>
   201→   790→          </div>
   202→   791→          <div className="flex flex-col sm:flex-row gap-2">
   203→   792→            <div className="relative flex-1 sm:w-64">
   204→   793→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-stone-400 pointer-events-none" />
   205→   794→              <Input
   206→   795→                value={searchQuery}
   207→   796→                onChange={(e) => setSearchQuery(e.target.value)}
   208→   797→                placeholder="Cari no. invoice / customer / barang…"
   209→   798→                aria-label="Cari invoice"
   210→   799→                className="pl-9 min-h-[44px]"
   211→   800→              />
   212→   801→            </div>
   213→   802→            <div className="flex gap-2">
   214→   803→              <Button
   215→   804→                onClick={onCreate}
   216→   805→                title="Buat invoice baru"
   217→   806→                className="bg-emerald-600 hover:bg-emerald-700 min-h-[44px] flex-1 sm:flex-none"
   218→   807→              >
   219→   808→                <Plus className="h-4 w-4" /> Buat Invoice
   220→   809→              </Button>
   221→   810→              <Button onClick={handleBackup} variant="outline" disabled={backupLoading === 'backup'} title="Backup riwayat invoice" className="min-h-[44px] flex-1 sm:flex-none">
   222→   811→                {backupLoading === 'backup' ? <Loader2 className="h-4 w-4 animate-spin" /> : <DatabaseBackup className="h-4 w-4" />} Backup
   223→   812→              </Button>
   224→   813→              <Button onClick={handleRestore} variant="outline" disabled={backupLoading === 'restore'} title="Restore riwayat invoice" className="min-h-[44px] flex-1 sm:flex-none">
   225→   814→                {backupLoading === 'restore' ? <Loader2 className="h-4 w-4 animate-spin" /> : <Upload className="h-4 w-4" />} Restore
   226→   815→              </Button>
   227→   816→            </div>
   228→   817→          </div>

   231→   820→        {/* === TABS: Riwayat Invoice | Riwayat Pembayaran | Sampah === */}

   234→   823→            { key: 'invoice' as const, label: 'Riwayat Invoice', icon: <FileText className="w-3.5 h-3.5" /> },
   235→   824→            { key: 'pembayaran' as const, label: 'Riwayat Pembayaran', icon: <Wallet className="w-3.5 h-3.5" /> },

   239→   828→              key={t.key}
   240→   829→              onClick={() => setActiveTab(t.key)}
   241→   830→              className={cn(
   242→   831→                'px-4 py-1.5 text-sm font-semibold rounded-lg border transition-colors whitespace-nowrap flex-shrink-0',
   243→   832→                activeTab === t.key
   244→   833→                  ? 'bg-violet-600 text-white border-violet-600 shadow-sm'
   245→   834→                  : 'bg-card text-slate-600 border-slate-200 hover:bg-slate-50 hover:border-slate-300'
   246→   835→              )}
   247→   836→            >
   248→   837→              <span className="inline-flex items-center gap-1.5">{t.icon}{t.label}</span>
   249→   838→            </button>

   253→   842→        {/* ================= TAB: RIWAYAT INVOICE ================= */}

   263→   852→                      <SelectValue placeholder="Semua Status" />
   264→   853→                    </SelectTrigger>

   268→   857→                      <SelectItem value="belum">Belum Lunas</SelectItem>

   281→   870→                {filtersActive && (
   282→   871→                  <Button variant="ghost" size="sm" className="lg:ml-auto mt-1 lg:mt-4 text-xs text-muted-foreground" onClick={() => { setStatusFilter('all'); setDateFrom(''); setDateTo(''); setSearchQuery('') }}>
   283→   872→                    <X className="h-3.5 w-3.5" /> Reset Filter

   286→   875→              </div>

   290→   879→              <>
   291→   880→                <div className="hidden md:block rounded-xl border border-stone-200 bg-white overflow-hidden p-4 space-y-3">
   292→   881→                  {[1, 2, 3, 4].map((i) => <Skeleton key={i} className="h-10 w-full" />)}
   293→   882→                </div>
   294→   883→                <div className="md:hidden space-y-3">
   295→   884→                  {[1, 2, 3].map((i) => <Skeleton key={i} className="h-40 w-full rounded-xl" />)}
   296→   885→                </div>

   301→   890→              <>
   302→   891→                {/* === SECTION: INVOICE === */}

   306→   895→                      <FileText className="h-4 w-4 text-violet-600 shrink-0" />

   330→   919→                              const st = getInvoiceStatus(info)
   331→   920→                              return (
   332→   921→                                <TableRow
   333→   922→                                  key={entry.id}

   352→   941→                      </div>
   353→   942→                    </div>

   355→   944→                    {/* Mobile cards — klik kartu → Detail Invoice */}
   356→   945→                    <div className="md:hidden space-y-3">
   357→   946→                      {dpInvoices.slice(0, 100).map((entry) => {

   359→   948→                        const st = getInvoiceStatus(info)
   360→   949→                        return (
   361→   950→                          <Card
