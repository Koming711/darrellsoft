    66→// ===== Helpers =====
    67→function formatRupiah(n: number): string {
    68→  return new Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR', minimumFractionDigits: 0 }).format(n || 0)
    69→}
    70→
    71→function formatDateID(iso: string): string {
    72→  if (!iso) return '-'
    73→  const d = new Date(iso + (iso.length === 10 ? 'T00:00:00' : ''))
    74→  if (isNaN(d.getTime())) return iso
    75→  return d.toLocaleDateString('id-ID', { day: '2-digit', month: 'short', year: 'numeric' })
    76→}
    77→
    78→function fmtISO(d: Date): string {
    79→  return new Date(d.getTime() - d.getTimezoneOffset() * 60000).toISOString().slice(0, 10)
    80→}
    81→
    82→function getFilterDates(period: PeriodType, customStart: string, customEnd: string): { start: string; end: string } {
    83→  const now = new Date()
    84→  const today = fmtISO(now)
    85→  if (period === 'today') return { start: today, end: today }
    86→  if (period === 'week') {
    87→    const day = now.getDay()
    88→    const monday = new Date(now)
    89→    monday.setDate(now.getDate() - (day === 0 ? 6 : day - 1))
    90→    const sunday = new Date(monday)
    91→    sunday.setDate(monday.getDate() + 6)
    92→    return { start: fmtISO(monday), end: fmtISO(sunday) }
    93→  }
    94→  if (period === 'month') {
    95→    return { start: fmtISO(new Date(now.getFullYear(), now.getMonth(), 1)), end: fmtISO(new Date(now.getFullYear(), now.getMonth() + 1, 0)) }
    96→  }
    97→  if (period === 'year') return { start: `${now.getFullYear()}-01-01`, end: `${now.getFullYear()}-12-31` }
    98→  return { start: customStart || '2000-01-01', end: customEnd || today }
    99→}
   100→
   101→function periodLabel(period: PeriodType, customStart: string, customEnd: string): string {
   102→  if (period === 'today') return 'Hari Ini'
   103→  if (period === 'week') return 'Minggu Ini'
   104→  if (period === 'month') return 'Bulan Ini'
   105→  if (period === 'year') return 'Tahun Ini'
   106→  return `${formatDateID(customStart)} s/d ${formatDateID(customEnd)}`
   107→}
   108→
   109→function StatusBadge({ status }: { status: SaleRow['status'] }) {
   110→  if (status === 'lunas') return <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200 hover:bg-emerald-100 text-[11px]">Lunas</Badge>
   111→  if (status === 'jatuh_tempo') return <Badge className="bg-red-100 text-red-700 border-red-200 hover:bg-red-100 text-[11px]">Jatuh Tempo</Badge>
   112→  if (status === 'dp') return <Badge className="bg-sky-100 text-sky-700 border-sky-200 hover:bg-sky-100 text-[11px]">DP</Badge>
   113→  return <Badge className="bg-amber-100 text-amber-700 border-amber-200 hover:bg-amber-100 text-[11px]">Belum Lunas</Badge>
   114→}

   445→                  </TableRow>
   446→                </TableHeader>
   447→                <TableBody>
   448→                  {rows.map((r, i) => (

   450→                      <TableCell className="text-muted-foreground">{i + 1}</TableCell>

   459→                      <TableCell className="text-right whitespace-nowrap">{r.pelunasan > 0 ? formatRupiah(r.pelunasan) : '-'}</TableCell>

   462→                      <TableCell className="text-right">

   472→                          </Button>
   473→                          <Button variant="ghost" size="icon" className="h-8 w-8 text-destructive hover:text-destructive" onClick={() => setDeleteRow(r)} aria-label={`Hapus ${r.nomor}`} title="Hapus">
   474→                            <Trash2 className="h-4 w-4" />
   475→                          </Button>
   476→                        </div>
   477→                      </TableCell>
   478→                    </TableRow>
   479→                  ))}
   480→                </TableBody>
   481→              </Table>
   482→            </div>
   483→          )}
   484→        </div>
   485→
   486→        {/* Mobile cards */}

   488→          {loading ? (
   489→            [1, 2, 3].map((i) => <Skeleton key={i} className="h-40 w-full rounded-xl" />)

   491→            <EmptyState />
   492→          ) : (
   493→            rows.map((r) => (
   494→              <Card key={r.id} className="p-0 gap-0">
   495→                <CardContent className="p-4 space-y-2">
   496→                  <div className="flex items-start justify-between gap-2">
   497→                    <div className="min-w-0">
   498→                      <p className="font-semibold truncate">{r.nomor}</p>

   512→                    </Button>
   513→                    <Button variant="outline" size="sm" className="flex-1 min-h-[40px]" onClick={() => openEdit(r)}>
   514→                      <Pencil className="h-4 w-4" /> Edit
   515→                    </Button>

   517→                      <Printer className="h-4 w-4" /> Print
   518→                    </Button>
   519→                    <Button
   520→                      variant="outline" size="sm"
   521→                      className="min-h-[40px] px-2.5 text-destructive border-stone-200 hover:bg-destructive/10 hover:text-destructive"

   524→                      <Trash2 className="h-4 w-4" />
   525→                    </Button>
   526→                  </div>
   527→                </CardContent>

   530→          )}
   531→        </div>

   541→            const hasModal = items.some((it) => Number((it as { modal?: number }).modal) > 0)

   578→                      </TableBody>
   579→                    </Table>
   580→                  </div>
   581→                  <div className="grid grid-cols-2 gap-2 text-sm">
   582→                    <div className="rounded-lg bg-stone-50 p-3 space-y-1.5">
   583→                      <p className="flex justify-between"><span className="text-muted-foreground">Total Penjualan</span><span className="font-bold">{formatRupiah(detailRow.total)}</span></p>

   586→                      <p className="flex justify-between border-t border-stone-200 pt-1.5"><span className="text-muted-foreground">Sisa Tagihan</span><span className="font-semibold text-amber-700">{formatRupiah(detailRow.sisa)}</span></p>
   587→                    </div>

   592→                      {detailRow.catatan && <p className="text-xs text-muted-foreground pt-1 border-t border-stone-200">Catatan: {detailRow.catatan}</p>}
   593→                    </div>
   594→                  </div>
   595→                </div>
   596→              </>
   597→            )
   598→          })()}
   599→        </DialogContent>

   698→    </DashboardLayout>
