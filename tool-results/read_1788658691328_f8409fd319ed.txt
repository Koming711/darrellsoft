    53→// --- Parse dataJson for document info ---
    54→function parseDocInfo(entry: HistoryEntry) {
    55→  try {
    56→    const parsed = JSON.parse(entry.dataJson)
    57→    const items = parsed.items || []
    58→    const firstItem = items[0]
    59→    const namaBarang = firstItem?.deskripsi || ''
    60→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0)
    61→    const referensi = parsed.referensi || ''
    62→    const noKendaraan = parsed.noKendaraan || ''
    63→    const pengemudi = parsed.pengemudi || ''
    64→    return { namaBarang, totalQty, referensi, noKendaraan, pengemudi }
    65→  } catch {
    66→    return { namaBarang: '', totalQty: 0, referensi: '', noKendaraan: '', pengemudi: '' }
    67→  }
    68→}
    69→
    70→// --- Parse dataJson for SuratJalanData ---
    71→function parseSuratJalanData(entry: HistoryEntry): SuratJalanData {
    72→  try {
    73→    const parsed = JSON.parse(entry.dataJson)
    74→    const company: CompanyInfo = {
    75→      nama: parsed.company?.nama || DEFAULT_COMPANY.nama,
    76→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
    77→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
    78→      email: parsed.company?.email || DEFAULT_COMPANY.email,
    79→      npwp: parsed.company?.npwp || '',
    80→      website: parsed.company?.website || '',
    81→      ppn: parsed.company?.ppn ?? 0,
    82→      logo: parsed.company?.logo || '',
    83→      bankName: parsed.company?.bankName || '',
    84→      bankAccount: parsed.company?.bankAccount || '',
    85→      bankHolder: parsed.company?.bankHolder || '',
    86→      bankName2: parsed.company?.bankName2 || '',
    87→      bankAccount2: parsed.company?.bankAccount2 || '',
    88→      bankHolder2: parsed.company?.bankHolder2 || '',
    89→    }
    90→    const penerima = parsed.penerima || { nama: '', kontak: '', alamat: '' }
    91→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number }, i: number) => ({
    92→      id: it.id || `item-${i}`,
    93→      deskripsi: it.deskripsi || '',
    94→      qty: it.qty || 0,
    95→      satuan: it.satuan || '',
    96→      harga: it.harga || 0,
    97→    }))
    98→    return {
    99→      type: 'surat-jalan',
   100→      company,
   101→      nomor: parsed.nomor || entry.nomor || '',
   102→      tanggal: parsed.tanggal || entry.tanggal || '',
   103→      referensi: parsed.referensi || '',
   104→      penerima,
   105→      items,
   106→      noKendaraan: parsed.noKendaraan || '',
   107→      pengemudi: parsed.pengemudi || '',
   108→      catatan: parsed.catatan || '',
   109→    }
   110→  } catch {
   111→    return {
   112→      type: 'surat-jalan',
   113→      company: { ...DEFAULT_COMPANY },
   114→      nomor: entry.nomor || '',
   115→      tanggal: entry.tanggal || '',
   116→      referensi: '',
   117→      penerima: { nama: '', kontak: '', alamat: '' },
   118→      items: [],
   119→      noKendaraan: '',
   120→      pengemudi: '',
   121→      catatan: '',
   122→    }
   123→  }
   124→}

   126→function SuratJalanRiwayatTab({ onRestore }: { onRestore: () => void }) {
   127→  const setSuratJalan = useDokuproStore((s) => s.setSuratJalan)
   128→  const [sjHistory, setSjHistory] = useState<HistoryEntry[]>([])
   129→  const [loading, setLoading] = useState(true)
   130→  const [searchQuery, setSearchQuery] = useState('')
   131→  const [deleteConfirmId, setDeleteConfirmId] = useState<string | null>(null)
   132→  const [previewItem, setPreviewItem] = useState<HistoryEntry | null>(null)
   133→  const [previewOpen, setPreviewOpen] = useState(false)
   134→  const [previewScale, setPreviewScale] = useState(1)
   135→  const [previewDims, setPreviewDims] = useState<{ w: number; h: number } | null>(null)
   136→  const previewWrapperRef = useRef<HTMLDivElement>(null)
   137→  const [sendingPdf, setSendingPdf] = useState(false)
   138→  const [backupLoading, setBackupLoading] = useState<string | null>(null)
   139→
   140→  const fetchHistory = useCallback(async () => {
   141→    try {
   142→      setLoading(true)
   143→      const headers = getAuthHeaders()
   144→      const res = await fetch('/api/history?docType=surat-jalan', { headers })
   145→      if (res.ok) {
   146→        const json = await res.json()
   147→        setSjHistory(json.data || [])
   148→      }
   149→    } catch (err) {
   150→      console.error('Failed to fetch surat jalan history:', err)
   151→    } finally {
   152→      setLoading(false)
   153→    }
   154→  }, [])
   155→
   156→  useEffect(() => {
   157→    fetchHistory()
   158→  }, [fetchHistory])
   159→
   160→  // Listen for save events
   161→  useEffect(() => {
   162→    const handler = () => fetchHistory()
   163→    window.addEventListener('dokupro:history-updated', handler)
   164→    return () => window.removeEventListener('dokupro:history-updated', handler)
   165→  }, [fetchHistory])
   166→
   167→  const sjData = useMemo(() => {
   168→    if (!previewItem) return null
   169→    return parseSuratJalanData(previewItem)
   170→  }, [previewItem])
   171→
   172→  // Measure actual rendered element and fit it to the available viewport space.
   173→  // Fixes mobile cut-off: uses real offsetWidth/offsetHeight (unaffected by
   174→  // transform), top-left origin, and a wrapper sized to the scaled dimensions
   175→  // so flex layout reserves the correct visual space.
   176→  useLayoutEffect(() => {
   177→    if (!previewOpen) {
   178→      setPreviewDims(null)
   179→      setPreviewScale(1)
   180→      return
   181→    }
   182→    const measureAndScale = () => {
   183→      const el = previewWrapperRef.current
   184→      if (!el) return
   185→      const naturalW = el.offsetWidth
   186→      const naturalH = el.offsetHeight
   187→      if (naturalW === 0 || naturalH === 0) {
   188→        requestAnimationFrame(measureAndScale)
   189→        return
   190→      }
   191→      const vw = window.innerWidth
   192→      const vh = window.innerHeight
   193→      const reservedH = 56 + 88 + 32
   194→      const reservedW = 32
   195→      const availW = Math.max(120, vw - reservedW)
   196→      const availH = Math.max(120, vh - reservedH)
   197→      const scale = Math.min(availW / naturalW, availH / naturalH, 1.4)
   198→      setPreviewScale(scale)
   199→      setPreviewDims({ w: naturalW * scale, h: naturalH * scale })
   200→    }
   201→    const t = setTimeout(measureAndScale, 50)
   202→    window.addEventListener('resize', measureAndScale)
   203→    return () => { clearTimeout(t); window.removeEventListener('resize', measureAndScale) }
   204→  }, [previewOpen, sjData])
   205→
   206→  const handleDelete = async (id: string) => {
   207→    try {
   208→      const res = await fetcher(`/api/history/${id}`, { method: 'DELETE', headers: getAuthHeaders() })
   209→      if (res.ok) {
   210→        toast.success('Surat Jalan berhasil dihapus')
   211→        notifyDataChange('surat-jalan')
   212→        fetchHistory()
   213→      } else {
   214→        toast.error('Gagal menghapus surat jalan')
   215→      }
   216→    } catch {
   217→      toast.error('Gagal menghapus surat jalan')
   218→    }
   219→    setDeleteConfirmId(null)
   220→  }
   221→
   222→  const handleSendJpg = useCallback(async () => {
   223→    if (!sjData) return
   224→    setSendingPdf(true)
   225→    try {
   226→      // Capture the preview DOM element as A5-sized JPG (matches print output)
   227→      const previewEl = document.querySelector('[data-document-preview]') as HTMLElement
   228→      if (previewEl) {
   229→        const blob = await captureElementAsJpg(previewEl)
   230→        const fileName = `${(sjData.nomor || 'draft').replace(/\//g, '-')}.jpg`
   231→        const phone = sjData.penerima?.kontak || ''
   232→
   233→        // No-API sharing: Web Share API first (auto-attaches file),
   234→        // then fallback to download + WhatsApp Web.
   235→        const result = await shareJpgToWhatsApp({
   236→          blob,
   237→          fileName,
   238→          documentLabel: `Surat Jalan ${sjData.nomor}`,
   239→          phone,
   240→        })
   241→
   242→        if (result.status === 'shared') {
   243→          toast.success('Gambar dibagikan ke WhatsApp')
   244→        } else if (result.status === 'cancelled') {
   245→          // silent
   246→        } else if (result.status === 'downloaded') {
   247→          toast.success(`${fileName} tersimpan ke perangkat`, {
   248→            description: 'File JPG telah diunduh ke folder Downloads.',
   249→          })
   250→        } else {
   251→          toast.error(result.error || 'Gagal memproses JPG')
   252→        }
   253→      } else {
   254→        toast.error('Preview tidak ditemukan')
   255→      }
   256→    } catch (err) {
   257→      console.error(err)
   258→      toast.error('Gagal mengirim gambar')
   259→    } finally {
   260→      setSendingPdf(false)
   261→    }
   262→  }, [sjData])
   263→
   264→  const handleBackup = async () => {
   265→    setBackupLoading('backup')
   266→    try {
   267→      const res = await authFetch(`/api/database/backup-master?table=surat_jalan_history`)
   268→      if (!res.ok) {
   269→        let errMsg = 'Gagal backup data riwayat surat jalan'
   270→        try { const errData = await res.json(); errMsg = errData?.error || errMsg } catch {}
   271→        toast.error(errMsg)
   272→        return
   273→      }
   274→      const blob = await res.blob()
   275→      if (blob.size === 0) {
   276→        toast.error('Backup kosong — tidak ada data')
   277→        return
   278→      }
   279→      const url = URL.createObjectURL(blob)
   280→      const a = document.createElement('a')
   281→      a.href = url
   282→      const disposition = res.headers.get('Content-Disposition')
   283→      const match = disposition?.match(/filename\*?=(?:UTF-8'')?"?([^";]+)"?/i)
   284→      a.download = match ? match[1] : `backup-surat-jalan-history-${Date.now()}.xlsx`
   285→      document.body.appendChild(a)
   286→      a.click()
   287→      document.body.removeChild(a)
   288→      URL.revokeObjectURL(url)
   289→      toast.success('Backup berhasil diunduh')
   290→    } catch (e) { console.error('Backup error:', e); toast.error('Gagal backup data riwayat surat jalan') }
   291→    setBackupLoading(null)
   292→  }
   293→
   294→  const handleRestore = async () => {
   295→    const input = document.createElement('input')
   296→    input.type = 'file'
   297→    input.accept = '.xlsx'
   298→    input.onchange = async (e) => {
   299→      const file = (e.target as HTMLInputElement).files?.[0]
   300→      if (!file) return
   301→      if (!confirm('Data riwayat surat jalan yang ada akan diganti dengan data dari file backup. Lanjutkan?')) return
   302→      setBackupLoading('restore')
   303→      try {
   304→        const fd = new FormData()
   305→        fd.append('file', file)
   306→        fd.append('table', 'surat_jalan_history')
   307→        const res = await authFetch('/api/database/restore-master', {
   308→          method: 'POST',
   309→          body: fd,
   310→        })
   311→        const data = await res.json()
   312→        if (res.ok && data.success) {
   313→          toast.success(`Restore berhasil (${data.count} data)`)
   314→          fetchHistory()
   315→          notifyDataChange('surat-jalan')
   316→        } else {
   317→          toast.error(data.error || 'Gagal restore data riwayat surat jalan')
   318→        }
   319→      } catch { toast.error('File backup tidak valid') }
   320→      setBackupLoading(null)
   321→    }
   322→    input.click()
   323→  }
   324→
   325→  // Filter by search
   326→  const filteredHistory = useMemo(() => {
   327→    if (!searchQuery.trim()) return sjHistory
   328→    const q = searchQuery.toLowerCase().trim()
   329→    return sjHistory.filter(entry => {
   330→      const info = parseDocInfo(entry)
   331→      return (
   332→        entry.nomor?.toLowerCase().includes(q) ||
   333→        entry.pihakKedua?.toLowerCase().includes(q) ||
   334→        info.namaBarang?.toLowerCase().includes(q) ||
   335→        entry.tanggal?.toLowerCase().includes(q)
   336→      )
   337→    })
   338→  }, [sjHistory, searchQuery])
   339→
   340→  return (
   341→    <>
   342→      <div className="bg-card rounded-2xl shadow-sm border border-slate-200 overflow-hidden">
   343→        {/* Header */}
   344→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2 px-4 py-3 border-b border-slate-200 bg-slate-50/60">
   345→          <div className="flex items-center gap-2 min-w-0">
   346→            <History className="w-4 h-4 text-amber-600 shrink-0" />
   347→            <h2 className="text-sm font-bold text-slate-700 uppercase tracking-wide truncate">Riwayat Surat Jalan</h2>
   348→            <span className="text-[10px] font-medium text-slate-500 bg-slate-100 px-2 py-0.5 rounded-full shrink-0">{sjHistory.length} data</span>
   349→          </div>
   350→          <div className="flex items-center gap-1.5 flex-wrap">
   351→            <Button
   352→              onClick={handleBackup}
   353→              variant="outline"
   354→              size="sm"
   355→              disabled={backupLoading === 'backup'}
   356→              className="h-7 gap-1.5 text-xs"
   357→            >
   358→              {backupLoading === 'backup' ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <DatabaseBackup className="w-3.5 h-3.5" />}
   359→              Backup
   360→            </Button>
   361→            <Button
   362→              onClick={handleRestore}
   363→              variant="outline"
   364→              size="sm"
   365→              disabled={backupLoading === 'restore'}
   366→              className="h-7 gap-1.5 text-xs"
   367→            >
   368→              {backupLoading === 'restore' ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <Upload className="w-3.5 h-3.5" />}
   369→              Restore
   370→            </Button>
   371→          </div>
   372→        </div>
   373→
   374→        {/* Search */}
   375→        {sjHistory.length > 0 && (
   376→          <div className="px-4 py-2 border-b border-slate-100 bg-white/50">
   377→            <div className="relative">
   378→              <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 w-3.5 h-3.5 text-slate-400" />
   379→              <input
   380→                type="text"
   381→                value={searchQuery}
   382→                onChange={(e) => setSearchQuery(e.target.value)}
   383→                placeholder="Cari no. SJ, customer, barang..."
   384→                className="w-full h-8 pl-8 pr-8 text-xs bg-white border border-slate-200 rounded-lg focus:outline-none focus:ring-1 focus:ring-amber-400 focus:border-amber-400 placeholder:text-slate-400"
   385→              />
   386→              {searchQuery && (
   387→                <button onClick={() => setSearchQuery('')} className="absolute right-2 top-1/2 -translate-y-1/2">
   388→                  <X className="w-3.5 h-3.5 text-slate-400 hover:text-slate-600" />
   389→                </button>
   390→              )}
   391→            </div>
   392→          </div>
   393→        )}
   394→
   395→        {/* Table */}
   396→        {loading ? (
   397→          <div className="px-4 py-6 text-center">
   398→            <Loader2 className="w-6 h-6 mx-auto text-amber-500 animate-spin" />
   399→            <p className="text-xs text-slate-400 mt-2">Memuat riwayat...</p>
   400→          </div>
   401→        ) : filteredHistory.length > 0 ? (
   402→          <>
   403→            {/* Mobile card layout */}
   404→            <div className="sm:hidden divide-y divide-slate-100">
   405→              {filteredHistory.slice(0, 100).map((entry) => {
   406→                const info = parseDocInfo(entry)
   407→                return (
   408→                  <div
   409→                    key={entry.id}
   410→                    className="px-4 py-3 hover:bg-amber-50/30 active:bg-amber-100/40 transition-colors cursor-pointer"
   411→                    onClick={() => { setPreviewItem(entry); setPreviewOpen(true) }}
   412→                  >
   413→                    <div className="flex items-start justify-between gap-2 mb-1">
   414→                      <div className="min-w-0">
   415→                        <p className="text-amber-700 font-semibold text-[13px] truncate">{entry.nomor || '-'}</p>
   416→                        <p className="text-slate-500 text-xs">{entry.tanggal ? formatTanggal(entry.tanggal) : entry.createdAt ? new Date(entry.createdAt).toLocaleDateString('id-ID', { day: '2-digit', month: 'short' }) : '-'}</p>
   417→                      </div>
   418→                      {info.totalQty > 0 && (
   419→                        <p className="text-slate-600 text-xs whitespace-nowrap">Qty: {info.totalQty.toLocaleString('id-ID')}</p>
   420→                      )}
   421→                    </div>
   422→                    <div className="flex items-center justify-between gap-2">
   423→                      <div className="min-w-0">
   424→                        <p className="text-slate-700 font-medium text-xs truncate">{entry.pihakKedua || '-'}</p>
   425→                        {info.namaBarang && <p className="text-slate-400 text-[11px] truncate">{info.namaBarang.split('\n')[0]}</p>}
   426→                      </div>
   427→                      <div className="flex items-center gap-1 shrink-0" onClick={(e) => e.stopPropagation()}>
   428→                        <button
   429→                          onClick={() => {
   430→                            const parsed = parseSuratJalanData(entry)
   431→                            setSuratJalan(parsed)
   432→                            onRestore()
   433→                            toast.success('Surat Jalan berhasil dimuat ke editor')
   434→                          }}
   435→                          className="inline-flex items-center justify-center w-7 h-7 bg-emerald-50 hover:bg-emerald-100 text-emerald-700 rounded-md border border-emerald-200 transition-colors"
   436→                          title="Restore ke Editor"
   437→                        >
   438→                          <RotateCcw className="w-3.5 h-3.5" />
   439→                        </button>
   440→                        <button
   441→                          onClick={() => setDeleteConfirmId(entry.id)}
   442→                          className="inline-flex items-center justify-center w-7 h-7 bg-red-50 hover:bg-red-100 text-red-600 rounded-md border border-red-200 transition-colors"
   443→                          title="Hapus"
   444→                        >
   445→                          <Trash2 className="w-3.5 h-3.5" />
   446→                        </button>
   447→                      </div>
   448→                    </div>
   449→                  </div>
   450→                )
   451→              })}
   452→            </div>
   453→            {/* Desktop table layout */}
   454→            <div className="hidden sm:block overflow-x-auto">
   455→              <table className="w-full text-[13px] min-w-[600px]">
   456→                <thead>
   457→                  <tr className="border-b border-slate-200 bg-slate-50/80">
   458→                    <th className="text-left py-3 px-3 text-slate-500 font-semibold whitespace-nowrap">No. SJ</th>
   459→                    <th className="text-left py-3 px-3 text-slate-500 font-semibold whitespace-nowrap">Tgl</th>

   466→                <tbody>
   467→                  {filteredHistory.slice(0, 100).map((entry, idx) => {

   469→                    return (

   474→                        <td className="py-3 px-3 text-slate-600 max-w-[180px] truncate" title={info.namaBarang}>{info.namaBarang ? info.namaBarang.split('\n')[0] : '-'}</td>
