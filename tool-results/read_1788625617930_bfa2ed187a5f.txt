    82→// --- Parse dataJson for document info ---
    83→function parseDocInfo(entry: HistoryEntry) {
    84→  try {
    85→    const parsed = JSON.parse(entry.dataJson)
    86→    const items = parsed.items || []
    87→    const itemCount = items.length
    88→    const firstItem = items[0]
    89→    const namaBarang = firstItem?.deskripsi || ''
    90→    const hargaSatuan = firstItem?.harga || 0
    91→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0)
    92→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0)
    93→    const ppn = parsed.ppn || 0
    94→    const dpPercent = parsed.dp || 0
    95→    const totalHarga = subtotal + (subtotal * ppn / 100)
    96→    // Always derive dpAmount from originalTotal
    97→    const originalTotal = parsed.originalTotal !== undefined ? parsed.originalTotal : totalHarga
    98→    const dpAmount = originalTotal * (dpPercent / 100)
    99→    const sisa = totalHarga - dpAmount
   100→    const lunas = parsed.lunas === true
   101→    const tanggalJatuhTempo = parsed.tanggalJatuhTempo || ''
   102→    const tanggalPelunasan = parsed.tanggalPelunasan || ''
   103→    const referensi = parsed.referensi || ''
   104→    const referensiInvoiceNomor = parsed.referensiInvoiceNomor || ''
   105→    const isPelunasan = entry.docType === 'invoice-pelunasan'
   106→    const uangCapek = parsed.uangCapek || 0
   107→    return { namaBarang, hargaSatuan, totalQty, totalHarga, dpPercent, dp: dpAmount, sisa, lunas, tanggalJatuhTempo, tanggalPelunasan, referensi, referensiInvoiceNomor, isPelunasan, originalTotal, uangCapek, itemCount }
   108→  } catch {
   109→    return { namaBarang: '', hargaSatuan: 0, totalQty: 0, totalHarga: 0, dpPercent: 0, dp: 0, sisa: 0, lunas: false, tanggalJatuhTempo: '', tanggalPelunasan: '', referensi: '', referensiInvoiceNomor: '', isPelunasan: false, originalTotal: 0, uangCapek: 0, itemCount: 0 }
   110→  }
   111→}

   113→// --- Parse dataJson for InvoiceData ---
   114→function parseInvoiceData(entry: HistoryEntry): InvoiceData {
   115→  try {
   116→    const parsed = JSON.parse(entry.dataJson)
   117→    const company: CompanyInfo = {
   118→      nama: parsed.company?.nama || DEFAULT_COMPANY.nama,
   119→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
   120→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
   121→      email: parsed.company?.email || DEFAULT_COMPANY.email,
   122→      npwp: parsed.company?.npwp || '',
   123→      website: parsed.company?.website || '',
   124→      ppn: parsed.company?.ppn ?? parsed.ppn ?? 11,
   125→      logo: parsed.company?.logo || '',
   126→      bankName: parsed.company?.bankName || '',
   127→      bankAccount: parsed.company?.bankAccount || '',
   128→      bankHolder: parsed.company?.bankHolder || '',
   129→      bankName2: parsed.company?.bankName2 || '',
   130→      bankAccount2: parsed.company?.bankAccount2 || '',
   131→      bankHolder2: parsed.company?.bankHolder2 || '',
   132→    }
   133→    const client = parsed.client || { nama: '', kontak: '', alamat: '' }
   134→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number }, i: number) => ({
   135→      id: it.id || `item-${i}`,
   136→      deskripsi: it.deskripsi || '',
   137→      qty: it.qty || 0,
   138→      satuan: it.satuan || '',
   139→      harga: it.harga || 0,
   140→    }))
   141→    const docType = (entry.docType === 'invoice-pelunasan' ? 'invoice-pelunasan' : 'invoice') as 'invoice' | 'invoice-pelunasan'
   142→    return {
   143→      type: docType,
   144→      company,
   145→      nomor: parsed.nomor || entry.nomor || '',
   146→      tanggal: parsed.tanggal || entry.tanggal || '',
   147→      referensi: parsed.referensi || '',
   148→      client,
   149→      items,
   150→      ppn: parsed.ppn ?? 11,
   151→      dp: parsed.dp || 0,
   152→      dpAmount: parsed.dpAmount,
   153→      catatan: parsed.catatan || '',
   154→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   155→      caraPembayaran: parsed.caraPembayaran || '',
   156→      tanggalGiro: parsed.tanggalGiro || '',
   157→      lunas: parsed.lunas === true,
   158→      tanggalPelunasan: parsed.tanggalPelunasan || '',
   159→      referensiInvoiceId: parsed.referensiInvoiceId || '',
   160→      referensiInvoiceNomor: parsed.referensiInvoiceNomor || '',
   161→      originalTotal: parsed.originalTotal,
   162→      uangCapek: parsed.uangCapek || 0,
   163→    }
   164→  } catch {
   165→    const docType = (entry.docType === 'invoice-pelunasan' ? 'invoice-pelunasan' : 'invoice') as 'invoice' | 'invoice-pelunasan'
   166→    return {
   167→      type: docType,
   168→      company: { ...DEFAULT_COMPANY },
   169→      nomor: entry.nomor || '',
   170→      tanggal: entry.tanggal || '',
   171→      referensi: '',
   172→      client: { nama: '', kontak: '', alamat: '' },
   173→      items: [],
   174→      ppn: 11,
   175→      dp: 0,
   176→      catatan: '',
   177→      tanggalJatuhTempo: '',
   178→      caraPembayaran: '',
   179→      tanggalGiro: '',
   180→      lunas: false,
   181→      tanggalPelunasan: '',
   182→      referensiInvoiceId: '',
   183→      referensiInvoiceNomor: '',
   184→      uangCapek: 0,
   185→    }
   186→  }
   187→}

   482→      return entry.nomor?.toLowerCase().includes(q) || entry.pihakKedua?.toLowerCase().includes(q) || info.namaBarang?.toLowerCase().includes(q) || entry.tanggal?.toLowerCase().includes(q)
   483→    })
   484→  }, [allHistory, searchQuery])

   486→  // Separate into DP invoices and Pelunasan invoices
   487→  const dpInvoices = useMemo(() => filteredHistory.filter(e => e.docType === 'invoice'), [filteredHistory])
   488→  const pelunasanInvoices = useMemo(() => filteredHistory.filter(e => e.docType === 'invoice-pelunasan'), [filteredHistory])
   489→
   490→  // Calculate profit per invoice — use saved value from dataJson, fallback to cetakan lookup
   491→  const invoiceUangCapek = useMemo(() => {
   492→    const result = new Map<string, number>()
   493→    // Build cetakan lookup by both nomorUrut and printName (fallback for older data)
   494→    const cetakanByRef = new Map<string, number>()
   495→    for (const c of cetakanList) {
   496→      if (c.nomorUrut) {
   497→        cetakanByRef.set(c.nomorUrut, (cetakanByRef.get(c.nomorUrut) || 0) + c.profitAmount)
   498→      }
   499→      if (c.printName && !cetakanByRef.has(c.printName)) {
   500→        cetakanByRef.set(c.printName, (cetakanByRef.get(c.printName) || 0) + c.profitAmount)
   501→      }
   502→    }
   503→    for (const inv of dpInvoices) {
   504→      const info = parseDocInfo(inv)
   505→      // Prefer saved uangCapek from dataJson, fallback to cetakan lookup
   506→      const uangCapek = info.uangCapek > 0 ? info.uangCapek : (info.referensi ? (cetakanByRef.get(info.referensi) || 0) : 0)
   507→      result.set(inv.id, uangCapek)
   508→    }
   509→    return result
   510→  }, [dpInvoices, cetakanList])
   511→
   512→  // ============================================================
   513→  // Merge invoice logic
   514→  // ============================================================
   515→  const selectedInvoices = useMemo(() => dpInvoices.filter(e => selectedIds.has(e.id)), [dpInvoices, selectedIds])
   516→
   517→  // Build a merged preview from selected invoices + chosen primary
   518→  const mergedPreview = useMemo(() => {
   519→    if (selectedInvoices.length < 2 || !mergePrimaryId) return null
   520→    const primary = selectedInvoices.find(e => e.id === mergePrimaryId)
   521→    if (!primary) return null
   522→    const primaryData = parseInvoiceData(primary)
   523→    // Calculate primary's original DP amount (already paid) — keep it FIXED after merge
   524→    const primaryInfo = parseDocInfo(primary)
   525→    const originalDpAmount = primaryInfo.dp > 0 ? primaryInfo.dp : 0
   526→    const originalDpPercent = primaryInfo.dpPercent
   527→
   528→    // Combine items: primary first, then append items from others (with new ids)
   529→    const allItems = [...primaryData.items]
   530→    const otherInvoices: HistoryEntry[] = []
   531→    for (const inv of selectedInvoices) {
   532→      if (inv.id === mergePrimaryId) continue
   533→      otherInvoices.push(inv)
   534→      const data = parseInvoiceData(inv)
   535→      for (const it of data.items) {
   536→        allItems.push({ ...it, id: `${it.id || 'item'}-${inv.id}` })
   537→      }
   538→    }
   539→    const subtotal = allItems.reduce((s, it) => s + (it.qty || 0) * (it.harga || 0), 0)
   540→    const ppn = primaryData.ppn || 0
   541→    const newTotal = subtotal + (subtotal * ppn / 100)
   542→    // Keep DP amount fixed; recalculate percentage so the already-paid amount stays the same
   543→    let newDpPercent = 0
   544→    let newDpAmount = 0
   545→    let newSisa = newTotal
   546→    if (originalDpAmount > 0 && newTotal > 0) {
   547→      newDpAmount = originalDpAmount
   548→      newDpPercent = Math.round((originalDpAmount / newTotal) * 10000) / 100 // 2 decimals
   549→      newSisa = newTotal - originalDpAmount
   550→    }
   551→    // Detect customer mismatch
   552→    const customers = new Set<string>()
   553→    for (const inv of selectedInvoices) {
   554→      const d = parseInvoiceData(inv)
   555→      if (d.client?.nama) customers.add(d.client.nama.trim().toLowerCase())
   556→    }
   557→    const customerMismatch = customers.size > 1
   558→
   559→    // Check if primary has a PEL child (referencing primary's nomor)
   560→    const pelChild = pelunasanHistory.find(p => {
   561→      const pInfo = parseDocInfo(p)
   562→      return pInfo.referensiInvoiceNomor === primary.nomor
   563→    })
   564→    const hasPelChild = !!pelChild
   565→    const pelChildLunas = pelChild ? parseDocInfo(pelChild).lunas : false
   566→
   567→    // Sum profit (uangCapek) from ALL selected invoices — primary + others.
   568→    // Uses the same lookup logic as invoiceUangCapek (prefer saved uangCapek,
   569→    // fallback to cetakan lookup by referensi). This ensures the merged invoice
   570→    // carries the COMBINED profit, not just the primary's.
   571→    const totalUangCapek = selectedInvoices.reduce((sum, inv) => {
   572→      return sum + (invoiceUangCapek.get(inv.id) ?? 0)
   573→    }, 0)
   574→
   575→    return {
   576→      primary, primaryData, otherInvoices, allItems,
   577→      subtotal, ppn, newTotal, newDpPercent, newDpAmount, newSisa,
   578→      originalDpAmount, originalDpPercent,
   579→      customerMismatch, pelChild, hasPelChild, pelChildLunas,
   580→      totalUangCapek,
   581→    }
   582→  }, [selectedInvoices, mergePrimaryId, pelunasanHistory, invoiceUangCapek])
   583→

   586→      const next = new Set(prev)

   588→      else next.add(id)
   589→      return next
   590→    })
   591→  }

   609→      .catch(() => { /* non-critical */ })
   610→  }

   612→  const handleMerge = async () => {
   613→    if (!mergedPreview) return
   614→    const { primary, primaryData, allItems, ppn, newTotal, newDpPercent, newDpAmount, otherInvoices, pelChild, hasPelChild, pelChildLunas, totalUangCapek } = mergedPreview
   615→    setMergeLoading(true)
   616→    try {
   617→      // 0. Generate a BRAND-NEW invoice number for the merged result.
   618→      //    The merged invoice must NOT keep the primary's old number — it gets a
   619→      //    fresh sequential number (e.g. merging INV/.../0001 + INV/.../0002
   620→      //    produces INV/.../0003). The counter is incremented atomically so the
   621→      //    number is reserved even if the merge fails halfway.
   622→      const genRes = await fetcher('/api/history/generate-number?docType=invoice', {
   623→        method: 'POST',
   624→        headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   625→      })
   626→      if (!genRes.ok) throw new Error('Gagal generate nomor invoice baru')
   627→      const { nomor: newNomor } = await genRes.json()
   628→      if (!newNomor) throw new Error('Nomor invoice baru kosong')
   629→
   630→      // Build merged InvoiceData — keep primary's metadata, replace items, adjust DP,
   631→      // set combined profit (uangCapek) from ALL merged invoices, AND assign the
   632→      // new nomor so the result is a fresh invoice number.
   633→      const merged: Record<string, unknown> = {
   634→        ...primaryData,
   635→        nomor: newNomor,
   636→        items: allItems,
   637→        ppn,
   638→        dp: newDpPercent,
   639→        uangCapek: totalUangCapek,

   642→        merged.dpAmount = newDpAmount
   643→        merged.originalTotal = newTotal

   645→        delete merged.dpAmount
   646→        delete merged.originalTotal
   647→      }

   649→      delete merged.statusPembayaran
   650→      const newDataJson = JSON.stringify(merged)
   651→
   652→      // 1. Update primary invoice (PUT) — pass `nomor` so BOTH the DB column
   653→      //    and dataJson.nomor are updated to the new number.
   654→      const putRes = await fetcher(`/api/history/${primary.id}`, {
   655→        method: 'PUT',
   656→        headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   657→        body: JSON.stringify({ dataJson: newDataJson, nomor: newNomor, total: String(Math.round(newTotal)) }),

   662→      //    totals + profit, repoint its referensiInvoiceNomor to the new nomor,

   665→      //    PEL nomors are DERIVED from INV nomors by replacing the INV→PEL prefix

   667→      //    during merge (e.g. INV/07/26/0001 → INV/07/26/0007), the PEL child
   668→      //    MUST also be renumbered (PEL/07/26/0001 → PEL/07/26/0007) to stay
   669→      //    in sync. Otherwise the PEL keeps its old number while the INV has a
   670→      //    new one — which is the bug "nomor invoice pelunasan sama, harusnya
   671→      //    beda".
   672→      if (hasPelChild && pelChild && !pelChildLunas) {
   673→        try {
   674→          const pelParsed = JSON.parse(pelChild.dataJson)
   675→          const newPelNomor = newNomor.replace(/^INV/, 'PEL')
   676→          pelParsed.nomor = newPelNomor
   677→          pelParsed.items = allItems
   678→          pelParsed.ppn = ppn
   679→          pelParsed.originalTotal = newTotal
   680→          pelParsed.uangCapek = totalUangCapek
   681→          pelParsed.referensiInvoiceNomor = newNomor

   683→            pelParsed.dp = newDpPercent
   684→            pelParsed.dpAmount = newDpAmount
   685→          }
   686→          delete pelParsed.statusPembayaran
   687→          await fetcher(`/api/history/${pelChild.id}`, {
   688→            method: 'PUT',
   689→            headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   690→            body: JSON.stringify({ dataJson: JSON.stringify(pelParsed), nomor: newPelNomor, total: String(Math.round(newTotal)) }),
   691→          })
   692→        } catch (e) {
   693→          console.error('Failed to sync PEL child:', e)
   694→        }
   695→      }
   696→
   697→      // 3. Delete other (non-primary) invoices if option is checked
   698→      //    Also delete their PEL children to avoid orphaned pelunasan entries
   699→      if (mergeDeleteOthers) {
   700→        for (const inv of otherInvoices) {
   701→          // Find and delete PEL children referencing this invoice (by nomor)
   702→          const childPelList = pelunasanHistory.filter(p => {
   703→            const pInfo = parseDocInfo(p)
   704→            return pInfo.referensiInvoiceNomor === inv.nomor
   705→          })
   706→          for (const childPel of childPelList) {
   707→            try {
   708→              await fetcher(`/api/history/${childPel.id}`, { method: 'DELETE', headers: getAuthHeaders() })
   709→            } catch (e) {
   710→              console.error('Failed to delete orphaned PEL child:', childPel.id, e)
   711→            }
   712→          }
   713→          // Delete the DP invoice itself
   714→          try {
   715→            await fetcher(`/api/history/${inv.id}`, { method: 'DELETE', headers: getAuthHeaders() })
   716→          } catch (e) {
   717→            console.error('Failed to delete merged invoice:', inv.id, e)
   718→          }
   719→        }
   720→      }
   721→
   722→      toast.success(`${selectedInvoices.length} invoice berhasil digabung menjadi ${newNomor}`)
   723→      setMergeDialogOpen(false)
   724→      setMergeMode(false)
   725→      setSelectedIds(new Set())
   726→      fetchHistory()
   727→      notifyDataChange('invoice')
   728→    } catch (err) {
   729→      console.error(err)
   730→      toast.error('Gagal menggabungkan invoice')
   731→    } finally {
   732→      setMergeLoading(false)
   733→    }
   734→  }
   735→
   736→  return (
   737→    <>
   738→      <div className="bg-white dark:bg-zinc-900 rounded-2xl shadow-sm border border-slate-200 dark:border-zinc-700 overflow-hidden">
   739→        {/* Header */}
   740→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2 px-4 py-3 border-b border-slate-200 dark:border-zinc-700 bg-white dark:bg-zinc-900">
   741→          <div className="flex items-center gap-2 min-w-0">
   742→            <History className="w-4 h-4 text-violet-600 shrink-0" />
   743→            <h2 className="text-sm font-bold text-slate-700 uppercase tracking-wide truncate">Riwayat Invoice</h2>
   744→            <span className="text-[10px] font-medium text-slate-500 bg-slate-100 px-2 py-0.5 rounded-full shrink-0">{allHistory.length} data</span>
   745→          </div>
   746→          <div className="flex items-center gap-1.5 flex-wrap">
   747→            {!mergeMode && (
   748→              <Button
   749→                onClick={onCreate}
   750→                size="sm"
   751→                className="h-7 gap-1.5 text-xs bg-blue-600 hover:bg-blue-700 text-white"
   752→                title="Buat invoice baru"
   753→              >
   754→                <Plus className="w-3.5 h-3.5" /> Buat Invoice
   755→              </Button>
   756→            )}
   757→            {!mergeMode && (
   758→              <Button 
   759→                onClick={() => { setMergeMode(true); setSelectedIds(new Set()) }} 
   760→                variant="outline" 
   761→                size="sm" 
   762→                disabled={dpInvoices.length < 2}
   763→                title={dpInvoices.length < 2 ? `Butuh minimal 2 invoice untuk digabungkan (saat ini: ${dpInvoices.length})` : 'Gabungkan 2 atau lebih invoice menjadi 1'}
   764→                className="h-7 gap-1.5 text-xs"
   765→              >
   766→                <Combine className="w-3.5 h-3.5" /> Gabungkan
   767→              </Button>
   768→            )}
   769→            <Button onClick={handleBackup} variant="outline" size="sm" disabled={backupLoading === 'backup' || mergeMode} className="h-7 gap-1.5 text-xs">
   770→              {backupLoading === 'backup' ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <DatabaseBackup className="w-3.5 h-3.5" />} Backup
   771→            </Button>
   772→            <Button onClick={handleRestore} variant="outline" size="sm" disabled={backupLoading === 'restore' || mergeMode} className="h-7 gap-1.5 text-xs">
   773→              {backupLoading === 'restore' ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <Upload className="w-3.5 h-3.5" />} Restore
   774→            </Button>
   775→          </div>
   776→        </div>
   777→
   778→        {/* Merge mode action bar */}
   779→        {mergeMode && (
   780→          <div className="flex items-center justify-between gap-2 px-4 py-2.5 border-b border-violet-200 bg-violet-50 dark:bg-violet-950/30 dark:border-violet-800">
   781→            <div className="flex items-center gap-2 min-w-0">
   782→              <Layers className="w-4 h-4 text-violet-600 shrink-0" />
   783→              <span className="text-xs font-semibold text-violet-800 dark:text-violet-200 truncate">
   784→                Mode Gabung: {selectedIds.size} invoice dipilih
   785→              </span>
   786→            </div>
   787→            <div className="flex items-center gap-1.5 shrink-0">
   788→              <Button onClick={() => { setMergeMode(false); setSelectedIds(new Set()) }} variant="outline" size="sm" className="h-7 text-xs">
   789→                Batal
   790→              </Button>
   791→              <Button onClick={openMergeDialog} size="sm" disabled={selectedIds.size < 2} className="h-7 gap-1.5 text-xs bg-violet-600 hover:bg-violet-700 text-white">
   792→                <Combine className="w-3.5 h-3.5" /> Gabungkan ({selectedIds.size})
   793→              </Button>
   794→            </div>
   795→          </div>
   796→        )}
   797→
   798→        {/* Search */}
   799→        {invoiceHistory.length > 0 && (
   800→          <div className="px-4 py-2 border-b border-slate-200 dark:border-zinc-700 bg-white dark:bg-zinc-900">
   801→            <div className="relative">

   804→              {searchQuery && (<button onClick={() => setSearchQuery('')} className="absolute right-2 top-1/2 -translate-y-1/2"><X className="w-3.5 h-3.5 text-slate-400 hover:text-slate-600" /></button>)}
   805→            </div>
   806→          </div>

   808→
   809→        {/* Table / Cards */}

   812→        ) : filteredHistory.length > 0 ? (
   813→          <>
   814→            {/* === INVOICE DP Section === */}

'bg-violet-50 dark:bg-violet-950/30' : 'hover:bg-violet-50/30 active:bg-violet-100/40'}`} onClick={() => mergeMode ? toggleSelect(entry.id) : (setPreviewItem(entry), setPreviewOpen(true))}>
