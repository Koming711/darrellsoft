     1→'use client';
     2→
     3→import { useEffect, useState, useCallback, useRef, useMemo } from 'react';
     4→import { useSearchParams, useRouter } from 'next/navigation';
     5→import { Input } from '@/components/ui/input';
     6→import { Label } from '@/components/ui/label';
     7→import { Textarea } from '@/components/ui/textarea';
     8→import { Popover, PopoverContent, PopoverAnchor } from '@/components/ui/popover';
     9→import { useDokuproStore } from '@/lib/store';
    10→import { CompanyFields } from './company-fields';
    11→import { ItemsFields } from './items-fields';
    12→import { InvoicePreview } from './invoice-preview';
    13→import { DocumentEditorLayout } from './document-editor-layout';
    14→import { DocumentActionButtons } from './document-action-buttons';
    15→import { formatRupiah } from '@/lib/format';
    16→import { getAuthHeaders } from '@/lib/auth';
    17→import { Truck } from 'lucide-react';
    18→import { Button } from '@/components/ui/button';
    19→import { toast } from 'sonner';
    20→import type { InvoiceData } from '@/lib/types';
    21→
    22→interface CustomerItem {
    23→  id: string;
    24→  name: string;
    25→  companyName?: string;
    26→  phone?: string;
    27→  email?: string;
    28→  address?: string;
    29→}
    30→
    31→interface RiwayatCetakanItem {
    32→  id: string;
    33→  nomorUrut: string;
    34→  printName: string;
    35→  customerName: string;
    36→  paperName: string;
    37→  paperGrammage: string;
    38→  cutWidth: string;
    39→  cutHeight: string;
    40→  quantity: string;
    41→  jumlahPesanan: string;
    42→  berapaMata: string;
    43→  warna: string;
    44→  machineName: string;
    45→  ongkosCetak: number;
    46→  machineName2: string;
    47→  ongkosCetak2: number;
    48→  totalPaperPrice: number;
    49→  pricePerSheet: number;
    50→  finishingNames: string;
    51→  finishingCost: number;
    52→  packingCost: number;
    53→  shippingCost: number;
    54→  otherCost: number;
    55→  otherCost2: number;
    56→  otherCostLabel: string;
    57→  otherCostLabel2: string;
    58→  glueCost: number;
    59→  glueBorongan: number;
    60→  subTotal: number;
    61→  profitPercent: number;
    62→  profitAmount: number;
    63→  grandTotal: number;
    64→  createdAt: string;
    65→}
    66→
    67→export function InvoiceEditor() {
    68→  const invoice = useDokuproStore((s) => s.invoice);
    69→  const setInvoice = useDokuproStore((s) => s.setInvoice);
    70→  const resetDocument = useDokuproStore((s) => s.resetDocument);
    71→  const loadCompanyFromAPI = useDokuproStore((s) => s.loadCompanyFromAPI);
    72→  const router = useRouter();
    73→
    74→  const searchParams = useSearchParams();
    75→  const riwayatIdFromUrl = searchParams.get('riwayatId');
    76→  const autoSelectDoneRef = useRef<string | null>(null); // track which riwayatId was auto-selected
    77→  const [savingSj, setSavingSj] = useState(false);
    78→
    79→  // Riwayat cetakan dropdown state
    80→  const [riwayatList, setRiwayatList] = useState<RiwayatCetakanItem[]>([]);
    81→  const [referensiInput, setReferensiInput] = useState(invoice.referensi);
    82→  const [dropdownOpen, setDropdownOpen] = useState(false);
    83→
    84→  // Customer dropdown state
    85→  const [customerList, setCustomerList] = useState<CustomerItem[]>([]);
    86→  const [clientInput, setClientInput] = useState(invoice.client.nama);
    87→  const [clientDropdownOpen, setClientDropdownOpen] = useState(false);
    88→  const [clientTyping, setClientTyping] = useState(false);
    89→
    90→  const fetchRiwayatCetakan = useCallback(async () => {
    91→    try {
    92→      const res = await fetch('/api/riwayat-cetakan', { headers: getAuthHeaders() });
    93→      if (res.ok) {
    94→        const data = await res.json();
    95→        setRiwayatList(data);
    96→        return data;
    97→      }
    98→    } catch {
    99→      // silently fail
   100→    }
   101→    return [];
   102→  }, []);
   103→
   104→  // Fetch customer list
   105→  const fetchCustomers = useCallback(async () => {
   106→    try {
   107→      const res = await fetch('/api/customers', { headers: getAuthHeaders() });
   108→      if (res.ok) {
   109→        const data = await res.json();
   110→        setCustomerList(Array.isArray(data) ? data : []);
   111→      }
   112→    } catch {
   113→      // silently fail
   114→    }
   115→  }, []);
   116→
   117→  useEffect(() => { loadCompanyFromAPI() }, [loadCompanyFromAPI]);
   118→  useEffect(() => { fetchRiwayatCetakan() }, [fetchRiwayatCetakan]);
   119→  useEffect(() => { fetchCustomers() }, [fetchCustomers]);
   120→
   121→  // Fetch next Invoice number from server
   122→  const fetchNextNumber = useCallback(() => {
   123→    fetch('/api/history?preview=next-number&docType=invoice', { headers: getAuthHeaders() })
   124→      .then(res => res.ok ? res.json() : null)
   125→      .then(data => { if (data?.nextNumber) setInvoice((prev) => ({ ...prev, nomor: data.nextNumber })) })
   126→      .catch(() => {});
   127→  }, [setInvoice]);
   128→
   129→  useEffect(() => { fetchNextNumber() }, [fetchNextNumber]);
   130→
   131→  // Re-fetch next number after a document is saved
   132→  useEffect(() => {
   133→    const handler = () => { fetchNextNumber() };
   134→    window.addEventListener('dokupro:history-updated', handler);
   135→    return () => { window.removeEventListener('dokupro:history-updated', handler) };
   136→  }, [fetchNextNumber]);
   137→
   138→  // Sync referensiInput when invoice.referensi changes externally
   139→  useEffect(() => { setReferensiInput(invoice.referensi) }, [invoice.referensi]);
   140→  // Sync clientInput when invoice.client.nama changes externally
   141→  useEffect(() => { setClientInput(invoice.client.nama) }, [invoice.client.nama]);
   142→
   143→  // Filter riwayat list — when dropdown is open show all, otherwise not used
   144→  const filteredRiwayatList = riwayatList;
   145→
   146→  const handleReferensiInputChange = (value: string) => {
   147→    setReferensiInput(value);
   148→    setInvoice((prev) => ({ ...prev, referensi: value }));
   149→    setDropdownOpen(true);
   150→  };
   151→
   152→  // Apply referensi data using functional setInvoice to avoid stale closure
   153→  const applyReferensi = (item: RiwayatCetakanItem) => {
   154→    const ref = item.nomorUrut || item.printName || '';
   155→    setReferensiInput(ref);
   156→
   157→    // Build single item with multi-line description
   158→    const descLines: string[] = [];
   159→    if (item.printName) descLines.push(item.printName);
   160→
   161→    const bahanParts: string[] = [];
   162→    if (item.paperName) bahanParts.push(item.paperName);
   163→    if (item.paperGrammage) bahanParts.push(item.paperGrammage + 'g');
   164→    if (bahanParts.length > 0) descLines.push(bahanParts.join(' '));
   165→
   166→    if (item.finishingNames) {
   167→      const finishings = item.finishingNames.split(',').map(f => f.trim()).filter(Boolean);
   168→      descLines.push('Finishing:');
   169→      finishings.forEach(f => descLines.push(f));
   170→    }
   171→
   172→    const deskripsi = descLines.join('\n');
   173→    const jumlahPesanan = parseInt(item.jumlahPesanan) || 1;
   174→    const hargaPerPcs = jumlahPesanan > 0 ? Math.round(item.grandTotal / jumlahPesanan) : 0;
   175→
   176→    const newItems: InvoiceData['items'] = [{
   177→      id: 'riwayat-0',
   178→      deskripsi: deskripsi || ref,
   179→      qty: jumlahPesanan,
   180→      satuan: 'pcs',
   181→      harga: hargaPerPcs,
   182→    }];
   183→
   184→    // Use functional form to always get the LATEST state
   185→    setInvoice((prev) => ({
   186→      ...prev,
   187→      referensi: ref,
   188→      client: {
   189→        ...prev.client,
   190→        nama: item.customerName || prev.client.nama,
   191→      },
   192→      items: newItems,
   193→      catatan: '',
   194→      uangCapek: item.profitAmount || 0,
   195→    }));
   196→    setDropdownOpen(false);
   197→  };
   198→
   199→  const handleReferensiSelect = (item: RiwayatCetakanItem) => {
   200→    applyReferensi(item);
   201→  };
   202→
   203→  // Auto-select referensi when coming from hitung cetakan with riwayatId
   204→  // Fetches the specific riwayat directly by ID for reliability (no race condition with list loading)
   205→  useEffect(() => {
   206→    if (!riwayatIdFromUrl) return;
   207→    if (autoSelectDoneRef.current === riwayatIdFromUrl) return;
   208→
   209→    let cancelled = false;
   210→
   211→    const fetchAndApply = async () => {
   212→      // First, try to find it in the already-loaded list (fast path)
   213→      const foundInList = riwayatList.find((r) => r.id === riwayatIdFromUrl);
   214→      if (foundInList) {
   215→        if (!cancelled) {
   216→          autoSelectDoneRef.current = riwayatIdFromUrl;
   217→          applyReferensi(foundInList);
   218→        }
   219→        return;
   220→      }
   221→
   222→      // Not in list yet — fetch directly by ID from the API (reliable path)
   223→      try {
   224→        const res = await fetch(`/api/riwayat-cetakan?id=${riwayatIdFromUrl}`, { headers: getAuthHeaders() });
   225→        if (res.ok && !cancelled) {
   226→          const data = await res.json();
   227→          if (data && !data.error) {
   228→            autoSelectDoneRef.current = riwayatIdFromUrl;
   229→            applyReferensi(data as RiwayatCetakanItem);
   230→          }
   231→        }
   232→      } catch {
   233→        // Fallback: retry once after a short delay
   234→        if (cancelled) return;
   235→        await new Promise(r => setTimeout(r, 800));
   236→        if (cancelled) return;
   237→        try {
   238→          const res = await fetch(`/api/riwayat-cetakan?id=${riwayatIdFromUrl}`, { headers: getAuthHeaders() });
   239→          if (res.ok) {
   240→            const data = await res.json();
   241→            if (data && !data.error) {
   242→              autoSelectDoneRef.current = riwayatIdFromUrl;
   243→              applyReferensi(data as RiwayatCetakanItem);
   244→            }
   245→          }
   246→        } catch {
   247→          // Give up silently
   248→        }
   249→      }
   250→    };
   251→
   252→    fetchAndApply();
   253→
   254→    return () => { cancelled = true; };
   255→  }, [riwayatIdFromUrl]);
   256→
   257→  // Filter customer list (only when user is actively typing)
   258→  const filteredCustomerList = useMemo(() =>
   259→    clientTyping
   260→      ? customerList.filter((c) => {
   261→          const search = clientInput.toLowerCase().trim();
   262→          if (!search) return true;
   263→          return c.name.toLowerCase().includes(search) ||
   264→                 (c.companyName || '').toLowerCase().includes(search) ||
   265→                 (c.phone || '').toLowerCase().includes(search) ||
   266→                 (c.address || '').toLowerCase().includes(search);
   267→        })
   268→      : customerList
   269→  , [clientTyping, customerList, clientInput]);
   270→
   271→  const handleClientInputChange = (value: string) => {
   272→    setClientInput(value);
   273→    setClientTyping(true);
   274→    updateClient('nama', value);
   275→    setClientDropdownOpen(true);
   276→  };
   277→
   278→  const handleClientSelect = (item: CustomerItem) => {
   279→    setClientInput(item.name);
   280→    setClientTyping(false);
   281→    setInvoice((prev) => ({
   282→      ...prev,
   283→      client: {
   284→        nama: item.name,
   285→        kontak: item.phone || item.email || '',
   286→        alamat: item.address || '',
   287→      },
   288→    }));
   289→    setClientDropdownOpen(false);
   290→  };
   291→
   292→  const updateCompany = (company: typeof invoice.company) => {
   293→    setInvoice((prev) => ({ ...prev, company }));
   294→  };
   295→
   296→  const updateClient = (field: string, value: string) => {
   297→    setInvoice((prev) => ({
   298→      ...prev,
   299→      client: { ...prev.client, [field]: value },
   300→    }));
   301→  };
   302→
   303→  const subtotal = invoice.items.reduce((sum, item) => sum + item.qty * item.harga, 0);
   304→  const ppnAmount = subtotal * (invoice.ppn / 100);
   305→  const total = subtotal + ppnAmount;
   306→  const dpPercent = invoice.dp || 0;
   307→  const dpAmount = total * (dpPercent / 100);
   308→  const sisa = total - dpAmount;
   309→
   310→  const handleSuratJalan = async () => {
   311→    // Check if data has content
   312→    const pihakKedua = invoice.client?.nama;
   313→    const hasItem = invoice.items?.some((item) => item.deskripsi.trim() !== '');
   314→    if (!pihakKedua?.trim() || !hasItem) {
   315→      toast.error('Lengkapi data invoice terlebih dahulu!');
   316→      return;
   317→    }
   318→
   319→    setSavingSj(true);
   320→    try {
   321→      // Save invoice to history first
   322→      const dataToSave = { ...invoice, dpAmount, originalTotal: total };
   323→      const res = await fetch('/api/history', {
   324→        method: 'POST',
   325→        headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   326→        body: JSON.stringify({
   327→          docType: 'invoice',
   328→          nomor: invoice.nomor || '-',
   329→          tanggal: invoice.tanggal || '',
   330→          pihakKedua: invoice.client?.nama || '-',
   331→          total: '-',
   332→          dataJson: JSON.stringify(dataToSave),
   333→        }),
   334→      });
   335→      if (res.ok) {
   336→        const saved = await res.json();
   337→        window.dispatchEvent(new CustomEvent('dokupro:history-updated'));
   338→
   339→        // If invoice has DP, also create an invoice-pelunasan entry
   340→        if (invoice.dp > 0) {
   341→          try {
   342→            // Derive PEL nomor from INV nomor (same number, different prefix)
   343→            const invNomor = saved.nomor || invoice.nomor;
   344→            const pelNomor = invNomor.replace(/^INV/, 'PEL');
   345→
   346→            const pelunasanData = {
   347→              ...dataToSave,
   348→              type: 'invoice-pelunasan',
   349→              referensiInvoiceId: saved.id,
   350→              referensiInvoiceNomor: invNomor,
   351→              lunas: false,
   352→              tanggalPelunasan: '',
   353→            };
   354→            await fetch('/api/history', {
   355→              method: 'POST',
   356→              headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   357→              body: JSON.stringify({
   358→                docType: 'invoice-pelunasan',
   359→                customNomor: pelNomor,
   360→                tanggal: invoice.tanggal || '',
   361→                pihakKedua: invoice.client?.nama || '-',
   362→                total: '-',
   363→                dataJson: JSON.stringify(pelunasanData),
   364→              }),
   365→            });
   366→          } catch {
   367→            console.error('Failed to create pelunasan invoice entry');
   368→          }
   369→        }
   370→
   371→        toast.success('Invoice disimpan ke riwayat');
   372→        // Reset invoice form after saving
   373→        resetDocument('invoice');
   374→        router.push(`/surat-jalan?invoiceId=${saved.id}`);
   375→      } else if (res.status === 409) {
   376→        // Invoice already saved, get existing ID and navigate
   377→        const errData = await res.json().catch(() => ({}));
   378→        if (errData.id) {
   379→          resetDocument('invoice');
   380→          router.push(`/surat-jalan?invoiceId=${errData.id}`);
   381→        } else {
   382→          toast('Invoice sudah ada di riwayat.');
   383→        }
   384→      } else {
   385→        toast.error('Gagal menyimpan invoice');
   386→      }
   387→    } catch {
   388→      toast.error('Gagal menyimpan invoice');
   389→    }
   390→    setSavingSj(false);
   391→  };
   392→
   393→  return (
   394→    <>
   395→      <DocumentEditorLayout
   396→        title="Invoice"
   397→        previewContent={<InvoicePreview data={invoice} />}
   398→        actions={
   399→          <div className="flex items-center gap-2 flex-wrap">
   400→            <DocumentActionButtons
   401→              docType="invoice"
   402→              documentLabel="Invoice"
   403→              currentData={invoice}
   404→              onReset={() => resetDocument('invoice')}
   405→            />
   406→            <Button
   407→              size="sm"
   408→              onClick={handleSuratJalan}
   409→              disabled={savingSj}
   410→              className="bg-orange-600 hover:bg-orange-700 h-8 sm:h-9"
   411→            >
   412→              <Truck className="mr-1.5 h-3.5 w-3.5" />
   413→              {savingSj ? 'Menyimpan...' : 'Surat Jalan'}
   414→            </Button>
   415→          </div>
   416→        }
   417→      >
   418→        <CompanyFields company={invoice.company} onChange={updateCompany} />
   419→
   420→        <div className="rounded-lg border bg-card p-3 sm:p-4 shadow-sm">
   421→          <h3 className="mb-3 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   422→            Detail Dokumen
   423→          </h3>
   424→          <div className="grid grid-cols-2 gap-2">
   425→            <div className="space-y-1.5">
   426→              <Label className="text-xs">No. Invoice</Label>
   427→              <Input
   428→                value={invoice.nomor}
   429→                readOnly
   430→                className="bg-slate-50 text-slate-500 cursor-not-allowed"
   431→              />
   432→            </div>
   433→            <div className="space-y-1.5">
   434→              <Label className="text-xs">Tanggal</Label>
   435→              <Input
   436→                type="date"
   437→                value={invoice.tanggal}
   438→                onChange={(e) => setInvoice((prev) => ({ ...prev, tanggal: e.target.value }))}
   439→              />
   440→            </div>
   441→          </div>
   442→          <div className="space-y-2">
   443→            <Label className="text-xs">Referensi (No. HC)</Label>
   444→            <Popover open={dropdownOpen} onOpenChange={setDropdownOpen}>
   445→              <PopoverAnchor asChild>
   446→                <button
   447→                  type="button"
   448→                  onClick={() => setDropdownOpen(true)}
   449→                  className="flex h-9 w-full items-center justify-between rounded-md border border-input bg-slate-50 px-3 py-2 text-sm shadow-xs outline-none cursor-pointer hover:bg-slate-100 transition-colors text-left"
   450→                >
   451→                  <span className={referensiInput ? 'text-slate-900' : 'text-slate-400'}>
   452→                    {referensiInput || 'Pilih No. HC...'}
   453→                  </span>
   454→                  <svg className="w-4 h-4 text-slate-400 shrink-0" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M19 9l-7 7-7-7" /></svg>
   455→                </button>
   456→              </PopoverAnchor>
   457→              <PopoverContent
   458→                align="start"
   459→                className="p-0 w-[var(--radix-popover-trigger-width)] max-h-52 overflow-y-auto"
   460→                onOpenAutoFocus={(e) => e.preventDefault()}
   461→              >
   462→                {filteredRiwayatList.length > 0 && (
   463→                  <div>
   464→                    <div className="px-3 py-1.5 text-[10px] font-semibold text-slate-400 uppercase bg-slate-50 border-b border-slate-100 sticky top-0">Riwayat Hitung Cetakan</div>
   465→                    {filteredRiwayatList.map((r) => (
   466→                      <button
   467→                        key={r.id}
   468→                        type="button"
   469→                        onMouseDown={(e) => { e.preventDefault(); handleReferensiSelect(r) }}
   470→                        className={`w-full text-left px-3 py-2 text-sm transition-colors ${invoice.referensi === r.nomorUrut ? 'bg-blue-50 text-blue-700 font-medium' : 'text-slate-700 hover:bg-blue-50 hover:text-blue-700'}`}
   471→                      >
   472→                        <span className="font-semibold text-xs text-emerald-700">{r.nomorUrut || '-'}</span>
   473→                        <span className="ml-1.5 truncate">{r.printName || ''}</span>
   474→                        {r.customerName && <span className="text-slate-400 ml-1.5 text-[11px]">({r.customerName})</span>}
   475→                      </button>
   476→                    ))}
   477→                  </div>
   478→                )}
   479→                {filteredRiwayatList.length === 0 && (
   480→                  <div className="px-3 py-3 text-sm text-slate-400 text-center">Tidak ada riwayat cetakan</div>
   481→                )}
   482→              </PopoverContent>
   483→            </Popover>
   484→          </div>
   485→        </div>
   486→
   487→        {/* Informasi Pembayaran - moved below Detail Dokumen */}
   488→        <div className="rounded-lg border bg-card p-3 sm:p-4 shadow-sm">
   489→          <h3 className="mb-3 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   490→            Informasi Pembayaran
   491→          </h3>
   492→          <div className="grid grid-cols-2 gap-3">
   493→            <div className="space-y-1.5">
   494→              <Label className="text-xs">Tgl. Jatuh Tempo</Label>
   495→              <Input
   496→                type="date"
   497→                value={invoice.tanggalJatuhTempo}
   498→                onChange={(e) => setInvoice((prev) => ({ ...prev, tanggalJatuhTempo: e.target.value }))}
   499→              />
   500→            </div>
   501→            <div className="space-y-1.5">
   502→              <Label className="text-xs">Cara Pembayaran</Label>
   503→              <div className="flex gap-1.5">
   504→                {(['cash', 'transfer', 'giro'] as const).map((method) => {
   505→                  const labels: Record<string, string> = { cash: 'Cash', transfer: 'Transfer', giro: 'Giro' };
   506→                  const isSelected = invoice.caraPembayaran === method;
   507→                  return (
   508→                    <button
   509→                      key={method}
   510→                      type="button"
   511→                      onClick={() => setInvoice((prev) => ({
   512→                        ...prev,
   513→                        caraPembayaran: prev.caraPembayaran === method ? '' : method,
   514→                        tanggalGiro: method === 'giro' && prev.caraPembayaran !== method ? prev.tanggalGiro : (method !== 'giro' ? '' : prev.tanggalGiro),
   515→                      }))}
   516→                      className={`flex-1 py-2 rounded-lg border text-[11px] font-semibold transition-colors ${
   517→                        isSelected
   518→                          ? 'bg-emerald-100 text-emerald-700 border-emerald-300 ring-2 ring-offset-1 ring-emerald-400 shadow-sm'
   519→                          : 'bg-slate-50 text-slate-500 border-slate-200 hover:bg-slate-100'
   520→                      }`}
   521→                    >
   522→                      {labels[method]}
   523→                    </button>
   524→                  );
   525→                })}
   526→              </div>
   527→            </div>
   528→          </div>
   529→          {invoice.caraPembayaran === 'giro' && (
   530→            <div className="mt-3 space-y-1.5">
