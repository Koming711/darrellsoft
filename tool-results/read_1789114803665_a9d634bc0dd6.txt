    69→/**
    70→ * Badge status invoice. Lunas → hijau, Belum Lunas → oranye,
    71→ * dengan info jatuh tempo bila tersedia.

    73→function StatusBadge({ status, dueDate }: { status: 'LUNAS' | 'BELUM_BAYAR'; dueDate: string | null }) {
    74→  if (status === 'LUNAS') {
    75→    return (
    76→      <Badge className="bg-emerald-100 text-emerald-800 border border-emerald-200 hover:bg-emerald-100 dark:bg-emerald-950 dark:text-emerald-300 dark:border-emerald-900">
    77→        Lunas
    78→      </Badge>
    79→    )
    80→  }
    81→  return (
    82→    <Badge
    83→      variant="outline"
    84→      className="bg-amber-50 text-amber-800 border-amber-200 hover:bg-amber-50 dark:bg-amber-950 dark:text-amber-300 dark:border-amber-900"
    85→      title={dueDate ? `Jatuh tempo ${dueDate}` : undefined}
    86→    >

   109→/** Warna angka sisa tagihan: 0 → hijau (lunas), lainnya → oranye */
   110→function sisaClass(sisa: number): string {
   111→  return sisa <= 0 ? 'text-emerald-700 dark:text-emerald-400' : 'text-amber-700 dark:text-amber-400'
   112→}

   114→/** Format tanggal ISO (YYYY-MM-DD) → "19 Jul 2026" */
   115→function formatDateID(iso: string): string {
   116→  if (!iso) return '-'
   117→  const d = new Date(iso)
   118→  if (Number.isNaN(d.getTime())) return iso
   119→  return d.toLocaleDateString('id-ID', { day: '2-digit', month: 'short', year: 'numeric' })
   120→}
   121→
   122→/** Parse daftar entry history menjadi baris terstruktur; skip JSON rusak. */
   123→function parseHistory<T>(entries: HistoryEntry[], mapFn: (parsed: Record<string, unknown>, entry: HistoryEntry) => T | null): T[] {
   124→  const out: T[] = []
   125→  for (const entry of entries) {
   126→    try {
   127→      const parsed = JSON.parse(entry.dataJson) as Record<string, unknown>
   128→      const row = mapFn(parsed, entry)
   129→      if (row) out.push(row)
   130→    } catch {
   131→      /* skip rusak */
   132→    }
   133→  }
   134→  return out
   135→}

   160→  }, [])

   167→        authFetch('/api/history?docType=invoice&startDate=2000-01-01&endDate=2099-12-31', { headers }),
   168→        authFetch('/api/history?docType=invoice-pelunasan&startDate=2000-01-01&endDate=2099-12-31', { headers }),
   169→      ])

   173→      const invEntries: HistoryEntry[] = invJson.data || []
   174→      const pelEntries: HistoryEntry[] = pelJson.data || []
   175→
   176→      // Pelunasan: referensi nomor → { lunas, nominal }
   177→      const pelunasanList = parseHistory(pelEntries, (p, entry) => {
   178→        const itemsRaw = Array.isArray(p.items) ? (p.items as Record<string, unknown>[]) : []
   179→        const subtotal = itemsRaw.reduce((acc, it) => acc + Number(it.qty ?? 0) * Number(it.harga ?? 0), 0)
   180→        const ppn = Number(p.ppn ?? 0)
   181→        const docTotal = Number(p.originalTotal ?? 0) > 0 ? Number(p.originalTotal) : subtotal + subtotal * (ppn / 100)
   182→        const dpAmount =
   183→          p.dpAmount !== undefined && p.dpAmount !== null && Number(p.dpAmount) > 0
   184→            ? Number(p.dpAmount)
   185→            : docTotal * (Number(p.dp ?? 0) / 100)
   186→        const referensiNomor = String(p.referensiInvoiceNomor ?? '')
   187→        if (!referensiNomor) return null
   188→        const row: PelunasanRow = {
   189→          id: entry.id,
   190→          nomor: entry.nomor,
   191→          tanggal: String(p.tanggal ?? entry.tanggal ?? '').slice(0, 10),
   192→          referensiNomor,
   193→          nominal: Math.max(0, docTotal - dpAmount),
   194→          lunas: p.lunas === true,
   195→          tanggalJatuhTempo: String(p.tanggalJatuhTempo ?? ''),
   196→          data: p,
   197→        }
   198→        return row
   199→      })
   200→      setSettlementRows(pelunasanList.sort((a, b) => (a.tanggal < b.tanggal ? 1 : -1)))
   201→
   202→      const settledMap = new Map<string, { lunas: boolean; nominal: number }>()
   203→      for (const p of pelunasanList) {
   204→        const prev = settledMap.get(p.referensiNomor)
   205→        if (!prev || (p.lunas && !prev.lunas)) {
   206→          settledMap.set(p.referensiNomor, { lunas: p.lunas, nominal: p.lunas ? p.nominal : prev?.nominal ?? 0 })
   207→        }
   208→      }
   209→
   210→      const dpList = parseHistory(invEntries, (p, entry) => {
   211→        const itemsRaw = Array.isArray(p.items) ? (p.items as Record<string, unknown>[]) : []
   212→        const subtotal = itemsRaw.reduce((acc, it) => acc + Number(it.qty ?? 0) * Number(it.harga ?? 0), 0)
   213→        const ppn = Number(p.ppn ?? 0)
   214→        const total = subtotal + subtotal * (ppn / 100)
   215→        const dpAmount =
   216→          p.dpAmount !== undefined && p.dpAmount !== null && Number(p.dpAmount) > 0
   217→            ? Number(p.dpAmount)
   218→            : total * (Number(p.dp ?? 0) / 100)
   219→        if (dpAmount <= 0) return null // hanya invoice dengan DP
   220→        const client = (p.client ?? {}) as Record<string, unknown>
   221→        const lunas = p.lunas === true
   222→        const settled = settledMap.get(entry.nomor)
   223→        const settledAmount = lunas ? total - dpAmount : settled?.lunas ? settled.nominal : 0
   224→        const sisa = Math.max(0, Math.round(total - dpAmount - settledAmount))

   226→          id: entry.id,
   227→          nomor: entry.nomor,
   228→          tanggal: String(p.tanggal ?? entry.tanggal ?? '').slice(0, 10),
   229→          customerName: entry.pihakKedua || String(client.nama ?? '') || '-',
   230→          total: Math.round(total),
   231→          dpAmount: Math.round(dpAmount),
   232→          settledAmount: Math.round(settledAmount),
   233→          sisa,
   234→          lunas: lunas || sisa <= 0,
   235→          tanggalJatuhTempo: String(p.tanggalJatuhTempo ?? ''),
   236→          data: p,
   237→        }
   238→        return row
   239→      })
   240→      setDpRows(dpList.sort((a, b) => (a.tanggal < b.tanggal ? 1 : -1)))
   241→    } catch (e) {
   242→      toast.error(e instanceof Error ? e.message : 'Terjadi kesalahan')
   243→    } finally {
   244→      setLoading(false)
   245→    }
   246→  }, [])
