    92→    92→  const d = new Date()
    93→    93→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`

   100→   100→// --- Parse dataJson for document info ---
   101→   101→function parseDocInfo(entry: HistoryEntry) {
   102→   102→  try {
   103→   103→    const parsed = JSON.parse(entry.dataJson)
   104→   104→    const items = parsed.items || []
   105→   105→    const itemCount = items.length
   106→   106→    const firstItem = items[0]
   107→   107→    const namaBarang = firstItem?.deskripsi || ''
   108→   108→    const hargaSatuan = firstItem?.harga || 0
   109→   109→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0)
   110→   110→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0)
   111→   111→    const ppn = parsed.ppn || 0
   112→   112→    const dpPercent = parsed.dp || 0
   113→   113→    const totalHarga = subtotal + (subtotal * ppn / 100)
   114→   114→    // Always derive dpAmount from originalTotal
   115→   115→    const originalTotal = parsed.originalTotal !== undefined ? parsed.originalTotal : totalHarga
   116→   116→    const dpAmount = originalTotal * (dpPercent / 100)
   117→   117→    const sisa = totalHarga - dpAmount
   118→   118→    const lunas = parsed.lunas === true
   119→   119→    const batal = parsed.batal === true
   120→   120→    const tanggalJatuhTempo = parsed.tanggalJatuhTempo || ''
   121→   121→    const tanggalPelunasan = parsed.tanggalPelunasan || ''
   122→   122→    const referensi = parsed.referensi || ''
   123→   123→    const referensiInvoiceNomor = parsed.referensiInvoiceNomor || ''
   124→   124→    const isPelunasan = entry.docType === 'invoice-pelunasan'
   125→   125→    const uangCapek = parsed.uangCapek || 0
   126→   126→    return { namaBarang, hargaSatuan, totalQty, totalHarga, dpPercent, dp: dpAmount, sisa, lunas, batal, tanggalJatuhTempo, tanggalPelunasan, referensi, referensiInvoiceNomor, isPelunasan, originalTotal, uangCapek, itemCount }
   127→   127→  } catch {
   128→   128→    return { namaBarang: '', hargaSatuan: 0, totalQty: 0, totalHarga: 0, dpPercent: 0, dp: 0, sisa: 0, lunas: false, batal: false, tanggalJatuhTempo: '', tanggalPelunasan: '', referensi: '', referensiInvoiceNomor: '', isPelunasan: false, originalTotal: 0, uangCapek: 0, itemCount: 0 }
   129→   129→  }
   130→   130→}

   132→   132→// --- Status invoice (batal > lunas > belum) ---
   133→   133→function getInvoiceStatus(info: ReturnType<typeof parseDocInfo>): InvoiceStatus {
   134→   134→  if (info.batal) return 'batal'
   135→   135→  if (info.isPelunasan) return info.lunas ? 'lunas' : 'belum'
   136→   136→  const hasDP = info.dpPercent > 0 || info.dp > 0
   137→   137→  return (hasDP ? info.lunas : (info.lunas || info.sisa <= 0)) ? 'lunas' : 'belum'
   138→   138→}

   141→   141→function parseInvoiceData(entry: HistoryEntry): InvoiceData {

   162→   162→      id: it.id || `item-${i}`,

   164→   164→      qty: it.qty || 0,
   165→   165→      satuan: it.satuan || '',
   166→   166→      harga: it.harga || 0,
   167→   167→      ...(it.modal !== undefined ? { modal: it.modal } : {}),

   169→   169→    const docType = (entry.docType === 'invoice-pelunasan' ? 'invoice-pelunasan' : 'invoice') as 'invoice' | 'invoice-pelunasan'
   170→   170→    return {
   171→   171→      type: docType,
   172→   172→      company,
   173→   173→      nomor: parsed.nomor || entry.nomor || '',
   174→   174→      tanggal: parsed.tanggal || entry.tanggal || '',

   178→   178→      ppn: parsed.ppn ?? 11,
   179→   179→      dp: parsed.dp || 0,
   180→   180→      dpAmount: parsed.dpAmount,
   181→   181→      catatan: parsed.catatan || '',
   182→   182→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   183→   183→      caraPembayaran: parsed.caraPembayaran || '',

   187→   187→      referensiInvoiceId: parsed.referensiInvoiceId || '',
   188→   188→      referensiInvoiceNomor: parsed.referensiInvoiceNomor || '',
   189→   189→      originalTotal: parsed.originalTotal,
   190→   190→      uangCapek: parsed.uangCapek || 0,
   191→   191→    }
   192→   192→  } catch {
   193→   193→    const docType = (entry.docType === 'invoice-pelunasan' ? 'invoice-pelunasan' : 'invoice') as 'invoice' | 'invoice-pelunasan'
   194→   194→    return {
   195→   195→      type: docType,
   196→   196→      company: { ...DEFAULT_COMPANY },
   197→   197→      nomor: entry.nomor || '',
   198→   198→      tanggal: entry.tanggal || '',
   199→   199→      referensi: '',
   200→   200→      client: { nama: '', kontak: '', alamat: '' },
   201→   201→      items: [],
   202→   202→      ppn: 11,
   203→   203→      dp: 0,
   204→   204→      catatan: '',
   205→   205→      tanggalJatuhTempo: '',
   206→   206→      caraPembayaran: '',
   207→   207→      tanggalGiro: '',
   208→   208→      lunas: false,
   209→   209→      tanggalPelunasan: '',
   210→   210→      referensiInvoiceId: '',
   211→   211→      referensiInvoiceNomor: '',
   212→   212→      uangCapek: 0,
   213→   213→    }
   214→   214→  }

   240→   240→// ============================================================
   241→   241→// DetailInvoiceView — "halaman pratinjau" setelah Simpan di Buat
   242→   242→// Invoice & halaman detail saat baris riwayat diklik.
   243→   243→// Tombol Cetak / JPG / Surat Jalan / Tandai Lunas / Hapus / Batal
   244→   244→// ada DI ATAS pratinjau. Pratinjau A5 ber-outline, +20% (desktop),
   245→   245→// full-width di mobile.
   246→   246→// ============================================================
   247→   247→function DetailInvoiceView({ id, onBack }: { id: string; onBack: () => void }) {
   248→   248→  const router = useRouter()
   249→   249→  const resetDocument = useDokuproStore((s) => s.resetDocument)
   250→   250→  const [entry, setEntry] = useState<HistoryEntry | null>(null)
   251→   251→  const [loading, setLoading] = useState(true)
   252→   252→  const [error, setError] = useState('')
   253→   253→  const [jpgGenerating, setJpgGenerating] = useState(false)
   254→   254→  const [updating, setUpdating] = useState(false)
   255→   255→
   256→   256→  // Dialog konfirmasi
   257→   257→  const [lunasOpen, setLunasOpen] = useState(false)
   258→   258→  const [batalOpen, setBatalOpen] = useState(false)
   259→   259→  const [hapusOpen, setHapusOpen] = useState(false)
   260→   260→
   261→   261→  // Pratinjau scaler
   262→   262→  const scalerRef = useRef<HTMLDivElement>(null)
   263→   263→
   264→   264→  const loadEntry = useCallback(async () => {
   265→   265→    setLoading(true)
   266→   266→    setError('')
   267→   267→    try {
   268→   268→      const res = await fetcher(`/api/history/${id}`, { headers: getAuthHeaders() })
   269→   269→      if (res.ok) {
   270→   270→        const json = await res.json()
   271→   271→        setEntry(json.data || null)
   272→   272→      } else {
   273→   273→        setError('Invoice tidak ditemukan')
   274→   274→      }
   275→   275→    } catch {
   276→   276→      setError('Gagal memuat invoice')
   277→   277→    } finally {
   278→   278→      setLoading(false)
   279→   279→    }
   280→   280→  }, [id])
   281→   281→
   282→   282→  useEffect(() => { loadEntry() }, [loadEntry])
   283→   283→
   284→   284→  const data = useMemo(() => (entry ? parseInvoiceData(entry) : null), [entry])
   285→   285→  const info = useMemo(() => (entry ? parseDocInfo(entry) : null), [entry])
   286→   286→  const status: InvoiceStatus = info ? getInvoiceStatus(info) : 'belum'
   287→   287→
   288→   288→  // Scale pratinjau A5 agar pas dengan container (desktop ~+20% dari ukuran
   289→   289→  // asli 148mm, mobile full-width). offsetWidth/Height tidak terpengaruh transform.
   290→   290→  useLayoutEffect(() => {
   291→   291→    if (!data) return
   292→   292→    const fit = () => {
   293→   293→      const wrapper = scalerRef.current
   294→   294→      if (!wrapper) return
   295→   295→      const a5 = wrapper.querySelector('.a5-page') as HTMLElement | null
   296→   296→      if (!a5) return
   297→   297→      const naturalW = a5.offsetWidth
   298→   298→      const naturalH = a5.offsetHeight
   299→   299→      if (naturalW === 0 || naturalH === 0) {
   300→   300→        requestAnimationFrame(fit)
   301→   301→        return
   302→   302→      }
   303→   303→      const availW = wrapper.parentElement?.clientWidth || wrapper.clientWidth
   304→   304→      if (availW === 0) {
   305→   305→        requestAnimationFrame(fit)
   306→   306→        return
   307→   307→      }
   308→   308→      const scale = availW / naturalW
   309→   309→      a5.style.transform = `scale(${scale})`
   310→   310→      a5.style.transformOrigin = 'top left'
   311→   311→      wrapper.style.width = `${naturalW * scale}px`
   312→   312→      wrapper.style.height = `${naturalH * scale}px`
   313→   313→    }
   314→   314→    fit()
   315→   315→    const raf = requestAnimationFrame(fit)
   316→   316→    const timer = setTimeout(fit, 250)
   317→   317→    window.addEventListener('resize', fit)
   318→   318→    return () => { cancelAnimationFrame(raf); clearTimeout(timer); window.removeEventListener('resize', fit) }
   319→   319→  }, [data])
   320→   320→
   321→   321→  const handleJpg = async () => {
   322→   322→    if (!data) return
   323→   323→    setJpgGenerating(true)
   324→   324→    try {
   325→   325→      const previewEl = document.querySelector('[data-document-preview]') as HTMLElement
   326→   326→      if (!previewEl) { toast.error('Pratinjau tidak ditemukan'); return }
   327→   327→      const blob = await captureElementAsJpg(previewEl)
   328→   328→      const fileName = `${(data.nomor || 'draft').replace(/\//g, '-')}.jpg`
   329→   329→      const phone = data.client?.kontak || ''
   330→   330→      const result = await shareJpgToWhatsApp({
   331→   331→        blob,
   332→   332→        fileName,
   333→   333→        documentLabel: `Invoice ${data.nomor}`,
   334→   334→        phone,
   335→   335→      })
   336→   336→      if (result.status === 'shared') toast.success('Gambar dibagikan ke WhatsApp')
   337→   337→      else if (result.status === 'cancelled') { /* silent */ }
   338→   338→      else if (result.status === 'downloaded') toast.success(`${fileName} tersimpan ke perangkat`, { description: 'File JPG telah diunduh ke folder Downloads.' })
   339→   339→      else toast.error(result.error || 'Gagal memproses JPG')
   340→   340→    } catch (err) {
   341→   341→      console.error(err)
   342→   342→      toast.error('Gagal membuat JPG')
   343→   343→    } finally {
   344→   344→      setJpgGenerating(false)
   345→   345→    }
   346→   346→  }
   347→   347→
   348→   348→  // Tandai Lunas — invoice dinyatakan lunas penuh hari ini (+ sinkron PEL)
   349→   349→  const handleTandaiLunas = async () => {
   350→   350→    if (!entry || !data) return
   351→   351→    setUpdating(true)
   352→   352→    try {
   353→   353→      const parsed = JSON.parse(entry.dataJson)
   354→   354→      parsed.lunas = true
   355→   355→      parsed.tanggalPelunasan = getTodayStr()
   356→   356→      delete parsed.statusPembayaran
   357→   357→      if ((parsed.dp || 0) > 0) {
   358→   358→        const sub = (parsed.items || []).reduce((s: number, it: { qty: number; harga: number }) => s + it.qty * it.harga, 0)
   359→   359→        const totalHarga = sub + (sub * (parsed.ppn || 0) / 100)
   360→   360→        parsed.originalTotal = parsed.originalTotal !== undefined ? parsed.originalTotal : totalHarga
   361→   361→        parsed.dpAmount = parsed.originalTotal * (parsed.dp / 100)
   362→   362→      }
   363→   363→      const res = await fetcher(`/api/history/${entry.id}`, {
   364→   364→        method: 'PUT',
   365→   365→        headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   366→   366→        body: JSON.stringify({ dataJson: JSON.stringify(parsed) }),
   367→   367→      })
   368→   368→      if (res.ok) {
   369→   369→        await syncLinkedPelunasan(entry.id, parsed.nomor || entry.nomor, parsed)
   370→   370→        toast.success('Invoice ditandai LUNAS')
   371→   371→        setLunasOpen(false)
   372→   372→        window.dispatchEvent(new CustomEvent('dokupro:history-updated'))
   373→   373→        notifyDataChange('invoice')
   374→   374→        loadEntry()
   375→   375→      } else {
   376→   376→        toast.error('Gagal menandai lunas')
   377→   377→      }
   378→   378→    } catch {
   379→   379→      toast.error('Gagal menandai lunas')
   380→   380→    } finally {
   381→   381→      setUpdating(false)
   382→   382→    }
   383→   383→  }
   384→   384→
   385→   385→  // Batal — invoice dibatalkan (status Batal di riwayat & detail)
   386→   386→  const handleBatal = async () => {
   387→   387→    if (!entry) return
   388→   388→    setUpdating(true)
   389→   389→    try {
   390→   390→      const parsed = JSON.parse(entry.dataJson)
   391→   391→      parsed.batal = true
   392→   392→      const res = await fetcher(`/api/history/${entry.id}`, {
   393→   393→        method: 'PUT',
   394→   394→        headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   395→   395→        body: JSON.stringify({ dataJson: JSON.stringify(parsed) }),
   396→   396→      })
   397→   397→      if (res.ok) {
   398→   398→        toast.success('Invoice dibatalkan')
   399→   399→        setBatalOpen(false)
   400→   400→        window.dispatchEvent(new CustomEvent('dokupro:history-updated'))
   401→   401→        notifyDataChange('invoice')
   402→   402→        loadEntry()
   403→   403→      } else {
   404→   404→        toast.error('Gagal membatalkan invoice')
   405→   405→      }
   406→   406→    } catch {
   407→   407→      toast.error('Gagal membatalkan invoice')
   408→   408→    } finally {
   409→   409→      setUpdating(false)
   410→   410→    }
   411→   411→  }
   412→   412→
   413→   413→  // Hapus — soft delete → masuk tab Sampah (bisa dipulihkan)
   414→   414→  const handleHapus = async () => {
   415→   415→    if (!entry) return
