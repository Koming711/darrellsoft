   169→function formatTanggalID(iso: string): string {
   170→  if (!iso) return '-'
   171→  const d = new Date(iso)
   172→  if (Number.isNaN(d.getTime())) return iso
   173→  return d.toLocaleDateString('id-ID', { day: '2-digit', month: 'short', year: 'numeric' })
   174→}
   175→
   176→/** Tanggal hari ini pada pukul 00:00 (untuk deteksi jatuh tempo). */
   177→function startOfToday(): number {
   178→  const n = new Date()
   179→  return new Date(n.getFullYear(), n.getMonth(), n.getDate()).getTime()
   180→}

   182→/** Format ringkas utk sumbu grafik: 850rb / 1,2jt / 2M */
   183→function compactIDR(v: number): string {
   184→  const trim = (n: number) => {
   185→    const s = n.toFixed(1)
   186→    return s.endsWith('.0') ? s.slice(0, -2) : s
   187→  }
   188→  if (v >= 1_000_000_000) return `${trim(v / 1_000_000_000)}M`
   189→  if (v >= 1_000_000) return `${trim(v / 1_000_000)}jt`
   190→  if (v >= 1_000) return `${Math.round(v / 1_000)}rb`
   191→  return `${Math.round(v)}`
   192→}

   194→/** Tooltip kustom grafik batang */
   195→function ChartTip({
   196→  active,
   197→  payload,
   198→  label,
   199→}: {
   200→  active?: boolean
   201→  payload?: Array<{ value?: number | string }>
   202→  label?: string
   203→}) {
   204→  if (!active || !payload?.length) return null
   205→  return (
   206→    <div className="rounded-lg border border-stone-200 bg-white px-3 py-2 shadow-md dark:border-zinc-700 dark:bg-zinc-800">
   207→      <p className="text-[11px] font-medium text-stone-500 dark:text-zinc-400">{label}</p>
   208→      <p className="text-sm font-semibold text-stone-900 dark:text-zinc-100">{formatRupiah(Number(payload[0]?.value ?? 0))}</p>
   209→    </div>
   210→  )
   211→}
