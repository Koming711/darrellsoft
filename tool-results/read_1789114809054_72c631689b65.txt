   109→   109→/** Warna angka sisa tagihan: 0 → hijau (lunas), lainnya → oranye */
   110→   110→function sisaClass(sisa: number): string {
   111→   111→  return sisa <= 0 ? 'text-emerald-700 dark:text-emerald-400' : 'text-amber-700 dark:text-amber-400'

   114→   114→/** Format tanggal ISO (YYYY-MM-DD) → "19 Jul 2026" */
   115→   115→function formatDateID(iso: string): string {
   116→   116→  if (!iso) return '-'
   117→   117→  const d = new Date(iso)
   118→   118→  if (Number.isNaN(d.getTime())) return iso
   119→   119→  return d.toLocaleDateString('id-ID', { day: '2-digit', month: 'short', year: 'numeric' })
   120→   120→}
   121→   121→
   122→   122→/** Parse daftar entry history menjadi baris terstruktur; skip JSON rusak. */
   123→   123→function parseHistory<T>(entries: HistoryEntry[], mapFn: (parsed: Record<string, unknown>, entry: HistoryEntry) => T | null): T[] {
   124→   124→  const out: T[] = []
   125→   125→  for (const entry of entries) {
   126→   126→    try {
   127→   127→      const parsed = JSON.parse(entry.dataJson) as Record<string, unknown>
   128→   128→      const row = mapFn(parsed, entry)
   129→   129→      if (row) out.push(row)
   130→   130→    } catch {
   131→   131→      /* skip rusak */
   132→   132→    }
   133→   133→  }
   134→   134→  return out
   135→   135→}
