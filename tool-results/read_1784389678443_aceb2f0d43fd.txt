    64→    64→// ===== Date filter helpers (same scheme as rekap-penjualan) =====
    65→    65→type FilterType = 'all' | 'today' | 'week' | 'month' | 'custom'
    66→    66→
    67→    67→function getFilterDates(
    68→    68→  filter: FilterType,
    69→    69→  customStart?: Date,
    70→    70→  customEnd?: Date
    71→    71→): { startDate: string; endDate: string } {
    72→    72→  const today = new Date()
    73→    73→  const fmt = (d: Date) => {
    74→    74→    const y = d.getFullYear()
    75→    75→    const m = String(d.getMonth() + 1).padStart(2, '0')
    76→    76→    const day = String(d.getDate()).padStart(2, '0')
    77→    77→    return `${y}-${m}-${day}`
    78→    78→  }
    79→    79→  switch (filter) {
    80→    80→    case 'all':
    81→    81→      return { startDate: '2000-01-01', endDate: '2099-12-31' }
    82→    82→    case 'today':
    83→    83→      return { startDate: fmt(today), endDate: fmt(today) }
    84→    84→    case 'week': {
    85→    85→      const s = new Date(today)
    86→    86→      const day = s.getDay()
    87→    87→      s.setDate(s.getDate() + (day === 0 ? -6 : 1 - day))
    88→    88→      return { startDate: fmt(s), endDate: fmt(today) }
    89→    89→    }
    90→    90→    case 'month': {
    91→    91→      const s = new Date(today.getFullYear(), today.getMonth(), 1)
    92→    92→      return { startDate: fmt(s), endDate: fmt(today) }
    93→    93→    }
    94→    94→    case 'custom':
    95→    95→      return {
    96→    96→        startDate: customStart ? fmt(customStart) : fmt(today),
    97→    97→        endDate: customEnd ? fmt(customEnd) : fmt(today),
    98→    98→      }
    99→    99→  }
   100→   100→}

   102→   102→function formatDateShort(iso: string): string {
   103→   103→  if (!iso) return '-'
   104→   104→  try {
   105→   105→    return new Date(iso).toLocaleDateString('id-ID', {
   106→   106→      day: '2-digit',
   107→   107→      month: 'short',
   108→   108→      year: 'numeric',
   109→   109→    })
   110→   110→  } catch {
   111→   111→    return iso
   112→   112→  }
   113→   113→}
   114→   114→
   115→   115→function formatDateShortId(dateStr: string): string {
   116→   116→  if (!dateStr) return '-'
   117→   117→  try {
   118→   118→    return new Date(dateStr + 'T00:00:00').toLocaleDateString('id-ID', {
   119→   119→      day: '2-digit',
   120→   120→      month: 'short',
   121→   121→      year: 'numeric',
   122→   122→    })
   123→   123→  } catch {
   124→   124→    return dateStr
   125→   125→  }
   126→   126→}
   127→   127→
   128→   128→function formatRupiahShort(n: number): string {
   129→   129→  return new Intl.NumberFormat('id-ID', {
   130→   130→    style: 'currency',
   131→   131→    currency: 'IDR',
   132→   132→    minimumFractionDigits: 0,
   133→   133→  }).format(n)
   134→   134→}
