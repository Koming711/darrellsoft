    12→// === MODULE-LEVEL SESSION CACHE ===
    13→// Persists across navigations so DashboardLayout doesn't need to re-verify on every mount
    14→interface CachedSession {
    15→  user: any
    16→  userProfile: { createdAt: string | null; validUntil: string | null } | null
    17→  autoLogoutMin: number
    18→  logoutWarningSec: number
    19→  sessionWarning: string | null
    20→  forceLogoutAvailable: boolean
    21→  accountExpired: boolean
    22→  cachedAt: number
    23→}
    24→let _sessionCache: CachedSession | null = null
    25→const SESSION_CACHE_TTL = 120_000 // 2 minutes
    26→
    27→function getSessionCache(): CachedSession | null {
    28→  // Check module-level cache first
    29→  if (_sessionCache && Date.now() - _sessionCache.cachedAt <= SESSION_CACHE_TTL) {
    30→    return _sessionCache
    31→  }
    32→  // Fallback to sessionStorage
    33→  if (typeof window !== 'undefined') {
    34→    try {
    35→      const stored = sessionStorage.getItem('__dashboard_session_cache')
    36→      if (stored) {
    37→        const parsed = JSON.parse(stored) as CachedSession
    38→        if (Date.now() - parsed.cachedAt <= SESSION_CACHE_TTL) {
    39→          _sessionCache = parsed
    40→          return parsed
    41→        }
    42→        sessionStorage.removeItem('__dashboard_session_cache')
    43→      }
    44→    } catch {}
    45→  }
    46→  _sessionCache = null
    47→  return null
    48→}
    49→
    50→function setSessionCache(data: Omit<CachedSession, 'cachedAt'>) {
    51→  const entry = { ...data, cachedAt: Date.now() }
    52→  _sessionCache = entry
    53→  if (typeof window !== 'undefined') {
    54→    try {
    55→      sessionStorage.setItem('__dashboard_session_cache', JSON.stringify(entry))
    56→    } catch {}
    57→  }
    58→}
    59→
    60→function invalidateSessionCache() {
    61→  _sessionCache = null
    62→  if (typeof window !== 'undefined') {
    63→    try {
    64→      sessionStorage.removeItem('__dashboard_session_cache')
    65→    } catch {}
    66→  }
    67→}

   547→      {/* ===== MODAL: ACCOUNT EXPIRED ===== */}
   548→      {accountExpired && sessionWarning && (
   549→        <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/50 backdrop-blur-sm px-4">
   550→          <div className="rounded-2xl shadow-2xl border border-red-200 dark:border-red-800 max-w-sm w-full p-6 animate-in fade-in zoom-in duration-200" style={{ backgroundColor: 'var(--card)' }}>
   551→            <div className="flex items-start gap-3 mb-4">
   552→              <div className="w-10 h-10 rounded-xl bg-red-100 dark:bg-red-900/30 flex items-center justify-center flex-shrink-0">
   553→                <TimerOff className="w-5 h-5 text-red-600 dark:text-red-400" />
   554→              </div>
   555→              <div>
   556→                <h3 className="text-lg font-bold text-foreground">Akun Kadaluarsa</h3>
   557→                <p className="text-xs text-muted-foreground mt-0.5">Masa berlaku akun telah habis</p>
   558→              </div>
   559→            </div>
   560→            <div className="bg-red-50 dark:bg-red-950 border border-red-200 dark:border-red-800 rounded-xl p-4 mb-5">
   561→              <p className="text-sm text-red-700 dark:text-red-400 leading-relaxed">{sessionWarning}</p>
   562→            </div>
   563→            <button
   564→              onClick={handleLogout}
   565→              className="w-full bg-red-600 hover:bg-red-700 text-white font-semibold py-2.5 rounded-xl transition-colors"
   566→              autoFocus
   567→            >
   568→              <span className="flex items-center justify-center gap-2">
   569→                <LogOut className="w-4 h-4" /> OK, Mengerti
   570→              </span>
   571→            </button>
   572→          </div>
   573→        </div>
   574→      )}
   575→
   576→      {/* ===== MODAL: SESSION WARNING (Multi-Device) ===== */}
   577→      {sessionWarning && !accountExpired && (
   578→        <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/50 backdrop-blur-sm px-4">
   579→          <div className="rounded-2xl shadow-2xl border border-red-200 dark:border-red-800 max-w-sm w-full p-6 animate-in fade-in zoom-in duration-200" style={{ backgroundColor: 'var(--card)' }}>
   580→            <div className="flex items-start gap-3 mb-4">
   581→              <div className="w-10 h-10 rounded-xl bg-red-100 dark:bg-red-900/30 flex items-center justify-center flex-shrink-0">
   582→                {forceLogoutAvailable ? <ShieldAlert className="w-5 h-5 text-red-600 dark:text-red-400" /> : <Smartphone className="w-5 h-5 text-red-600 dark:text-red-400" />}
   583→              </div>
   584→              <div>
   585→                <h3 className="text-lg font-bold text-foreground">Peringatan Keamanan</h3>
   586→                <p className="text-xs text-muted-foreground mt-0.5">Akun digunakan di perangkat lain</p>
   587→              </div>
   588→            </div>
   589→            <div className="bg-red-50 dark:bg-red-950 border border-red-200 dark:border-red-800 rounded-xl p-4 mb-5">
   590→              <p className="text-sm text-red-700 dark:text-red-400 leading-relaxed">{sessionWarning}</p>
   591→            </div>
   592→            {forceLogoutAvailable ? (
   593→              <div className="space-y-2.5">
   594→                <button
   595→                  onClick={async () => {
   596→                    setIsReclaiming(true)
   597→                    try {
   598→                      const authUser = getAuthUser()
   599→                      if (!authUser?.sessionId) {
   600→                        toast.error('Data sesi tidak ditemukan, silakan login ulang')
   601→                        setIsReclaiming(false)
   602→                        handleLogout()
   603→                        return
   604→                      }
   605→                      const res = await authFetch('/api/auth/reclaim-session', {
   606→                        method: 'POST',
   607→                        headers: { 'Content-Type': 'application/json' },
   608→                        body: JSON.stringify({ username: authUser.username, sessionId: authUser.sessionId, userId: authUser.id }),
   609→                      })
   610→                      const data = await res.json()
   611→                      if (res.ok) {
   612→                        setSessionWarning(null)
   613→                        setForceLogoutAvailable(false)
   614→                        toast.success('Perangkat lain berhasil di-logout')
   615→                      } else {
   616→                        toast.error(data.error || 'Gagal mengklaim sesi. Silakan coba lagi.')
   617→                      }
   618→                    } catch {
   619→                      toast.error('Terjadi kesalahan jaringan')
   620→                    }
   621→                    setIsReclaiming(false)
   622→                  }}
   623→                  disabled={isReclaiming}
   624→                  className="w-full flex items-center justify-center gap-2 bg-amber-500 hover:bg-amber-600 disabled:bg-amber-400 text-white font-semibold py-2.5 rounded-xl transition-colors"
   625→                  autoFocus
   626→                >
   627→                  {isReclaiming ? (
   628→                    <><div className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin" /> Memproses...</>
   629→                  ) : (
   630→                    <><ShieldAlert className="w-4 h-4" /> Paksa Logout Perangkat Lain</>
   631→                  )}
   632→                </button>
   633→                <button
   634→                  onClick={handleLogout}
   635→                  className="w-full flex items-center justify-center gap-2 border-2 border-red-200 dark:border-red-800 text-red-600 dark:text-red-400 hover:bg-red-50 dark:hover:bg-red-950 font-semibold py-2.5 rounded-xl transition-colors"
   636→                >
   637→                  <LogOut className="w-4 h-4" /> Logout dari Sini
   638→                </button>
   639→              </div>
   640→            ) : (
   641→              <button
   642→                onClick={handleLogout}
   643→                className="w-full bg-red-600 hover:bg-red-700 text-white font-semibold py-2.5 rounded-xl transition-colors"
   644→                autoFocus
   645→              >
   646→                OK, Logout
   647→              </button>
   648→            )}
   649→          </div>
   650→        </div>
   651→      )}
   652→
   653→      {/* ===== MODAL: AUTO-LOGOUT COUNTDOWN ===== */}
   654→      {showCountdown && (
   655→        <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/50 backdrop-blur-sm px-4">
   656→          <div className="rounded-2xl shadow-2xl border border-amber-200 dark:border-amber-800 max-w-sm w-full p-6 animate-in fade-in zoom-in duration-200" style={{ backgroundColor: 'var(--card)' }}>
   657→            <div className="flex items-start gap-3 mb-4">
   658→              <div className="w-10 h-10 rounded-xl bg-amber-100 dark:bg-amber-900/30 flex items-center justify-center flex-shrink-0">
   659→                <AlertTriangle className="w-5 h-5 text-amber-600 dark:text-amber-400" />
   660→              </div>
   661→              <div>
   662→                <h3 className="text-lg font-bold text-foreground">Sesi Akan Berakhir</h3>
   663→                <p className="text-xs text-muted-foreground mt-0.5">Anda tidak aktif, akan otomatis logout</p>
   664→              </div>
   665→            </div>
   666→            <div className="text-center py-5">
   667→              <div className="text-6xl font-bold text-amber-600 dark:text-amber-400 tabular-nums">{countdown}</div>
   668→              <p className="text-sm text-muted-foreground mt-2">detik tersisa</p>
   669→            </div>
   670→            <button
   671→              onClick={handleStayLoggedIn}
   672→              className="w-full bg-amber-500 hover:bg-amber-600 text-white font-semibold py-2.5 rounded-xl transition-colors"
   673→              autoFocus
   674→            >
   675→              Tetap Login
   676→            </button>
   677→          </div>
   678→        </div>
   679→      )}
   680→    </div>
   681→  )
