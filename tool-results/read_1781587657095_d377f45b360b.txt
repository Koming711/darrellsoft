    12→// === MODULE-LEVEL SESSION CACHE ===
    13→// Persists across navigations so DashboardLayout doesn't need to re-verify on every mount
    14→interface CachedSession {
    15→  user: any
    16→  userProfile: { createdAt: string | null; validUntil: string | null } | null
    17→  autoLogoutMin: number
    18→  logoutWarningSec: number
    19→  sessionWarning: string | null
    20→  forceLogoutAvailable: boolean
    21→  accountExpired: boolean
    22→  cachedAt: number
    23→}
    24→let _sessionCache: CachedSession | null = null
    25→const SESSION_CACHE_TTL = 30_000 // 30 seconds
    26→
    27→function getSessionCache(): CachedSession | null {
    28→  if (!_sessionCache) return null
    29→  if (Date.now() - _sessionCache.cachedAt > SESSION_CACHE_TTL) {
    30→    _sessionCache = null
    31→    return null
    32→  }
    33→  return _sessionCache
    34→}
    35→
    36→function setSessionCache(data: Omit<CachedSession, 'cachedAt'>) {
    37→  _sessionCache = { ...data, cachedAt: Date.now() }
    38→}
    39→
    40→function invalidateSessionCache() {
    41→  _sessionCache = null
    42→}

    63→  const sessionIntervalRef = useRef<ReturnType<typeof setInterval> | null>(null)
    64→
    65→  // === AUTO-LOGOUT STATE ===
    66→  const lastActivityRef = useRef(Date.now())
    67→  const countdownActiveRef = useRef(false)
    68→  const [autoLogoutMin, setAutoLogoutMin] = useState(cache?.autoLogoutMin ?? 0)
    69→  const [logoutWarningSec, setLogoutWarningSec] = useState(cache?.logoutWarningSec ?? 0)

    71→  const [countdown, setCountdown] = useState(0)

    78→  const initDoneRef = useRef(false)

    82→

   111→    setShowCountdown(false)

   113→  }, [])
   114→

   118→    initDoneRef.current = true

   120→    // If we have a valid cache, we're already ready — just do a background refresh

   137→            setLogoutWarningSec(data.securitySettings.logout_warning_sec || 0)

   140→            saveRolePermissions(authUser.role, data.permissions.features, data.permissions.subPermissions)

   148→              setSessionWarning(data.warningMessage)

   158→            sessionWarning: data.valid ? null : (data.warningMessage || null),
   159→            forceLogoutAvailable: data.valid ? false : !!data.forceLogoutAvailable,
   160→            accountExpired: data.valid ? false : !!data.expired,

   163→          // Background refresh failed — keep using cached data
   164→        }

   166→      bgRefresh()
   167→      return
   168→    }
   169→
   170→    let cancelled = false

   172→    const init = async () => {

   174→        const authUser = getAuthUser()

   176→          if (!cancelled) setReady(true)
   177→          return
   178→        }
   179→        if (cancelled) return

   181→

   183→        authFetch('/api/auth/me')

   188→                setSessionWarning(data.error || 'Akun sudah expired. Silahkan diperpanjang lagi akunnya.')

   190→              return null

   192→            return r.ok ? r.json() : null
   193→          })
   194→          .then(data => { if (data && !cancelled) {
   195→            setUserProfile({ createdAt: data.createdAt || null, validUntil: data.validUntil || null })
   196→            // Update cache with profile data
   197→            const currentCache = getSessionCache()
   198→            if (currentCache) {
   199→              setSessionCache({ ...currentCache, userProfile: { createdAt: data.createdAt || null, validUntil: data.validUntil || null } })
   200→            }
   201→          } })
   202→          .catch(() => {})
   203→
   204→        // Wait for session verification BEFORE checking permissions
   205→        try {
   206→          const sessionRes = await authFetch('/api/auth/verify-session', {
   207→            method: 'POST',
   208→            headers: { 'Content-Type': 'application/json' },
   209→            body: JSON.stringify({ username: authUser.username, sessionId: authUser.sessionId, role: authUser.role }),
   210→          })
   211→          if (cancelled) return
   212→          const data = await sessionRes.json()
   213→          if (cancelled) return
   214→
   215→          // Load security settings from verify-session response
   216→          if (data.securitySettings) {
   217→            setAutoLogoutMin(data.securitySettings.auto_logout_min || 0)
   218→            setLogoutWarningSec(data.securitySettings.logout_warning_sec || 0)
   219→          }
   220→          // Sync latest permissions from server to localStorage
   221→          if (data.permissions && authUser.role) {
   222→            saveRolePermissions(authUser.role, data.permissions.features, data.permissions.subPermissions)
   223→            setPermVersion(v => v + 1)
   224→          }

   226→            if (data.expired) {
   227→              setAccountExpired(true)
   228→              setSessionWarning(data.warningMessage || 'Akun sudah expired.')
   229→            } else if (data.warningMessage) {
   230→              setSessionWarning(data.warningMessage)
   231→              setForceLogoutAvailable(!!data.forceLogoutAvailable)
   232→            }
   233→          }
   234→
   235→          // Update session cache with verified data
   236→          setSessionCache({
   237→            user: authUser,
   238→            userProfile: null,
   239→            autoLogoutMin: data.securitySettings?.auto_logout_min || 0,
   240→            logoutWarningSec: data.securitySettings?.logout_warning_sec || 0,
   241→            sessionWarning: data.valid ? null : (data.warningMessage || null),
   242→            forceLogoutAvailable: data.valid ? false : !!data.forceLogoutAvailable,
   243→            accountExpired: data.valid ? false : !!data.expired,
   244→          })
   245→
   246→          // === AUTO-BACKUP CHECK ===
   247→          // Check if auto-backup is due and trigger it silently
   248→          if (data.valid && authUser.role !== 'superadmin') {
   249→            authFetch('/api/database/auto-backup', { method: 'POST' })

   254→                }

   256→              .catch(() => {}) // silent fail
   257→          }
   258→        } catch {
   259→          // Session verification failed - continue anyway
   260→        }
   261→
   262→      } catch {
   263→        // On error, don't redirect — just show the page
   264→      } finally {
   265→        if (!cancelled) {
   266→          setReady(true)
   267→          // Cache session data for instant navigation
   268→          const authUser = getAuthUser()
   269→          if (authUser) {
   270→            setSessionCache({
   271→              user: authUser,
   272→              userProfile: null, // will be updated by authFetch below
   273→              autoLogoutMin: 0,
   274→              logoutWarningSec: 0,
   275→              sessionWarning: null,
   276→              forceLogoutAvailable: false,
   277→              accountExpired: false,
   278→            })
   279→          }
   280→        }
   281→      }
   282→    }
   283→
   284→    init()
   285→
   286→    return () => { cancelled = true }
   287→  }, [])
   288→
   289→  // === PERMISSION CHECK ON ROUTE CHANGE ===
   290→  useEffect(() => {
   291→    if (!user || !ready) return
   292→    if (user.role === 'superadmin') { setNoAccess(false); return }
   293→
   294→    const featureId = getFeatureIdForPath(pathname)
   295→    if (featureId) {
   296→      const allowed = hasFeatureAccess(user.role, featureId)
   297→      setNoAccess(!allowed)
   298→    } else {
   299→      setNoAccess(false)
   300→    }
   301→  }, [pathname, user, ready, permVersion])
   302→
   303→  // === PERIODIC SESSION CHECK (every 10s) ===
   304→  useEffect(() => {
   305→    if (!ready || !user) return
   306→
   307→    sessionIntervalRef.current = setInterval(() => {
   308→      const authUser = getAuthUser()
   309→      if (!authUser || !authUser.sessionId) return
   310→
   311→      authFetch('/api/auth/verify-session', {
   312→        method: 'POST',
   313→        headers: { 'Content-Type': 'application/json' },
   314→        body: JSON.stringify({ username: authUser.username, sessionId: authUser.sessionId, role: authUser.role }),
   315→      })
   316→      .then(r => r.json())
   317→      .then(data => {
   318→        if (!data.valid) {
   319→          if (data.expired) {
   320→            setAccountExpired(true)
   321→            setSessionWarning(data.warningMessage || 'Akun sudah expired.')
   322→          } else if (data.warningMessage) {
   323→            setSessionWarning(data.warningMessage)
   324→            setForceLogoutAvailable(!!data.forceLogoutAvailable)
   325→          }
   326→        }
   327→        if (data.permissions && authUser.role) {
   328→          saveRolePermissions(authUser.role, data.permissions.features, data.permissions.subPermissions)
   329→          setPermVersion(v => v + 1)
   330→        }
   331→        if (data.securitySettings) {
   332→          setAutoLogoutMin(data.securitySettings.auto_logout_min || 0)
   333→          setLogoutWarningSec(data.securitySettings.logout_warning_sec || 0)
   334→        }
   335→      })
   336→      .catch(() => {})
   337→    }, 10000)
   338→
   339→    return () => {
   340→      if (sessionIntervalRef.current) {
   341→        clearInterval(sessionIntervalRef.current)
   342→        sessionIntervalRef.current = null
   343→      }
   344→    }
   345→  }, [ready, user])
   346→
   347→  // === ACTIVITY LISTENERS ===
   348→  useEffect(() => {
   349→    const events = ['mousemove', 'keydown', 'scroll', 'click', 'touchstart'] as const
   350→    const handler = () => { lastActivityRef.current = Date.now() }
   351→    events.forEach(e => window.addEventListener(e, handler, { passive: true }))
   352→    return () => events.forEach(e => window.removeEventListener(e, handler))
   353→  }, [])
   354→
   355→  // === AUTO-LOGOUT INTERVAL ===
   356→  useEffect(() => {
   357→    if (!ready) return
   358→
   359→    autoLogoutIntervalRef.current = setInterval(() => {
   360→      if (autoLogoutMin === 0) return
   361→
   362→      const elapsed = (Date.now() - lastActivityRef.current) / 1000
   363→      const autoLogoutSec = autoLogoutMin * 60
   364→
   365→      if (elapsed >= autoLogoutSec) {
   366→        handleLogout()
   367→        return
   368→      }
   369→
   370→      if (logoutWarningSec > 0 && elapsed >= autoLogoutSec - logoutWarningSec) {
   371→        countdownActiveRef.current = true
   372→        setShowCountdown(true)
   373→        setCountdown(Math.ceil(autoLogoutSec - elapsed))
   374→      } else if (countdownActiveRef.current) {
   375→        countdownActiveRef.current = false
   376→        setShowCountdown(false)
   377→        setCountdown(0)
   378→      }
   379→    }, 1000)
   380→
   381→    return () => {
   382→      if (autoLogoutIntervalRef.current) {
   383→        clearInterval(autoLogoutIntervalRef.current)
   384→        autoLogoutIntervalRef.current = null
   385→      }
   386→    }
   387→  }, [ready, autoLogoutMin, logoutWarningSec, handleLogout])
   388→

   390→  if (!ready) {
   391→    return (
   392→      <div className="min-h-screen bg-background flex items-center justify-center">
   393→        <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-blue-600"></div>
   394→      </div>
   395→    )
   396→  }
   397→
   398→  if (!user) {
   399→    return (
   400→      <div className="min-h-screen bg-background flex items-center justify-center">
   401→        <div className="text-center p-8">

   411→            }}
   412→            className="px-6 py-2.5 bg-blue-600 hover:bg-blue-700 text-white font-semibold rounded-xl transition-colors"
   413→          >
   414→            Masuk

   435→    '/riwayat-pembelian': berandaPath,
   436→    '/riwayat-penjualan': berandaPath,
   437→  }
   438→  const backLabel = backLabelMap[pathname] || 'Kembali ke Beranda'
   439→  const backPath = backPathMap[pathname] || (user ? (getFirstAccessiblePath(user.role) || '/pembukaan') : '/pembukaan')
   440→
   441→  if (noAccess) {
   442→    return (
   443→      <div className="min-h-screen" style={{ backgroundColor: 'var(--app-content-bg)' }}>
   444→        <Sidebar
   445→          username={user?.username || 'User'}
   446→          role={user?.role}
   447→          onLogout={handleLogout}
   448→          isOpen={sidebarOpen}
   449→          onToggle={() => setSidebarOpen(!sidebarOpen)}
   450→        />
   451→        <div className="lg:ml-16 transition-all duration-300">
   452→          <MobileHeader username={user?.username} title={title} subtitle={subtitle} userProfile={userProfile} />
   453→          <main className="p-4 pb-20 lg:p-8 lg:pb-8">
   454→            <div className="flex flex-col items-center justify-center min-h-[60vh] text-center">
   455→              <div className="w-20 h-20 rounded-full bg-amber-50 dark:bg-amber-900/20 flex items-center justify-center mb-5 relative">
   456→                <Lock className="w-9 h-9 text-amber-500" />
   457→                <div className="absolute -top-1 -right-1 w-7 h-7 rounded-full bg-amber-500 flex items-center justify-center">
   458→                  <Crown className="w-4 h-4 text-white" />
   459→                </div>
   460→              </div>
   461→              <div className="inline-flex items-center gap-1.5 px-3 py-1 rounded-full bg-amber-100 dark:bg-amber-900/30 mb-3">
   462→                <span className="text-[11px] font-bold text-amber-700 dark:text-amber-400 uppercase tracking-wider">Fitur PRO</span>
   463→              </div>
   464→              <h2 className="text-xl font-bold text-foreground mb-2">Fitur Belum Tersedia</h2>
   465→              <p className="text-sm text-muted-foreground max-w-md">
   466→                Fitur <span className="font-semibold text-foreground">{title}</span> belum diaktifkan untuk akun Anda.
   467→                Hubungi administrator untuk mendapatkan akses.
   468→              </p>
   469→              <a
   470→                href="/pembukaan"
   471→                onClick={(e) => {

   473→                  window.location.replace('/pembukaan')
   474→                }}
   475→                className="mt-5 px-5 py-2.5 bg-primary text-primary-foreground text-sm font-medium rounded-lg hover:bg-primary/90 transition-colors no-underline inline-block cursor-pointer"
   476→              >
   477→                {backLabel}
   478→              </a>

   485→
   486→  // === RENDER ===
   487→  return (

   489→      <Sidebar
   490→        username={user?.username || 'User'}

   498→      <div className="lg:ml-16 transition-all duration-300">

   501→          title={title}

   527→            </div>
   528→            <div className="bg-red-50 dark:bg-red-950 border border-red-200 dark:border-red-800 rounded-xl p-4 mb-5">
   529→              <p className="text-sm text-red-700 dark:text-red-400 leading-relaxed">{sessionWarning}</p>
   530→            </div>
   531→            <button
   532→              onClick={handleLogout}
   533→              className="w-full bg-red-600 hover:bg-red-700 text-white font-semibold py-2.5 rounded-xl transition-colors"
   534→              autoFocus
   535→            >
   536→              <span className="flex items-center justify-center gap-2">
   537→                <LogOut className="w-4 h-4" /> OK, Mengerti
   538→              </span>
   539→            </button>
   540→          </div>
   541→        </div>
   542→      )}
   543→
   544→      {/* ===== MODAL: SESSION WARNING (Multi-Device) ===== */}
   545→      {sessionWarning && !accountExpired && (
   546→        <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/50 backdrop-blur-sm px-4">
   547→          <div className="rounded-2xl shadow-2xl border border-red-200 dark:border-red-800 max-w-sm w-full p-6 animate-in fade-in zoom-in duration-200" style={{ backgroundColor: 'var(--card)' }}>
   548→            <div className="flex items-start gap-3 mb-4">
   549→              <div className="w-10 h-10 rounded-xl bg-red-100 dark:bg-red-900/30 flex items-center justify-center flex-shrink-0">
   550→                {forceLogoutAvailable ? <ShieldAlert className="w-5 h-5 text-red-600 dark:text-red-400" /> : <Smartphone className="w-5 h-5 text-red-600 dark:text-red-400" />}
   551→              </div>
   552→              <div>
   553→                <h3 className="text-lg font-bold text-foreground">Peringatan Keamanan</h3>
   554→                <p className="text-xs text-muted-foreground mt-0.5">Akun digunakan di perangkat lain</p>
   555→              </div>
   556→            </div>
   557→            <div className="bg-red-50 dark:bg-red-950 border border-red-200 dark:border-red-800 rounded-xl p-4 mb-5">
   558→              <p className="text-sm text-red-700 dark:text-red-400 leading-relaxed">{sessionWarning}</p>
   559→            </div>
   560→            {forceLogoutAvailable ? (
   561→              <div className="space-y-2.5">
   562→                <button
   563→                  onClick={async () => {
   564→                    setIsReclaiming(true)
   565→                    try {
   566→                      const authUser = getAuthUser()
   567→                      if (!authUser?.sessionId) {
   568→                        toast.error('Data sesi tidak ditemukan, silakan login ulang')
   569→                        setIsReclaiming(false)
   570→                        handleLogout()
   571→                        return
   572→                      }
   573→                      const res = await authFetch('/api/auth/reclaim-session', {
   574→                        method: 'POST',
   575→                        headers: { 'Content-Type': 'application/json' },
   576→                        body: JSON.stringify({ username: authUser.username, sessionId: authUser.sessionId, userId: authUser.id }),
   577→                      })
   578→                      const data = await res.json()
   579→                      if (res.ok) {
   580→                        setSessionWarning(null)
   581→                        setForceLogoutAvailable(false)
   582→                        toast.success('Perangkat lain berhasil di-logout')
   583→                      } else {
   584→                        toast.error(data.error || 'Gagal mengklaim sesi. Silakan coba lagi.')
   585→                      }
   586→                    } catch {
   587→                      toast.error('Terjadi kesalahan jaringan')
   588→                    }
   589→                    setIsReclaiming(false)
   590→                  }}
   591→                  disabled={isReclaiming}
   592→                  className="w-full flex items-center justify-center gap-2 bg-amber-500 hover:bg-amber-600 disabled:bg-amber-400 text-white font-semibold py-2.5 rounded-xl transition-colors"
   593→                  autoFocus
   594→                >
   595→                  {isReclaiming ? (
   596→                    <><div className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin" /> Memproses...</>
   597→                  ) : (
   598→                    <><ShieldAlert className="w-4 h-4" /> Paksa Logout Perangkat Lain</>
   599→                  )}
   600→                </button>
   601→                <button
   602→                  onClick={handleLogout}
   603→                  className="w-full flex items-center justify-center gap-2 border-2 border-red-200 dark:border-red-800 text-red-600 dark:text-red-400 hover:bg-red-50 dark:hover:bg-red-950 font-semibold py-2.5 rounded-xl transition-colors"
   604→                >
   605→                  <LogOut className="w-4 h-4" /> Logout dari Sini
   606→                </button>
   607→              </div>
   608→            ) : (
   609→              <button
   610→                onClick={handleLogout}
   611→                className="w-full bg-red-600 hover:bg-red-700 text-white font-semibold py-2.5 rounded-xl transition-colors"
   612→                autoFocus
   613→              >
   614→                OK, Logout
   615→              </button>
   616→            )}
   617→          </div>
   618→        </div>
   619→      )}
   620→
   621→      {/* ===== MODAL: AUTO-LOGOUT COUNTDOWN ===== */}
   622→      {showCountdown && (
   623→        <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/50 backdrop-blur-sm px-4">
   624→          <div className="rounded-2xl shadow-2xl border border-amber-200 dark:border-amber-800 max-w-sm w-full p-6 animate-in fade-in zoom-in duration-200" style={{ backgroundColor: 'var(--card)' }}>
   625→            <div className="flex items-start gap-3 mb-4">
   626→              <div className="w-10 h-10 rounded-xl bg-amber-100 dark:bg-amber-900/30 flex items-center justify-center flex-shrink-0">
   627→                <AlertTriangle className="w-5 h-5 text-amber-600 dark:text-amber-400" />
   628→              </div>
   629→              <div>
   630→                <h3 className="text-lg font-bold text-foreground">Sesi Akan Berakhir</h3>
   631→                <p className="text-xs text-muted-foreground mt-0.5">Anda tidak aktif, akan otomatis logout</p>
   632→              </div>
   633→            </div>
   634→            <div className="text-center py-5">
   635→              <div className="text-6xl font-bold text-amber-600 dark:text-amber-400 tabular-nums">{countdown}</div>
   636→              <p className="text-sm text-muted-foreground mt-2">detik tersisa</p>
   637→            </div>
   638→            <button
   639→              onClick={handleStayLoggedIn}
   640→              className="w-full bg-amber-500 hover:bg-amber-600 text-white font-semibold py-2.5 rounded-xl transition-colors"
   641→              autoFocus
   642→            >
   643→              Tetap Login
   644→            </button>
   645→          </div>
   646→        </div>
   647→      )}
   648→    </div>
   649→  )
