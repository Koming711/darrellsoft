    35→// --- Types ---

    37→  id: string;

    47→// --- Helper ---
    48→function getTodayStr(): string {
    49→  const d = new Date();
    50→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
    51→}
    52→
    53→// --- Parse dataJson for document info ---
    54→function parseDocInfo(entry: HistoryEntry) {
    55→  try {
    56→    const parsed = JSON.parse(entry.dataJson);
    57→    const items = parsed.items || [];
    58→    const firstItem = items[0];
    59→    const namaBarang = firstItem?.deskripsi || '';
    60→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0);
    61→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0);
    62→    const ppn = parsed.ppn || 0;
    63→    const dpPercent = parsed.dp || 0;
    64→    const totalHarga = subtotal + (subtotal * ppn / 100);
    65→    // Always derive dpAmount from originalTotal
    66→    const originalTotal = parsed.originalTotal !== undefined ? parsed.originalTotal : totalHarga;
    67→    const dpAmount = originalTotal * (dpPercent / 100);
    68→    const sisa = totalHarga - dpAmount;
    69→    const lunas = parsed.lunas === true;
    70→    const tanggalJatuhTempo = parsed.tanggalJatuhTempo || '';
    71→    const tanggalPelunasan = parsed.tanggalPelunasan || '';
    72→    const referensiInvoiceNomor = parsed.referensiInvoiceNomor || '';
    73→    return { namaBarang, totalQty, totalHarga, dpPercent, dp: dpAmount, sisa, lunas, tanggalJatuhTempo, tanggalPelunasan, referensiInvoiceNomor, originalTotal };
    74→  } catch {
    75→    return { namaBarang: '', totalQty: 0, totalHarga: 0, dpPercent: 0, dp: 0, sisa: 0, lunas: false, tanggalJatuhTempo: '', tanggalPelunasan: '', referensiInvoiceNomor: '', originalTotal: 0 };
    76→  }
    77→}
    78→
    79→// --- Parse dataJson for InvoiceData ---
    80→function parseInvoiceData(entry: HistoryEntry): InvoiceData {
    81→  try {
    82→    const parsed = JSON.parse(entry.dataJson);
    83→    const company: CompanyInfo = {
    84→      nama: parsed.company?.nama || DEFAULT_COMPANY.nama,
    85→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
    86→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
    87→      email: parsed.company?.email || DEFAULT_COMPANY.email,
    88→      npwp: parsed.company?.npwp || '',
    89→      website: parsed.company?.website || '',
    90→      ppn: parsed.company?.ppn ?? parsed.ppn ?? 11,
    91→      logo: parsed.company?.logo || '',
    92→      bankName: parsed.company?.bankName || '',
    93→      bankAccount: parsed.company?.bankAccount || '',
    94→      bankHolder: parsed.company?.bankHolder || '',
    95→      bankName2: parsed.company?.bankName2 || '',
    96→      bankAccount2: parsed.company?.bankAccount2 || '',
    97→      bankHolder2: parsed.company?.bankHolder2 || '',
    98→    };
    99→    const client = parsed.client || { nama: '', kontak: '', alamat: '' };
   100→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number }, i: number) => ({
   101→      id: it.id || `item-${i}`,
   102→      deskripsi: it.deskripsi || '',
   103→      qty: it.qty || 0,
   104→      satuan: it.satuan || '',
   105→      harga: it.harga || 0,
   106→    }));
   107→    return {
   108→      type: 'invoice-pelunasan',
   109→      company,
   110→      nomor: parsed.nomor || entry.nomor || '',
   111→      tanggal: parsed.tanggal || entry.tanggal || '',
   112→      referensi: parsed.referensi || '',
   113→      client,
   114→      items,
   115→      ppn: parsed.ppn ?? 11,
   116→      dp: parsed.dp || 0,
   117→      dpAmount: parsed.dpAmount,
   118→      catatan: parsed.catatan || '',
   119→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   120→      caraPembayaran: parsed.caraPembayaran || '',
   121→      tanggalGiro: parsed.tanggalGiro || '',
   122→      lunas: parsed.lunas === true,
   123→      tanggalPelunasan: parsed.tanggalPelunasan || '',
   124→      referensiInvoiceId: parsed.referensiInvoiceId || '',
   125→      referensiInvoiceNomor: parsed.referensiInvoiceNomor || '',
   126→      originalTotal: parsed.originalTotal,
   127→    };
   128→  } catch {
   129→    return {
   130→      type: 'invoice-pelunasan',
   131→      company: { ...DEFAULT_COMPANY },
   132→      nomor: entry.nomor || '',
   133→      tanggal: entry.tanggal || '',
   134→      referensi: '',
   135→      client: { nama: '', kontak: '', alamat: '' },
   136→      items: [],
   137→      ppn: 11,
   138→      dp: 0,
   139→      catatan: '',
   140→      tanggalJatuhTempo: '',
   141→      caraPembayaran: '',
   142→      tanggalGiro: '',
   143→      lunas: false,
   144→      tanggalPelunasan: '',
   145→      referensiInvoiceId: '',
   146→      referensiInvoiceNomor: '',
   147→    };
   148→  }
   149→}

   151→export function InvoicePelunasanEditor() {
   152→  // Local invoice state (not using global store to avoid conflicts with regular editor)
   153→  const [invoiceData, setInvoiceData] = useState<InvoiceData | null>(null);
   154→  const [selectedEntry, setSelectedEntry] = useState<HistoryEntry | null>(null);
   155→  const [invoiceHistory, setInvoiceHistory] = useState<HistoryEntry[]>([]);
   156→  const [loading, setLoading] = useState(true);

   158→  const [dropdownOpen, setDropdownOpen] = useState(false);
   159→  const [pelunasanSaving, setPelunasanSaving] = useState(false);
   160→  const [jpgGenerating, setJpgGenerating] = useState(false);

   162→  // Pelunasan fields

   167→  const [tanggalGiro, setTanggalGiro] = useState('');
   168→

   170→  const [originalDpAmount, setOriginalDpAmount] = useState(0);

   184→      console.error('Failed to fetch pelunasan invoice history:', err);
   185→    } finally {

   187→    }

   189→
   190→  useEffect(() => {
   191→    fetchHistory();
   192→  }, [fetchHistory]);
   193→

   201→  const pendingInvoices = useMemo(() => {
   202→    return invoiceHistory.filter(entry => {
   203→      const info = parseDocInfo(entry);
   204→      return !info.lunas;
   205→    });
   206→  }, [invoiceHistory]);
   207→
   208→  // Filter by search
   209→  const filteredPending = useMemo(() => {
   210→    if (!searchQuery.trim()) return pendingInvoices;
   211→    const q = searchQuery.toLowerCase().trim();
   212→    return pendingInvoices.filter(entry => {
   213→      const info = parseDocInfo(entry);
   214→      return entry.nomor?.toLowerCase().includes(q) || entry.pihakKedua?.toLowerCase().includes(q) || info.namaBarang?.toLowerCase().includes(q) || info.referensiInvoiceNomor?.toLowerCase().includes(q);
   215→    });
   216→  }, [pendingInvoices, searchQuery]);
   217→
   218→  // Select an invoice
   219→  const selectInvoice = (entry: HistoryEntry) => {
   220→    const parsed = parseInvoiceData(entry);
   221→    const info = parseDocInfo(entry);
   222→    setInvoiceData(parsed);
   223→    setSelectedEntry(entry);
   224→    setLunasToggle(info.lunas);
   225→    setTanggalPelunasan(info.tanggalPelunasan || getTodayStr());
   226→    setTanggalJatuhTempo(info.tanggalJatuhTempo || '');
   227→    setCaraPembayaran(parsed.caraPembayaran || '');
   228→    setTanggalGiro(parsed.tanggalGiro || '');
   229→    // Fix: store original DP amount so it doesn't change when items are added
   230→    setOriginalDpAmount(info.dp);
   231→    setOriginalTotal(info.originalTotal);
   232→    setDropdownOpen(false);
   233→  };
   234→
   235→  // Clear selection
   236→  const clearSelection = () => {
   237→    setInvoiceData(null);
   238→    setSelectedEntry(null);
   239→    setLunasToggle(false);
   240→    setTanggalPelunasan('');
   241→    setTanggalJatuhTempo('');
   242→    setCaraPembayaran('');
   243→    setTanggalGiro('');
   244→    setOriginalDpAmount(0);
   245→    setOriginalTotal(0);
   246→  };
   247→
   248→  // Update invoice data locally
   249→  const updateInvoice = (updates: Partial<InvoiceData>) => {
   250→    setInvoiceData(prev => prev ? { ...prev, ...updates } : null);
   251→  };
   252→
   253→  const updateCompany = (company: CompanyInfo) => {
   254→    setInvoiceData(prev => prev ? { ...prev, company } : null);
   255→  };
   256→
   257→  const updateClient = (field: string, value: string) => {
   258→    setInvoiceData(prev => prev ? { ...prev, client: { ...prev.client, [field]: value } } : null);
   259→  };
   260→
   261→  // Build preview data with pelunasan overrides
   262→  const previewData: InvoiceData | null = useMemo(() => {
   263→    if (!invoiceData) {
   264→      // Default preview when no invoice is selected
   265→      return {
   266→        type: 'invoice-pelunasan',
   267→        company: { ...DEFAULT_COMPANY },
   268→        nomor: '-',
   269→        tanggal: getTodayStr(),
   270→        referensi: '',
   271→        client: { nama: '', kontak: '', alamat: '' },
   272→        items: [],
   273→        ppn: 11,
   274→        dp: 0,
   275→        catatan: '',
   276→        tanggalJatuhTempo: '',
   277→        caraPembayaran: '',
   278→        tanggalGiro: '',
   279→        lunas: false,
   280→        tanggalPelunasan: '',
   281→      };
   282→    }
   283→    return {
   284→      ...invoiceData,
   285→      tanggalJatuhTempo,
   286→      caraPembayaran: caraPembayaran as InvoiceData['caraPembayaran'],
   287→      tanggalGiro,
   288→      lunas: lunasToggle,
   289→      tanggalPelunasan: lunasToggle ? tanggalPelunasan : '',
   290→    };
   291→  }, [invoiceData, tanggalJatuhTempo, caraPembayaran, tanggalGiro, lunasToggle, tanggalPelunasan]);
   292→
   293→  // Calculate amounts — DP is FIXED at original amount, only subtotal and sisa change
   294→  const subtotal = invoiceData?.items.reduce((sum, item) => sum + item.qty * item.harga, 0) || 0;
   295→  const ppnAmount = subtotal * ((invoiceData?.ppn || 0) / 100);
   296→  const total = subtotal + ppnAmount;
   297→  const dpPercent = invoiceData?.dp || 0;
   298→  // DP amount stays fixed at original value, doesn't recalculate when items change
   299→  const dpAmount = originalDpAmount;
   300→  const sisa = total - dpAmount;
   301→
   302→  // Generate JPG from preview and send to WhatsApp
   303→  const handleGenerateJpg = async () => {
   304→    const previewEl = document.querySelector('[data-document-preview]') as HTMLElement;
   305→    if (!previewEl) {
   306→      toast.error('Preview tidak ditemukan');
   307→      return;
   308→    }
   309→    setJpgGenerating(true);
   310→    try {
   311→      const dataUrl = await toJpeg(previewEl, {
   312→        quality: 0.95,
   313→        pixelRatio: 2,
   314→        backgroundColor: '#ffffff',
   315→      });
   316→
   317→      // Convert data URL to Blob for sharing
   318→      const res = await fetch(dataUrl);
   319→      const blob = await res.blob();
   320→      const fileName = `Invoice-PEL-${invoiceData?.nomor || 'draft'}.jpg`;
   321→      const file = new File([blob], fileName, { type: 'image/jpeg' });
   322→
   323→      // Try Web Share API (mobile) — can share file directly to WhatsApp
   324→      if (navigator.share && navigator.canShare?.({ files: [file] })) {
   325→        await navigator.share({
   326→          files: [file],
   327→          title: `Invoice ${invoiceData?.nomor || ''}`,
   328→          text: `Berikut invoice pelunasan ${invoiceData?.nomor || ''} dari ${invoiceData?.company?.nama || ''}`,
   329→        });
   330→        toast.success('JPG berhasil dikirim ke WhatsApp');
   331→      } else {
   332→        // Fallback (desktop): download JPG + open WhatsApp link
   333→        const link = document.createElement('a');
   334→        link.download = fileName;
   335→        link.href = dataUrl;
   336→        link.click();
   337→
   338→        const phone = invoiceData?.client?.kontak?.replace(/\D/g, '') || '';
   339→        const message = encodeURIComponent(
   340→          `Berikut invoice pelunasan ${invoiceData?.nomor || ''} dari ${invoiceData?.company?.nama || ''}\n\nJPG invoice sudah didownload, silakan lampirkan ke chat ini.`
   341→        );
   342→        const waUrl = phone
   343→          ? `https://wa.me/${phone}?text=${message}`
   344→          : `https://wa.me/?text=${message}`;
   345→        window.open(waUrl, '_blank');
   346→        toast.success('JPG didownload & WhatsApp terbuka');
   347→      }
   348→    } catch (err) {
   349→      console.error('Failed to generate JPG:', err);
   350→      toast.error('Gagal membuat JPG');
   351→    } finally {
   352→      setJpgGenerating(false);
   353→    }
   354→  };
   355→
   356→  // Save pelunasan — update the existing history entry
   357→  const handleSavePelunasan = async () => {
   358→    if (!selectedEntry || !invoiceData) return;
   359→    setPelunasanSaving(true);
   360→    try {
   361→      const parsed = JSON.parse(selectedEntry.dataJson);
   362→      // Update pelunasan fields
   363→      parsed.tanggalJatuhTempo = tanggalJatuhTempo;
   364→      parsed.caraPembayaran = caraPembayaran;
   365→      parsed.tanggalGiro = tanggalGiro;
   366→      if (lunasToggle) {
   367→        parsed.lunas = true;
   368→        parsed.tanggalPelunasan = tanggalPelunasan || getTodayStr();
   369→      } else {
   370→        parsed.lunas = false;
   371→        parsed.tanggalPelunasan = '';
   372→      }
   373→      // Also update any edited fields from the form
   374→      parsed.company = invoiceData.company;
   375→      parsed.client = invoiceData.client;
   376→      parsed.items = invoiceData.items;
   377→      parsed.ppn = invoiceData.ppn;
   378→      parsed.dp = invoiceData.dp;
   379→      // Save fixed DP amount so it doesn't change on restore
   380→      parsed.dpAmount = originalDpAmount;
   381→      parsed.originalTotal = originalTotal;
   382→      parsed.catatan = invoiceData.catatan;
   383→      parsed.nomor = invoiceData.nomor;
   384→      parsed.tanggal = invoiceData.tanggal;
   385→      parsed.referensi = invoiceData.referensi;
   386→      parsed.type = 'invoice-pelunasan';
   387→      delete parsed.statusPembayaran;
   388→
   389→      const newDataJson = JSON.stringify(parsed);
   390→      const res = await fetcher(`/api/history/${selectedEntry.id}`, {
   391→        method: 'PUT',
   392→        headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   393→        body: JSON.stringify({ dataJson: newDataJson }),
   394→      });
   395→      if (res.ok) {
   396→        toast.success(lunasToggle ? 'Pelunasan berhasil dicatat!' : 'Invoice berhasil diperbarui');
   397→        clearSelection();
   398→        fetchHistory();
   399→        notifyDataChange('invoice');
   400→      } else {
   401→        toast.error('Gagal menyimpan perubahan');
   402→      }
   403→    } catch {
   404→      toast.error('Gagal menyimpan perubahan');
   405→    } finally {
   406→      setPelunasanSaving(false);
   407→    }
   408→  };
   409→
   410→  return (

   412→      title="Invoice Pelunasan"

   416→          <div className="flex items-center gap-2 flex-wrap">
   417→            <Button
   418→              variant="outline"
   419→              size="sm"
   420→              onClick={clearSelection}
   421→              className="border-gray-200 text-gray-600 hover:bg-gray-50"
   422→            >
   423→              <X className="mr-1.5 h-3.5 w-3.5" />

   426→            <Button
   427→              size="sm"

   431→                document.title = ' ';

   433→                  window.print();
   434→                  document.title = origTitle;
   435→                }, 100);
   436→              }}
   437→              className="bg-emerald-600 hover:bg-emerald-700"
   438→            >
   439→              Cetak
   440→            </Button>
   441→            <Button
   442→              size="sm"
   443→              onClick={handleGenerateJpg}
   444→              disabled={jpgGenerating}
   445→              className="bg-violet-600 hover:bg-violet-700 text-white"
   446→            >
   447→              {jpgGenerating ? (
   448→                <><Loader2 className="mr-1.5 h-3.5 h-3.5 animate-spin" /> Membuat...</>
   449→              ) : (
   450→                <><ImageIcon className="mr-1.5 h-3.5 w-3.5" /> JPG</>
   451→              )}
   452→            </Button>
   453→            <Button
   454→              size="sm"
   455→              onClick={handleSavePelunasan}
   456→              disabled={pelunasanSaving}
   457→              className={cn(
   458→                'text-white',
   459→                lunasToggle
   460→                  ? 'bg-green-600 hover:bg-green-700'
   461→                  : 'bg-blue-600 hover:bg-blue-700'
   462→              )}
   463→            >
   464→              {pelunasanSaving ? (
   465→                <><Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" /> Menyimpan...</>
   466→              ) : (
   467→                <><CheckCircle2 className="mr-1.5 h-3.5 w-3.5" /> {lunasToggle ? 'Simpan Pelunasan' : 'Simpan Perubahan'}</>
   468→              )}
   469→            </Button>
   470→          </div>
   471→        ) : undefined
   472→      }
   473→    >
   474→      {/* Invoice Selector */}
   475→      <div className="rounded-lg border bg-card p-3 sm:p-4 shadow-sm">
   476→        <h3 className="mb-3 text-xs font-semibold uppercase tracking-wider text-muted-foreground flex items-center gap-1.5">
   477→          <Wallet className="w-3.5 h-3.5 text-amber-600" />
   478→          Pilih Invoice Pelunasan
   479→        </h3>
   480→        <Popover open={dropdownOpen} onOpenChange={setDropdownOpen}>
   481→          <PopoverAnchor asChild>
   482→            <div className="relative">
   483→              <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 w-3.5 h-3.5 text-slate-400" />
   484→              <input
   485→                type="text"
   486→                placeholder={selectedEntry ? `${selectedEntry.nomor} — ${selectedEntry.pihakKedua}` : "Cari no. invoice, customer..."}
   487→                value={selectedEntry ? `${selectedEntry.nomor} — ${selectedEntry.pihakKedua}` : searchQuery}

   492→              {selectedEntry && (
   493→                <button
   494→                  type="button"
   495→                  onClick={(e) => { e.preventDefault(); clearSelection(); }}
   496→                  className="absolute right-2 top-1/2 -translate-y-1/2 text-slate-400 hover:text-slate-600"
   497→                >
   498→                  <X className="w-4 h-4" />
   499→                </button>
   500→              )}
   501→            </div>

   504→            align="start"
   505→            className="p-0 w-[var(--radix-popover-trigger-width)] max-h-64 overflow-y-auto"
