    91→function getTodayStr(): string {
    92→  const d = new Date()
    93→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
    94→}

    96→function formatRupiahShort(n: number): string {
    97→  return new Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR', minimumFractionDigits: 0 }).format(n)
    98→}

   100→// --- Parse dataJson for document info ---
   101→function parseDocInfo(entry: HistoryEntry) {
   102→  try {
   103→    const parsed = JSON.parse(entry.dataJson)
   104→    const items = parsed.items || []
   105→    const itemCount = items.length
   106→    const firstItem = items[0]
   107→    const namaBarang = firstItem?.deskripsi || ''
   108→    const hargaSatuan = firstItem?.harga || 0
   109→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0)
   110→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0)
   111→    const ppn = parsed.ppn || 0
   112→    const dpPercent = parsed.dp || 0
   113→    const totalHarga = subtotal + (subtotal * ppn / 100)
   114→    // Always derive dpAmount from originalTotal
   115→    const originalTotal = parsed.originalTotal !== undefined ? parsed.originalTotal : totalHarga
   116→    const dpAmount = originalTotal * (dpPercent / 100)
   117→    const sisa = totalHarga - dpAmount
   118→    const lunas = parsed.lunas === true
   119→    const batal = parsed.batal === true
   120→    const tanggalJatuhTempo = parsed.tanggalJatuhTempo || ''
   121→    const tanggalPelunasan = parsed.tanggalPelunasan || ''
   122→    const referensi = parsed.referensi || ''
   123→    const referensiInvoiceNomor = parsed.referensiInvoiceNomor || ''
   124→    const isPelunasan = entry.docType === 'invoice-pelunasan'
   125→    const uangCapek = parsed.uangCapek || 0
   126→    return { namaBarang, hargaSatuan, totalQty, totalHarga, dpPercent, dp: dpAmount, sisa, lunas, batal, tanggalJatuhTempo, tanggalPelunasan, referensi, referensiInvoiceNomor, isPelunasan, originalTotal, uangCapek, itemCount }
   127→  } catch {
   128→    return { namaBarang: '', hargaSatuan: 0, totalQty: 0, totalHarga: 0, dpPercent: 0, dp: 0, sisa: 0, lunas: false, batal: false, tanggalJatuhTempo: '', tanggalPelunasan: '', referensi: '', referensiInvoiceNomor: '', isPelunasan: false, originalTotal: 0, uangCapek: 0, itemCount: 0 }
   129→  }
   130→}

   132→// --- Status invoice (batal > lunas > belum) ---
   133→function getInvoiceStatus(info: ReturnType<typeof parseDocInfo>): InvoiceStatus {
   134→  if (info.batal) return 'batal'
   135→  if (info.isPelunasan) return info.lunas ? 'lunas' : 'belum'
   136→  const hasDP = info.dpPercent > 0 || info.dp > 0
   137→  return (hasDP ? info.lunas : (info.lunas || info.sisa <= 0)) ? 'lunas' : 'belum'
   138→}

   140→// --- Parse dataJson for InvoiceData ---
   141→function parseInvoiceData(entry: HistoryEntry): InvoiceData {
   142→  try {
   143→    const parsed = JSON.parse(entry.dataJson)
   144→    const company: CompanyInfo = {
   145→      nama: parsed.company?.nama || DEFAULT_COMPANY.nama,
   146→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
   147→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
   148→      email: parsed.company?.email || DEFAULT_COMPANY.email,
   149→      npwp: parsed.company?.npwp || '',
   150→      website: parsed.company?.website || '',
   151→      ppn: parsed.company?.ppn ?? parsed.ppn ?? 11,
   152→      logo: parsed.company?.logo || '',
   153→      bankName: parsed.company?.bankName || '',
   154→      bankAccount: parsed.company?.bankAccount || '',
   155→      bankHolder: parsed.company?.bankHolder || '',
   156→      bankName2: parsed.company?.bankName2 || '',
   157→      bankAccount2: parsed.company?.bankAccount2 || '',
   158→      bankHolder2: parsed.company?.bankHolder2 || '',
   159→    }
   160→    const client = parsed.client || { nama: '', kontak: '', alamat: '' }
   161→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number; modal?: number }, i: number) => ({
   162→      id: it.id || `item-${i}`,
   163→      deskripsi: it.deskripsi || '',
   164→      qty: it.qty || 0,
   165→      satuan: it.satuan || '',
   166→      harga: it.harga || 0,
   167→      ...(it.modal !== undefined ? { modal: it.modal } : {}),
   168→    }))
   169→    const docType = (entry.docType === 'invoice-pelunasan' ? 'invoice-pelunasan' : 'invoice') as 'invoice' | 'invoice-pelunasan'
   170→    return {
   171→      type: docType,
   172→      company,
   173→      nomor: parsed.nomor || entry.nomor || '',
   174→      tanggal: parsed.tanggal || entry.tanggal || '',
   175→      referensi: parsed.referensi || '',
   176→      client,
   177→      items,
   178→      ppn: parsed.ppn ?? 11,
   179→      dp: parsed.dp || 0,
   180→      dpAmount: parsed.dpAmount,
   181→      catatan: parsed.catatan || '',
   182→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   183→      caraPembayaran: parsed.caraPembayaran || '',
   184→      tanggalGiro: parsed.tanggalGiro || '',
   185→      lunas: parsed.lunas === true,
   186→      tanggalPelunasan: parsed.tanggalPelunasan || '',
   187→      referensiInvoiceId: parsed.referensiInvoiceId || '',
   188→      referensiInvoiceNomor: parsed.referensiInvoiceNomor || '',
   189→      originalTotal: parsed.originalTotal,
   190→      uangCapek: parsed.uangCapek || 0,
   191→    }
   192→  } catch {
   193→    const docType = (entry.docType === 'invoice-pelunasan' ? 'invoice-pelunasan' : 'invoice') as 'invoice' | 'invoice-pelunasan'
   194→    return {
   195→      type: docType,
   196→      company: { ...DEFAULT_COMPANY },
   197→      nomor: entry.nomor || '',
   198→      tanggal: entry.tanggal || '',
   199→      referensi: '',
   200→      client: { nama: '', kontak: '', alamat: '' },
   201→      items: [],
   202→      ppn: 11,
   203→      dp: 0,
   204→      catatan: '',
   205→      tanggalJatuhTempo: '',
   206→      caraPembayaran: '',
   207→      tanggalGiro: '',
   208→      lunas: false,
   209→      tanggalPelunasan: '',
   210→      referensiInvoiceId: '',
   211→      referensiInvoiceNomor: '',
   212→      uangCapek: 0,
   213→    }
   214→  }
   215→}

   240→// ============================================================
   241→// DetailInvoiceView — "halaman pratinjau" setelah Simpan di Buat
   242→// Invoice & halaman detail saat baris riwayat diklik.
   243→// Tombol Cetak / JPG / Surat Jalan / Tandai Lunas / Hapus / Batal
   244→// ada DI ATAS pratinjau. Pratinjau A5 ber-outline, +20% (desktop),
   245→// full-width di mobile.
   246→// ============================================================
   247→function DetailInvoiceView({ id, onBack }: { id: string; onBack: () => void }) {
   248→  const router = useRouter()
   249→  const resetDocument = useDokuproStore((s) => s.resetDocument)
   250→  const [entry, setEntry] = useState<HistoryEntry | null>(null)
   251→  const [loading, setLoading] = useState(true)
   252→  const [error, setError] = useState('')
   253→  const [jpgGenerating, setJpgGenerating] = useState(false)
   254→  const [updating, setUpdating] = useState(false)
   255→
   256→  // Dialog konfirmasi
   257→  const [lunasOpen, setLunasOpen] = useState(false)
   258→  const [batalOpen, setBatalOpen] = useState(false)
   259→  const [hapusOpen, setHapusOpen] = useState(false)
   260→
   261→  // Pratinjau scaler
   262→  const scalerRef = useRef<HTMLDivElement>(null)
   263→
   264→  const loadEntry = useCallback(async () => {
   265→    setLoading(true)
   266→    setError('')
   267→    try {
   268→      const res = await fetcher(`/api/history/${id}`, { headers: getAuthHeaders() })
   269→      if (res.ok) {
   270→        const json = await res.json()
   271→        setEntry(json.data || null)
   272→      } else {
   273→        setError('Invoice tidak ditemukan')
   274→      }
   275→    } catch {
   276→      setError('Gagal memuat invoice')
   277→    } finally {
   278→      setLoading(false)
   279→    }
   280→  }, [id])
   281→
   282→  useEffect(() => { loadEntry() }, [loadEntry])
   283→
   284→  const data = useMemo(() => (entry ? parseInvoiceData(entry) : null), [entry])
   285→  const info = useMemo(() => (entry ? parseDocInfo(entry) : null), [entry])
   286→  const status: InvoiceStatus = info ? getInvoiceStatus(info) : 'belum'
   287→
   288→  // Scale pratinjau A5 agar pas dengan container (desktop ~+20% dari ukuran
   289→  // asli 148mm, mobile full-width). offsetWidth/Height tidak terpengaruh transform.
   290→  useLayoutEffect(() => {
   291→    if (!data) return
   292→    const fit = () => {
   293→      const wrapper = scalerRef.current
   294→      if (!wrapper) return
   295→      const a5 = wrapper.querySelector('.a5-page') as HTMLElement | null
   296→      if (!a5) return
   297→      const naturalW = a5.offsetWidth
   298→      const naturalH = a5.offsetHeight
   299→      if (naturalW === 0 || naturalH === 0) {
   300→        requestAnimationFrame(fit)
   301→        return
   302→      }
   303→      const availW = wrapper.parentElement?.clientWidth || wrapper.clientWidth
   304→      if (availW === 0) {
   305→        requestAnimationFrame(fit)
   306→        return
   307→      }
   308→      const scale = availW / naturalW
   309→      a5.style.transform = `scale(${scale})`
   310→      a5.style.transformOrigin = 'top left'
   311→      wrapper.style.width = `${naturalW * scale}px`
   312→      wrapper.style.height = `${naturalH * scale}px`
   313→    }
   314→    fit()
   315→    const raf = requestAnimationFrame(fit)
   316→    const timer = setTimeout(fit, 250)
   317→    window.addEventListener('resize', fit)
   318→    return () => { cancelAnimationFrame(raf); clearTimeout(timer); window.removeEventListener('resize', fit) }
   319→  }, [data])
   320→
   321→  const handleJpg = async () => {
   322→    if (!data) return
   323→    setJpgGenerating(true)
   324→    try {
   325→      const previewEl = document.querySelector('[data-document-preview]') as HTMLElement
   326→      if (!previewEl) { toast.error('Pratinjau tidak ditemukan'); return }
   327→      const blob = await captureElementAsJpg(previewEl)
   328→      const fileName = `${(data.nomor || 'draft').replace(/\//g, '-')}.jpg`
   329→      const phone = data.client?.kontak || ''
   330→      const result = await shareJpgToWhatsApp({
   331→        blob,
   332→        fileName,
   333→        documentLabel: `Invoice ${data.nomor}`,
   334→        phone,
   335→      })
   336→      if (result.status === 'shared') toast.success('Gambar dibagikan ke WhatsApp')
   337→      else if (result.status === 'cancelled') { /* silent */ }
   338→      else if (result.status === 'downloaded') toast.success(`${fileName} tersimpan ke perangkat`, { description: 'File JPG telah diunduh ke folder Downloads.' })
   339→      else toast.error(result.error || 'Gagal memproses JPG')
   340→    } catch (err) {
   341→      console.error(err)
   342→      toast.error('Gagal membuat JPG')
   343→    } finally {
   344→      setJpgGenerating(false)
   345→    }
   346→  }
   347→
   348→  // Tandai Lunas — invoice dinyatakan lunas penuh hari ini (+ sinkron PEL)
   349→  const handleTandaiLunas = async () => {
   350→    if (!entry || !data) return
   351→    setUpdating(true)
   352→    try {
   353→      const parsed = JSON.parse(entry.dataJson)
   354→      parsed.lunas = true
   355→      parsed.tanggalPelunasan = getTodayStr()
   356→      delete parsed.statusPembayaran
   357→      if ((parsed.dp || 0) > 0) {
   358→        const sub = (parsed.items || []).reduce((s: number, it: { qty: number; harga: number }) => s + it.qty * it.harga, 0)
   359→        const totalHarga = sub + (sub * (parsed.ppn || 0) / 100)
   360→        parsed.originalTotal = parsed.originalTotal !== undefined ? parsed.originalTotal : totalHarga
   361→        parsed.dpAmount = parsed.originalTotal * (parsed.dp / 100)
   362→      }
   363→      const res = await fetcher(`/api/history/${entry.id}`, {
   364→        method: 'PUT',
   365→        headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   366→        body: JSON.stringify({ dataJson: JSON.stringify(parsed) }),
   367→      })
   368→      if (res.ok) {
   369→        await syncLinkedPelunasan(entry.id, parsed.nomor || entry.nomor, parsed)
   370→        toast.success('Invoice ditandai LUNAS')
   371→        setLunasOpen(false)
   372→        window.dispatchEvent(new CustomEvent('dokupro:history-updated'))
   373→        notifyDataChange('invoice')
   374→        loadEntry()
   375→      } else {
   376→        toast.error('Gagal menandai lunas')
   377→      }
   378→    } catch {
   379→      toast.error('Gagal menandai lunas')
   380→    } finally {
   381→      setUpdating(false)
   382→    }
   383→  }
   384→
   385→  // Batal — invoice dibatalkan (status Batal di riwayat & detail)
   386→  const handleBatal = async () => {
   387→    if (!entry) return
   388→    setUpdating(true)
   389→    try {
   390→      const parsed = JSON.parse(entry.dataJson)
   391→      parsed.batal = true
   392→      const res = await fetcher(`/api/history/${entry.id}`, {
   393→        method: 'PUT',
   394→        headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   395→        body: JSON.stringify({ dataJson: JSON.stringify(parsed) }),
   396→      })
   397→      if (res.ok) {
   398→        toast.success('Invoice dibatalkan')
   399→        setBatalOpen(false)
   400→        window.dispatchEvent(new CustomEvent('dokupro:history-updated'))
   401→        notifyDataChange('invoice')
   402→        loadEntry()
   403→      } else {
   404→        toast.error('Gagal membatalkan invoice')
   405→      }
   406→    } catch {
   407→      toast.error('Gagal membatalkan invoice')
   408→    } finally {
   409→      setUpdating(false)
   410→    }
   411→  }
   412→
   413→  // Hapus — soft delete → masuk tab Sampah (bisa dipulihkan)
   414→  const handleHapus = async () => {
   415→    if (!entry) return
   416→    setUpdating(true)
   417→    try {
   418→      const res = await fetcher(`/api/history/${entry.id}`, { method: 'DELETE', headers: getAuthHeaders() })
   419→      if (res.ok) {
   420→        toast.success('Invoice dipindahkan ke Sampah')
   421→        setHapusOpen(false)
   422→        window.dispatchEvent(new CustomEvent('dokupro:history-updated'))
   423→        notifyDataChange('invoice')
   424→        resetDocument('invoice')
   425→        onBack()
   426→      } else {
   427→        toast.error('Gagal menghapus invoice')
   428→      }
   429→    } catch {
   430→      toast.error('Gagal menghapus invoice')
   431→    } finally {
   432→      setUpdating(false)
   433→    }
   434→  }
   435→
   436→  const actionButtons = (
   437→    <div className="flex flex-wrap gap-2 print:hidden">
   438→      <Button size="sm" onClick={() => { toast.dismiss(); const origTitle = document.title; document.title = ' '; setTimeout(() => { window.print(); document.title = origTitle }, 100) }} className="bg-emerald-600 hover:bg-emerald-700 min-h-[36px]">
   439→        <Printer className="mr-1.5 h-3.5 w-3.5" /> Cetak
   440→      </Button>
   441→      <Button size="sm" onClick={handleJpg} disabled={jpgGenerating} className="bg-green-600 hover:bg-green-700 min-h-[36px]">
   442→        {jpgGenerating ? <><Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" /> JPG...</> : <><ImageIcon className="mr-1.5 h-3.5 w-3.5" /> JPG</>}
   443→      </Button>
   444→      <Button size="sm" onClick={() => router.push(`/surat-jalan?invoiceId=${id}`)} className="bg-orange-600 hover:bg-orange-700 min-h-[36px]">
   445→        <Truck className="mr-1.5 h-3.5 w-3.5" /> Surat Jalan
   446→      </Button>
   447→      {status !== 'lunas' && status !== 'batal' && (
   448→        <Button size="sm" onClick={() => setLunasOpen(true)} className="bg-violet-600 hover:bg-violet-700 min-h-[36px]">
   449→          <CheckCircle2 className="mr-1.5 h-3.5 w-3.5" /> Tandai Lunas
   450→        </Button>
   451→      )}
   452→      {status !== 'batal' && (
   453→        <Button size="sm" variant="outline" onClick={() => setBatalOpen(true)} className="border-amber-300 text-amber-700 hover:bg-amber-50 min-h-[36px]">
   454→          <Ban className="mr-1.5 h-3.5 w-3.5" /> Batal
   455→        </Button>
   456→      )}
   457→      <Button size="sm" variant="outline" onClick={() => setHapusOpen(true)} className="border-red-200 text-destructive hover:bg-red-50 hover:text-destructive min-h-[36px]">
   458→        <Trash2 className="mr-1.5 h-3.5 w-3.5" /> Hapus
   459→      </Button>
   460→    </div>
   461→  )
   462→
   463→  return (
   464→    <div>
   465→      {/* Header — Kembali + judul "Detail Invoice" + status */}
   466→      <div className="flex items-center gap-2 mb-3 print:hidden">
   467→        <Button onClick={onBack} variant="outline" size="sm" className="h-9 gap-1.5 text-xs">
   468→          <ArrowLeft className="w-3.5 h-3.5" /> Kembali
   469→        </Button>
   470→        <h2 className="text-xl md:text-2xl font-bold tracking-tight text-foreground truncate">Detail Invoice</h2>
   471→        <StatusBadge status={status} />
   472→      </div>
   473→
   474→      {loading ? (
   475→        <div className="space-y-3">
   476→          <Skeleton className="h-10 w-full" />
