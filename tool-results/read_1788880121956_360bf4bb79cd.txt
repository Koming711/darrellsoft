   605→  const [dateFrom, setDateFrom] = useState('')
   606→  const [dateTo, setDateTo] = useState('')
   607→  const [purgeTarget, setPurgeTarget] = useState<HistoryEntry | null>(null)
   608→  const [backupLoading, setBackupLoading] = useState<string | null>(null)
   609→
   610→  const fetchHistory = useCallback(async () => {
   611→    try {
   612→      setLoading(true)
   613→      const headers = getAuthHeaders()
   614→      // cache: 'no-store' MANDATORY — hindari stale cache setelah PUT/DELETE.
   615→      const [invRes, pelRes] = await Promise.all([
   616→        fetch('/api/history?docType=invoice', { headers, cache: 'no-store' }),

   622→      setPelunasanHistory(pelData)

   626→      setLoading(false)
   627→    }

   635→      setSampahList(json.data || [])

   638→    } finally {
   639→      setSampahLoading(false)
   640→    }

   644→
   645→  useEffect(() => {
   646→    if (activeTab === 'sampah') fetchSampah()

   650→    const handler = () => { fetchHistory(); if (activeTab === 'sampah') fetchSampah() }

   652→    return () => window.removeEventListener('dokupro:history-updated', handler)
   653→  }, [fetchHistory, fetchSampah, activeTab])
   654→
   655→  const allHistory = useMemo(() => [...invoiceHistory, ...pelunasanHistory], [invoiceHistory, pelunasanHistory])
   656→
   657→  // Filter pencarian + status + rentang tanggal
   658→  const matchesFilters = useCallback((entry: HistoryEntry) => {
   659→    const info = parseDocInfo(entry)
   660→    const st = getInvoiceStatus(info)

   695→        toast.error('Gagal memulihkan data')
   696→      }
   697→    } catch {
   698→      toast.error('Gagal memulihkan data')
   699→    } finally {
   700→      setSampahActionId(null)
   701→    }

   710→        toast.success('Data dihapus permanen')
   711→        setPurgeTarget(null)
   712→        window.dispatchEvent(new CustomEvent('dokupro:history-updated'))
   713→        fetchSampah()
   714→      } else {
   715→        toast.error('Gagal menghapus permanen')
   716→      }
   717→    } catch {

   719→    } finally {
   720→      setSampahActionId(null)
   721→    }

   732→        toast.error(errMsg)
   733→        return

   738→      const a = document.createElement('a')
   739→      a.href = url
   740→      const disposition = res.headers.get('Content-Disposition')
   741→      const match = disposition?.match(/filename\*?=(?:UTF-8'')?"?([^";]+)"?/i)
   742→      a.download = match ? match[1] : `backup-invoice-history-${Date.now()}.xlsx`
   743→      document.body.appendChild(a); a.click(); document.body.removeChild(a)
   744→      URL.revokeObjectURL(url)
   745→      toast.success('Backup berhasil diunduh')
   746→    } catch (e) { console.error('Backup error:', e); toast.error('Gagal backup data') }
   747→    setBackupLoading(null)

   778→
   779→  const hasResults = dpInvoices.length > 0 || pelunasanInvoices.length > 0

   781→
   782→  return (
   783→    <>
   784→      <div className="space-y-5">

   788→            <h1 className="text-xl md:text-2xl font-bold tracking-tight">Riwayat Invoice</h1>
   789→            <p className="text-sm text-muted-foreground mt-1">{countLabel}</p>
   790→          </div>
   791→          <div className="flex flex-col sm:flex-row gap-2">
   792→            <div className="relative flex-1 sm:w-64">
   793→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-stone-400 pointer-events-none" />

   795→                value={searchQuery}
   796→                onChange={(e) => setSearchQuery(e.target.value)}
   797→                placeholder="Cari no. invoice / customer / barang…"
   798→                aria-label="Cari invoice"
   799→                className="pl-9 min-h-[44px]"
   800→              />
   801→            </div>
   802→            <div className="flex gap-2">
   803→              <Button
   804→                onClick={onCreate}
   805→                title="Buat invoice baru"
   806→                className="bg-emerald-600 hover:bg-emerald-700 min-h-[44px] flex-1 sm:flex-none"
   807→              >
   808→                <Plus className="h-4 w-4" /> Buat Invoice
   809→              </Button>
   810→              <Button onClick={handleBackup} variant="outline" disabled={backupLoading === 'backup'} title="Backup riwayat invoice" className="min-h-[44px] flex-1 sm:flex-none">
   811→                {backupLoading === 'backup' ? <Loader2 className="h-4 w-4 animate-spin" /> : <DatabaseBackup className="h-4 w-4" />} Backup
   812→              </Button>

   816→            </div>
   817→          </div>
   818→        </div>
   819→
   820→        {/* === TABS: Riwayat Invoice | Riwayat Pembayaran | Sampah === */}
   821→        <div className="flex items-center gap-2 overflow-x-auto">
   822→          {([
   823→            { key: 'invoice' as const, label: 'Riwayat Invoice', icon: <FileText className="w-3.5 h-3.5" /> },
   824→            { key: 'pembayaran' as const, label: 'Riwayat Pembayaran', icon: <Wallet className="w-3.5 h-3.5" /> },
   825→            { key: 'sampah' as const, label: 'Sampah', icon: <Trash2 className="w-3.5 h-3.5" /> },
   826→          ]).map(t => (
   827→            <button
   828→              key={t.key}
   829→              onClick={() => setActiveTab(t.key)}
   830→              className={cn(
   831→                'px-4 py-1.5 text-sm font-semibold rounded-lg border transition-colors whitespace-nowrap flex-shrink-0',
   832→                activeTab === t.key
   833→                  ? 'bg-violet-600 text-white border-violet-600 shadow-sm'
   834→                  : 'bg-card text-slate-600 border-slate-200 hover:bg-slate-50 hover:border-slate-300'
   835→              )}
   836→            >
   837→              <span className="inline-flex items-center gap-1.5">{t.icon}{t.label}</span>
   838→            </button>
   839→          ))}
   840→        </div>
   841→
   842→        {/* ================= TAB: RIWAYAT INVOICE ================= */}
   843→        {activeTab === 'invoice' && (
   844→          <>
   845→            {/* Filter: status + rentang tanggal */}
   846→            <div className="rounded-xl border border-stone-200 bg-white p-3 md:px-4 md:py-3">
   847→              <div className="flex flex-col gap-2.5 lg:flex-row lg:items-center lg:gap-4">
   848→                <div className="lg:w-44">
   849→                  <Label className="text-[11px] uppercase tracking-wide text-muted-foreground">Status</Label>
   850→                  <Select value={statusFilter} onValueChange={(v) => setStatusFilter((v || 'all') as 'all' | InvoiceStatus)}>
   851→                    <SelectTrigger className="w-full min-h-[40px] bg-white mt-1" aria-label="Filter status">
   852→                      <SelectValue placeholder="Semua Status" />
   853→                    </SelectTrigger>
   854→                    <SelectContent>
   855→                      <SelectItem value="all">Semua Status</SelectItem>
   856→                      <SelectItem value="lunas">Lunas</SelectItem>
   857→                      <SelectItem value="belum">Belum Lunas</SelectItem>
   858→                      <SelectItem value="batal">Batal</SelectItem>
   859→                    </SelectContent>
   860→                  </Select>
   861→                </div>
   862→                <div className="lg:w-40">
   863→                  <Label className="text-[11px] uppercase tracking-wide text-muted-foreground">Dari Tanggal</Label>
   864→                  <Input type="date" value={dateFrom} onChange={(e) => setDateFrom(e.target.value)} className="mt-1 min-h-[40px] bg-white" aria-label="Dari tanggal" />
   865→                </div>
   866→                <div className="lg:w-40">
   867→                  <Label className="text-[11px] uppercase tracking-wide text-muted-foreground">Sampai Tanggal</Label>
   868→                  <Input type="date" value={dateTo} onChange={(e) => setDateTo(e.target.value)} className="mt-1 min-h-[40px] bg-white" aria-label="Sampai tanggal" />
   869→                </div>
   870→                {filtersActive && (
   871→                  <Button variant="ghost" size="sm" className="lg:ml-auto mt-1 lg:mt-4 text-xs text-muted-foreground" onClick={() => { setStatusFilter('all'); setDateFrom(''); setDateTo(''); setSearchQuery('') }}>
   872→                    <X className="h-3.5 w-3.5" /> Reset Filter
   873→                  </Button>
   874→                )}
   875→              </div>
   876→            </div>
   877→
   878→            {loading ? (
   879→              <>
   880→                <div className="hidden md:block rounded-xl border border-stone-200 bg-white overflow-hidden p-4 space-y-3">
   881→                  {[1, 2, 3, 4].map((i) => <Skeleton key={i} className="h-10 w-full" />)}
   882→                </div>
   883→                <div className="md:hidden space-y-3">
   884→                  {[1, 2, 3].map((i) => <Skeleton key={i} className="h-40 w-full rounded-xl" />)}
   885→                </div>

   888→              <EmptyState filtered={filtersActive} />

   989→                    {/* Desktop table — klik baris → Detail Invoice */}
   990→                    <div className="hidden md:block rounded-xl border border-stone-200 bg-white overflow-hidden">
