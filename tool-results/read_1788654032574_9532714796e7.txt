    69→    69→export function InvoiceEditor({ dpDisabled = false }: { dpDisabled?: boolean }) {
    70→    70→  const invoice = useDokuproStore((s) => s.invoice);
    71→    71→  const setInvoice = useDokuproStore((s) => s.setInvoice);
    72→    72→  const resetDocument = useDokuproStore((s) => s.resetDocument);
    73→    73→  const loadCompanyFromAPI = useDokuproStore((s) => s.loadCompanyFromAPI);
    74→    74→  const invoiceEditingId = useDokuproStore((s) => s.invoiceEditingId);
    75→    75→  const setInvoiceEditingId = useDokuproStore((s) => s.setInvoiceEditingId);
    76→    76→  const router = useRouter();
    77→    77→  const searchParams = useSearchParams();
    78→    78→  const riwayatIdFromUrl = searchParams.get('riwayatId');
    79→    79→  const autoSelectDoneRef = useRef<string | null>(null); // track which riwayatId was auto-selected
    80→    80→  const [savingSj, setSavingSj] = useState(false);
    81→    81→
    82→    82→  // Riwayat cetakan dropdown state
    83→    83→  const [riwayatList, setRiwayatList] = useState<RiwayatCetakanItem[]>([]);
    84→    84→  const [referensiInput, setReferensiInput] = useState(invoice.referensi);
    85→    85→  const [dropdownOpen, setDropdownOpen] = useState(false);
    86→    86→
    87→    87→  // Customer dropdown state
    88→    88→  const [customerList, setCustomerList] = useState<CustomerItem[]>([]);
    89→    89→  const [clientInput, setClientInput] = useState(invoice.client.nama);
    90→    90→  const [clientDropdownOpen, setClientDropdownOpen] = useState(false);
    91→    91→  const [clientTyping, setClientTyping] = useState(false);
    92→    92→
    93→    93→  const fetchRiwayatCetakan = useCallback(async () => {
    94→    94→    try {
    95→    95→      const res = await fetch('/api/riwayat-cetakan', { headers: getAuthHeaders() });
    96→    96→      if (res.ok) {
    97→    97→        const data = await res.json();
    98→    98→        setRiwayatList(data);
    99→    99→        return data;
   100→   100→      }
   101→   101→    } catch {
   102→   102→      // silently fail
   103→   103→    }
   104→   104→    return [];
   105→   105→  }, []);
   106→   106→
   107→   107→  // Fetch customer list
   108→   108→  const fetchCustomers = useCallback(async () => {
   109→   109→    try {
   110→   110→      const res = await fetch('/api/customers', { headers: getAuthHeaders() });
   111→   111→      if (res.ok) {
   112→   112→        const data = await res.json();
   113→   113→        setCustomerList(Array.isArray(data) ? data : []);
   114→   114→      }
   115→   115→    } catch {
   116→   116→      // silently fail
   117→   117→    }
   118→   118→  }, []);
   119→   119→
   120→   120→  useEffect(() => { loadCompanyFromAPI() }, [loadCompanyFromAPI]);
   121→   121→  useEffect(() => { fetchRiwayatCetakan() }, [fetchRiwayatCetakan]);
   122→   122→  useEffect(() => { fetchCustomers() }, [fetchCustomers]);
   123→   123→
   124→   124→  // Fetch next Invoice number from server (only when creating new, not editing)
   125→   125→  const fetchNextNumber = useCallback(() => {
   126→   126→    // Read current editingId from store to avoid stale closure
   127→   127→    const currentEditingId = useDokuproStore.getState().invoiceEditingId;
   128→   128→    if (currentEditingId) return; // Don't overwrite nomor when editing
   129→   129→    fetch('/api/history?preview=next-number&docType=invoice', { headers: getAuthHeaders() })
   130→   130→      .then(res => res.ok ? res.json() : null)
   131→   131→      .then(data => { if (data?.nextNumber) setInvoice((prev) => ({ ...prev, nomor: data.nextNumber })) })
   132→   132→      .catch(() => {});
   133→   133→  }, [setInvoice]);
   134→   134→
   135→   135→  useEffect(() => { fetchNextNumber() }, [fetchNextNumber]);
   136→   136→
   137→   137→  // Re-fetch next number after a document is saved
   138→   138→  useEffect(() => {
   139→   139→    const handler = () => { fetchNextNumber() };
   140→   140→    window.addEventListener('dokupro:history-updated', handler);
   141→   141→    return () => { window.removeEventListener('dokupro:history-updated', handler) };
   142→   142→  }, [fetchNextNumber]);
   143→   143→
   144→   144→  // Mode Regular (dpDisabled): paksa DP = 0 dan sembunyikan input DP
   145→   145→  useEffect(() => {
   146→   146→    if (dpDisabled) {
   147→   147→      setInvoice((prev) => (prev.dp === 0 ? prev : ({ ...prev, dp: 0 })));
   148→   148→    }
   149→   149→  }, [dpDisabled, setInvoice]);
   150→   150→
   151→   151→  // Sync referensiInput when invoice.referensi changes externally
   152→   152→  useEffect(() => { setReferensiInput(invoice.referensi) }, [invoice.referensi]);
   153→   153→  // Sync clientInput when invoice.client.nama changes externally
   154→   154→  useEffect(() => { setClientInput(invoice.client.nama) }, [invoice.client.nama]);
   155→   155→
   156→   156→  // Filter riwayat list — when dropdown is open show all, otherwise not used
   157→   157→  const filteredRiwayatList = riwayatList;
   158→   158→
   159→   159→  const handleReferensiInputChange = (value: string) => {
   160→   160→    setReferensiInput(value);
   161→   161→    setInvoice((prev) => ({ ...prev, referensi: value }));
   162→   162→    setDropdownOpen(true);
   163→   163→  };
   164→   164→
   165→   165→  // Apply referensi data using functional setInvoice to avoid stale closure
   166→   166→  const applyReferensi = (item: RiwayatCetakanItem) => {
   167→   167→    const ref = item.nomorUrut || item.printName || '';
   168→   168→    setReferensiInput(ref);
   169→   169→
   170→   170→    // Build single item with multi-line description
   171→   171→    const descLines: string[] = [];
   172→   172→    if (item.printName) descLines.push(item.printName);
   173→   173→
   174→   174→    const bahanParts: string[] = [];
   175→   175→    if (item.paperName) bahanParts.push(item.paperName);
   176→   176→    if (item.paperGrammage) bahanParts.push(item.paperGrammage + 'g');
   177→   177→    if (bahanParts.length > 0) descLines.push(bahanParts.join(' '));
   178→   178→
   179→   179→    if (item.finishingNames) {
   180→   180→      const finishings = item.finishingNames.split(',').map(f => f.trim()).filter(Boolean);
   181→   181→      descLines.push('Finishing:');
   182→   182→      finishings.forEach(f => descLines.push(f));
   183→   183→    }
   184→   184→
   185→   185→    const deskripsi = descLines.join('\n');
   186→   186→    const jumlahPesanan = parseInt(item.jumlahPesanan) || 1;
   187→   187→    const hargaPerPcs = jumlahPesanan > 0 ? Math.round(item.grandTotal / jumlahPesanan) : 0;
   188→   188→
   189→   189→    const newItems: InvoiceData['items'] = [{
   190→   190→      id: 'riwayat-0',
   191→   191→      deskripsi: deskripsi || ref,
   192→   192→      qty: jumlahPesanan,
   193→   193→      satuan: 'pcs',
   194→   194→      harga: hargaPerPcs,
   195→   195→    }];
   196→   196→
   197→   197→    // Use functional form to always get the LATEST state
   198→   198→    setInvoice((prev) => ({
   199→   199→      ...prev,
   200→   200→      referensi: ref,
   201→   201→      client: {
   202→   202→        ...prev.client,
   203→   203→        nama: item.customerName || prev.client.nama,
   204→   204→      },
   205→   205→      items: newItems,
   206→   206→      catatan: '',
   207→   207→      uangCapek: item.profitAmount || 0,
   208→   208→    }));
   209→   209→    setDropdownOpen(false);
   210→   210→  };
   211→   211→
   212→   212→  const handleReferensiSelect = (item: RiwayatCetakanItem) => {
   213→   213→    applyReferensi(item);
   214→   214→  };
   215→   215→
   216→   216→  // Auto-select referensi when coming from hitung cetakan with riwayatId
   217→   217→  // Fetches the specific riwayat directly by ID for reliability (no race condition with list loading)
   218→   218→  useEffect(() => {
   219→   219→    if (!riwayatIdFromUrl) return;
   220→   220→    if (autoSelectDoneRef.current === riwayatIdFromUrl) return;
   221→   221→
   222→   222→    let cancelled = false;
   223→   223→
   224→   224→    const fetchAndApply = async () => {
   225→   225→      // First, try to find it in the already-loaded list (fast path)
   226→   226→      const foundInList = riwayatList.find((r) => r.id === riwayatIdFromUrl);
   227→   227→      if (foundInList) {
   228→   228→        if (!cancelled) {
   229→   229→          autoSelectDoneRef.current = riwayatIdFromUrl;
   230→   230→          applyReferensi(foundInList);
   231→   231→        }
   232→   232→        return;
   233→   233→      }
   234→   234→
   235→   235→      // Not in list yet — fetch directly by ID from the API (reliable path)
   236→   236→      try {
   237→   237→        const res = await fetch(`/api/riwayat-cetakan?id=${riwayatIdFromUrl}`, { headers: getAuthHeaders() });
   238→   238→        if (res.ok && !cancelled) {
   239→   239→          const data = await res.json();
   240→   240→          if (data && !data.error) {
   241→   241→            autoSelectDoneRef.current = riwayatIdFromUrl;
   242→   242→            applyReferensi(data as RiwayatCetakanItem);
   243→   243→          }
   244→   244→        }
   245→   245→      } catch {
   246→   246→        // Fallback: retry once after a short delay
   247→   247→        if (cancelled) return;
   248→   248→        await new Promise(r => setTimeout(r, 800));
   249→   249→        if (cancelled) return;
   250→   250→        try {
   251→   251→          const res = await fetch(`/api/riwayat-cetakan?id=${riwayatIdFromUrl}`, { headers: getAuthHeaders() });
   252→   252→          if (res.ok) {
   253→   253→            const data = await res.json();
   254→   254→            if (data && !data.error) {
   255→   255→              autoSelectDoneRef.current = riwayatIdFromUrl;
   256→   256→              applyReferensi(data as RiwayatCetakanItem);
   257→   257→            }
   258→   258→          }
   259→   259→        } catch {
   260→   260→          // Give up silently
   261→   261→        }
   262→   262→      }
   263→   263→    };
   264→   264→
   265→   265→    fetchAndApply();
   266→   266→
   267→   267→    return () => { cancelled = true; };
   268→   268→  }, [riwayatIdFromUrl]);
   269→   269→
   270→   270→  // Filter customer list (only when user is actively typing)
   271→   271→  const filteredCustomerList = useMemo(() =>
   272→   272→    clientTyping
   273→   273→      ? customerList.filter((c) => {
   274→   274→          const search = clientInput.toLowerCase().trim();
   275→   275→          if (!search) return true;
   276→   276→          return c.name.toLowerCase().includes(search) ||
   277→   277→                 (c.companyName || '').toLowerCase().includes(search) ||
   278→   278→                 (c.phone || '').toLowerCase().includes(search) ||
   279→   279→                 (c.address || '').toLowerCase().includes(search);
   280→   280→        })
   281→   281→      : customerList
   282→   282→  , [clientTyping, customerList, clientInput]);
   283→   283→
   284→   284→  const handleClientInputChange = (value: string) => {
   285→   285→    setClientInput(value);
   286→   286→    setClientTyping(true);
   287→   287→    updateClient('nama', value);
   288→   288→    setClientDropdownOpen(true);
   289→   289→  };
   290→   290→
   291→   291→  const handleClientSelect = (item: CustomerItem) => {
   292→   292→    setClientInput(item.name);
   293→   293→    setClientTyping(false);
   294→   294→    setInvoice((prev) => ({
   295→   295→      ...prev,
   296→   296→      client: {
   297→   297→        nama: item.name,
   298→   298→        kontak: item.phone || item.email || '',
   299→   299→        alamat: item.address || '',
   300→   300→      },
   301→   301→    }));
   302→   302→    setClientDropdownOpen(false);
   303→   303→  };
   304→   304→
   305→   305→  const updateCompany = (company: typeof invoice.company) => {
   306→   306→    setInvoice((prev) => ({ ...prev, company }));
   307→   307→  };
   308→   308→
   309→   309→  const updateClient = (field: string, value: string) => {
   310→   310→    setInvoice((prev) => ({
   311→   311→      ...prev,
   312→   312→      client: { ...prev.client, [field]: value },
   313→   313→    }));
   314→   314→  };
   315→   315→
   316→   316→  const subtotal = invoice.items.reduce((sum, item) => sum + item.qty * item.harga, 0);
   317→   317→  const ppnAmount = subtotal * (invoice.ppn / 100);
   318→   318→  const total = subtotal + ppnAmount;
   319→   319→  const dpPercent = invoice.dp || 0;
   320→   320→  const dpAmount = total * (dpPercent / 100);
   321→   321→  const sisa = total - dpAmount;
   322→   322→
   323→   323→  const handleSuratJalan = async () => {
   324→   324→    // Check if data has content
   325→   325→    const pihakKedua = invoice.client?.nama;
   326→   326→    const hasItem = invoice.items?.some((item) => item.deskripsi.trim() !== '');
   327→   327→    if (!pihakKedua?.trim() || !hasItem) {
   328→   328→      toast.error('Lengkapi data invoice terlebih dahulu!');
   329→   329→      return;
   330→   330→    }
   331→   331→
   332→   332→    setSavingSj(true);
   333→   333→    try {
   334→   334→      // Save invoice to history first
   335→   335→      const dataToSave = { ...invoice, dpAmount, originalTotal: total };
   336→   336→
   337→   337→      if (invoiceEditingId) {
   338→   338→        // Update existing record
   339→   339→        const res = await fetch(`/api/history/${invoiceEditingId}`, {
   340→   340→          method: 'PUT',
   341→   341→          headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   342→   342→          body: JSON.stringify({
   343→   343→            nomor: invoice.nomor || '-',
   344→   344→            tanggal: invoice.tanggal || '',
   345→   345→            pihakKedua: invoice.client?.nama || '-',
   346→   346→            total: '-',
   347→   347→            dataJson: JSON.stringify(dataToSave),
   348→   348→          }),
   349→   349→        });
   350→   350→        if (res.ok) {
   351→   351→          // If DP was added/updated during this edit, ensure a linked
   352→   352→          // invoice-pelunasan entry exists & is in sync so it shows up in
   353→   353→          // the Editor Pelunasan tab. (Previously this only happened on CREATE.)
   354→   354→          await syncLinkedPelunasan(invoiceEditingId, invoice.nomor || '-', dataToSave);
   355→   355→          window.dispatchEvent(new CustomEvent('dokupro:history-updated'));
   356→   356→          toast.success('Invoice diperbarui');
   357→   357→          const invoiceId = invoiceEditingId;
   358→   358→          setInvoiceEditingId(null);
   359→   359→          resetDocument('invoice');
   360→   360→          router.push(`/surat-jalan?invoiceId=${invoiceId}`);
   361→   361→        } else {
   362→   362→          toast.error('Gagal memperbarui invoice');
   363→   363→        }
   364→   364→      } else {
   365→   365→        const res = await fetch('/api/history', {
   366→   366→          method: 'POST',
   367→   367→          headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   368→   368→          body: JSON.stringify({
   369→   369→            docType: 'invoice',
   370→   370→            nomor: invoice.nomor || '-',
   371→   371→            tanggal: invoice.tanggal || '',
   372→   372→            pihakKedua: invoice.client?.nama || '-',
   373→   373→            total: '-',
   374→   374→            dataJson: JSON.stringify(dataToSave),
   375→   375→          }),
   376→   376→        });
   377→   377→        if (res.ok) {
   378→   378→          const saved = await res.json();
   379→   379→          window.dispatchEvent(new CustomEvent('dokupro:history-updated'));
   380→   380→
   381→   381→          // If invoice has DP, ensure a linked invoice-pelunasan entry exists
   382→   382→          // (creates if missing, updates if already present)
   383→   383→          await syncLinkedPelunasan(saved.id, saved.nomor || invoice.nomor || '-', dataToSave);
   384→   384→
   385→   385→          toast.success('Invoice disimpan ke riwayat');
   386→   386→          // Reset invoice form after saving
   387→   387→          resetDocument('invoice');
   388→   388→          router.push(`/surat-jalan?invoiceId=${saved.id}`);
   389→   389→        } else if (res.status === 409) {
   390→   390→          // Invoice already saved, get existing ID and navigate
   391→   391→          const errData = await res.json().catch(() => ({}));
   392→   392→          if (errData.id) {
   393→   393→            resetDocument('invoice');
   394→   394→            router.push(`/surat-jalan?invoiceId=${errData.id}`);
   395→   395→          } else {
   396→   396→            toast('Invoice sudah ada di riwayat.');
   397→   397→          }
   398→   398→        } else {
   399→   399→          toast.error('Gagal menyimpan invoice');
   400→   400→        }
   401→   401→      }
   402→   402→    } catch {
   403→   403→      toast.error('Gagal menyimpan invoice');
   404→   404→    }
   405→   405→    setSavingSj(false);
   406→   406→  };
   407→   407→
   408→   408→  return (
   409→   409→    <>
   410→   410→      <DocumentEditorLayout
   411→   411→        title="Invoice"
   412→   412→        previewMode="popup"
   413→   413→        previewContent={<InvoicePreview data={invoice} />}
   414→   414→        actions={
   415→   415→          <div className="flex items-center gap-2 flex-wrap">
   416→   416→            <DocumentActionButtons
   417→   417→              docType="invoice"
   418→   418→              documentLabel="Invoice"
   419→   419→              currentData={invoice}
   420→   420→              onReset={() => resetDocument('invoice')}
   421→   421→              editingId={invoiceEditingId}
   422→   422→              onUpdateSuccess={() => setInvoiceEditingId(null)}
   423→   423→            />
   424→   424→            <Button
   425→   425→              size="sm"
   426→   426→              onClick={handleSuratJalan}
   427→   427→              disabled={savingSj}
   428→   428→              className="bg-orange-600 hover:bg-orange-700 h-8 sm:h-9"
   429→   429→            >
   430→   430→              <Truck className="mr-1.5 h-3.5 w-3.5" />
   431→   431→              {savingSj ? 'Menyimpan...' : 'Surat Jalan'}
   432→   432→            </Button>
   433→   433→          </div>
   434→   434→        }
   435→   435→      >
   436→   436→        <CompanyFields company={invoice.company} onChange={updateCompany} />
   437→   437→
   438→   438→        <div className="rounded-lg border bg-card p-3 sm:p-4 shadow-sm">
   439→   439→          <h3 className="mb-3 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   440→   440→            Detail Dokumen
   441→   441→          </h3>
   442→   442→          <div className="grid grid-cols-2 gap-2">
   443→   443→            <div className="space-y-1.5">
   444→   444→              <Label className="text-xs">No. Invoice</Label>
   445→   445→              <Input
   446→   446→                value={invoice.nomor}
   447→   447→                readOnly
   448→   448→                className="bg-slate-50 text-slate-500 cursor-not-allowed"
   449→   449→              />
   450→   450→            </div>
   451→   451→            <div className="space-y-1.5">
   452→   452→              <Label className="text-xs">Tanggal</Label>
   453→   453→              <Input
