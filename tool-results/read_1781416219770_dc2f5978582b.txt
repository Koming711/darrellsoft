    58→    58→function parseDocInfo(entry: HistoryEntry): ParsedInfo {
    59→    59→  try {
    60→    60→    const parsed = JSON.parse(entry.dataJson)
    61→    61→    const items = parsed.items || []
    62→    62→    const firstItem = items[0]
    63→    63→    const namaBarang = firstItem?.deskripsi || ''
    64→    64→    const hargaSatuan = firstItem?.harga || 0
    65→    65→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0)
    66→    66→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0)
    67→    67→    const ppn = parsed.ppn || 0
    68→    68→    const totalHarga = subtotal + (subtotal * ppn / 100)
    69→    69→    const referensi = parsed.referensi || ''
    70→    70→    const client = parsed.client || parsed.penerima || parsed.pemasok || {}
    71→    71→    return {
    72→    72→      namaBarang, hargaSatuan, totalQty, totalHarga, referensi, items,
    73→    73→      ppn,
    74→    74→      catatan: parsed.catatan || '',
    75→    75→      client: { nama: client.nama || '', kontak: client.kontak || '', alamat: client.alamat || '' },
    76→    76→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
    77→    77→      caraPembayaran: parsed.caraPembayaran || '',
    78→    78→      tanggalGiro: parsed.tanggalGiro || '',
    79→    79→    }
    80→    80→  } catch {
    81→    81→    return { namaBarang: '', hargaSatuan: 0, totalQty: 0, totalHarga: 0, referensi: '', items: [], ppn: 0, catatan: '', client: { nama: '', kontak: '', alamat: '' }, tanggalJatuhTempo: '', caraPembayaran: '', tanggalGiro: '' }
    82→    82→  }
    83→    83→}

   143→   143→  }

   158→   158→// --- Date filter ---
   159→   159→type FilterType = 'today' | 'week' | 'month' | 'custom'
   160→   160→
   161→   161→function getFilterDates(filter: FilterType, customStart?: Date, customEnd?: Date): { startDate: string; endDate: string } {
   162→   162→  const today = new Date()
   163→   163→  const fmt = (d: Date) => {
   164→   164→    const y = d.getFullYear()
   165→   165→    const m = String(d.getMonth() + 1).padStart(2, '0')
   166→   166→    const day = String(d.getDate()).padStart(2, '0')
   167→   167→    return `${y}-${m}-${day}`
   168→   168→  }
   169→   169→  switch (filter) {
   170→   170→    case 'today': return { startDate: fmt(today), endDate: fmt(today) }
   171→   171→    case 'week': {
   172→   172→      const s = new Date(today)
   173→   173→      const day = s.getDay()
   174→   174→      s.setDate(s.getDate() + (day === 0 ? -6 : 1 - day))
   175→   175→      return { startDate: fmt(s), endDate: fmt(today) }
   176→   176→    }
   177→   177→    case 'month': {
   178→   178→      const s = new Date(today.getFullYear(), today.getMonth(), 1)
   179→   179→      return { startDate: fmt(s), endDate: fmt(today) }
   180→   180→    }
   181→   181→    case 'custom': {
   182→   182→      return {
   183→   183→        startDate: customStart ? fmt(customStart) : fmt(today),
   184→   184→        endDate: customEnd ? fmt(customEnd) : fmt(today),
   185→   185→      }
   186→   186→    }
   187→   187→  }
   188→   188→}
