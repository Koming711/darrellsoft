    43→function FadeIn({
    44→  children,
    45→  delay = 0,
    46→  className = '',
    47→  direction = 'up',
    48→}: {
    49→  children: React.ReactNode;
    50→  delay?: number;
    51→  className?: string;
    52→  direction?: 'up' | 'down' | 'left' | 'right';
    53→}) {
    54→  const ref = useRef<HTMLDivElement>(null);
    55→  const isInView = useInView(ref, { once: true });
    56→
    57→  const dirMap = {
    58→    up: { y: 40, x: 0 },
    59→    down: { y: -40, x: 0 },
    60→    left: { x: 40, y: 0 },
    61→    right: { x: -40, y: 0 },
    62→  };
    63→  const { x, y } = dirMap[direction];
    64→
    65→  return (
    66→    <motion.div
    67→      ref={ref}
    68→      className={className}
    69→      initial={isInView ? { opacity: 0, x, y } : false}
    70→      animate={{ opacity: 1, x: 0, y: 0 }}
    71→      transition={{ duration: 0.6, delay, ease: 'easeOut' }}
    72→    >
    73→      {children}
    74→    </motion.div>
    75→  );
    76→}

    78→function CountUp({ end, suffix = '', prefix = '', duration = 2000 }: { end: number; suffix?: string; prefix?: string; duration?: number }) {
    79→  const ref = useRef<HTMLSpanElement>(null);
    80→  const isInView = useInView(ref, { once: true });
    81→
    82→  return (
    83→    <motion.span
    84→      ref={ref}
    85→      initial={false}
    86→      animate={{ opacity: 1 }}
    87→      transition={{ duration: 0.4 }}
    88→    >
    89→      {isInView ? (
    90→        <Counter end={end} suffix={suffix} prefix={prefix} duration={duration} />
    91→      ) : (
    92→        <span>{prefix}{end.toLocaleString('id-ID')}{suffix}</span>
    93→      )}
    94→    </motion.span>
    95→  );
    96→}
    97→
    98→function Counter({ end, suffix, prefix, duration }: { end: number; suffix: string; prefix: string; duration: number }) {
    99→  const [count, setCount] = useState(0);
   100→  const ref = useRef<HTMLSpanElement>(null);
   101→  const started = useRef(false);
   102→
   103→  useInView(ref, { once: true });
   104→
   105→  if (!started.current) {
   106→    started.current = true;
   107→    const startTime = Date.now();
   108→    const step = () => {
   109→      const elapsed = Date.now() - startTime;
   110→      const progress = Math.min(elapsed / duration, 1);
   111→      const eased = 1 - Math.pow(1 - progress, 3);
   112→      setCount(Math.round(eased * end));
   113→      if (progress < 1) requestAnimationFrame(step);
   114→    };
   115→    requestAnimationFrame(step);
   116→  }
   117→
   118→  return <span ref={ref}>{prefix}{count.toLocaleString('id-ID')}{suffix}</span>;
   119→}

   243→              popular
   244→                ? 'cta-glow bg-gradient-to-r from-blue-600 to-sky-400 hover:from-blue-700 hover:to-sky-500 text-white shadow-lg shadow-blue-600/25 hover:shadow-xl hover:shadow-blue-600/30'

   249→          </Button>

   251→      </Card>
   252→      </motion.div>
   253→    </FadeIn>
   254→  );
