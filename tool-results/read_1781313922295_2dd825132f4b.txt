   363→      // Only set company_name if it doesn't exist yet (create-only, no overwrite)

   365→      if (!existingCompanyName) {

   369→      if (!existingCompanyLogo) {

   374→    }

   378→
   379→  if (!globalSeedChecked) {

   389→  // Seed per-user master data if userId is provided
