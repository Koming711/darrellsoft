    65→function getTodayStr(): string {
    66→  const d = new Date()
    67→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
    68→}

    70→function formatRupiahShort(n: number): string {
    71→  return new Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR', minimumFractionDigits: 0 }).format(n)
    72→}

    74→// --- Parse dataJson for document info ---
    75→function parseDocInfo(entry: HistoryEntry) {
    76→  try {
    77→    const parsed = JSON.parse(entry.dataJson)
    78→    const items = parsed.items || []
    79→    const firstItem = items[0]
    80→    const namaBarang = firstItem?.deskripsi || ''
    81→    const hargaSatuan = firstItem?.harga || 0
    82→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0)
    83→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0)
    84→    const ppn = parsed.ppn || 0
    85→    const dpPercent = parsed.dp || 0
    86→    const totalHarga = subtotal + (subtotal * ppn / 100)
    87→    const dpAmount = parsed.dpAmount !== undefined ? parsed.dpAmount : totalHarga * (dpPercent / 100)
    88→    const sisa = totalHarga - dpAmount
    89→    const lunas = parsed.lunas === true
    90→    const tanggalJatuhTempo = parsed.tanggalJatuhTempo || ''
    91→    const tanggalPelunasan = parsed.tanggalPelunasan || ''
    92→    const referensi = parsed.referensi || ''
    93→    return { namaBarang, hargaSatuan, totalQty, totalHarga, dpPercent, dp: dpAmount, sisa, lunas, tanggalJatuhTempo, tanggalPelunasan, referensi }
    94→  } catch {
    95→    return { namaBarang: '', hargaSatuan: 0, totalQty: 0, totalHarga: 0, dpPercent: 0, dp: 0, sisa: 0, lunas: false, tanggalJatuhTempo: '', tanggalPelunasan: '', referensi: '' }
    96→  }
    97→}

    99→// --- Parse dataJson for InvoiceData ---
   100→function parseInvoiceData(entry: HistoryEntry): InvoiceData {
   101→  try {
   102→    const parsed = JSON.parse(entry.dataJson)
   103→    const company: CompanyInfo = {
   104→      nama: parsed.company?.nama || DEFAULT_COMPANY.nama,
   105→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
   106→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
   107→      email: parsed.company?.email || DEFAULT_COMPANY.email,
   108→      npwp: parsed.company?.npwp || '',
   109→      website: parsed.company?.website || '',
   110→      ppn: parsed.company?.ppn ?? parsed.ppn ?? 11,
   111→      logo: parsed.company?.logo || '',
   112→      bankName: parsed.company?.bankName || '',
   113→      bankAccount: parsed.company?.bankAccount || '',
   114→      bankHolder: parsed.company?.bankHolder || '',
   115→      bankName2: parsed.company?.bankName2 || '',
   116→      bankAccount2: parsed.company?.bankAccount2 || '',
   117→      bankHolder2: parsed.company?.bankHolder2 || '',
   118→    }
   119→    const client = parsed.client || { nama: '', kontak: '', alamat: '' }
   120→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number }, i: number) => ({
   121→      id: it.id || `item-${i}`,
   122→      deskripsi: it.deskripsi || '',
   123→      qty: it.qty || 0,
   124→      satuan: it.satuan || '',
   125→      harga: it.harga || 0,
   126→    }))
   127→    return {
   128→      type: 'invoice',
   129→      company,
   130→      nomor: parsed.nomor || entry.nomor || '',
   131→      tanggal: parsed.tanggal || entry.tanggal || '',
   132→      referensi: parsed.referensi || '',
   133→      client,
   134→      items,
   135→      ppn: parsed.ppn ?? 11,
   136→      dp: parsed.dp || 0,
   137→      catatan: parsed.catatan || '',
   138→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   139→      caraPembayaran: parsed.caraPembayaran || '',
   140→      tanggalGiro: parsed.tanggalGiro || '',
   141→      lunas: parsed.lunas === true,
   142→      tanggalPelunasan: parsed.tanggalPelunasan || '',
   143→    }
   144→  } catch {
   145→    return {
   146→      type: 'invoice',
   147→      company: { ...DEFAULT_COMPANY },
   148→      nomor: entry.nomor || '',
   149→      tanggal: entry.tanggal || '',
   150→      referensi: '',
   151→      client: { nama: '', kontak: '', alamat: '' },
   152→      items: [],
   153→      ppn: 11,
   154→      dp: 0,
   155→      catatan: '',
   156→      tanggalJatuhTempo: '',
   157→      caraPembayaran: '',
   158→      tanggalGiro: '',
   159→      lunas: false,
   160→      tanggalPelunasan: '',
   161→    }
   162→  }
   163→}
