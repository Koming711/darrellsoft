     4→import {
     5→  Package, Pencil, Plus, Search, Trash2, Users, X,
     6→} from 'lucide-react'
     7→import { toast } from 'sonner'
     8→import { apiFetch } from '@/lib/client'
     9→import { formatIDR, formatNum } from '@/lib/format'
    10→import { UNIT_OPTIONS, type Item, type SessionUser } from '@/lib/types'
    11→import { Badge } from '@/components/ui/badge'
    12→import { Button } from '@/components/ui/button'
    13→import { Card, CardContent } from '@/components/ui/card'
    14→import {
    15→  AlertDialog,
    16→  AlertDialogAction,

    18→  AlertDialogContent,
    19→  AlertDialogDescription,
    20→  AlertDialogFooter,
    21→  AlertDialogHeader,
    22→  AlertDialogTitle,
    23→} from '@/components/ui/alert-dialog'
    24→import {
    25→  Dialog,

    28→  DialogFooter,
    29→  DialogHeader,
    30→  DialogTitle,
    31→} from '@/components/ui/dialog'
    32→import { Input } from '@/components/ui/input'
    33→import { Label } from '@/components/ui/label'
    34→import { Skeleton } from '@/components/ui/skeleton'
    35→import { Switch } from '@/components/ui/switch'
    36→import {
    37→  Select,
    38→  SelectContent,
    39→  SelectItem,
    40→  SelectTrigger,
    41→  SelectValue,
    42→} from '@/components/ui/select'
    43→import {
    44→  Table,
    45→  TableBody,

    47→  TableHead,
    48→  TableHeader,
    49→  TableRow,
    50→} from '@/components/ui/table'
    51→
    52→function errText(e: unknown): string {
    53→  return e instanceof Error ? e.message : 'Terjadi kesalahan'
    54→}
    55→
    56→interface CustomerOption {
    57→  id: string
    58→  name: string
    59→  companyName: string | null
    60→}
    61→
    62→interface ItemFormState {
    63→  name: string
    64→  unit: string
    65→  standardPrice: string
    66→  hpp: string
    67→  keterangan: string
    68→  isActive: boolean
    69→}
    70→
    71→const EMPTY_FORM: ItemFormState = {
    72→  name: '', unit: 'pcs', standardPrice: '', hpp: '', keterangan: '', isActive: true,
    73→}
    74→
    75→function ActiveBadge({ active }: { active: boolean }) {
    76→  return active
    77→    ? <Badge variant="outline" className="bg-emerald-50 text-emerald-700 border-emerald-200 text-[11px] shrink-0">Aktif</Badge>
    78→    : <Badge variant="outline" className="bg-stone-100 text-stone-500 border-stone-200 text-[11px] shrink-0">Nonaktif</Badge>
    79→}
    80→
    81→interface ItemsViewProps {
    82→  user: SessionUser
    83→  /** Izin granular Matriks Hak Akses (master-barang-tambah/edit/hapus); undefined = fallback role lama */
    84→  canAdd?: boolean
    85→  canEdit?: boolean
    86→  canDelete?: boolean
    87→}
    88→
    89→export default function ItemsView({ user, canAdd: canAddProp, canEdit: canEditProp, canDelete: canDeleteProp }: ItemsViewProps) {
    90→  const canManage = user.role !== 'KASIR'
    91→  const canAdd = canAddProp ?? canManage
    92→  const canEditItem = canEditProp ?? canManage
    93→  const canDelete = canDeleteProp ?? canManage
    94→  const showHpp = user.role !== 'KASIR'
    95→
    96→  const [items, setItems] = useState<Item[]>([])
    97→  const [loading, setLoading] = useState(true)
    98→  const [searchInput, setSearchInput] = useState('')
    99→  const [query, setQuery] = useState('')
   100→
   101→  // Pilih Customer — filter daftar barang per pelanggan (barang terdaftar)
   102→  const [customers, setCustomers] = useState<CustomerOption[]>([])
   103→  const [customerId, setCustomerId] = useState<string>('all')
   104→
   105→  const [dialogOpen, setDialogOpen] = useState(false)
   106→  const [editing, setEditing] = useState<Item | null>(null)
   107→  const [form, setForm] = useState<ItemFormState>(EMPTY_FORM)
   108→  const [saving, setSaving] = useState(false)
   109→
   110→  const [deleteTarget, setDeleteTarget] = useState<Item | null>(null)
   111→  const [deleting, setDeleting] = useState(false)
   112→
   113→  useEffect(() => {
   114→    const t = setTimeout(() => setQuery(searchInput.trim()), 300)
   115→    return () => clearTimeout(t)
   116→  }, [searchInput])
   117→
   118→  useEffect(() => {
   119→    apiFetch<CustomerOption[]>('/api/customers')
   120→      .then((data) => setCustomers(Array.isArray(data) ? data : []))
   121→      .catch(() => setCustomers([]))
   122→  }, [])
   123→
   124→  const load = useCallback(async () => {
   125→    setLoading(true)
   126→    try {
   127→      // active=all agar item nonaktif tetap terlihat (bisa diaktifkan kembali)
   128→      const cs = customerId !== 'all' ? `&customerId=${encodeURIComponent(customerId)}` : ''
   129→      const data = await apiFetch<{ items: Item[] }>(
   130→        `/api/items?q=${encodeURIComponent(query)}&active=all${cs}`
   131→      )
   132→      setItems(data.items)
   133→    } catch (e) {
   134→      toast.error(errText(e))
   135→    } finally {
   136→      setLoading(false)
   137→    }
   138→  }, [query, customerId])
   139→
   140→  useEffect(() => {
   141→    void load()
   142→  }, [load])
   143→
   144→  const openCreate = () => {
   145→    setEditing(null)
   146→    setForm(EMPTY_FORM)
   147→    setDialogOpen(true)
   148→  }
   149→
   150→  const openEdit = (it: Item) => {
   151→    setEditing(it)
   152→    setForm({
   153→      name: it.name,
   154→      unit: it.unit,
   155→      standardPrice: String(it.standardPrice),
   156→      hpp: it.hpp != null ? String(it.hpp) : '',
   157→      keterangan: it.keterangan ?? '',
   158→      isActive: it.isActive,
   159→    })
   160→    setDialogOpen(true)
   161→  }
   162→
   163→  // Profit = harga jual − HPP; Margin = profit / harga jual × 100
   164→  const marginInfo = (() => {
   165→    const std = Number(form.standardPrice)
   166→    const hpp = Number(form.hpp)
   167→    if (form.standardPrice === '' || form.hpp === '' || !Number.isFinite(std) || !Number.isFinite(hpp) || std <= 0) {
   168→      return null
   169→    }
   170→    const profit = std - hpp
   171→    const margin = (profit / std) * 100
   172→    return { profit, margin, negative: margin < 0 }
   173→  })()
   174→
   175→  const handleSave = async () => {
   176→    const std = Number(form.standardPrice)
   177→    if (!form.name.trim()) {
   178→      toast.error('Nama barang wajib diisi')
   179→      return
   180→    }
   181→    if (form.standardPrice === '' || !Number.isFinite(std) || std < 0) {
   182→      toast.error('Harga jual wajib diisi (min 0)')
   183→      return
   184→    }
   185→    const hppNum = form.hpp === '' ? null : Number(form.hpp)
   186→    if (hppNum !== null && (!Number.isFinite(hppNum) || hppNum < 0)) {
   187→      toast.error('HPP tidak boleh negatif')
   188→      return
   189→    }
   190→    setSaving(true)
   191→    try {
   192→      const body = {
   193→        name: form.name.trim(),
   194→        unit: form.unit,
   195→        standardPrice: std,
   196→        hpp: showHpp ? hppNum : null,
   197→        keterangan: form.keterangan.trim(),
   198→        ...(editing ? { isActive: form.isActive } : { customerId: customerId !== 'all' ? customerId : undefined }),
   199→      }
   200→      const selectedCustomer = customers.find((c) => c.id === customerId)
   201→      if (editing) {
   202→        await apiFetch<{ item: Item }>(`/api/items/${editing.id}`, {
   203→          method: 'PUT',
   204→          body: JSON.stringify(body),
   205→        })
   206→        toast.success('Barang berhasil diperbarui')
   207→      } else {
   208→        await apiFetch<{ item: Item }>('/api/items', {
   209→          method: 'POST',
   210→          body: JSON.stringify(body),
   211→        })
   212→        toast.success(
   213→          selectedCustomer
   214→            ? `Barang berhasil ditambahkan untuk ${selectedCustomer.name}`
   215→            : 'Barang berhasil ditambahkan'
   216→        )
   217→      }
   218→      setDialogOpen(false)
   219→      void load()
   220→    } catch (e) {
   221→      toast.error(errText(e))
   222→    } finally {
   223→      setSaving(false)
   224→    }
   225→  }
   226→
   227→  const handleDelete = async () => {
   228→    if (!deleteTarget) return
   229→    setDeleting(true)
   230→    try {
   231→      await apiFetch<{ ok: boolean }>(`/api/items/${deleteTarget.id}`, { method: 'DELETE' })
   232→      toast.success('Barang berhasil dihapus')
   233→      setDeleteTarget(null)
   234→      void load()
   235→    } catch (e) {
   236→      // 409 → item dipakai di invoice
   237→      toast.error(errText(e))
   238→    } finally {
   239→      setDeleting(false)
   240→    }
   241→  }
   242→
   243→  const hasResults = items.length > 0
   244→  const selectedCustomerName = customers.find((c) => c.id === customerId)?.name ?? null
   245→  // Tambah/Edit hanya tampil saat pelanggan spesifik dipilih — bukan "Semua Barang"
   246→  const showTambah = canAdd && customerId !== 'all'
   247→  const showEdit = canEditItem && customerId !== 'all'
   248→  // Hapus SELALU tampil di tabel (selaras Master Pelanggan), termasuk mode "Semua Barang"
   249→  const showHapus = canDelete
   250→  const countLabel = loading
   251→    ? 'Memuat data…'
   252→    : selectedCustomerName
   253→      ? `${items.length} barang untuk ${selectedCustomerName}`
   254→      : `${items.length} barang`
   255→
   256→  return (
   257→    <div className="space-y-5">
   258→      {/* Header */}
   259→      <div className="flex flex-col gap-3 md:flex-row md:items-end md:justify-between">
   260→        <div>
   261→          <h1 className="text-xl md:text-2xl font-bold tracking-tight">Master Barang</h1>
   262→          <p className="text-sm text-muted-foreground mt-1">{countLabel}</p>
   263→        </div>
   264→        <div className="flex flex-col sm:flex-row sm:items-center gap-2">
   265→          <div className="relative flex-1 sm:w-64">
   266→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-stone-400 pointer-events-none" />
   267→            <Input
   268→              value={searchInput}
   269→              onChange={(e) => setSearchInput(e.target.value)}
   270→              placeholder="Cari nama / kode barang…"
   271→              aria-label="Cari barang"
   272→              className="pl-9 min-h-[44px]"
   273→            />
   274→          </div>
   275→          {showTambah && (
   276→            <Button
   277→              onClick={openCreate}
   278→              title="Tambah barang untuk pelanggan ini"
   279→              className="bg-emerald-600 hover:bg-emerald-700 min-h-[44px] w-full sm:w-auto"
   280→            >
   281→              <Plus className="h-4 w-4" /> Tambah
   282→            </Button>
   283→          )}
   284→        </div>
   285→      </div>
   286→
   287→      {/* Pilih Pelanggan — kotak filter tepat di bawah judul Master Barang */}
   288→      <section
   289→        aria-label="Filter pelanggan"
   290→        className="rounded-xl border border-stone-200 bg-white p-3 md:px-4 md:py-3"
   291→      >
   292→        <div className="flex flex-col gap-2.5 lg:flex-row lg:items-center lg:gap-4">
   293→          <div className="flex items-center gap-2.5 lg:shrink-0">
   294→            <span className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg border border-stone-200 bg-stone-100">
   295→              <Users className="h-4 w-4 text-stone-500" aria-hidden="true" />
   296→            </span>
   297→            <div className="min-w-0">
   298→              <Label
   299→                htmlFor="pilih-pelanggan"
   300→                className="block text-sm font-semibold text-stone-800 leading-tight"
   301→              >
   302→                Pilih Pelanggan
   303→              </Label>
   304→              <p className="text-[11px] text-muted-foreground leading-tight mt-0.5">
   305→                Tampilkan barang per pelanggan
   306→              </p>
   307→            </div>
   308→          </div>
   309→          <div className="lg:w-80">
   310→            <Select value={customerId} onValueChange={(v) => setCustomerId(v || 'all')}>
   311→              <SelectTrigger
   312→                id="pilih-pelanggan"
   313→                className="w-full min-h-[44px] bg-white"
   314→                aria-label="Pilih Pelanggan"
   315→              >
   316→                <SelectValue placeholder="Pilih pelanggan" />
   317→              </SelectTrigger>
   318→              <SelectContent>
   319→                <SelectItem value="all">Semua Barang</SelectItem>
   320→                {customers.map((c) => (
   321→                  <SelectItem key={c.id} value={c.id}>
   322→                    {c.name}{c.companyName ? ` — ${c.companyName}` : ''}
   323→                  </SelectItem>
   324→                ))}
   325→              </SelectContent>
   326→            </Select>
   327→          </div>
   328→          {canAdd && customerId === 'all' && (
   329→            <p className="text-xs text-muted-foreground lg:ml-auto self-center">
   330→              Pilih nama pelanggan untuk menambah / mengelola barang
   331→            </p>
   332→          )}
   333→          {customerId !== 'all' && selectedCustomerName && (
   334→            <div className="flex items-center gap-2 lg:ml-auto">
   335→              <Badge
   336→                variant="outline"
   337→                className="bg-emerald-50 text-emerald-700 border-emerald-200 text-xs max-w-[240px]"
   338→              >
   339→                <span className="truncate font-medium">{selectedCustomerName}</span>
   340→              </Badge>
   341→              <Button
   342→                variant="ghost"
   343→                size="sm"
   344→                className="h-8 px-2 text-xs text-muted-foreground hover:text-stone-800"
   345→                onClick={() => setCustomerId('all')}
   346→                aria-label="Reset filter pelanggan"
   347→              >
   348→                <X className="h-3.5 w-3.5" /> Reset
   349→              </Button>
   350→            </div>
   351→          )}
   352→        </div>
   353→      </section>
   354→
   355→      {/* Desktop table */}
   356→      <div className="hidden md:block rounded-xl border border-stone-200 bg-white overflow-hidden">
   357→        {loading ? (
   358→          <div className="p-4 space-y-3">
   359→            {[1, 2, 3, 4].map((i) => <Skeleton key={i} className="h-10 w-full" />)}
   360→          </div>
   361→        ) : !hasResults ? (
   362→          <EmptyState
   363→            filtered={query !== ''}
   364→            customerName={selectedCustomerName}
   365→            action={
   366→              showTambah && !query ? (
   367→                <Button
   368→                  onClick={openCreate}
   369→                  className="bg-emerald-600 hover:bg-emerald-700 min-h-[44px]"
   370→                >
   371→                  <Plus className="h-4 w-4" /> Tambah Barang
   372→                </Button>
   373→              ) : undefined
   374→            }
   375→          />
   376→        ) : (
   377→          <div className="max-h-96 overflow-y-auto scrollbar-thin">
   378→            <Table>
   379→              <TableHeader className="sticky top-0 z-10 bg-stone-50">
   380→                <TableRow className="bg-stone-50 hover:bg-stone-50">
   381→                  <TableHead>Kode</TableHead>
   382→                  <TableHead>Nama</TableHead>
   383→                  <TableHead className="text-center">Satuan</TableHead>
   384→                  <TableHead className="text-right">Harga Jual</TableHead>
   385→                  {showHpp && <TableHead className="text-right">HPP</TableHead>}
   386→                  <TableHead>Keterangan</TableHead>
   387→                  <TableHead>Status</TableHead>
   388→                  {(showEdit || showHapus) && <TableHead className="text-right">Aksi</TableHead>}
   389→                </TableRow>
   390→              </TableHeader>
   391→              <TableBody>
   392→                {items.map((it) => (
   393→                  <TableRow key={it.id}>
   394→                    <TableCell className="font-mono text-xs text-muted-foreground whitespace-nowrap">{it.code}</TableCell>
   395→                    <TableCell className="font-medium">{it.name}</TableCell>
   396→                    <TableCell className="text-center">{it.unit}</TableCell>
   397→                    <TableCell className="text-right whitespace-nowrap">{formatIDR(it.standardPrice)}</TableCell>
   398→                    {showHpp && (
   399→                      <TableCell className="text-right whitespace-nowrap">
   400→                        {it.hpp != null ? formatIDR(it.hpp) : '-'}
   401→                      </TableCell>
   402→                    )}
   403→                    <TableCell className="max-w-[200px]">
   404→                      <span
   405→                        className="block truncate text-sm text-muted-foreground"
   406→                        title={it.keterangan || undefined}
   407→                      >
   408→                        {it.keterangan || '-'}
   409→                      </span>
   410→                    </TableCell>
   411→                    <TableCell><ActiveBadge active={it.isActive} /></TableCell>
   412→                    {(showEdit || showHapus) && (
   413→                      <TableCell className="text-right">
   414→                        <div className="flex justify-end gap-1">
   415→                          {showEdit && (
   416→                            <Button
   417→                              variant="ghost"
   418→                              size="icon"
   419→                              className="h-9 w-9"
   420→                              onClick={() => openEdit(it)}
   421→                              aria-label={`Edit ${it.name}`}
   422→                              title="Edit"
   423→                            >
   424→                              <Pencil className="h-4 w-4" />
   425→                            </Button>
   426→                          )}
   427→                          {showHapus && (
   428→                            <Button
   429→                              variant="ghost"
   430→                              size="icon"
   431→                              className="h-9 w-9 text-destructive hover:text-destructive"
   432→                              onClick={() => setDeleteTarget(it)}
   433→                              aria-label={`Hapus ${it.name}`}
   434→                              title="Hapus"
   435→                            >
   436→                              <Trash2 className="h-4 w-4" />
   437→                            </Button>
   438→                          )}
   439→                        </div>
   440→                      </TableCell>
   441→                    )}
   442→                  </TableRow>
   443→                ))}
   444→              </TableBody>
   445→            </Table>
   446→          </div>
   447→        )}
   448→      </div>
   449→
   450→      {/* Mobile cards */}
   451→      <div className="md:hidden space-y-3">
   452→        {loading ? (
   453→          [1, 2, 3].map((i) => <Skeleton key={i} className="h-36 w-full rounded-xl" />)
   454→        ) : !hasResults ? (
   455→          <EmptyState
   456→            filtered={query !== ''}
   457→            customerName={selectedCustomerName}
   458→            action={
   459→              showTambah && !query ? (
   460→                <Button
   461→                  onClick={openCreate}
   462→                  className="bg-emerald-600 hover:bg-emerald-700 min-h-[44px] w-full sm:w-auto"
   463→                >
   464→                  <Plus className="h-4 w-4" /> Tambah Barang
   465→                </Button>
   466→              ) : undefined
   467→            }
   468→          />
   469→        ) : (
   470→          items.map((it) => (
   471→            <Card key={it.id} className="p-0 gap-0">
   472→              <CardContent className="p-4 space-y-2">
   473→                <div className="flex items-start justify-between gap-2">
   474→                  <div className="min-w-0">
   475→                    <p className="font-medium truncate">{it.name}</p>
   476→                    <p className="text-xs text-muted-foreground font-mono">{it.code}</p>
   477→                  </div>
   478→                  <ActiveBadge active={it.isActive} />
   479→                </div>
   480→                <div className="flex items-center gap-4 text-sm">
   481→                  <p className="text-muted-foreground">Satuan: <span className="font-medium text-stone-700">{it.unit}</span></p>
   482→                  <p className="text-muted-foreground">Harga Jual: <span className="font-semibold text-stone-700">{formatIDR(it.standardPrice)}</span></p>
   483→                </div>
   484→                {showHpp && (
   485→                  <p className="text-sm text-muted-foreground">
   486→                    HPP: <span className="font-medium text-stone-700">{it.hpp != null ? formatIDR(it.hpp) : '-'}</span>
   487→                  </p>
   488→                )}
   489→                {it.keterangan && (
   490→                  <p className="text-sm text-muted-foreground">
   491→                    Keterangan: <span className="text-stone-700">{it.keterangan}</span>
   492→                  </p>
   493→                )}
   494→                {(showEdit || showHapus) && (
   495→                  <div className="flex gap-2 pt-1">
   496→                    {showEdit && (
   497→                      <Button
   498→                        variant="outline"
   499→                        size="sm"
   500→                        className="flex-1 min-h-[44px]"
   501→                        onClick={() => openEdit(it)}
   502→                      >
   503→                        <Pencil className="h-4 w-4" /> Edit
   504→                      </Button>
   505→                    )}
   506→                    {showHapus && (
   507→                      <Button
   508→                        variant="outline"
   509→                        size="sm"
   510→                        className="flex-1 min-h-[44px] text-destructive border-stone-200 hover:bg-destructive/10 hover:text-destructive"
   511→                        onClick={() => setDeleteTarget(it)}
   512→                      >
   513→                        <Trash2 className="h-4 w-4" /> Hapus
   514→                      </Button>
   515→                    )}
   516→                  </div>
   517→                )}
   518→              </CardContent>
   519→            </Card>
   520→          ))
   521→        )}
   522→      </div>
   523→
   524→      {/* Dialog create / edit */}
   525→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   526→        <DialogContent className="sm:max-w-lg">
   527→          <DialogHeader>
   528→            <DialogTitle>{editing ? 'Edit Barang' : 'Tambah Barang'}</DialogTitle>
   529→            <DialogDescription>
   530→              {editing
   531→                ? `Kode ${editing.code} — perbarui data barang.`
   532→                : selectedCustomerName
   533→                  ? `Kode barang dibuat otomatis. Barang akan didaftarkan untuk ${selectedCustomerName}.`
   534→                  : 'Kode barang dibuat otomatis oleh sistem.'}
   535→            </DialogDescription>
   536→          </DialogHeader>
   537→          <div className="grid gap-4">
   538→            <div className="grid gap-1.5">
   539→              <Label htmlFor="item-name">Nama Barang <span className="text-destructive">*</span></Label>
   540→              <Input
   541→                id="item-name"
   542→                value={form.name}
   543→                onChange={(e) => setForm((f) => ({ ...f, name: e.target.value }))}
   544→                placeholder="Nama barang / layanan"
   545→                autoComplete="off"
   546→              />
   547→            </div>
   548→            <div className="grid sm:grid-cols-2 gap-4">
   549→              <div className="grid gap-1.5">
   550→                <Label htmlFor="item-unit">Satuan</Label>
   551→                <Select value={form.unit} onValueChange={(v) => setForm((f) => ({ ...f, unit: v }))}>
   552→                  <SelectTrigger id="item-unit" className="w-full min-h-[44px]" aria-label="Satuan barang">

   557→                      <SelectItem key={u} value={u}>{u}</SelectItem>

   559→                  </SelectContent>
   560→                </Select>
   561→              </div>
   562→              <div className="grid gap-1.5">
   563→                <Label htmlFor="item-price">Harga Jual (Rp) <span className="text-destructive">*</span></Label>
   564→                <Input
   565→                  id="item-price"
   566→                  type="number"
   567→                  inputMode="numeric"
   568→                  min={0}
   569→                  step="any"
   570→                  value={form.standardPrice}
   571→                  onChange={(e) => setForm((f) => ({ ...f, standardPrice: e.target.value }))}
   572→                  placeholder="0"
   573→                />
   574→              </div>
   575→            </div>
   576→            {showHpp && (
   577→              <div className="grid gap-1.5">

   582→                  inputMode="numeric"

   584→                  step="any"

   587→                  placeholder="0"
   588→                />

   596→            )}
   597→            <div className="grid gap-1.5">

   618→                  className="data-[state=checked]:bg-emerald-600"
   619→                />

   621→            )}
   622→          </div>
   623→          <DialogFooter className="gap-2">
   624→            <Button variant="outline" onClick={() => setDialogOpen(false)} disabled={saving} className="min-h-[44px]">

   626→            </Button>
   627→            <Button
   628→              onClick={() => void handleSave()}
   629→              disabled={saving}
   630→              className="bg-emerald-600 hover:bg-emerald-700 min-h-[44px]"
   631→            >
   632→              {saving ? 'Menyimpan…' : 'Simpan'}
   633→            </Button>
   634→          </DialogFooter>
   635→        </DialogContent>
   636→      </Dialog>
   637→
   638→      {/* AlertDialog hapus */}
   639→      <AlertDialog open={!!deleteTarget} onOpenChange={(o) => { if (!o) setDeleteTarget(null) }}>
   640→        <AlertDialogContent>
   641→          <AlertDialogHeader>
   642→            <AlertDialogTitle>Hapus barang?</AlertDialogTitle>
   643→            <AlertDialogDescription>
   644→              {deleteTarget?.name} akan dihapus permanen dari daftar beserta daftar harga khususnya.

   646→            </AlertDialogDescription>
   647→          </AlertDialogHeader>
   648→          <AlertDialogFooter>
   649→            <AlertDialogCancel disabled={deleting}>Batal</AlertDialogCancel>
   650→            <AlertDialogAction
   651→              disabled={deleting}
   652→              className="bg-destructive text-white hover:bg-destructive/90"
   653→              onClick={(e) => { e.preventDefault(); void handleDelete() }}
   654→            >
   655→              {deleting ? 'Menghapus…' : 'Hapus'}
   656→            </AlertDialogAction>
   657→          </AlertDialogFooter>
   658→        </AlertDialogContent>
   659→      </AlertDialog>
   660→    </div>
   661→  )
