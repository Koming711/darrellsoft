    43→interface ParsedInfo {

    58→function parseDocInfo(entry: HistoryEntry): ParsedInfo {
    59→  try {
    60→    const parsed = JSON.parse(entry.dataJson)
    61→    const items = parsed.items || []
    62→    const firstItem = items[0]
    63→    const namaBarang = firstItem?.deskripsi || ''
    64→    const hargaSatuan = firstItem?.harga || 0
    65→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0)
    66→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0)
    67→    const ppn = parsed.ppn || 0
    68→    const totalHarga = subtotal + (subtotal * ppn / 100)
    69→    const referensi = parsed.referensi || ''
    70→    const client = parsed.client || parsed.penerima || parsed.pemasok || {}
    71→    return {
    72→      namaBarang, hargaSatuan, totalQty, totalHarga, referensi, items,
    73→      ppn,
    74→      catatan: parsed.catatan || '',
    75→      client: { nama: client.nama || '', kontak: client.kontak || '', alamat: client.alamat || '' },
    76→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
    77→      caraPembayaran: parsed.caraPembayaran || '',
    78→      tanggalGiro: parsed.tanggalGiro || '',
    79→    }
    80→  } catch {
    81→    return { namaBarang: '', hargaSatuan: 0, totalQty: 0, totalHarga: 0, referensi: '', items: [], ppn: 0, catatan: '', client: { nama: '', kontak: '', alamat: '' }, tanggalJatuhTempo: '', caraPembayaran: '', tanggalGiro: '' }
    82→  }
    83→}

    85→function parseInvoiceData(entry: HistoryEntry): InvoiceData {
    86→  try {
    87→    const parsed = JSON.parse(entry.dataJson)
    88→    const company: CompanyInfo = {
    89→      nama: parsed.company?.nama || DEFAULT_COMPANY.nama,
    90→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
    91→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
    92→      email: parsed.company?.email || DEFAULT_COMPANY.email,
    93→      npwp: parsed.company?.npwp || '',
    94→      website: parsed.company?.website || '',
    95→      ppn: parsed.company?.ppn ?? parsed.ppn ?? 11,
    96→      logo: parsed.company?.logo || '',
    97→      bankName: parsed.company?.bankName || '',
    98→      bankAccount: parsed.company?.bankAccount || '',
    99→      bankHolder: parsed.company?.bankHolder || '',
   100→      bankName2: parsed.company?.bankName2 || '',
   101→      bankAccount2: parsed.company?.bankAccount2 || '',
   102→      bankHolder2: parsed.company?.bankHolder2 || '',
   103→    }
   104→    const client = parsed.client || { nama: '', kontak: '', alamat: '' }
   105→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number }, i: number) => ({
   106→      id: it.id || `item-${i}`,
   107→      deskripsi: it.deskripsi || '',
   108→      qty: it.qty || 0,
   109→      satuan: it.satuan || '',
   110→      harga: it.harga || 0,
   111→    }))
   112→    return {
   113→      type: 'invoice',
   114→      company,
   115→      nomor: parsed.nomor || entry.nomor || '',
   116→      tanggal: parsed.tanggal || entry.tanggal || '',
   117→      referensi: parsed.referensi || '',
   118→      client,
   119→      items,
   120→      ppn: parsed.ppn ?? 11,
   121→      dp: parsed.dp ?? 0,
   122→      catatan: parsed.catatan || '',
   123→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   124→      caraPembayaran: parsed.caraPembayaran || '',
   125→      tanggalGiro: parsed.tanggalGiro || '',
   126→    }
   127→  } catch {
   128→    return {
   129→      type: 'invoice',
   130→      company: { ...DEFAULT_COMPANY },
   131→      nomor: entry.nomor || '',
   132→      tanggal: entry.tanggal || '',
   133→      referensi: '',
   134→      client: { nama: '', kontak: '', alamat: '' },
   135→      items: [],
   136→      ppn: 11,
   137→      dp: 0,
   138→      catatan: '',
   139→      tanggalJatuhTempo: '',
   140→      caraPembayaran: '',
   141→      tanggalGiro: '',
   142→    }
   143→  }
   144→}

   146→function formatRp(n: number) {
   147→  return `Rp ${n.toLocaleString('id-ID')}`
   148→}

   150→function formatRupiahShort(n: number): string {
   151→  return new Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR', minimumFractionDigits: 0 }).format(n)
   152→}

   154→function formatDateShort(iso: string): string {
   155→  return new Date(iso).toLocaleDateString('id-ID', { day: '2-digit', month: 'short', year: 'numeric' })
   156→}

   158→// --- Date filter ---
   159→type FilterType = 'today' | 'week' | 'month' | 'custom'
   160→
   161→function getFilterDates(filter: FilterType, customStart?: Date, customEnd?: Date): { startDate: string; endDate: string } {
   162→  const today = new Date()
   163→  const fmt = (d: Date) => {
   164→    const y = d.getFullYear()
   165→    const m = String(d.getMonth() + 1).padStart(2, '0')
   166→    const day = String(d.getDate()).padStart(2, '0')
   167→    return `${y}-${m}-${day}`
   168→  }
   169→  switch (filter) {
   170→    case 'today': return { startDate: fmt(today), endDate: fmt(today) }
   171→    case 'week': {
   172→      const s = new Date(today)
   173→      const day = s.getDay()
   174→      s.setDate(s.getDate() + (day === 0 ? -6 : 1 - day))
   175→      return { startDate: fmt(s), endDate: fmt(today) }
   176→    }
   177→    case 'month': {
   178→      const s = new Date(today.getFullYear(), today.getMonth(), 1)
   179→      return { startDate: fmt(s), endDate: fmt(today) }
   180→    }
   181→    case 'custom': {
   182→      return {
   183→        startDate: customStart ? fmt(customStart) : fmt(today),
   184→        endDate: customEnd ? fmt(customEnd) : fmt(today),
   185→      }
   186→    }
   187→  }
   188→}
