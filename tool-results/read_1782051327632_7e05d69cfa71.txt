   148→type FilterType = 'today' | 'week' | 'month' | 'custom'

   150→function getFilterDates(filter: FilterType, customStart?: Date, customEnd?: Date): { startDate: string; endDate: string } {
   151→  const today = new Date()
   152→  const formatDate = (d: Date) => {
   153→    const y = d.getFullYear()
   154→    const m = String(d.getMonth() + 1).padStart(2, '0')
   155→    const day = String(d.getDate()).padStart(2, '0')
   156→    return `${y}-${m}-${day}`
   157→  }
   158→
   159→  switch (filter) {
   160→    case 'today': {
   161→      return { startDate: formatDate(today), endDate: formatDate(today) }
   162→    }
   163→    case 'week': {
   164→      const startOfWeek = new Date(today)
   165→      // Get Monday of current week
   166→      const day = startOfWeek.getDay()
   167→      const diff = day === 0 ? -6 : 1 - day
   168→      startOfWeek.setDate(startOfWeek.getDate() + diff)
   169→      return { startDate: formatDate(startOfWeek), endDate: formatDate(today) }
   170→    }
   171→    case 'month': {
   172→      const startOfMonth = new Date(today.getFullYear(), today.getMonth(), 1)
   173→      return { startDate: formatDate(startOfMonth), endDate: formatDate(today) }
   174→    }
   175→    case 'custom': {
   176→      return {
   177→        startDate: customStart ? formatDate(customStart) : formatDate(today),
   178→        endDate: customEnd ? formatDate(customEnd) : formatDate(today),
   179→      }
   180→    }
   181→  }
   182→}

   195→// --- Greeting ---
   196→function getGreetingKey(): 'greeting_morning' | 'greeting_afternoon' | 'greeting_evening' | 'greeting_night' {
   197→  const hour = new Date().getHours()
   198→  if (hour >= 0 && hour < 11) return 'greeting_morning'
   199→  if (hour >= 11 && hour < 15) return 'greeting_afternoon'
   200→  if (hour >= 15 && hour < 18) return 'greeting_evening'
   201→  return 'greeting_night'
   202→}

   237→// --- Parse dataJson for document info ---
   238→function parseDocInfo(entry: HistoryEntry) {
   239→  try {
   240→    const parsed = JSON.parse(entry.dataJson);
   241→    const items = parsed.items || [];
   242→    const firstItem = items[0];
   243→    const namaBarang = firstItem?.deskripsi || '';
   244→    const hargaSatuan = firstItem?.harga || 0;
   245→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0);
   246→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0);
   247→    const ppn = parsed.ppn || 0;
   248→    const totalHarga = subtotal + (subtotal * ppn / 100);
   249→    const dp = parsed.dp || 0;
   250→    const originalTotal = parsed.originalTotal !== undefined ? parsed.originalTotal : totalHarga;
   251→    const grandTotal = dp > 0 ? originalTotal : totalHarga;
   252→    const referensi = parsed.referensi || '';
   253→    return { namaBarang, hargaSatuan, totalQty, totalHarga, dp, originalTotal, grandTotal, referensi };
   254→  } catch {
   255→    return { namaBarang: '', hargaSatuan: 0, totalQty: 0, totalHarga: 0, dp: 0, originalTotal: 0, grandTotal: 0, referensi: '' };
   256→  }
   257→}

   375→// Calculate uang capek (profit) for each invoice by matching referensi with riwayat cetakan profitAmount
   376→// Referensi can be either a nomorUrut (e.g. "HC/05/2026/0007") or a printName (e.g. "Brosur")
   377→function calculateUangCapek(invoices: HistoryEntry[], cetakanRecords: { nomorUrut: string; printName: string; profitAmount: number }[]): Map<string, number> {
   378→  const result = new Map<string, number>()
   379→  
   380→  // Build cetakan lookup by both nomorUrut and printName
   381→  const cetakanByRef = new Map<string, number>()
   382→  for (const c of cetakanRecords) {
   383→    // Index by nomorUrut (preferred match)
   384→    if (c.nomorUrut) {
   385→      cetakanByRef.set(c.nomorUrut, (cetakanByRef.get(c.nomorUrut) || 0) + c.profitAmount)
   386→    }
   387→    // Also index by printName (fallback for older data)
   388→    if (c.printName && !cetakanByRef.has(c.printName)) {
   389→      cetakanByRef.set(c.printName, (cetakanByRef.get(c.printName) || 0) + c.profitAmount)
   390→    }
   391→  }
   392→
   393→  // Calculate uang capek per invoice
   394→  for (const inv of invoices) {
   395→    const info = parseDocInfo(inv)
   396→    const uangCapek = info.referensi ? (cetakanByRef.get(info.referensi) || 0) : 0
   397→    result.set(inv.id, uangCapek)
   398→  }
   399→
   400→  return result
   401→}
