    57→function getInitialFormState(): FormData {
    58→  if (typeof window === 'undefined') {
    59→    return {
    60→      paperWidth: '', paperHeight: '', cutWidth: '', cutHeight: '',
    61→      selectedCustomerId: '', selectedPaperId: '', grammage: '', pricePerSheet: '',
    62→      quantity: '', jumlahPesanan: '', berapaMata: '', setelanKertas: '', printName: '', isCustomPaper: false, optimizationMode: 'maximal',
    63→    }
    64→  }
    65→  try {
    66→    const savedVersion = localStorage.getItem(STORAGE_VERSION_KEY())
    67→    if (savedVersion && savedVersion !== STORAGE_VERSION) {
    68→      localStorage.removeItem(STORAGE_KEY())
    69→      localStorage.removeItem(STORAGE_RESULTS_KEY())
    70→      localStorage.setItem(STORAGE_VERSION_KEY(), STORAGE_VERSION)
    71→    }
    72→    const saved = localStorage.getItem(STORAGE_KEY())
    73→    if (saved) return JSON.parse(saved)
    74→  } catch {}
    75→  return {
    76→    paperWidth: '', paperHeight: '', cutWidth: '', cutHeight: '',
    77→    selectedCustomerId: '', selectedPaperId: '', grammage: '', pricePerSheet: '',
    78→    quantity: '', jumlahPesanan: '', berapaMata: '', setelanKertas: '', printName: '', isCustomPaper: false, optimizationMode: 'maximal',
