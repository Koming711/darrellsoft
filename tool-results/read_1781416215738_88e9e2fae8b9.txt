     1→'use client';
     2→
     3→import { useEffect, useState, useCallback, useRef, useMemo } from 'react';
     4→import { useSearchParams, useRouter } from 'next/navigation';
     5→import { Input } from '@/components/ui/input';
     6→import { Label } from '@/components/ui/label';
     7→import { Textarea } from '@/components/ui/textarea';
     8→import { Popover, PopoverContent, PopoverAnchor } from '@/components/ui/popover';
     9→import { useDokuproStore } from '@/lib/store';
    10→import { CompanyFields } from './company-fields';
    11→import { ItemsFields } from './items-fields';
    12→import { InvoicePreview } from './invoice-preview';
    13→import { DocumentEditorLayout } from './document-editor-layout';
    14→import { DocumentActionButtons } from './document-action-buttons';
    15→import { formatRupiah } from '@/lib/format';
    16→import { getAuthHeaders } from '@/lib/auth';
    17→import { Truck } from 'lucide-react';
    18→import { Button } from '@/components/ui/button';
    19→import { toast } from 'sonner';
    20→import type { InvoiceData } from '@/lib/types';
    21→
    22→interface CustomerItem {
    23→  id: string;
    24→  name: string;
    25→  companyName?: string;
    26→  phone?: string;
    27→  email?: string;
    28→  address?: string;
    29→}
    30→
    31→interface RiwayatCetakanItem {
    32→  id: string;
    33→  nomorUrut: string;
    34→  printName: string;
    35→  customerName: string;
    36→  paperName: string;
    37→  paperGrammage: string;
    38→  cutWidth: string;
    39→  cutHeight: string;
    40→  quantity: string;
    41→  jumlahPesanan: string;
    42→  berapaMata: string;
    43→  warna: string;
    44→  machineName: string;
    45→  ongkosCetak: number;
    46→  machineName2: string;
    47→  ongkosCetak2: number;
    48→  totalPaperPrice: number;
    49→  pricePerSheet: number;
    50→  finishingNames: string;
    51→  finishingCost: number;
    52→  packingCost: number;
    53→  shippingCost: number;
    54→  otherCost: number;
    55→  otherCost2: number;
    56→  otherCostLabel: string;
    57→  otherCostLabel2: string;
    58→  glueCost: number;
    59→  glueBorongan: number;
    60→  subTotal: number;
    61→  profitPercent: number;
    62→  profitAmount: number;
    63→  grandTotal: number;
    64→  createdAt: string;
    65→}
    66→
    67→export function InvoiceEditor() {
    68→  const invoice = useDokuproStore((s) => s.invoice);
    69→  const setInvoice = useDokuproStore((s) => s.setInvoice);
    70→  const resetDocument = useDokuproStore((s) => s.resetDocument);
    71→  const loadCompanyFromAPI = useDokuproStore((s) => s.loadCompanyFromAPI);
    72→  const invoiceEditingId = useDokuproStore((s) => s.invoiceEditingId);
    73→  const setInvoiceEditingId = useDokuproStore((s) => s.setInvoiceEditingId);
    74→  const router = useRouter();
    75→
    76→  const searchParams = useSearchParams();
    77→  const riwayatIdFromUrl = searchParams.get('riwayatId');
    78→  const autoSelectDoneRef = useRef<string | null>(null); // track which riwayatId was auto-selected
    79→  const [savingSj, setSavingSj] = useState(false);
    80→
    81→  // Riwayat cetakan dropdown state
    82→  const [riwayatList, setRiwayatList] = useState<RiwayatCetakanItem[]>([]);
    83→  const [referensiInput, setReferensiInput] = useState(invoice.referensi);
    84→  const [dropdownOpen, setDropdownOpen] = useState(false);
    85→
    86→  // Customer dropdown state
    87→  const [customerList, setCustomerList] = useState<CustomerItem[]>([]);
    88→  const [clientInput, setClientInput] = useState(invoice.client.nama);
    89→  const [clientDropdownOpen, setClientDropdownOpen] = useState(false);
    90→  const [clientTyping, setClientTyping] = useState(false);
    91→
    92→  const fetchRiwayatCetakan = useCallback(async () => {
    93→    try {
    94→      const res = await fetch('/api/riwayat-cetakan', { headers: getAuthHeaders() });
    95→      if (res.ok) {
    96→        const data = await res.json();
    97→        setRiwayatList(data);
    98→        return data;
    99→      }
   100→    } catch {
   101→      // silently fail
   102→    }
   103→    return [];
   104→  }, []);
   105→
   106→  // Fetch customer list
   107→  const fetchCustomers = useCallback(async () => {
   108→    try {
   109→      const res = await fetch('/api/customers', { headers: getAuthHeaders() });
   110→      if (res.ok) {
   111→        const data = await res.json();
   112→        setCustomerList(Array.isArray(data) ? data : []);
   113→      }
   114→    } catch {
   115→      // silently fail
   116→    }
   117→  }, []);
   118→
   119→  useEffect(() => { loadCompanyFromAPI() }, [loadCompanyFromAPI]);
   120→  useEffect(() => { fetchRiwayatCetakan() }, [fetchRiwayatCetakan]);
   121→  useEffect(() => { fetchCustomers() }, [fetchCustomers]);
   122→
   123→  // Fetch next Invoice number from server (only when creating new, not editing)
   124→  const fetchNextNumber = useCallback(() => {
   125→    // Read current editingId from store to avoid stale closure
   126→    const currentEditingId = useDokuproStore.getState().invoiceEditingId;
   127→    if (currentEditingId) return; // Don't overwrite nomor when editing
   128→    fetch('/api/history?preview=next-number&docType=invoice', { headers: getAuthHeaders() })
   129→      .then(res => res.ok ? res.json() : null)
   130→      .then(data => { if (data?.nextNumber) setInvoice((prev) => ({ ...prev, nomor: data.nextNumber })) })
   131→      .catch(() => {});
   132→  }, [setInvoice]);
   133→
   134→  useEffect(() => { fetchNextNumber() }, [fetchNextNumber]);
   135→
   136→  // Re-fetch next number after a document is saved
   137→  useEffect(() => {
   138→    const handler = () => { fetchNextNumber() };
   139→    window.addEventListener('dokupro:history-updated', handler);
   140→    return () => { window.removeEventListener('dokupro:history-updated', handler) };
   141→  }, [fetchNextNumber]);
   142→
   143→  // Sync referensiInput when invoice.referensi changes externally
   144→  useEffect(() => { setReferensiInput(invoice.referensi) }, [invoice.referensi]);
   145→  // Sync clientInput when invoice.client.nama changes externally
   146→  useEffect(() => { setClientInput(invoice.client.nama) }, [invoice.client.nama]);
   147→
   148→  // Filter riwayat list — when dropdown is open show all, otherwise not used
   149→  const filteredRiwayatList = riwayatList;
   150→
   151→  const handleReferensiInputChange = (value: string) => {
   152→    setReferensiInput(value);
   153→    setInvoice((prev) => ({ ...prev, referensi: value }));
   154→    setDropdownOpen(true);
   155→  };
   156→
   157→  // Apply referensi data using functional setInvoice to avoid stale closure
   158→  const applyReferensi = (item: RiwayatCetakanItem) => {
   159→    const ref = item.nomorUrut || item.printName || '';
   160→    setReferensiInput(ref);
   161→
   162→    // Build single item with multi-line description
   163→    const descLines: string[] = [];
   164→    if (item.printName) descLines.push(item.printName);
   165→
   166→    const bahanParts: string[] = [];
   167→    if (item.paperName) bahanParts.push(item.paperName);
   168→    if (item.paperGrammage) bahanParts.push(item.paperGrammage + 'g');
   169→    if (bahanParts.length > 0) descLines.push(bahanParts.join(' '));
   170→
   171→    if (item.finishingNames) {
   172→      const finishings = item.finishingNames.split(',').map(f => f.trim()).filter(Boolean);
   173→      descLines.push('Finishing:');
   174→      finishings.forEach(f => descLines.push(f));
   175→    }
   176→
   177→    const deskripsi = descLines.join('\n');
   178→    const jumlahPesanan = parseInt(item.jumlahPesanan) || 1;
   179→    const hargaPerPcs = jumlahPesanan > 0 ? Math.round(item.grandTotal / jumlahPesanan) : 0;
   180→
   181→    const newItems: InvoiceData['items'] = [{
   182→      id: 'riwayat-0',
   183→      deskripsi: deskripsi || ref,
   184→      qty: jumlahPesanan,
   185→      satuan: 'pcs',
   186→      harga: hargaPerPcs,
   187→    }];
   188→
   189→    // Use functional form to always get the LATEST state
   190→    setInvoice((prev) => ({
   191→      ...prev,
   192→      referensi: ref,
   193→      client: {
   194→        ...prev.client,
   195→        nama: item.customerName || prev.client.nama,
   196→      },
   197→      items: newItems,
   198→      catatan: '',
   199→      uangCapek: item.profitAmount || 0,
   200→    }));
   201→    setDropdownOpen(false);
   202→  };
   203→
   204→  const handleReferensiSelect = (item: RiwayatCetakanItem) => {
   205→    applyReferensi(item);
   206→  };
   207→
   208→  // Auto-select referensi when coming from hitung cetakan with riwayatId
   209→  // Fetches the specific riwayat directly by ID for reliability (no race condition with list loading)
   210→  useEffect(() => {
   211→    if (!riwayatIdFromUrl) return;
   212→    if (autoSelectDoneRef.current === riwayatIdFromUrl) return;
   213→
   214→    let cancelled = false;
   215→
   216→    const fetchAndApply = async () => {
   217→      // First, try to find it in the already-loaded list (fast path)
   218→      const foundInList = riwayatList.find((r) => r.id === riwayatIdFromUrl);
   219→      if (foundInList) {
   220→        if (!cancelled) {
   221→          autoSelectDoneRef.current = riwayatIdFromUrl;
   222→          applyReferensi(foundInList);
   223→        }
   224→        return;
   225→      }
   226→
   227→      // Not in list yet — fetch directly by ID from the API (reliable path)
   228→      try {
   229→        const res = await fetch(`/api/riwayat-cetakan?id=${riwayatIdFromUrl}`, { headers: getAuthHeaders() });
   230→        if (res.ok && !cancelled) {
   231→          const data = await res.json();
   232→          if (data && !data.error) {
   233→            autoSelectDoneRef.current = riwayatIdFromUrl;
   234→            applyReferensi(data as RiwayatCetakanItem);
   235→          }
   236→        }
   237→      } catch {
   238→        // Fallback: retry once after a short delay
   239→        if (cancelled) return;
   240→        await new Promise(r => setTimeout(r, 800));
   241→        if (cancelled) return;
   242→        try {
   243→          const res = await fetch(`/api/riwayat-cetakan?id=${riwayatIdFromUrl}`, { headers: getAuthHeaders() });
   244→          if (res.ok) {
   245→            const data = await res.json();
   246→            if (data && !data.error) {
   247→              autoSelectDoneRef.current = riwayatIdFromUrl;
   248→              applyReferensi(data as RiwayatCetakanItem);
   249→            }
   250→          }
   251→        } catch {
   252→          // Give up silently
   253→        }
   254→      }
   255→    };
   256→
   257→    fetchAndApply();
   258→
   259→    return () => { cancelled = true; };
   260→  }, [riwayatIdFromUrl]);
   261→
   262→  // Filter customer list (only when user is actively typing)
   263→  const filteredCustomerList = useMemo(() =>
   264→    clientTyping
   265→      ? customerList.filter((c) => {
   266→          const search = clientInput.toLowerCase().trim();
   267→          if (!search) return true;
   268→          return c.name.toLowerCase().includes(search) ||
   269→                 (c.companyName || '').toLowerCase().includes(search) ||
   270→                 (c.phone || '').toLowerCase().includes(search) ||
   271→                 (c.address || '').toLowerCase().includes(search);
   272→        })
   273→      : customerList
   274→  , [clientTyping, customerList, clientInput]);
   275→
   276→  const handleClientInputChange = (value: string) => {
   277→    setClientInput(value);
   278→    setClientTyping(true);
   279→    updateClient('nama', value);
   280→    setClientDropdownOpen(true);
   281→  };
   282→
   283→  const handleClientSelect = (item: CustomerItem) => {
   284→    setClientInput(item.name);
   285→    setClientTyping(false);
   286→    setInvoice((prev) => ({
   287→      ...prev,
   288→      client: {
   289→        nama: item.name,
   290→        kontak: item.phone || item.email || '',
   291→        alamat: item.address || '',
   292→      },
   293→    }));
   294→    setClientDropdownOpen(false);
   295→  };
   296→
   297→  const updateCompany = (company: typeof invoice.company) => {
   298→    setInvoice((prev) => ({ ...prev, company }));
   299→  };
   300→
   301→  const updateClient = (field: string, value: string) => {
   302→    setInvoice((prev) => ({
   303→      ...prev,
   304→      client: { ...prev.client, [field]: value },
   305→    }));
   306→  };
   307→
   308→  const subtotal = invoice.items.reduce((sum, item) => sum + item.qty * item.harga, 0);
   309→  const ppnAmount = subtotal * (invoice.ppn / 100);
   310→  const total = subtotal + ppnAmount;
   311→  const dpPercent = invoice.dp || 0;
   312→  const dpAmount = total * (dpPercent / 100);
   313→  const sisa = total - dpAmount;
   314→
   315→  const handleSuratJalan = async () => {
   316→    // Check if data has content
   317→    const pihakKedua = invoice.client?.nama;
   318→    const hasItem = invoice.items?.some((item) => item.deskripsi.trim() !== '');
   319→    if (!pihakKedua?.trim() || !hasItem) {
   320→      toast.error('Lengkapi data invoice terlebih dahulu!');
   321→      return;
   322→    }
   323→
   324→    setSavingSj(true);
   325→    try {
   326→      // Save invoice to history first
   327→      const dataToSave = { ...invoice, dpAmount, originalTotal: total };
   328→
   329→      if (invoiceEditingId) {
   330→        // Update existing record
   331→        const res = await fetch(`/api/history/${invoiceEditingId}`, {
   332→          method: 'PUT',
   333→          headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   334→          body: JSON.stringify({
   335→            nomor: invoice.nomor || '-',
   336→            tanggal: invoice.tanggal || '',
   337→            pihakKedua: invoice.client?.nama || '-',
   338→            total: '-',
   339→            dataJson: JSON.stringify(dataToSave),
   340→          }),
   341→        });
   342→        if (res.ok) {
   343→          window.dispatchEvent(new CustomEvent('dokupro:history-updated'));
   344→          toast.success('Invoice diperbarui');
   345→          const invoiceId = invoiceEditingId;
   346→          setInvoiceEditingId(null);
   347→          resetDocument('invoice');
   348→          router.push(`/surat-jalan?invoiceId=${invoiceId}`);
   349→        } else {
   350→          toast.error('Gagal memperbarui invoice');
   351→        }
   352→      } else {
   353→        const res = await fetch('/api/history', {
   354→          method: 'POST',
   355→          headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   356→          body: JSON.stringify({
   357→            docType: 'invoice',
   358→            nomor: invoice.nomor || '-',
   359→            tanggal: invoice.tanggal || '',
   360→            pihakKedua: invoice.client?.nama || '-',
   361→            total: '-',
   362→            dataJson: JSON.stringify(dataToSave),
   363→          }),
   364→        });
   365→        if (res.ok) {
   366→          const saved = await res.json();
   367→          window.dispatchEvent(new CustomEvent('dokupro:history-updated'));
   368→
   369→          // If invoice has DP, also create an invoice-pelunasan entry
   370→          if (invoice.dp > 0) {
   371→            try {
   372→              // Derive PEL nomor from INV nomor (same number, different prefix)
   373→              const invNomor = saved.nomor || invoice.nomor;
   374→              const pelNomor = invNomor.replace(/^INV/, 'PEL');
   375→
   376→              const pelunasanData = {
   377→                ...dataToSave,
   378→                type: 'invoice-pelunasan',
   379→                referensiInvoiceId: saved.id,
   380→                referensiInvoiceNomor: invNomor,
   381→                lunas: false,
   382→                tanggalPelunasan: '',
   383→              };
   384→              await fetch('/api/history', {
   385→                method: 'POST',
   386→                headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   387→                body: JSON.stringify({
   388→                  docType: 'invoice-pelunasan',
   389→                  customNomor: pelNomor,
   390→                  tanggal: invoice.tanggal || '',
   391→                  pihakKedua: invoice.client?.nama || '-',
   392→                  total: '-',
   393→                  dataJson: JSON.stringify(pelunasanData),
   394→                }),
   395→              });
   396→            } catch {
   397→              console.error('Failed to create pelunasan invoice entry');
   398→            }
   399→          }
   400→
   401→          toast.success('Invoice disimpan ke riwayat');
   402→          // Reset invoice form after saving
   403→          resetDocument('invoice');
   404→          router.push(`/surat-jalan?invoiceId=${saved.id}`);
   405→        } else if (res.status === 409) {
   406→          // Invoice already saved, get existing ID and navigate
   407→          const errData = await res.json().catch(() => ({}));
   408→          if (errData.id) {
   409→            resetDocument('invoice');
   410→            router.push(`/surat-jalan?invoiceId=${errData.id}`);
   411→          } else {
   412→            toast('Invoice sudah ada di riwayat.');
   413→          }
   414→        } else {
   415→          toast.error('Gagal menyimpan invoice');
   416→        }
   417→      }
   418→    } catch {
   419→      toast.error('Gagal menyimpan invoice');
   420→    }
   421→    setSavingSj(false);
   422→  };
   423→
   424→  return (
   425→    <>
   426→      <DocumentEditorLayout
   427→        title="Invoice"
   428→        previewContent={<InvoicePreview data={invoice} />}
   429→        actions={
   430→          <div className="flex items-center gap-2 flex-wrap">
   431→            {invoiceEditingId && (
   432→              <span className="inline-flex items-center gap-1 px-2.5 py-1 rounded-full bg-amber-100 text-amber-700 text-xs font-semibold border border-amber-200">
   433→                <svg className="w-3 h-3" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M11 5H6a2 2 0 00-2 2v11a2 2 0 002 2h11a2 2 0 002-2v-5m-1.414-9.414a2 2 0 112.828 2.828L11.828 15H9v-2.828l8.586-8.586z" /></svg>
   434→                Mode Edit
   435→              </span>
   436→            )}
   437→            <DocumentActionButtons
   438→              docType="invoice"
   439→              documentLabel="Invoice"
   440→              currentData={invoice}
   441→              onReset={() => resetDocument('invoice')}
   442→              editingId={invoiceEditingId}
   443→              onUpdateSuccess={() => setInvoiceEditingId(null)}
   444→            />
   445→            <Button
   446→              size="sm"
   447→              onClick={handleSuratJalan}
   448→              disabled={savingSj}
   449→              className="bg-orange-600 hover:bg-orange-700 h-8 sm:h-9"
   450→            >
   451→              <Truck className="mr-1.5 h-3.5 w-3.5" />
   452→              {savingSj ? 'Menyimpan...' : 'Surat Jalan'}
   453→            </Button>
   454→          </div>
   455→        }
   456→      >
   457→        <CompanyFields company={invoice.company} onChange={updateCompany} />
   458→
   459→        <div className="rounded-lg border bg-card p-3 sm:p-4 shadow-sm">
   460→          <h3 className="mb-3 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   461→            Detail Dokumen
   462→          </h3>
   463→          <div className="grid grid-cols-2 gap-2">
   464→            <div className="space-y-1.5">
   465→              <Label className="text-xs">No. Invoice</Label>
   466→              <Input
   467→                value={invoice.nomor}
   468→                readOnly
   469→                className="bg-slate-50 text-slate-500 cursor-not-allowed"
   470→              />
   471→            </div>
   472→            <div className="space-y-1.5">
   473→              <Label className="text-xs">Tanggal</Label>
   474→              <Input
   475→                type="date"
   476→                value={invoice.tanggal}
   477→                onChange={(e) => setInvoice((prev) => ({ ...prev, tanggal: e.target.value }))}
   478→              />
   479→            </div>
   480→          </div>
   481→          <div className="space-y-2">
   482→            <Label className="text-xs">Referensi (No. HC)</Label>
   483→            <Popover open={dropdownOpen} onOpenChange={setDropdownOpen}>
   484→              <PopoverAnchor asChild>
   485→                <button
   486→                  type="button"
   487→                  onClick={() => setDropdownOpen(true)}
   488→                  className="flex h-9 w-full items-center justify-between rounded-md border border-input bg-slate-50 px-3 py-2 text-sm shadow-xs outline-none cursor-pointer hover:bg-slate-100 transition-colors text-left"
   489→                >
   490→                  <span className={referensiInput ? 'text-slate-900' : 'text-slate-400'}>
   491→                    {referensiInput || 'Pilih No. HC...'}
   492→                  </span>
   493→                  <svg className="w-4 h-4 text-slate-400 shrink-0" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M19 9l-7 7-7-7" /></svg>
   494→                </button>
   495→              </PopoverAnchor>
   496→              <PopoverContent
   497→                align="start"
   498→                className="p-0 w-[var(--radix-popover-trigger-width)] max-h-52 overflow-y-auto"
   499→                onOpenAutoFocus={(e) => e.preventDefault()}
   500→              >
   501→                {filteredRiwayatList.length > 0 && (
   502→                  <div>
   503→                    <div className="px-3 py-1.5 text-[10px] font-semibold text-slate-400 uppercase bg-slate-50 border-b border-slate-100 sticky top-0">Riwayat Hitung Cetakan</div>
   504→                    {filteredRiwayatList.map((r) => (
   505→                      <button
   506→                        key={r.id}
   507→                        type="button"
   508→                        onMouseDown={(e) => { e.preventDefault(); handleReferensiSelect(r) }}
   509→                        className={`w-full text-left px-3 py-2 text-sm transition-colors ${invoice.referensi === r.nomorUrut ? 'bg-blue-50 text-blue-700 font-medium' : 'text-slate-700 hover:bg-blue-50 hover:text-blue-700'}`}
   510→                      >
   511→                        <span className="font-semibold text-xs text-emerald-700">{r.nomorUrut || '-'}</span>
   512→                        <span className="ml-1.5 truncate">{r.printName || ''}</span>
   513→                        {r.customerName && <span className="text-slate-400 ml-1.5 text-[11px]">({r.customerName})</span>}
   514→                      </button>
   515→                    ))}
   516→                  </div>
   517→                )}
   518→                {filteredRiwayatList.length === 0 && (
   519→                  <div className="px-3 py-3 text-sm text-slate-400 text-center">Tidak ada riwayat cetakan</div>
   520→                )}
   521→              </PopoverContent>
   522→            </Popover>
   523→          </div>
   524→        </div>
   525→
   526→        {/* Informasi Pembayaran - moved below Detail Dokumen */}
