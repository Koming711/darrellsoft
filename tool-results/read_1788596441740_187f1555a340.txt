    16→// === MODULE-LEVEL SESSION CACHE ===
    17→// Persists across navigations so DashboardLayout doesn't need to re-verify on every mount
    18→interface CachedSession {
    19→  user: any
    20→  userProfile: { createdAt: string | null; validUntil: string | null } | null
    21→  autoLogoutMin: number
    22→  logoutWarningSec: number
    23→  sessionWarning: string | null
    24→  forceLogoutAvailable: boolean
    25→  accountExpired: boolean
    26→  cachedAt: number
    27→}
    28→let _sessionCache: CachedSession | null = null
    29→const SESSION_CACHE_TTL = 120_000 // 2 minutes
    30→
    31→function getSessionCache(): CachedSession | null {
    32→  // Check module-level cache first
    33→  if (_sessionCache && Date.now() - _sessionCache.cachedAt <= SESSION_CACHE_TTL) {
    34→    return _sessionCache
    35→  }
    36→  // Fallback to sessionStorage
    37→  if (typeof window !== 'undefined') {
    38→    try {
    39→      const stored = sessionStorage.getItem('__dashboard_session_cache')
    40→      if (stored) {
    41→        const parsed = JSON.parse(stored) as CachedSession
    42→        if (Date.now() - parsed.cachedAt <= SESSION_CACHE_TTL) {
    43→          _sessionCache = parsed
    44→          return parsed
    45→        }
    46→        sessionStorage.removeItem('__dashboard_session_cache')
    47→      }
    48→    } catch {}
    49→  }
    50→  _sessionCache = null
    51→  return null
    52→}
    53→
    54→function setSessionCache(data: Omit<CachedSession, 'cachedAt'>) {
    55→  const entry = { ...data, cachedAt: Date.now() }
    56→  _sessionCache = entry
    57→  if (typeof window !== 'undefined') {
    58→    try {
    59→      sessionStorage.setItem('__dashboard_session_cache', JSON.stringify(entry))
    60→    } catch {}
    61→  }
    62→}
    63→
    64→function invalidateSessionCache() {
    65→  _sessionCache = null
    66→  if (typeof window !== 'undefined') {
    67→    try {
    68→      sessionStorage.removeItem('__dashboard_session_cache')
    69→    } catch {}
    70→  }
    71→}

    85→  const pathname = usePathname()

    87→  const { t } = useLanguage()
    88→  const { collapsed: sidebarCollapsed } = useSidebarCollapse()
    89→  // Desktop content left-margin follows the sidebar width (w-52 expanded / w-14 collapsed)
    90→  const desktopMargin = sidebarCollapsed ? 'lg:ml-14' : 'lg:ml-52'
    91→
    92→  // === SESSION CHECK STATE ===
    93→  const [sessionWarning, setSessionWarning] = useState<string | null>(null)
    94→  const [forceLogoutAvailable, setForceLogoutAvailable] = useState(false)
    95→  const [isReclaiming, setIsReclaiming] = useState(false)
    96→  const [accountExpired, setAccountExpired] = useState(false)
    97→  const sessionIntervalRef = useRef<ReturnType<typeof setInterval> | null>(null)
    98→
    99→  // === AUTO-LOGOUT STATE ===
   100→  const lastActivityRef = useRef(Date.now())
   101→  const countdownActiveRef = useRef(false)
   102→  const [autoLogoutMin, setAutoLogoutMin] = useState(0)
   103→  const [logoutWarningSec, setLogoutWarningSec] = useState(0)
   104→  const [showCountdown, setShowCountdown] = useState(false)
   105→  const [countdown, setCountdown] = useState(0)
   106→  const autoLogoutIntervalRef = useRef<ReturnType<typeof setInterval> | null>(null)
   107→
   108→  // === NO ACCESS STATE ===
   109→  const [noAccess, setNoAccess] = useState(false)
   110→
   111→  // === INIT GUARD (prevent double init in StrictMode) ===
   112→  const initDoneRef = useRef(false)
   113→
   114→  // === USER PROFILE (for header dates) ===
   115→  const [userProfile, setUserProfile] = useState<{ createdAt: string | null; validUntil: string | null } | null>(null)
   116→
   117→  // === PERMISSION VERSION (forces re-render of Sidebar when permissions update) ===
   118→  const [permVersion, setPermVersion] = useState(0)
   119→
   120→  // === INSTANT RESTORE FROM CACHE (useLayoutEffect runs before browser paint) ===
   121→  // This eliminates the loading spinner flash on client-side navigation.
   122→  // On SSR, cache is null so this is a no-op (spinner shows on first load as expected).
   123→  useLayoutEffect(() => {
   124→    const cache = getSessionCache()
   125→    if (cache) {
   126→      setUser(cache.user)
   127→      setSessionWarning(cache.sessionWarning ?? null)
   128→      setForceLogoutAvailable(cache.forceLogoutAvailable ?? false)
   129→      setAccountExpired(cache.accountExpired ?? false)
   130→      setAutoLogoutMin(cache.autoLogoutMin ?? 0)
   131→      setLogoutWarningSec(cache.logoutWarningSec ?? 0)
   132→      setUserProfile(cache.userProfile ?? null)
   133→      setReady(true)
   134→    }
   135→  }, [])
   136→
   137→  // Listen for permission updates from hak-akses page (immediate re-render)
   138→  useEffect(() => {
   139→    const handler = () => setPermVersion(v => v + 1)
   140→    window.addEventListener('permissions-updated', handler)
   141→    return () => window.removeEventListener('permissions-updated', handler)
   142→  }, [])
   143→
   144→  // === LOGOUT HANDLER ===
   145→  const handleLogout = useCallback(() => {
   146→    clearAuthUser()
   147→    invalidateSessionCache()
   148→    setAccountExpired(false)
   149→    setSessionWarning(null)
   150→    setForceLogoutAvailable(false)
   151→    if (sessionIntervalRef.current) clearInterval(sessionIntervalRef.current)
   152→    if (autoLogoutIntervalRef.current) clearInterval(autoLogoutIntervalRef.current)
   153→    sessionIntervalRef.current = null
   154→    autoLogoutIntervalRef.current = null
   155→    window.location.href = '/'
   156→  }, [])
   157→
   158→  // Stay logged in handler (resets activity + dismisses countdown)
   159→  const handleStayLoggedIn = useCallback(() => {
   160→    lastActivityRef.current = Date.now()
   161→    countdownActiveRef.current = false
   162→    setShowCountdown(false)
   163→    setCountdown(0)
   164→  }, [])
   165→
   166→  // === INITIAL AUTH + SETTINGS LOAD (runs once) ===
   167→  useEffect(() => {
   168→    if (initDoneRef.current) return
   169→    initDoneRef.current = true
   170→
   171→    // If cache was already restored by useLayoutEffect, skip the full init.
   172→    // Just do a background refresh to keep session data up-to-date.
   173→    const existingCache = getSessionCache()
   174→    if (existingCache) {
   175→      // State was already restored by useLayoutEffect — just do background refresh
   176→      // Do a background session refresh without blocking UI
   177→      const bgRefresh = async () => {
   178→        const authUser = getAuthUser()
   179→        if (!authUser) return
   180→        try {
   181→          const sessionRes = await authFetch('/api/auth/verify-session', {
   182→            method: 'POST',
   183→            headers: { 'Content-Type': 'application/json' },
   184→            body: JSON.stringify({ username: authUser.username, sessionId: authUser.sessionId, role: authUser.role }),
   185→          })
   186→          const data = await sessionRes.json()
   187→          if (data.securitySettings) {
   188→            setAutoLogoutMin(data.securitySettings.auto_logout_min || 0)
   189→            setLogoutWarningSec(data.securitySettings.logout_warning_sec || 0)
   190→          }
   191→          if (data.permissions && authUser.role) {
   192→            saveRolePermissions(authUser.role, data.permissions.features, data.permissions.subPermissions)
   193→            setPermVersion(v => v + 1)
   194→          }
   195→          if (!data.valid) {
   196→            if (data.expired) {
   197→              setAccountExpired(true)
   198→              setSessionWarning(data.warningMessage || t('akun_sudah_expired'))
   199→            } else if (data.warningMessage) {
   200→              setSessionWarning(data.warningMessage)
   201→              setForceLogoutAvailable(!!data.forceLogoutAvailable)
   202→            }
   203→          }
   204→          // Update cache with fresh data
   205→          setSessionCache({
   206→            user: authUser,
   207→            userProfile: existingCache.userProfile,
   208→            autoLogoutMin: data.securitySettings?.auto_logout_min || 0,
   209→            logoutWarningSec: data.securitySettings?.logout_warning_sec || 0,
   210→            sessionWarning: data.valid ? null : (data.warningMessage || null),
   211→            forceLogoutAvailable: data.valid ? false : !!data.forceLogoutAvailable,
   212→            accountExpired: data.valid ? false : !!data.expired,
   213→          })
   214→        } catch {
   215→          // Background refresh failed — keep using cached data
   216→        }
   217→      }
   218→      bgRefresh()
   219→      return
   220→    }
   221→
   222→    let cancelled = false
   223→
   224→    const init = async () => {

   228→          if (!cancelled) setReady(true)
   229→          return
   230→        }

   238→              if (data?.expired && !cancelled) {
   239→                setAccountExpired(true)
   240→                setSessionWarning(data.error || t('akun_expired_perpanjang'))
   241→              }
   242→              return null
   243→            })
   244→            return r.ok ? r.json() : null
   245→          })
   246→          .then(data => { if (data && !cancelled) {
   247→            setUserProfile({ createdAt: data.createdAt || null, validUntil: data.validUntil || null })
   248→            // Update cache with profile data
   249→            const currentCache = getSessionCache()
   250→            if (currentCache) {
   251→              setSessionCache({ ...currentCache, userProfile: { createdAt: data.createdAt || null, validUntil: data.validUntil || null } })
   252→            }
   253→          } })
   254→          .catch(() => {})
   255→
   256→        // Wait for session verification BEFORE checking permissions
   257→        try {
   258→          const sessionRes = await authFetch('/api/auth/verify-session', {
   259→            method: 'POST',
   260→            headers: { 'Content-Type': 'application/json' },
   261→            body: JSON.stringify({ username: authUser.username, sessionId: authUser.sessionId, role: authUser.role }),
   262→          })
   263→          if (cancelled) return
   264→          const data = await sessionRes.json()
   265→          if (cancelled) return
   266→
   267→          // Load security settings from verify-session response
   268→          if (data.securitySettings) {
   269→            setAutoLogoutMin(data.securitySettings.auto_logout_min || 0)
   270→            setLogoutWarningSec(data.securitySettings.logout_warning_sec || 0)
   271→          }
   272→          // Sync latest permissions from server to localStorage
   273→          if (data.permissions && authUser.role) {
   274→            saveRolePermissions(authUser.role, data.permissions.features, data.permissions.subPermissions)
   275→            setPermVersion(v => v + 1)
   276→          }
   277→          if (!data.valid) {
   278→            if (data.expired) {
   279→              setAccountExpired(true)
   280→              setSessionWarning(data.warningMessage || t('akun_sudah_expired'))
   281→            } else if (data.warningMessage) {
   282→              setSessionWarning(data.warningMessage)
   283→              setForceLogoutAvailable(!!data.forceLogoutAvailable)
   284→            }
   285→          }
   286→
   287→          // Update session cache with verified data
   288→          setSessionCache({
   289→            user: authUser,
   290→            userProfile: null,
   291→            autoLogoutMin: data.securitySettings?.auto_logout_min || 0,
   292→            logoutWarningSec: data.securitySettings?.logout_warning_sec || 0,
   293→            sessionWarning: data.valid ? null : (data.warningMessage || null),
   294→            forceLogoutAvailable: data.valid ? false : !!data.forceLogoutAvailable,
   295→            accountExpired: data.valid ? false : !!data.expired,
   296→          })
   297→
   298→          // === AUTO-BACKUP CHECK ===
   299→          // Check if auto-backup is due and trigger it silently
   300→          if (data.valid && authUser.role !== 'superadmin') {
   301→            authFetch('/api/database/auto-backup', { method: 'POST' })
   302→              .then(r => r.ok ? r.json() : null)
   303→              .then(backupData => {
   304→                if (backupData?.success && !backupData?.skipped) {
   305→                  console.log(`✅ Auto backup: ${backupData.fileName} (${backupData.totalRows} records)`)
   306→                }
   307→              })
   308→              .catch(() => {}) // silent fail
   309→          }
   310→        } catch {
   311→          // Session verification failed - continue anyway
   312→        }
   313→
   314→      } catch {
   315→        // On error, don't redirect — just show the page
   316→      } finally {
   317→        if (!cancelled) {
   318→          setReady(true)
   319→          // Cache session data for instant navigation
   320→          const authUser = getAuthUser()
   321→          if (authUser) {
   322→            setSessionCache({
   323→              user: authUser,
   324→              userProfile: null, // will be updated by authFetch below
   325→              autoLogoutMin: 0,
   326→              logoutWarningSec: 0,
   327→              sessionWarning: null,
   328→              forceLogoutAvailable: false,
   329→              accountExpired: false,
   330→            })
   331→          }
   332→        }
   333→      }
   334→    }
   335→

   339→  }, [])

   384→          setAutoLogoutMin(data.securitySettings.auto_logout_min || 0)
   385→          setLogoutWarningSec(data.securitySettings.logout_warning_sec || 0)
   386→        }
   387→      })
   388→      .catch(() => {})
   389→    }, 10000)

   391→    return () => {
   392→      if (sessionIntervalRef.current) {
   393→        clearInterval(sessionIntervalRef.current)
   394→        sessionIntervalRef.current = null
   395→      }
   396→    }
   397→  }, [ready, user])

   399→  // === ACTIVITY LISTENERS ===

   425→        setCountdown(Math.ceil(autoLogoutSec - elapsed))

   534→  }

   536→  // === RENDER ===
   537→  return (
   538→    <div className="min-h-screen" style={{ backgroundColor: 'var(--app-content-bg)' }}>
   539→      <Sidebar
   540→        username={user?.username || 'User'}
   541→        role={user?.role}
   542→        onLogout={handleLogout}
   543→        permVersion={permVersion}
   544→      />
   545→
   546→      <div className={`${desktopMargin} transition-all duration-300`}>
   547→        <MobileHeader
   548→          username={user?.username}
   549→          role={user?.role}
   550→          title={title}
   551→          subtitle={subtitle}
   552→          userProfile={userProfile}
   553→        />
   554→
   555→        {/* Main Content — extra bottom padding on mobile for bottom nav + safe area */}
   556→        <main className="p-4 pb-20 lg:p-8 lg:pb-8">
   557→          {children}
   558→        </main>
   559→      </div>
   560→
   561→      {/* ===== Mobile Bottom Navigation ===== */}
   562→      <MobileBottomNav role={user?.role} onMoreClick={() => setSidebarOpen(true)} username={user?.username} onLogout={handleLogout} />
   563→
   564→      {/* ===== MODAL: ACCOUNT EXPIRED ===== */}
   565→      {accountExpired && sessionWarning && (
   566→        <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/50 backdrop-blur-sm px-4">
   567→          <div className="rounded-2xl shadow-2xl border border-red-200 dark:border-red-800 max-w-sm w-full p-6 animate-in fade-in zoom-in duration-200" style={{ backgroundColor: 'var(--card)' }}>
   568→            <div className="flex items-start gap-3 mb-4">
   569→              <div className="w-10 h-10 rounded-xl bg-red-100 dark:bg-red-900/30 flex items-center justify-center flex-shrink-0">
   570→                <TimerOff className="w-5 h-5 text-red-600 dark:text-red-400" />
   571→              </div>
   572→              <div>
   573→                <h3 className="text-lg font-bold text-foreground">{t('account_expired')}</h3>
   574→                <p className="text-xs text-muted-foreground mt-0.5">{t('account_expired_desc')}</p>
   575→              </div>
   576→            </div>
   577→            <div className="bg-red-50 dark:bg-red-950 border border-red-200 dark:border-red-800 rounded-xl p-4 mb-5">
   578→              <p className="text-sm text-red-700 dark:text-red-400 leading-relaxed">{sessionWarning}</p>
   579→            </div>
   580→            <button
   581→              onClick={handleLogout}
   582→              className="w-full bg-red-600 hover:bg-red-700 text-white font-semibold py-2.5 rounded-xl transition-colors"
   583→              autoFocus
   584→            >
   585→              <span className="flex items-center justify-center gap-2">
   586→                <LogOut className="w-4 h-4" /> {t('ok_mengerti')}
   587→              </span>
   588→            </button>
   589→          </div>
   590→        </div>
   591→      )}
   592→
   593→      {/* ===== MODAL: SESSION WARNING (Multi-Device) ===== */}
   594→      {sessionWarning && !accountExpired && (
   595→        <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/50 backdrop-blur-sm px-4">
   596→          <div className="rounded-2xl shadow-2xl border border-red-200 dark:border-red-800 max-w-sm w-full p-6 animate-in fade-in zoom-in duration-200" style={{ backgroundColor: 'var(--card)' }}>
   597→            <div className="flex items-start gap-3 mb-4">
   598→              <div className="w-10 h-10 rounded-xl bg-red-100 dark:bg-red-900/30 flex items-center justify-center flex-shrink-0">
   599→                {forceLogoutAvailable ? <ShieldAlert className="w-5 h-5 text-red-600 dark:text-red-400" /> : <Smartphone className="w-5 h-5 text-red-600 dark:text-red-400" />}
   600→              </div>
   601→              <div>
   602→                <h3 className="text-lg font-bold text-foreground">{t('peringatan_keamanan')}</h3>
   603→                <p className="text-xs text-muted-foreground mt-0.5">{t('akun_digunakan_di_perangkat_lain')}</p>
   604→              </div>
   605→            </div>
   606→            <div className="bg-red-50 dark:bg-red-950 border border-red-200 dark:border-red-800 rounded-xl p-4 mb-5">
   607→              <p className="text-sm text-red-700 dark:text-red-400 leading-relaxed">{sessionWarning}</p>
   608→            </div>
   609→            {forceLogoutAvailable ? (
   610→              <div className="space-y-2.5">
   611→                <button
   612→                  onClick={async () => {
   613→                    setIsReclaiming(true)
   614→                    try {
   615→                      const authUser = getAuthUser()
   616→                      if (!authUser?.sessionId) {
   617→                        toast.error(t('session_data_not_found'))
   618→                        setIsReclaiming(false)
   619→                        handleLogout()
   620→                        return
   621→                      }
   622→                      const res = await authFetch('/api/auth/reclaim-session', {
   623→                        method: 'POST',
   624→                        headers: { 'Content-Type': 'application/json' },
   625→                        body: JSON.stringify({ username: authUser.username, sessionId: authUser.sessionId, userId: authUser.id }),
   626→                      })
   627→                      const data = await res.json()
   628→                      if (res.ok) {
   629→                        setSessionWarning(null)
   630→                        setForceLogoutAvailable(false)
   631→                        toast.success(t('other_device_logged_out'))
   632→                      } else {
   633→                        toast.error(data.error || t('reclaim_session_failed'))
   634→                      }
   635→                    } catch {
   636→                      toast.error(t('terjadi_kesalahan_jaringan'))
   637→                    }
   638→                    setIsReclaiming(false)
   639→                  }}
   640→                  disabled={isReclaiming}
   641→                  className="w-full flex items-center justify-center gap-2 bg-amber-500 hover:bg-amber-600 disabled:bg-amber-400 text-white font-semibold py-2.5 rounded-xl transition-colors"
   642→                  autoFocus
   643→                >
   644→                  {isReclaiming ? (
   645→                    <><div className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin" /> {t('memproses')}</>
   646→                  ) : (
   647→                    <><ShieldAlert className="w-4 h-4" /> {t('paksa_logout_perangkat_lain')}</>
   648→                  )}
   649→                </button>
   650→                <button
   651→                  onClick={handleLogout}
   652→                  className="w-full flex items-center justify-center gap-2 border-2 border-red-200 dark:border-red-800 text-red-600 dark:text-red-400 hover:bg-red-50 dark:hover:bg-red-950 font-semibold py-2.5 rounded-xl transition-colors"
   653→                >
   654→                  <LogOut className="w-4 h-4" /> {t('logout_dari_sini')}
   655→                </button>
   656→              </div>
   657→            ) : (
   658→              <button
   659→                onClick={handleLogout}
   660→                className="w-full bg-red-600 hover:bg-red-700 text-white font-semibold py-2.5 rounded-xl transition-colors"
   661→                autoFocus
   662→              >
   663→                {t('ok_logout')}
   664→              </button>
   665→            )}
   666→          </div>
   667→        </div>
   668→      )}
   669→
   670→      {/* ===== MODAL: AUTO-LOGOUT COUNTDOWN ===== */}
   671→      {showCountdown && (
   672→        <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/50 backdrop-blur-sm px-4">
   673→          <div className="rounded-2xl shadow-2xl border border-amber-200 dark:border-amber-800 max-w-sm w-full p-6 animate-in fade-in zoom-in duration-200" style={{ backgroundColor: 'var(--card)' }}>
   674→            <div className="flex items-start gap-3 mb-4">
   675→              <div className="w-10 h-10 rounded-xl bg-amber-100 dark:bg-amber-900/30 flex items-center justify-center flex-shrink-0">
   676→                <AlertTriangle className="w-5 h-5 text-amber-600 dark:text-amber-400" />
   677→              </div>
   678→              <div>
   679→                <h3 className="text-lg font-bold text-foreground">{t('session_ending')}</h3>
   680→                <p className="text-xs text-muted-foreground mt-0.5">{t('session_ending_desc')}</p>
   681→              </div>
   682→            </div>
   683→            <div className="text-center py-5">
   684→              <div className="text-6xl font-bold text-amber-600 dark:text-amber-400 tabular-nums">{countdown}</div>
   685→              <p className="text-sm text-muted-foreground mt-2">{t('detik_tersisa')}</p>
   686→            </div>
   687→            <button
   688→              onClick={handleStayLoggedIn}
   689→              className="w-full bg-amber-500 hover:bg-amber-600 text-white font-semibold py-2.5 rounded-xl transition-colors"
   690→              autoFocus
   691→            >
   692→              {t('tetap_login')}
   693→            </button>
   694→          </div>
   695→        </div>
   696→      )}
   697→
   698→      {/* ===== MODAL: COMPANY DATA POPUP (untuk demo user baru dari checkout) ===== */}
   699→      {user && <CompanyDataPopup />}
   700→    </div>
   701→  )
   702→}
