    48→    48→function getTodayStr(): string {
    49→    49→  const d = new Date();
    50→    50→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
    51→    51→}

    53→    53→// --- Parse dataJson for document info ---
    54→    54→function parseDocInfo(entry: HistoryEntry) {
    55→    55→  try {
    56→    56→    const parsed = JSON.parse(entry.dataJson);
    57→    57→    const items = parsed.items || [];
    58→    58→    const firstItem = items[0];
    59→    59→    const namaBarang = firstItem?.deskripsi || '';
    60→    60→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0);
    61→    61→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0);
    62→    62→    const ppn = parsed.ppn || 0;
    63→    63→    const dpPercent = parsed.dp || 0;
    64→    64→    const totalHarga = subtotal + (subtotal * ppn / 100);
    65→    65→    // Use saved dpAmount if available
    66→    66→    // If not, use originalTotal * dpPercent to get the correct original DP (not affected by pelunasan additions)
    67→    67→    // Only fall back to totalHarga * dpPercent if originalTotal is also missing
    68→    68→    const originalTotal = parsed.originalTotal !== undefined ? parsed.originalTotal : totalHarga;
    69→    69→    const dpAmount = parsed.dpAmount !== undefined ? parsed.dpAmount : originalTotal * (dpPercent / 100);
    70→    70→    const sisa = totalHarga - dpAmount;
    71→    71→    const lunas = parsed.lunas === true;
    72→    72→    const tanggalJatuhTempo = parsed.tanggalJatuhTempo || '';
    73→    73→    const tanggalPelunasan = parsed.tanggalPelunasan || '';
    74→    74→    return { namaBarang, totalQty, totalHarga, dpPercent, dp: dpAmount, sisa, lunas, tanggalJatuhTempo, tanggalPelunasan };
    75→    75→  } catch {
    76→    76→    return { namaBarang: '', totalQty: 0, totalHarga: 0, dpPercent: 0, dp: 0, sisa: 0, lunas: false, tanggalJatuhTempo: '', tanggalPelunasan: '' };
    77→    77→  }
    78→    78→}

    80→    80→// --- Parse dataJson for InvoiceData ---
    81→    81→function parseInvoiceData(entry: HistoryEntry): InvoiceData {
    82→    82→  try {
    83→    83→    const parsed = JSON.parse(entry.dataJson);
    84→    84→    const company: CompanyInfo = {
    85→    85→      nama: parsed.company?.nama || DEFAULT_COMPANY.nama,
    86→    86→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
    87→    87→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
    88→    88→      email: parsed.company?.email || DEFAULT_COMPANY.email,
    89→    89→      npwp: parsed.company?.npwp || '',
    90→    90→      website: parsed.company?.website || '',
    91→    91→      ppn: parsed.company?.ppn ?? parsed.ppn ?? 11,
    92→    92→      logo: parsed.company?.logo || '',
    93→    93→      bankName: parsed.company?.bankName || '',
    94→    94→      bankAccount: parsed.company?.bankAccount || '',
    95→    95→      bankHolder: parsed.company?.bankHolder || '',
    96→    96→      bankName2: parsed.company?.bankName2 || '',
    97→    97→      bankAccount2: parsed.company?.bankAccount2 || '',
    98→    98→      bankHolder2: parsed.company?.bankHolder2 || '',
    99→    99→    };
   100→   100→    const client = parsed.client || { nama: '', kontak: '', alamat: '' };
   101→   101→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number }, i: number) => ({
   102→   102→      id: it.id || `item-${i}`,
   103→   103→      deskripsi: it.deskripsi || '',
   104→   104→      qty: it.qty || 0,
   105→   105→      satuan: it.satuan || '',
   106→   106→      harga: it.harga || 0,
   107→   107→    }));
   108→   108→    return {
   109→   109→      type: 'invoice',
   110→   110→      company,
   111→   111→      nomor: parsed.nomor || entry.nomor || '',
   112→   112→      tanggal: parsed.tanggal || entry.tanggal || '',
   113→   113→      referensi: parsed.referensi || '',
   114→   114→      client,
   115→   115→      items,
   116→   116→      ppn: parsed.ppn ?? 11,
   117→   117→      dp: parsed.dp || 0,
   118→   118→      dpAmount: parsed.dpAmount,
   119→   119→      originalTotal: parsed.originalTotal,
   120→   120→      catatan: parsed.catatan || '',
   121→   121→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   122→   122→      caraPembayaran: parsed.caraPembayaran || '',
   123→   123→      tanggalGiro: parsed.tanggalGiro || '',
   124→   124→      lunas: parsed.lunas === true,
   125→   125→      tanggalPelunasan: parsed.tanggalPelunasan || '',
   126→   126→    };
   127→   127→  } catch {
   128→   128→    return {
   129→   129→      type: 'invoice',
   130→   130→      company: { ...DEFAULT_COMPANY },
   131→   131→      nomor: entry.nomor || '',
   132→   132→      tanggal: entry.tanggal || '',
   133→   133→      referensi: '',
   134→   134→      client: { nama: '', kontak: '', alamat: '' },
   135→   135→      items: [],
   136→   136→      ppn: 11,
   137→   137→      dp: 0,
   138→   138→      catatan: '',
   139→   139→      tanggalJatuhTempo: '',
   140→   140→      caraPembayaran: '',
   141→   141→      tanggalGiro: '',
   142→   142→      lunas: false,
   143→   143→      tanggalPelunasan: '',
   144→   144→    };
   145→   145→  }
   146→   146→}

   290→   290→        caraPembayaran: '',

   308→   308→  const ppnAmount = subtotal * ((invoiceData?.ppn || 0) / 100);

   310→   310→  const dpPercent = invoiceData?.dp || 0;
   311→   311→  // DP amount stays fixed at original value, doesn't recalculate when items change

   390→   390→      } else {
   391→   391→        parsed.lunas = false;
   392→   392→        parsed.tanggalPelunasan = '';
   393→   393→      }
   394→   394→      // Also update any edited fields from the form
   395→   395→      parsed.company = invoiceData.company;
   396→   396→      parsed.client = invoiceData.client;
   397→   397→      parsed.items = invoiceData.items;
   398→   398→      parsed.ppn = invoiceData.ppn;
   399→   399→      parsed.dp = invoiceData.dp;
   400→   400→      // Save fixed DP amount so it doesn't change on restore
   401→   401→      parsed.dpAmount = originalDpAmount;
   402→   402→      parsed.catatan = invoiceData.catatan;
   403→   403→      parsed.nomor = invoiceData.nomor;
