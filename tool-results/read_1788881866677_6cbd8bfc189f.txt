    61→/** Format tanggal pendek gaya arsip: "07 Sep 2026" */
    62→function formatDate(iso: string | Date | null | undefined): string {
    63→  if (!iso) return '-'
    64→  const d = new Date(iso)
    65→  if (isNaN(d.getTime())) return '-'
    66→  return d.toLocaleDateString('id-ID', { day: '2-digit', month: 'short', year: 'numeric' })
    67→}
    68→
    69→/** Format tanggal + jam untuk kop cetak (arsip formatDateTime) */
    70→function formatDateTime(iso: string | Date | null | undefined): string {
    71→  if (!iso) return '-'
    72→  const d = new Date(iso)
    73→  if (isNaN(d.getTime())) return '-'
    74→  return d.toLocaleString('id-ID', {
    75→    day: '2-digit', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit',
    76→  })
    77→}
    78→
    79→/** ISO string -> yyyy-mm-dd untuk <input type="date"> (arsip toInputDate) */
    80→function toInputDate(iso: string | Date | null | undefined): string {
    81→  if (!iso) return ''
    82→  const d = new Date(iso)
    83→  if (isNaN(d.getTime())) return ''
    84→  const pad = (n: number) => String(n).padStart(2, '0')
    85→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`
    86→}

   156→function yearOptions(): number[] {

   160→  return out

   296→/** Teks periode untuk subjudul & kop cetak (arsip periodText). */
   297→function periodText(
   298→  period: ReportPeriod,
   299→  from: string,
   300→  to: string,
   301→  month: number | null,
   302→  year: number | null
   303→): string {
   304→  const now = new Date()
   305→  const m = month ?? now.getMonth() + 1
   306→  const y = year ?? now.getFullYear()
   307→  switch (period) {
   308→    case 'today':
   309→      return formatDate(now)
   310→    case 'week': {
   311→      const day = (now.getDay() + 6) % 7 // 0 = Senin
   312→      const start = new Date(now.getFullYear(), now.getMonth(), now.getDate() - day)
   313→      const end = new Date(start.getFullYear(), start.getMonth(), start.getDate() + 6)
   314→      return `${formatDate(start)} – ${formatDate(end)}`
   315→    }
   316→    case 'date':
   317→      return `${formatDate(from)} – ${formatDate(to)}`
   318→    case 'month':
   319→      return `${MONTH_LABELS[m - 1] ?? ''} ${y}`
   320→    case 'year':
   321→      return String(y)
   322→    default:
   323→      return 'Semua Periode'
   324→  }
   325→}

   444→function buildInvoiceData(parsed: Record<string, unknown>, fallbackNomor: string, fallbackTanggal: string): InvoiceData {
   445→  const company: CompanyInfo = {
   446→    nama: (parsed.company as CompanyInfo | undefined)?.nama || 'Darrell Soft',
   447→    telepon: (parsed.company as CompanyInfo | undefined)?.telepon || '',
   448→    alamat: (parsed.company as CompanyInfo | undefined)?.alamat || '',
   449→    email: (parsed.company as CompanyInfo | undefined)?.email || '',
   450→    npwp: (parsed.company as CompanyInfo | undefined)?.npwp || '',
   451→    website: (parsed.company as CompanyInfo | undefined)?.website || '',
   452→    ppn: (parsed.company as CompanyInfo | undefined)?.ppn ?? (parsed.ppn as number | undefined) ?? 11,
   453→    logo: (parsed.company as CompanyInfo | undefined)?.logo || '',
   454→    bankName: (parsed.company as CompanyInfo | undefined)?.bankName || '',
   455→    bankAccount: (parsed.company as CompanyInfo | undefined)?.bankAccount || '',
   456→    bankHolder: (parsed.company as CompanyInfo | undefined)?.bankHolder || '',
   457→    bankName2: (parsed.company as CompanyInfo | undefined)?.bankName2 || '',
   458→    bankAccount2: (parsed.company as CompanyInfo | undefined)?.bankAccount2 || '',
   459→    bankHolder2: (parsed.company as CompanyInfo | undefined)?.bankHolder2 || '',
   460→  }
   461→  const client = (parsed.client as InvoiceData['client']) || { nama: '', kontak: '', alamat: '' }
   462→  const items = (Array.isArray(parsed.items) ? parsed.items : []).map(
   463→    (it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number; modal?: number }, i: number) => ({
   464→      id: it.id || `item-${i}`,
   465→      deskripsi: it.deskripsi || '',
   466→      qty: it.qty || 0,
   467→      satuan: it.satuan || '',
   468→      harga: it.harga || 0,
   469→      ...(it.modal !== undefined ? { modal: it.modal } : {}),
   470→    })
   471→  )
   472→  return {
   473→    type: 'invoice',
   474→    company,
   475→    nomor: (parsed.nomor as string) || fallbackNomor,
   476→    tanggal: (parsed.tanggal as string) || fallbackTanggal,
   477→    referensi: (parsed.referensi as string) || '',
   478→    client,
   479→    items,
   480→    ppn: (parsed.ppn as number) ?? 11,
   481→    dp: (parsed.dp as number) || 0,
   482→    dpAmount: parsed.dpAmount !== undefined ? (parsed.dpAmount as number) : undefined,
   483→    catatan: (parsed.catatan as string) || '',
   484→    tanggalJatuhTempo: (parsed.tanggalJatuhTempo as string) || '',
   485→    caraPembayaran: ((parsed.caraPembayaran as InvoiceData['caraPembayaran']) || ''),
   486→    tanggalGiro: (parsed.tanggalGiro as string) || '',
   487→    lunas: parsed.lunas === true,
   488→    tanggalPelunasan: (parsed.tanggalPelunasan as string) || '',
   489→    referensiInvoiceId: (parsed.referensiInvoiceId as string) || undefined,
   490→    referensiInvoiceNomor: (parsed.referensiInvoiceNomor as string) || undefined,
   491→    originalTotal: parsed.originalTotal !== undefined ? (parsed.originalTotal as number) : undefined,
   492→  }
   493→}
