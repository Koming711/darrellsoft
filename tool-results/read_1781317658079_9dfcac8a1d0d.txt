    33→// --- Types ---
    34→interface HistoryEntry {
    35→  id: string;
    36→  docType: string;
    37→  nomor: string;
    38→  tanggal: string;
    39→  pihakKedua: string;
    40→  total: string;

    44→
    45→// --- Helper ---
    46→function getTodayStr(): string {
    47→  const d = new Date();
    48→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
    49→}
    50→
    51→// --- Parse dataJson for document info ---
    52→function parseDocInfo(entry: HistoryEntry) {
    53→  try {
    54→    const parsed = JSON.parse(entry.dataJson);
    55→    const items = parsed.items || [];
    56→    const firstItem = items[0];
    57→    const namaBarang = firstItem?.deskripsi || '';
    58→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0);
    59→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0);
    60→    const ppn = parsed.ppn || 0;
    61→    const dpPercent = parsed.dp || 0;
    62→    const totalHarga = subtotal + (subtotal * ppn / 100);
    63→    const dpAmount = totalHarga * (dpPercent / 100);
    64→    const sisa = totalHarga - dpAmount;
    65→    const lunas = parsed.lunas === true;
    66→    const tanggalJatuhTempo = parsed.tanggalJatuhTempo || '';
    67→    const tanggalPelunasan = parsed.tanggalPelunasan || '';
    68→    return { namaBarang, totalQty, totalHarga, dpPercent, dp: dpAmount, sisa, lunas, tanggalJatuhTempo, tanggalPelunasan };
    69→  } catch {
    70→    return { namaBarang: '', totalQty: 0, totalHarga: 0, dpPercent: 0, dp: 0, sisa: 0, lunas: false, tanggalJatuhTempo: '', tanggalPelunasan: '' };
    71→  }
    72→}
    73→
    74→// --- Parse dataJson for InvoiceData ---
    75→function parseInvoiceData(entry: HistoryEntry): InvoiceData {
    76→  try {
    77→    const parsed = JSON.parse(entry.dataJson);
    78→    const company: CompanyInfo = {
    79→      nama: parsed.company?.nama || DEFAULT_COMPANY.nama,
    80→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
    81→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
    82→      email: parsed.company?.email || DEFAULT_COMPANY.email,
    83→      npwp: parsed.company?.npwp || '',
    84→      website: parsed.company?.website || '',
    85→      ppn: parsed.company?.ppn ?? parsed.ppn ?? 11,
    86→      logo: parsed.company?.logo || '',
    87→      bankName: parsed.company?.bankName || '',
    88→      bankAccount: parsed.company?.bankAccount || '',
    89→      bankHolder: parsed.company?.bankHolder || '',
    90→      bankName2: parsed.company?.bankName2 || '',
    91→      bankAccount2: parsed.company?.bankAccount2 || '',
    92→      bankHolder2: parsed.company?.bankHolder2 || '',
    93→    };
    94→    const client = parsed.client || { nama: '', kontak: '', alamat: '' };
    95→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number }, i: number) => ({
    96→      id: it.id || `item-${i}`,
    97→      deskripsi: it.deskripsi || '',
    98→      qty: it.qty || 0,
    99→      satuan: it.satuan || '',
   100→      harga: it.harga || 0,
   101→    }));
   102→    return {
   103→      type: 'invoice',
   104→      company,
   105→      nomor: parsed.nomor || entry.nomor || '',
   106→      tanggal: parsed.tanggal || entry.tanggal || '',
   107→      referensi: parsed.referensi || '',
   108→      client,
   109→      items,
   110→      ppn: parsed.ppn ?? 11,
   111→      dp: parsed.dp || 0,
   112→      catatan: parsed.catatan || '',
   113→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   114→      caraPembayaran: parsed.caraPembayaran || '',
   115→      tanggalGiro: parsed.tanggalGiro || '',
   116→      lunas: parsed.lunas === true,
   117→      tanggalPelunasan: parsed.tanggalPelunasan || '',
   118→    };
   119→  } catch {
   120→    return {
   121→      type: 'invoice',
   122→      company: { ...DEFAULT_COMPANY },
   123→      nomor: entry.nomor || '',
   124→      tanggal: entry.tanggal || '',
   125→      referensi: '',
   126→      client: { nama: '', kontak: '', alamat: '' },
   127→      items: [],
   128→      ppn: 11,
   129→      dp: 0,
   130→      catatan: '',
   131→      tanggalJatuhTempo: '',
   132→      caraPembayaran: '',
   133→      tanggalGiro: '',
   134→      lunas: false,
   135→      tanggalPelunasan: '',
   136→    };
   137→  }
   138→}

   140→export function InvoicePelunasanEditor() {
   141→  // Local invoice state (not using global store to avoid conflicts with regular editor)
   142→  const [invoiceData, setInvoiceData] = useState<InvoiceData | null>(null);
   143→  const [selectedEntry, setSelectedEntry] = useState<HistoryEntry | null>(null);
   144→  const [invoiceHistory, setInvoiceHistory] = useState<HistoryEntry[]>([]);
   145→  const [loading, setLoading] = useState(true);
   146→  const [searchQuery, setSearchQuery] = useState('');
   147→  const [dropdownOpen, setDropdownOpen] = useState(false);
   148→  const [pelunasanSaving, setPelunasanSaving] = useState(false);
   149→
   150→  // Pelunasan fields
   151→  const [lunasToggle, setLunasToggle] = useState(false);
   152→  const [tanggalPelunasan, setTanggalPelunasan] = useState('');
   153→  const [tanggalJatuhTempo, setTanggalJatuhTempo] = useState('');
   154→  const [caraPembayaran, setCaraPembayaran] = useState('');
   155→  const [tanggalGiro, setTanggalGiro] = useState('');
   156→
   157→  const fetchHistory = useCallback(async () => {
   158→    try {
   159→      setLoading(true);
   160→      const headers = getAuthHeaders();
   161→      const res = await fetch('/api/history?docType=invoice', { headers });
   162→      if (res.ok) {
   163→        const json = await res.json();
   164→        setInvoiceHistory(json.data || []);
   165→      }
   166→    } catch (err) {
   167→      console.error('Failed to fetch invoice history:', err);
   168→    } finally {
   169→      setLoading(false);
   170→    }
   171→  }, []);
   172→
   173→  useEffect(() => {
   174→    fetchHistory();
   175→  }, [fetchHistory]);
   176→
   177→  useEffect(() => {
   178→    const handler = () => fetchHistory();
   179→    window.addEventListener('dokupro:history-updated', handler);
   180→    return () => window.removeEventListener('dokupro:history-updated', handler);
   181→  }, [fetchHistory]);
   182→
   183→  // Filter pending invoices (with DP and not yet lunas)
   184→  const pendingInvoices = useMemo(() => {
   185→    return invoiceHistory.filter(entry => {
   186→      const info = parseDocInfo(entry);
   187→      return info.dpPercent > 0 && !info.lunas && info.sisa > 0;
   188→    });
   189→  }, [invoiceHistory]);
   190→
   191→  // Filter by search
   192→  const filteredPending = useMemo(() => {
   193→    if (!searchQuery.trim()) return pendingInvoices;
   194→    const q = searchQuery.toLowerCase().trim();
   195→    return pendingInvoices.filter(entry => {
   196→      const info = parseDocInfo(entry);
   197→      return entry.nomor?.toLowerCase().includes(q) || entry.pihakKedua?.toLowerCase().includes(q) || info.namaBarang?.toLowerCase().includes(q);
   198→    });
   199→  }, [pendingInvoices, searchQuery]);
   200→
   201→  // Select an invoice
   202→  const selectInvoice = (entry: HistoryEntry) => {
   203→    const parsed = parseInvoiceData(entry);
   204→    const info = parseDocInfo(entry);
   205→    setInvoiceData(parsed);
   206→    setSelectedEntry(entry);
   207→    setLunasToggle(info.lunas);
   208→    setTanggalPelunasan(info.tanggalPelunasan || getTodayStr());
   209→    setTanggalJatuhTempo(info.tanggalJatuhTempo || '');
   210→    setCaraPembayaran(parsed.caraPembayaran || '');
   211→    setTanggalGiro(parsed.tanggalGiro || '');
   212→    setDropdownOpen(false);
   213→  };
   214→
   215→  // Clear selection
   216→  const clearSelection = () => {
   217→    setInvoiceData(null);
   218→    setSelectedEntry(null);
   219→    setLunasToggle(false);
   220→    setTanggalPelunasan('');
   221→    setTanggalJatuhTempo('');
   222→    setCaraPembayaran('');
   223→    setTanggalGiro('');
   224→  };
   225→
   226→  // Update invoice data locally
   227→  const updateInvoice = (updates: Partial<InvoiceData>) => {
   228→    setInvoiceData(prev => prev ? { ...prev, ...updates } : null);
   229→  };
   230→
   231→  const updateCompany = (company: CompanyInfo) => {
   232→    setInvoiceData(prev => prev ? { ...prev, company } : null);
   233→  };
   234→
   235→  const updateClient = (field: string, value: string) => {
   236→    setInvoiceData(prev => prev ? { ...prev, client: { ...prev.client, [field]: value } } : null);
   237→  };
   238→
   239→  // Build preview data with pelunasan overrides
   240→  const previewData: InvoiceData | null = useMemo(() => {
   241→    if (!invoiceData) {
   242→      // Default preview when no invoice is selected
   243→      return {
   244→        type: 'invoice',
   245→        company: { ...DEFAULT_COMPANY },
   246→        nomor: '-',
   247→        tanggal: getTodayStr(),
   248→        referensi: '',
   249→        client: { nama: '', kontak: '', alamat: '' },
   250→        items: [],
   251→        ppn: 11,
   252→        dp: 0,
   253→        catatan: '',
   254→        tanggalJatuhTempo: '',
   255→        caraPembayaran: '',
   256→        tanggalGiro: '',
   257→        lunas: false,
   258→        tanggalPelunasan: '',
   259→      };
   260→    }
   261→    return {
   262→      ...invoiceData,
   263→      tanggalJatuhTempo,
   264→      caraPembayaran: caraPembayaran as InvoiceData['caraPembayaran'],
   265→      tanggalGiro,
   266→      lunas: lunasToggle,
   267→      tanggalPelunasan: lunasToggle ? tanggalPelunasan : '',
   268→    };
   269→  }, [invoiceData, tanggalJatuhTempo, caraPembayaran, tanggalGiro, lunasToggle, tanggalPelunasan]);
   270→
   271→  // Calculate amounts
   272→  const subtotal = invoiceData?.items.reduce((sum, item) => sum + item.qty * item.harga, 0) || 0;
   273→  const ppnAmount = subtotal * ((invoiceData?.ppn || 0) / 100);
   274→  const total = subtotal + ppnAmount;
   275→  const dpPercent = invoiceData?.dp || 0;
   276→  const dpAmount = total * (dpPercent / 100);
   277→  const sisa = total - dpAmount;
   278→
   279→  // Save pelunasan — update the existing history entry
   280→  const handleSavePelunasan = async () => {
   281→    if (!selectedEntry || !invoiceData) return;
   282→    setPelunasanSaving(true);
   283→    try {
   284→      const parsed = JSON.parse(selectedEntry.dataJson);
   285→      // Update pelunasan fields
   286→      parsed.tanggalJatuhTempo = tanggalJatuhTempo;
   287→      parsed.caraPembayaran = caraPembayaran;
   288→      parsed.tanggalGiro = tanggalGiro;
   289→      if (lunasToggle) {
   290→        parsed.lunas = true;
   291→        parsed.tanggalPelunasan = tanggalPelunasan || getTodayStr();
   292→      } else {
   293→        parsed.lunas = false;
   294→        parsed.tanggalPelunasan = '';
   295→      }
   296→      // Also update any edited fields from the form
   297→      parsed.company = invoiceData.company;
   298→      parsed.client = invoiceData.client;
   299→      parsed.items = invoiceData.items;
   300→      parsed.ppn = invoiceData.ppn;
   301→      parsed.dp = invoiceData.dp;
   302→      parsed.catatan = invoiceData.catatan;
   303→      parsed.nomor = invoiceData.nomor;
   304→      parsed.tanggal = invoiceData.tanggal;
   305→      parsed.referensi = invoiceData.referensi;
   306→      delete parsed.statusPembayaran;
   307→
   308→      const newDataJson = JSON.stringify(parsed);
   309→      const res = await fetcher(`/api/history/${selectedEntry.id}`, {
   310→        method: 'PUT',
   311→        headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   312→        body: JSON.stringify({ dataJson: newDataJson }),
   313→      });
   314→      if (res.ok) {
   315→        toast.success(lunasToggle ? 'Pelunasan berhasil dicatat!' : 'Invoice berhasil diperbarui');
   316→        clearSelection();
   317→        fetchHistory();
   318→        notifyDataChange('invoice');
   319→      } else {
   320→        toast.error('Gagal menyimpan perubahan');
   321→      }
   322→    } catch {
   323→      toast.error('Gagal menyimpan perubahan');
   324→    } finally {
   325→      setPelunasanSaving(false);
   326→    }
   327→  };
   328→
   329→  return (
   330→    <DocumentEditorLayout
   331→      title="Invoice Pelunasan"
   332→      previewContent={<InvoicePreview data={previewData!} />}
   333→      actions={
   334→        invoiceData ? (
   335→          <div className="flex items-center gap-2 flex-wrap">
   336→            <Button
   337→              variant="outline"
   338→              size="sm"
   339→              onClick={clearSelection}
   340→              className="border-gray-200 text-gray-600 hover:bg-gray-50"
   341→            >
   342→              <X className="mr-1.5 h-3.5 w-3.5" />
   343→              Batal
   344→            </Button>
   345→            <Button
   346→              size="sm"
   347→              onClick={() => {
   348→                toast.dismiss();
   349→                const origTitle = document.title;
   350→                document.title = ' ';
   351→                setTimeout(() => {
   352→                  window.print();
   353→                  document.title = origTitle;
   354→                }, 100);
   355→              }}
   356→              className="bg-emerald-600 hover:bg-emerald-700"
   357→            >
   358→              Cetak
   359→            </Button>
   360→            <Button
   361→              size="sm"
   362→              onClick={handleSavePelunasan}
   363→              disabled={pelunasanSaving}
   364→              className={cn(
   365→                'text-white',
   366→                lunasToggle
   367→                  ? 'bg-green-600 hover:bg-green-700'
   368→                  : 'bg-blue-600 hover:bg-blue-700'
   369→              )}
   370→            >
   371→              {pelunasanSaving ? (
   372→                <><Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" /> Menyimpan...</>
   373→              ) : (
   374→                <><CheckCircle2 className="mr-1.5 h-3.5 w-3.5" /> {lunasToggle ? 'Simpan Pelunasan' : 'Simpan Perubahan'}</>
   375→              )}
   376→            </Button>
   377→          </div>
   378→        ) : undefined
   379→      }
   380→    >
   381→      {/* Invoice Selector */}
   382→      <div className="rounded-lg border bg-card p-3 sm:p-4 shadow-sm">
   383→        <h3 className="mb-3 text-xs font-semibold uppercase tracking-wider text-muted-foreground flex items-center gap-1.5">
   384→          <Wallet className="w-3.5 h-3.5 text-amber-600" />
   385→          Pilih Invoice Belum Lunas
   386→        </h3>
   387→        <Popover open={dropdownOpen} onOpenChange={setDropdownOpen}>
   388→          <PopoverAnchor asChild>
   389→            <div className="relative">
   390→              <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 w-3.5 h-3.5 text-slate-400" />
   391→              <input
   392→                type="text"
   393→                placeholder={selectedEntry ? `${selectedEntry.nomor} — ${selectedEntry.pihakKedua}` : "Cari no. invoice, customer..."}
   394→                value={selectedEntry ? `${selectedEntry.nomor} — ${selectedEntry.pihakKedua}` : searchQuery}
   395→                onChange={(e) => { setSearchQuery(e.target.value); if (selectedEntry) { clearSelection(); } }}
   396→                onFocus={() => { if (selectedEntry) { clearSelection(); } setDropdownOpen(true); }}
   397→                className="flex h-9 w-full rounded-md border border-input bg-transparent pl-8 pr-8 py-2 text-sm shadow-xs transition-[color,box-shadow] outline-none focus-visible:border-ring focus-visible:ring-ring/50 focus-visible:ring-[3px]"
   398→              />
   399→              {selectedEntry && (
   400→                <button
   401→                  type="button"
   402→                  onClick={(e) => { e.preventDefault(); clearSelection(); }}
   403→                  className="absolute right-2 top-1/2 -translate-y-1/2 text-slate-400 hover:text-slate-600"
   404→                >
   405→                  <X className="w-4 h-4" />
   406→                </button>
   407→              )}
   408→            </div>
   409→          </PopoverAnchor>
   410→          <PopoverContent
   411→            align="start"
   412→            className="p-0 w-[var(--radix-popover-trigger-width)] max-h-64 overflow-y-auto"
   413→            onOpenAutoFocus={(e) => e.preventDefault()}
   414→          >
   415→            {loading ? (
   416→              <div className="px-3 py-4 text-center"><Loader2 className="w-5 h-5 mx-auto text-blue-500 animate-spin" /><p className="text-xs text-slate-400 mt-1">Memuat...</p></div>
   417→            ) : filteredPending.length > 0 ? (
   418→              <div>
   419→                <div className="px-3 py-1.5 text-[10px] font-semibold text-slate-400 uppercase bg-amber-50 border-b border-amber-100 sticky top-0">
   420→                  Menunggu Pelunasan ({filteredPending.length})
   421→                </div>
   422→                {filteredPending.map((entry) => {
   423→                  const info = parseDocInfo(entry);
   424→                  const isOverdue = info.tanggalJatuhTempo && new Date(info.tanggalJatuhTempo) < new Date(getTodayStr());
   425→                  return (
   426→                    <button
   427→                      key={entry.id}
   428→                      type="button"
   429→                      onMouseDown={(e) => { e.preventDefault(); selectInvoice(entry); }}
   430→                      className={cn(
   431→                        'w-full text-left px-3 py-2.5 text-sm transition-colors border-b border-slate-50 last:border-b-0',
   432→                        selectedEntry?.id === entry.id ? 'bg-amber-50' : 'hover:bg-amber-50/50'
   433→                      )}
   434→                    >
   435→                      <div className="flex items-center justify-between gap-2">
   436→                        <div className="min-w-0">
   437→                          <div className="flex items-center gap-1.5">
   438→                            <span className="font-semibold text-violet-700 text-xs">{entry.nomor || '-'}</span>
   439→                            {isOverdue && <AlertTriangle className="w-3 h-3 text-red-500" />}
   440→                          </div>
   441→                          <p className="text-slate-600 text-[11px] truncate">{entry.pihakKedua || '-'}</p>
   442→                        </div>
   443→                        <div className="text-right shrink-0">
   444→                          <p className="text-red-600 font-bold text-xs">{formatRupiah(info.sisa)}</p>
   445→                          <p className="text-[9px] text-violet-500">DP {info.dpPercent}%</p>
   446→                        </div>
   447→                      </div>
   448→                    </button>
   449→                  );
   450→                })}
   451→              </div>
   452→            ) : (
   453→              <div className="px-3 py-4 text-center">
   454→                <CheckCircle2 className="w-8 h-8 mx-auto text-green-300 mb-2" />
   455→                <p className="text-xs text-slate-400">Semua invoice sudah lunas</p>
   456→              </div>
   457→            )}
   458→          </PopoverContent>
   459→        </Popover>
   460→
   461→        {/* Quick list when no invoice selected */}
   462→        {!selectedEntry && !loading && pendingInvoices.length > 0 && (
   463→          <div className="mt-3 divide-y divide-slate-100 max-h-48 overflow-y-auto rounded-lg border border-slate-200">
   464→            {pendingInvoices.slice(0, 10).map((entry) => {
   465→              const info = parseDocInfo(entry);
   466→              const isOverdue = info.tanggalJatuhTempo && new Date(info.tanggalJatuhTempo) < new Date(getTodayStr());
   467→              return (
   468→                <div
   469→                  key={entry.id}
   470→                  onClick={() => selectInvoice(entry)}
   471→                  className="px-3 py-2.5 hover:bg-amber-50/50 cursor-pointer transition-colors"
   472→                >
   473→                  <div className="flex items-center justify-between gap-2">
   474→                    <div className="min-w-0">
   475→                      <div className="flex items-center gap-1.5">
   476→                        <span className="font-semibold text-violet-700 text-xs">{entry.nomor || '-'}</span>
   477→                        <CircleDot className="w-2.5 h-2.5 text-red-500" />
   478→                        {isOverdue && <AlertTriangle className="w-3 h-3 text-red-500" />}
   479→                      </div>
   480→                      <p className="text-slate-600 text-[11px] truncate">{entry.pihakKedua || '-'}</p>
   481→                    </div>
   482→                    <div className="text-right shrink-0">
   483→                      <p className="text-red-600 font-bold text-xs">{formatRupiah(info.sisa)}</p>
   484→                      <p className="text-[9px] text-violet-500">DP {info.dpPercent}%</p>
   485→                    </div>
   486→                  </div>
   487→                </div>
   488→              );
   489→            })}
   490→          </div>
   491→        )}
   492→      </div>
   493→
   494→      {/* When an invoice is selected, show the editor form */}
   495→      {invoiceData ? (
   496→        <>
   497→          <CompanyFields company={invoiceData.company} onChange={updateCompany} />
