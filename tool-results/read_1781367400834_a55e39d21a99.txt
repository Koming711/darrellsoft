    65→function getTodayStr(): string {
    66→  const d = new Date()
    67→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
    68→}

    74→// --- Parse dataJson for document info ---
    75→function parseDocInfo(entry: HistoryEntry) {
    76→  try {
    77→    const parsed = JSON.parse(entry.dataJson)
    78→    const items = parsed.items || []
    79→    const firstItem = items[0]
    80→    const namaBarang = firstItem?.deskripsi || ''
    81→    const hargaSatuan = firstItem?.harga || 0
    82→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0)
    83→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0)
    84→    const ppn = parsed.ppn || 0
    85→    const dpPercent = parsed.dp || 0
    86→    const totalHarga = subtotal + (subtotal * ppn / 100)
    87→    // Always derive dpAmount from originalTotal
    88→    const originalTotal = parsed.originalTotal !== undefined ? parsed.originalTotal : totalHarga
    89→    const dpAmount = originalTotal * (dpPercent / 100)
    90→    const sisa = totalHarga - dpAmount
    91→    const lunas = parsed.lunas === true
    92→    const tanggalJatuhTempo = parsed.tanggalJatuhTempo || ''
    93→    const tanggalPelunasan = parsed.tanggalPelunasan || ''
    94→    const referensi = parsed.referensi || ''
    95→    const referensiInvoiceNomor = parsed.referensiInvoiceNomor || ''
    96→    const isPelunasan = entry.docType === 'invoice-pelunasan'
    97→    const uangCapek = parsed.uangCapek || 0
    98→    return { namaBarang, hargaSatuan, totalQty, totalHarga, dpPercent, dp: dpAmount, sisa, lunas, tanggalJatuhTempo, tanggalPelunasan, referensi, referensiInvoiceNomor, isPelunasan, originalTotal, uangCapek }
    99→  } catch {
   100→    return { namaBarang: '', hargaSatuan: 0, totalQty: 0, totalHarga: 0, dpPercent: 0, dp: 0, sisa: 0, lunas: false, tanggalJatuhTempo: '', tanggalPelunasan: '', referensi: '', referensiInvoiceNomor: '', isPelunasan: false, originalTotal: 0, uangCapek: 0 }
   101→  }
   102→}

   104→// --- Parse dataJson for InvoiceData ---
   105→function parseInvoiceData(entry: HistoryEntry): InvoiceData {
   106→  try {
   107→    const parsed = JSON.parse(entry.dataJson)
   108→    const company: CompanyInfo = {
   109→      nama: parsed.company?.nama || DEFAULT_COMPANY.nama,
   110→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
   111→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
   112→      email: parsed.company?.email || DEFAULT_COMPANY.email,
   113→      npwp: parsed.company?.npwp || '',
   114→      website: parsed.company?.website || '',
   115→      ppn: parsed.company?.ppn ?? parsed.ppn ?? 11,
   116→      logo: parsed.company?.logo || '',
   117→      bankName: parsed.company?.bankName || '',
   118→      bankAccount: parsed.company?.bankAccount || '',
   119→      bankHolder: parsed.company?.bankHolder || '',
   120→      bankName2: parsed.company?.bankName2 || '',
   121→      bankAccount2: parsed.company?.bankAccount2 || '',
   122→      bankHolder2: parsed.company?.bankHolder2 || '',
   123→    }
   124→    const client = parsed.client || { nama: '', kontak: '', alamat: '' }
   125→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number }, i: number) => ({
   126→      id: it.id || `item-${i}`,
   127→      deskripsi: it.deskripsi || '',
   128→      qty: it.qty || 0,
   129→      satuan: it.satuan || '',
   130→      harga: it.harga || 0,
   131→    }))
   132→    const docType = (entry.docType === 'invoice-pelunasan' ? 'invoice-pelunasan' : 'invoice') as 'invoice' | 'invoice-pelunasan'
   133→    return {
   134→      type: docType,
   135→      company,
   136→      nomor: parsed.nomor || entry.nomor || '',
   137→      tanggal: parsed.tanggal || entry.tanggal || '',
   138→      referensi: parsed.referensi || '',
   139→      client,
   140→      items,
   141→      ppn: parsed.ppn ?? 11,
   142→      dp: parsed.dp || 0,
   143→      dpAmount: parsed.dpAmount,
   144→      catatan: parsed.catatan || '',
   145→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   146→      caraPembayaran: parsed.caraPembayaran || '',
   147→      tanggalGiro: parsed.tanggalGiro || '',
   148→      lunas: parsed.lunas === true,
   149→      tanggalPelunasan: parsed.tanggalPelunasan || '',
   150→      referensiInvoiceId: parsed.referensiInvoiceId || '',
   151→      referensiInvoiceNomor: parsed.referensiInvoiceNomor || '',
   152→      originalTotal: parsed.originalTotal,
   153→      uangCapek: parsed.uangCapek || 0,
   154→    }
   155→  } catch {
   156→    const docType = (entry.docType === 'invoice-pelunasan' ? 'invoice-pelunasan' : 'invoice') as 'invoice' | 'invoice-pelunasan'
   157→    return {
   158→      type: docType,
   159→      company: { ...DEFAULT_COMPANY },
   160→      nomor: entry.nomor || '',
   161→      tanggal: entry.tanggal || '',
   162→      referensi: '',
   163→      client: { nama: '', kontak: '', alamat: '' },
   164→      items: [],
   165→      ppn: 11,
   166→      dp: 0,
   167→      catatan: '',
   168→      tanggalJatuhTempo: '',
   169→      caraPembayaran: '',
   170→      tanggalGiro: '',
   171→      lunas: false,
   172→      tanggalPelunasan: '',
   173→      referensiInvoiceId: '',
   174→      referensiInvoiceNomor: '',
   175→      uangCapek: 0,
   176→    }
   177→  }
   178→}
