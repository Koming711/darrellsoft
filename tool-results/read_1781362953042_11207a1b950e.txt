    65→function getTodayStr(): string {
    66→  const d = new Date()

    74→// --- Parse dataJson for document info ---
    75→function parseDocInfo(entry: HistoryEntry) {
    76→  try {
    77→    const parsed = JSON.parse(entry.dataJson)
    78→    const items = parsed.items || []
    79→    const firstItem = items[0]
    80→    const namaBarang = firstItem?.deskripsi || ''
    81→    const hargaSatuan = firstItem?.harga || 0
    82→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0)
    83→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0)
    84→    const ppn = parsed.ppn || 0
    85→    const dpPercent = parsed.dp || 0
    86→    const totalHarga = subtotal + (subtotal * ppn / 100)
    87→    // Always derive dpAmount from originalTotal
    88→    const originalTotal = parsed.originalTotal !== undefined ? parsed.originalTotal : totalHarga
    89→    const dpAmount = originalTotal * (dpPercent / 100)
    90→    const sisa = totalHarga - dpAmount
    91→    const lunas = parsed.lunas === true
    92→    const tanggalJatuhTempo = parsed.tanggalJatuhTempo || ''
    93→    const tanggalPelunasan = parsed.tanggalPelunasan || ''
    94→    const referensi = parsed.referensi || ''
    95→    const referensiInvoiceNomor = parsed.referensiInvoiceNomor || ''
    96→    const isPelunasan = entry.docType === 'invoice-pelunasan'
    97→    return { namaBarang, hargaSatuan, totalQty, totalHarga, dpPercent, dp: dpAmount, sisa, lunas, tanggalJatuhTempo, tanggalPelunasan, referensi, referensiInvoiceNomor, isPelunasan, originalTotal }
    98→  } catch {
    99→    return { namaBarang: '', hargaSatuan: 0, totalQty: 0, totalHarga: 0, dpPercent: 0, dp: 0, sisa: 0, lunas: false, tanggalJatuhTempo: '', tanggalPelunasan: '', referensi: '', referensiInvoiceNomor: '', isPelunasan: false, originalTotal: 0 }
   100→  }
   101→}

   103→// --- Parse dataJson for InvoiceData ---
   104→function parseInvoiceData(entry: HistoryEntry): InvoiceData {
   105→  try {
   106→    const parsed = JSON.parse(entry.dataJson)
   107→    const company: CompanyInfo = {
   108→      nama: parsed.company?.nama || DEFAULT_COMPANY.nama,
   109→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
   110→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
   111→      email: parsed.company?.email || DEFAULT_COMPANY.email,
   112→      npwp: parsed.company?.npwp || '',
   113→      website: parsed.company?.website || '',
   114→      ppn: parsed.company?.ppn ?? parsed.ppn ?? 11,
   115→      logo: parsed.company?.logo || '',
   116→      bankName: parsed.company?.bankName || '',
   117→      bankAccount: parsed.company?.bankAccount || '',
   118→      bankHolder: parsed.company?.bankHolder || '',
   119→      bankName2: parsed.company?.bankName2 || '',
   120→      bankAccount2: parsed.company?.bankAccount2 || '',
   121→      bankHolder2: parsed.company?.bankHolder2 || '',
   122→    }
   123→    const client = parsed.client || { nama: '', kontak: '', alamat: '' }
   124→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number }, i: number) => ({
   125→      id: it.id || `item-${i}`,
   126→      deskripsi: it.deskripsi || '',
   127→      qty: it.qty || 0,
   128→      satuan: it.satuan || '',
   129→      harga: it.harga || 0,
   130→    }))
   131→    const docType = (entry.docType === 'invoice-pelunasan' ? 'invoice-pelunasan' : 'invoice') as 'invoice' | 'invoice-pelunasan'
   132→    return {
   133→      type: docType,
   134→      company,
   135→      nomor: parsed.nomor || entry.nomor || '',
   136→      tanggal: parsed.tanggal || entry.tanggal || '',
   137→      referensi: parsed.referensi || '',
   138→      client,
   139→      items,
   140→      ppn: parsed.ppn ?? 11,
   141→      dp: parsed.dp || 0,
   142→      dpAmount: parsed.dpAmount,
   143→      catatan: parsed.catatan || '',
   144→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   145→      caraPembayaran: parsed.caraPembayaran || '',
   146→      tanggalGiro: parsed.tanggalGiro || '',
   147→      lunas: parsed.lunas === true,
   148→      tanggalPelunasan: parsed.tanggalPelunasan || '',
   149→      referensiInvoiceId: parsed.referensiInvoiceId || '',
   150→      referensiInvoiceNomor: parsed.referensiInvoiceNomor || '',
   151→      originalTotal: parsed.originalTotal,
   152→    }
   153→  } catch {
   154→    const docType = (entry.docType === 'invoice-pelunasan' ? 'invoice-pelunasan' : 'invoice') as 'invoice' | 'invoice-pelunasan'
   155→    return {
   156→      type: docType,
   157→      company: { ...DEFAULT_COMPANY },
   158→      nomor: entry.nomor || '',
   159→      tanggal: entry.tanggal || '',
   160→      referensi: '',
   161→      client: { nama: '', kontak: '', alamat: '' },
   162→      items: [],
   163→      ppn: 11,
   164→      dp: 0,
   165→      catatan: '',
   166→      tanggalJatuhTempo: '',
   167→      caraPembayaran: '',
   168→      tanggalGiro: '',
   169→      lunas: false,
   170→      tanggalPelunasan: '',
   171→      referensiInvoiceId: '',
   172→      referensiInvoiceNomor: '',
   173→    }
   174→  }
   175→}
