    64→    64→function formatRupiah(n: number): string {
    65→    65→  return new Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR', minimumFractionDigits: 0 }).format(n || 0)
    66→    66→}

    68→    68→function formatDateID(iso: string): string {
    69→    69→  if (!iso) return '-'
    70→    70→  const d = new Date(iso + (iso.length === 10 ? 'T00:00:00' : ''))
    71→    71→  if (isNaN(d.getTime())) return iso
    72→    72→  return d.toLocaleDateString('id-ID', { day: '2-digit', month: 'short', year: 'numeric' })
    73→    73→}
    74→    74→
    75→    75→function todayISO(): string {
    76→    76→  const d = new Date()
    77→    77→  const tz = d.getTimezoneOffset() * 60000
    78→    78→  return new Date(d.getTime() - tz).toISOString().slice(0, 10)
    79→    79→}
