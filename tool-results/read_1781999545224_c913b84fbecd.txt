    14→// === MODULE-LEVEL SESSION CACHE ===
    15→// Persists across navigations so DashboardLayout doesn't need to re-verify on every mount
    16→interface CachedSession {
    17→  user: any
    18→  userProfile: { createdAt: string | null; validUntil: string | null } | null
    19→  autoLogoutMin: number
    20→  logoutWarningSec: number
    21→  sessionWarning: string | null
    22→  forceLogoutAvailable: boolean
    23→  accountExpired: boolean
    24→  cachedAt: number
    25→}
    26→let _sessionCache: CachedSession | null = null
    27→const SESSION_CACHE_TTL = 120_000 // 2 minutes
    28→
    29→function getSessionCache(): CachedSession | null {
    30→  // Check module-level cache first
    31→  if (_sessionCache && Date.now() - _sessionCache.cachedAt <= SESSION_CACHE_TTL) {
    32→    return _sessionCache
    33→  }
    34→  // Fallback to sessionStorage
    35→  if (typeof window !== 'undefined') {
    36→    try {
    37→      const stored = sessionStorage.getItem('__dashboard_session_cache')
    38→      if (stored) {
    39→        const parsed = JSON.parse(stored) as CachedSession
    40→        if (Date.now() - parsed.cachedAt <= SESSION_CACHE_TTL) {
    41→          _sessionCache = parsed
    42→          return parsed
    43→        }
    44→        sessionStorage.removeItem('__dashboard_session_cache')
    45→      }
    46→    } catch {}
    47→  }
    48→  _sessionCache = null
    49→  return null
    50→}
    51→
    52→function setSessionCache(data: Omit<CachedSession, 'cachedAt'>) {
    53→  const entry = { ...data, cachedAt: Date.now() }
    54→  _sessionCache = entry
    55→  if (typeof window !== 'undefined') {
    56→    try {
    57→      sessionStorage.setItem('__dashboard_session_cache', JSON.stringify(entry))
    58→    } catch {}
    59→  }
    60→}
    61→
    62→function invalidateSessionCache() {
    63→  _sessionCache = null
    64→  if (typeof window !== 'undefined') {
    65→    try {
    66→      sessionStorage.removeItem('__dashboard_session_cache')
    67→    } catch {}
    68→  }
    69→}

   540→        permVersion={permVersion}
   541→      />
   542→
   543→      <div className={`${desktopMargin} transition-all duration-300`}>
   544→        <MobileHeader
   545→          username={user?.username}
   546→          title={title}
   547→          subtitle={subtitle}
   548→          userProfile={userProfile}
   549→        />
   550→
   551→        {/* Main Content — extra bottom padding on mobile for bottom nav + safe area */}
   552→        <main className="p-4 pb-20 lg:p-8 lg:pb-8">
   553→          {children}
   554→        </main>
   555→      </div>
   556→
   557→      {/* ===== Mobile Bottom Navigation ===== */}
   558→      <MobileBottomNav role={user?.role} onMoreClick={() => setSidebarOpen(true)} username={user?.username} onLogout={handleLogout} />
   559→
   560→      {/* ===== MODAL: ACCOUNT EXPIRED ===== */}
   561→      {accountExpired && sessionWarning && (
   562→        <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/50 backdrop-blur-sm px-4">
   563→          <div className="rounded-2xl shadow-2xl border border-red-200 dark:border-red-800 max-w-sm w-full p-6 animate-in fade-in zoom-in duration-200" style={{ backgroundColor: 'var(--card)' }}>
   564→            <div className="flex items-start gap-3 mb-4">
   565→              <div className="w-10 h-10 rounded-xl bg-red-100 dark:bg-red-900/30 flex items-center justify-center flex-shrink-0">
   566→                <TimerOff className="w-5 h-5 text-red-600 dark:text-red-400" />
   567→              </div>
   568→              <div>
   569→                <h3 className="text-lg font-bold text-foreground">Akun Kadaluarsa</h3>
   570→                <p className="text-xs text-muted-foreground mt-0.5">Masa berlaku akun telah habis</p>
   571→              </div>
   572→            </div>
   573→            <div className="bg-red-50 dark:bg-red-950 border border-red-200 dark:border-red-800 rounded-xl p-4 mb-5">
   574→              <p className="text-sm text-red-700 dark:text-red-400 leading-relaxed">{sessionWarning}</p>
   575→            </div>
   576→            <button
   577→              onClick={handleLogout}
   578→              className="w-full bg-red-600 hover:bg-red-700 text-white font-semibold py-2.5 rounded-xl transition-colors"
   579→              autoFocus
   580→            >
   581→              <span className="flex items-center justify-center gap-2">
   582→                <LogOut className="w-4 h-4" /> OK, Mengerti
   583→              </span>
   584→            </button>
   585→          </div>
   586→        </div>
   587→      )}
   588→
   589→      {/* ===== MODAL: SESSION WARNING (Multi-Device) ===== */}
   590→      {sessionWarning && !accountExpired && (
   591→        <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/50 backdrop-blur-sm px-4">
   592→          <div className="rounded-2xl shadow-2xl border border-red-200 dark:border-red-800 max-w-sm w-full p-6 animate-in fade-in zoom-in duration-200" style={{ backgroundColor: 'var(--card)' }}>
   593→            <div className="flex items-start gap-3 mb-4">
   594→              <div className="w-10 h-10 rounded-xl bg-red-100 dark:bg-red-900/30 flex items-center justify-center flex-shrink-0">
   595→                {forceLogoutAvailable ? <ShieldAlert className="w-5 h-5 text-red-600 dark:text-red-400" /> : <Smartphone className="w-5 h-5 text-red-600 dark:text-red-400" />}
   596→              </div>
   597→              <div>
   598→                <h3 className="text-lg font-bold text-foreground">Peringatan Keamanan</h3>
   599→                <p className="text-xs text-muted-foreground mt-0.5">Akun digunakan di perangkat lain</p>
   600→              </div>
   601→            </div>
   602→            <div className="bg-red-50 dark:bg-red-950 border border-red-200 dark:border-red-800 rounded-xl p-4 mb-5">
   603→              <p className="text-sm text-red-700 dark:text-red-400 leading-relaxed">{sessionWarning}</p>
   604→            </div>
   605→            {forceLogoutAvailable ? (
   606→              <div className="space-y-2.5">
   607→                <button
   608→                  onClick={async () => {
   609→                    setIsReclaiming(true)
   610→                    try {
   611→                      const authUser = getAuthUser()
   612→                      if (!authUser?.sessionId) {
   613→                        toast.error('Data sesi tidak ditemukan, silakan login ulang')
   614→                        setIsReclaiming(false)
   615→                        handleLogout()
   616→                        return
   617→                      }
   618→                      const res = await authFetch('/api/auth/reclaim-session', {
   619→                        method: 'POST',
   620→                        headers: { 'Content-Type': 'application/json' },
   621→                        body: JSON.stringify({ username: authUser.username, sessionId: authUser.sessionId, userId: authUser.id }),
   622→                      })
   623→                      const data = await res.json()
   624→                      if (res.ok) {
   625→                        setSessionWarning(null)
   626→                        setForceLogoutAvailable(false)
   627→                        toast.success('Perangkat lain berhasil di-logout')
   628→                      } else {
   629→                        toast.error(data.error || 'Gagal mengklaim sesi. Silakan coba lagi.')
   630→                      }
   631→                    } catch {
   632→                      toast.error('Terjadi kesalahan jaringan')
   633→                    }
   634→                    setIsReclaiming(false)
   635→                  }}
   636→                  disabled={isReclaiming}
   637→                  className="w-full flex items-center justify-center gap-2 bg-amber-500 hover:bg-amber-600 disabled:bg-amber-400 text-white font-semibold py-2.5 rounded-xl transition-colors"
   638→                  autoFocus
   639→                >
   640→                  {isReclaiming ? (
   641→                    <><div className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin" /> Memproses...</>
   642→                  ) : (
   643→                    <><ShieldAlert className="w-4 h-4" /> Paksa Logout Perangkat Lain</>
   644→                  )}
   645→                </button>
   646→                <button
   647→                  onClick={handleLogout}
   648→                  className="w-full flex items-center justify-center gap-2 border-2 border-red-200 dark:border-red-800 text-red-600 dark:text-red-400 hover:bg-red-50 dark:hover:bg-red-950 font-semibold py-2.5 rounded-xl transition-colors"
   649→                >
   650→                  <LogOut className="w-4 h-4" /> Logout dari Sini
   651→                </button>
   652→              </div>
   653→            ) : (
   654→              <button
   655→                onClick={handleLogout}
   656→                className="w-full bg-red-600 hover:bg-red-700 text-white font-semibold py-2.5 rounded-xl transition-colors"
   657→                autoFocus
   658→              >
   659→                OK, Logout
   660→              </button>
   661→            )}
   662→          </div>
   663→        </div>
   664→      )}
   665→
   666→      {/* ===== MODAL: AUTO-LOGOUT COUNTDOWN ===== */}
   667→      {showCountdown && (
   668→        <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/50 backdrop-blur-sm px-4">
   669→          <div className="rounded-2xl shadow-2xl border border-amber-200 dark:border-amber-800 max-w-sm w-full p-6 animate-in fade-in zoom-in duration-200" style={{ backgroundColor: 'var(--card)' }}>
   670→            <div className="flex items-start gap-3 mb-4">
   671→              <div className="w-10 h-10 rounded-xl bg-amber-100 dark:bg-amber-900/30 flex items-center justify-center flex-shrink-0">
   672→                <AlertTriangle className="w-5 h-5 text-amber-600 dark:text-amber-400" />
   673→              </div>
   674→              <div>
   675→                <h3 className="text-lg font-bold text-foreground">Sesi Akan Berakhir</h3>
   676→                <p className="text-xs text-muted-foreground mt-0.5">Anda tidak aktif, akan otomatis logout</p>
   677→              </div>
   678→            </div>
   679→            <div className="text-center py-5">
   680→              <div className="text-6xl font-bold text-amber-600 dark:text-amber-400 tabular-nums">{countdown}</div>
   681→              <p className="text-sm text-muted-foreground mt-2">detik tersisa</p>
   682→            </div>
   683→            <button
   684→              onClick={handleStayLoggedIn}
   685→              className="w-full bg-amber-500 hover:bg-amber-600 text-white font-semibold py-2.5 rounded-xl transition-colors"
   686→              autoFocus
   687→            >
   688→              Tetap Login
   689→            </button>
   690→          </div>
   691→        </div>
   692→      )}
   693→    </div>
   694→  )
