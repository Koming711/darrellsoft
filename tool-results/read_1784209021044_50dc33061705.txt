    70→function getTodayStr(): string {
    71→  const d = new Date()

    79→// --- Parse dataJson for document info ---
    80→function parseDocInfo(entry: HistoryEntry) {
    81→  try {
    82→    const parsed = JSON.parse(entry.dataJson)
    83→    const items = parsed.items || []
    84→    const itemCount = items.length
    85→    const firstItem = items[0]
    86→    const namaBarang = firstItem?.deskripsi || ''
    87→    const hargaSatuan = firstItem?.harga || 0
    88→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0)
    89→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0)
    90→    const ppn = parsed.ppn || 0
    91→    const dpPercent = parsed.dp || 0
    92→    const totalHarga = subtotal + (subtotal * ppn / 100)
    93→    // Always derive dpAmount from originalTotal
    94→    const originalTotal = parsed.originalTotal !== undefined ? parsed.originalTotal : totalHarga
    95→    const dpAmount = originalTotal * (dpPercent / 100)
    96→    const sisa = totalHarga - dpAmount
    97→    const lunas = parsed.lunas === true
    98→    const tanggalJatuhTempo = parsed.tanggalJatuhTempo || ''
    99→    const tanggalPelunasan = parsed.tanggalPelunasan || ''
   100→    const referensi = parsed.referensi || ''
   101→    const referensiInvoiceNomor = parsed.referensiInvoiceNomor || ''
   102→    const isPelunasan = entry.docType === 'invoice-pelunasan'
   103→    const uangCapek = parsed.uangCapek || 0
   104→    return { namaBarang, hargaSatuan, totalQty, totalHarga, dpPercent, dp: dpAmount, sisa, lunas, tanggalJatuhTempo, tanggalPelunasan, referensi, referensiInvoiceNomor, isPelunasan, originalTotal, uangCapek, itemCount }
   105→  } catch {
   106→    return { namaBarang: '', hargaSatuan: 0, totalQty: 0, totalHarga: 0, dpPercent: 0, dp: 0, sisa: 0, lunas: false, tanggalJatuhTempo: '', tanggalPelunasan: '', referensi: '', referensiInvoiceNomor: '', isPelunasan: false, originalTotal: 0, uangCapek: 0, itemCount: 0 }
   107→  }
   108→}

   110→// --- Parse dataJson for InvoiceData ---
   111→function parseInvoiceData(entry: HistoryEntry): InvoiceData {
   112→  try {
   113→    const parsed = JSON.parse(entry.dataJson)
   114→    const company: CompanyInfo = {
   115→      nama: parsed.company?.nama || DEFAULT_COMPANY.nama,
   116→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
   117→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
   118→      email: parsed.company?.email || DEFAULT_COMPANY.email,
   119→      npwp: parsed.company?.npwp || '',
   120→      website: parsed.company?.website || '',
   121→      ppn: parsed.company?.ppn ?? parsed.ppn ?? 11,
   122→      logo: parsed.company?.logo || '',
   123→      bankName: parsed.company?.bankName || '',
   124→      bankAccount: parsed.company?.bankAccount || '',
   125→      bankHolder: parsed.company?.bankHolder || '',
   126→      bankName2: parsed.company?.bankName2 || '',
   127→      bankAccount2: parsed.company?.bankAccount2 || '',
   128→      bankHolder2: parsed.company?.bankHolder2 || '',
   129→    }
   130→    const client = parsed.client || { nama: '', kontak: '', alamat: '' }
   131→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number }, i: number) => ({
   132→      id: it.id || `item-${i}`,
   133→      deskripsi: it.deskripsi || '',
   134→      qty: it.qty || 0,
   135→      satuan: it.satuan || '',
   136→      harga: it.harga || 0,
   137→    }))
   138→    const docType = (entry.docType === 'invoice-pelunasan' ? 'invoice-pelunasan' : 'invoice') as 'invoice' | 'invoice-pelunasan'
   139→    return {
   140→      type: docType,
   141→      company,
   142→      nomor: parsed.nomor || entry.nomor || '',
   143→      tanggal: parsed.tanggal || entry.tanggal || '',
   144→      referensi: parsed.referensi || '',
   145→      client,
   146→      items,
   147→      ppn: parsed.ppn ?? 11,
   148→      dp: parsed.dp || 0,
   149→      dpAmount: parsed.dpAmount,
   150→      catatan: parsed.catatan || '',
   151→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   152→      caraPembayaran: parsed.caraPembayaran || '',
   153→      tanggalGiro: parsed.tanggalGiro || '',
   154→      lunas: parsed.lunas === true,
   155→      tanggalPelunasan: parsed.tanggalPelunasan || '',
   156→      referensiInvoiceId: parsed.referensiInvoiceId || '',
   157→      referensiInvoiceNomor: parsed.referensiInvoiceNomor || '',
   158→      originalTotal: parsed.originalTotal,
   159→      uangCapek: parsed.uangCapek || 0,
   160→    }
   161→  } catch {
   162→    const docType = (entry.docType === 'invoice-pelunasan' ? 'invoice-pelunasan' : 'invoice') as 'invoice' | 'invoice-pelunasan'
   163→    return {
   164→      type: docType,
   165→      company: { ...DEFAULT_COMPANY },
   166→      nomor: entry.nomor || '',
   167→      tanggal: entry.tanggal || '',
   168→      referensi: '',
   169→      client: { nama: '', kontak: '', alamat: '' },
   170→      items: [],
   171→      ppn: 11,
   172→      dp: 0,
   173→      catatan: '',
   174→      tanggalJatuhTempo: '',
   175→      caraPembayaran: '',
   176→      tanggalGiro: '',
   177→      lunas: false,
   178→      tanggalPelunasan: '',
   179→      referensiInvoiceId: '',
   180→      referensiInvoiceNomor: '',
   181→      uangCapek: 0,
   182→    }
   183→  }
   184→}
