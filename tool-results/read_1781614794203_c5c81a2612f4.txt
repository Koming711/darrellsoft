    54→// --- Parse dataJson for document info ---
    55→function parseDocInfo(entry: HistoryEntry) {
    56→  try {
    57→    const parsed = JSON.parse(entry.dataJson)
    58→    const items = parsed.items || []
    59→    const firstItem = items[0]
    60→    const namaBarang = firstItem?.deskripsi || ''
    61→    const hargaSatuan = firstItem?.harga || 0
    62→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0)
    63→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0)
    64→    const ppn = parsed.ppn || 0
    65→    const totalHarga = subtotal + (subtotal * ppn / 100)
    66→    const referensi = parsed.referensi || ''
    67→    const tanggalJatuhTempo = parsed.tanggalJatuhTempo || ''
    68→    return { namaBarang, hargaSatuan, totalQty, totalHarga, referensi, tanggalJatuhTempo }
    69→  } catch {
    70→    return { namaBarang: '', hargaSatuan: 0, totalQty: 0, totalHarga: 0, referensi: '', tanggalJatuhTempo: '' }
    71→  }
    72→}

    74→// --- Parse dataJson for PurchaseOrderData ---
    75→function parsePurchaseOrderData(entry: HistoryEntry): PurchaseOrderData {
    76→  try {
    77→    const parsed = JSON.parse(entry.dataJson)
    78→    const company: CompanyInfo = {
    79→      nama: parsed.company?.nama || DEFAULT_COMPANY.nama,
    80→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
    81→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
    82→      email: parsed.company?.email || DEFAULT_COMPANY.email,
    83→      npwp: parsed.company?.npwp || '',
    84→      website: parsed.company?.website || '',
    85→      ppn: parsed.company?.ppn ?? parsed.ppn ?? 0,
    86→      logo: parsed.company?.logo || '',
    87→      bankName: parsed.company?.bankName || '',
    88→      bankAccount: parsed.company?.bankAccount || '',
    89→      bankHolder: parsed.company?.bankHolder || '',
    90→      bankName2: parsed.company?.bankName2 || '',
    91→      bankAccount2: parsed.company?.bankAccount2 || '',
    92→      bankHolder2: parsed.company?.bankHolder2 || '',
    93→    }
    94→    const pemasok = parsed.pemasok || { nama: '', jenisBarang: '', kontak: '', alamat: '' }
    95→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number }, i: number) => ({
    96→      id: it.id || `item-${i}`,
    97→      deskripsi: it.deskripsi || '',
    98→      qty: it.qty || 0,
    99→      satuan: it.satuan || '',
   100→      harga: it.harga || 0,
   101→    }))
   102→    return {
   103→      type: 'purchase-order',
   104→      company,
   105→      nomor: parsed.nomor || entry.nomor || '',
   106→      tanggal: parsed.tanggal || entry.tanggal || '',
   107→      referensi: parsed.referensi || '',
   108→      pemasok,
   109→      items,
   110→      ppn: parsed.ppn ?? 0,
   111→      catatan: parsed.catatan || '',
   112→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   113→      riwayatPotongKertasId: parsed.riwayatPotongKertasId || '',
   114→    }
   115→  } catch {
   116→    return {
   117→      type: 'purchase-order',
   118→      company: { ...DEFAULT_COMPANY },
   119→      nomor: entry.nomor || '',
   120→      tanggal: entry.tanggal || '',
   121→      referensi: '',
   122→      pemasok: { nama: '', jenisBarang: '', kontak: '', alamat: '' },
   123→      items: [],
   124→      ppn: 0,
   125→      catatan: '',
   126→      tanggalJatuhTempo: '',
   127→    }
   128→  }
   129→}
