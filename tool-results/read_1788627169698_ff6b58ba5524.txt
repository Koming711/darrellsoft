    76→    76→// --- Parse dataJson for document info ---
    77→    77→function parseDocInfo(entry: HistoryEntry) {
    78→    78→  try {
    79→    79→    const parsed = JSON.parse(entry.dataJson)
    80→    80→    const items = parsed.items || []
    81→    81→    const itemCount = items.length
    82→    82→    const firstItem = items[0]
    83→    83→    const namaBarang = firstItem?.deskripsi || ''
    84→    84→    const hargaSatuan = firstItem?.harga || 0
    85→    85→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0)
    86→    86→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0)
    87→    87→    const ppn = parsed.ppn || 0
    88→    88→    const dpPercent = parsed.dp || 0
    89→    89→    const totalHarga = subtotal + (subtotal * ppn / 100)
    90→    90→    // Always derive dpAmount from originalTotal
    91→    91→    const originalTotal = parsed.originalTotal !== undefined ? parsed.originalTotal : totalHarga
    92→    92→    const dpAmount = originalTotal * (dpPercent / 100)
    93→    93→    const sisa = totalHarga - dpAmount
    94→    94→    const lunas = parsed.lunas === true
    95→    95→    const tanggalJatuhTempo = parsed.tanggalJatuhTempo || ''
    96→    96→    const tanggalPelunasan = parsed.tanggalPelunasan || ''
    97→    97→    const referensi = parsed.referensi || ''
    98→    98→    const referensiInvoiceNomor = parsed.referensiInvoiceNomor || ''
    99→    99→    const isPelunasan = entry.docType === 'invoice-pelunasan'
   100→   100→    const uangCapek = parsed.uangCapek || 0
   101→   101→    return { namaBarang, hargaSatuan, totalQty, totalHarga, dpPercent, dp: dpAmount, sisa, lunas, tanggalJatuhTempo, tanggalPelunasan, referensi, referensiInvoiceNomor, isPelunasan, originalTotal, uangCapek, itemCount }
   102→   102→  } catch {
   103→   103→    return { namaBarang: '', hargaSatuan: 0, totalQty: 0, totalHarga: 0, dpPercent: 0, dp: 0, sisa: 0, lunas: false, tanggalJatuhTempo: '', tanggalPelunasan: '', referensi: '', referensiInvoiceNomor: '', isPelunasan: false, originalTotal: 0, uangCapek: 0, itemCount: 0 }
   104→   104→  }
   105→   105→}

   108→   108→function parseInvoiceData(entry: HistoryEntry): InvoiceData {
   109→   109→  try {
   110→   110→    const parsed = JSON.parse(entry.dataJson)
   111→   111→    const company: CompanyInfo = {

   113→   113→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
   114→   114→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
   115→   115→      email: parsed.company?.email || DEFAULT_COMPANY.email,
   116→   116→      npwp: parsed.company?.npwp || '',

   118→   118→      ppn: parsed.company?.ppn ?? parsed.ppn ?? 11,
   119→   119→      logo: parsed.company?.logo || '',
   120→   120→      bankName: parsed.company?.bankName || '',
   121→   121→      bankAccount: parsed.company?.bankAccount || '',
   122→   122→      bankHolder: parsed.company?.bankHolder || '',
   123→   123→      bankName2: parsed.company?.bankName2 || '',
   124→   124→      bankAccount2: parsed.company?.bankAccount2 || '',
   125→   125→      bankHolder2: parsed.company?.bankHolder2 || '',
   126→   126→    }
   127→   127→    const client = parsed.client || { nama: '', kontak: '', alamat: '' }
   128→   128→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number }, i: number) => ({
   129→   129→      id: it.id || `item-${i}`,
   130→   130→      deskripsi: it.deskripsi || '',
   131→   131→      qty: it.qty || 0,
   132→   132→      satuan: it.satuan || '',
   133→   133→      harga: it.harga || 0,
   134→   134→    }))
   135→   135→    const docType = (entry.docType === 'invoice-pelunasan' ? 'invoice-pelunasan' : 'invoice') as 'invoice' | 'invoice-pelunasan'
   136→   136→    return {
   137→   137→      type: docType,
   138→   138→      company,
   139→   139→      nomor: parsed.nomor || entry.nomor || '',
   140→   140→      tanggal: parsed.tanggal || entry.tanggal || '',
   141→   141→      referensi: parsed.referensi || '',
   142→   142→      client,
   143→   143→      items,
   144→   144→      ppn: parsed.ppn ?? 11,
   145→   145→      dp: parsed.dp || 0,
   146→   146→      dpAmount: parsed.dpAmount,
   147→   147→      catatan: parsed.catatan || '',
   148→   148→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   149→   149→      caraPembayaran: parsed.caraPembayaran || '',
   150→   150→      tanggalGiro: parsed.tanggalGiro || '',
   151→   151→      lunas: parsed.lunas === true,
   152→   152→      tanggalPelunasan: parsed.tanggalPelunasan || '',
   153→   153→      referensiInvoiceId: parsed.referensiInvoiceId || '',
   154→   154→      referensiInvoiceNomor: parsed.referensiInvoiceNomor || '',
   155→   155→      originalTotal: parsed.originalTotal,
   156→   156→      uangCapek: parsed.uangCapek || 0,
   157→   157→    }
   158→   158→  } catch {
   159→   159→    const docType = (entry.docType === 'invoice-pelunasan' ? 'invoice-pelunasan' : 'invoice') as 'invoice' | 'invoice-pelunasan'
   160→   160→    return {
   161→   161→      type: docType,
   162→   162→      company: { ...DEFAULT_COMPANY },
   163→   163→      nomor: entry.nomor || '',
   164→   164→      tanggal: entry.tanggal || '',
   165→   165→      referensi: '',
   166→   166→      client: { nama: '', kontak: '', alamat: '' },
   167→   167→      items: [],
   168→   168→      ppn: 11,
   169→   169→      dp: 0,
   170→   170→      catatan: '',
   171→   171→      tanggalJatuhTempo: '',
   172→   172→      caraPembayaran: '',
   173→   173→      tanggalGiro: '',
   174→   174→      lunas: false,
   175→   175→      tanggalPelunasan: '',
   176→   176→      referensiInvoiceId: '',
   177→   177→      referensiInvoiceNomor: '',
   178→   178→      uangCapek: 0,
   179→   179→    }
   180→   180→  }
   181→   181→}
