   374→  const getFinishingCost = (finishing: Finishing): { cost: number; isMin: boolean; breakdown: string } => {
   375→    const qty = parseInt(formData.quantity) || 0
   376→    const cw = parseFloat(formData.cutWidth) || 0
   377→    const ch = parseFloat(formData.cutHeight) || 0
   378→
   379→    const minSheets = finishing.minimumSheets
   380→    const minPrice = finishing.minimumPrice
   381→    const hargaLebih = finishing.additionalPrice
   382→    const hargaPerCm = finishing.pricePerCm
   383→    const isPond = finishing.name.toLowerCase().includes('pond')
   384→
   385→    if (qty <= 0) {
   386→      return { cost: minPrice, isMin: true, breakdown: `Harga minimum: Rp ${Math.round(minPrice).toLocaleString('id-ID')}` }
   387→    }
   388→
   389→    if (isPond) {
   390→      if (qty <= minSheets) {
   391→        return {
   392→          cost: minPrice,
   393→          isMin: true,
   394→          breakdown: `Qty ${qty} ≤ minim ${minSheets} → Harga minimum: Rp ${Math.round(minPrice).toLocaleString('id-ID')}`
   395→        }
   396→      }
   397→      const selisih = qty - minSheets
   398→      const tambahan = selisih * hargaLebih
   399→      const total = minPrice + tambahan
   400→      return {
   401→        cost: total,
   402→        isMin: false,
   403→        breakdown: `Harga minimum Rp ${Math.round(minPrice).toLocaleString('id-ID')} + ((${qty} - ${minSheets}) × Rp ${Math.round(hargaLebih).toLocaleString('id-ID')}) = Rp ${Math.round(total).toLocaleString('id-ID')}`
   404→      }
   405→    }
   406→
   407→    let part1 = 0
   408→    let part1Text = ''
   409→    if (qty > minSheets && hargaLebih > 0) {
   410→      const selisih = qty - minSheets
   411→      part1 = selisih * hargaLebih
   412→      part1Text = `(${qty} - ${minSheets}) × Rp ${Math.round(hargaLebih).toLocaleString('id-ID')} = Rp ${Math.round(part1).toLocaleString('id-ID')}`
   413→    }
   414→
   415→    let part2 = 0
   416→    let part2Text = ''
   417→    if (cw > 0 && ch > 0 && hargaPerCm > 0) {
   418→      const areaCost = cw * ch * hargaPerCm
   419→      part2 = areaCost * qty
   420→      part2Text = `(${cw} × ${ch}) × Rp ${Math.round(hargaPerCm).toLocaleString('id-ID')} × ${qty} = Rp ${Math.round(part2).toLocaleString('id-ID')}`
   421→    }
   422→
   423→    const total = part1 + part2
   424→
   425→    if (total <= minPrice) {
   426→      const parts = [part1Text, part2Text].filter(Boolean)
   427→      const calcText = parts.length > 0 ? `${parts.join(' + ')} = Rp ${Math.round(total).toLocaleString('id-ID')} ≤ Rp ${Math.round(minPrice).toLocaleString('id-ID')}` : ''
   428→      return {
   429→        cost: minPrice,
   430→        isMin: true,
   431→        breakdown: calcText ? `${calcText} → Harga minimum: Rp ${Math.round(minPrice).toLocaleString('id-ID')}` : `Harga minimum: Rp ${Math.round(minPrice).toLocaleString('id-ID')}`
   432→      }
   433→    }
   434→
   435→    const parts = [part1Text, part2Text].filter(Boolean)
   436→    const breakdown = parts.join(' + ') + ` = Rp ${Math.round(total).toLocaleString('id-ID')}`
   437→
   438→    return { cost: total, isMin: false, breakdown }
   439→  }
