    33→// --- Types ---
    34→interface HistoryEntry {
    35→  id: string;
    36→  docType: string;
    37→  nomor: string;
    38→  tanggal: string;
    39→  pihakKedua: string;
    40→  total: string;

    44→
    45→// --- Helper ---
    46→function getTodayStr(): string {
    47→  const d = new Date();
    48→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
    49→}
    50→
    51→// --- Parse dataJson for document info ---
    52→function parseDocInfo(entry: HistoryEntry) {
    53→  try {
    54→    const parsed = JSON.parse(entry.dataJson);
    55→    const items = parsed.items || [];
    56→    const firstItem = items[0];
    57→    const namaBarang = firstItem?.deskripsi || '';
    58→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0);
    59→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0);
    60→    const ppn = parsed.ppn || 0;
    61→    const dpPercent = parsed.dp || 0;
    62→    const totalHarga = subtotal + (subtotal * ppn / 100);
    63→    const dpAmount = totalHarga * (dpPercent / 100);
    64→    const sisa = totalHarga - dpAmount;
    65→    const lunas = parsed.lunas === true;
    66→    const tanggalJatuhTempo = parsed.tanggalJatuhTempo || '';
    67→    const tanggalPelunasan = parsed.tanggalPelunasan || '';
    68→    return { namaBarang, totalQty, totalHarga, dpPercent, dp: dpAmount, sisa, lunas, tanggalJatuhTempo, tanggalPelunasan };
    69→  } catch {
    70→    return { namaBarang: '', totalQty: 0, totalHarga: 0, dpPercent: 0, dp: 0, sisa: 0, lunas: false, tanggalJatuhTempo: '', tanggalPelunasan: '' };
    71→  }
    72→}
    73→
    74→// --- Parse dataJson for InvoiceData ---
    75→function parseInvoiceData(entry: HistoryEntry): InvoiceData {
    76→  try {
    77→    const parsed = JSON.parse(entry.dataJson);
    78→    const company: CompanyInfo = {
    79→      nama: parsed.company?.nama || DEFAULT_COMPANY.nama,
    80→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
    81→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
    82→      email: parsed.company?.email || DEFAULT_COMPANY.email,
    83→      npwp: parsed.company?.npwp || '',
    84→      website: parsed.company?.website || '',
    85→      ppn: parsed.company?.ppn ?? parsed.ppn ?? 11,
    86→      logo: parsed.company?.logo || '',
    87→      bankName: parsed.company?.bankName || '',
    88→      bankAccount: parsed.company?.bankAccount || '',
    89→      bankHolder: parsed.company?.bankHolder || '',
    90→      bankName2: parsed.company?.bankName2 || '',
    91→      bankAccount2: parsed.company?.bankAccount2 || '',
    92→      bankHolder2: parsed.company?.bankHolder2 || '',
    93→    };
    94→    const client = parsed.client || { nama: '', kontak: '', alamat: '' };
    95→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number }, i: number) => ({
    96→      id: it.id || `item-${i}`,
    97→      deskripsi: it.deskripsi || '',
    98→      qty: it.qty || 0,
    99→      satuan: it.satuan || '',
   100→      harga: it.harga || 0,
   101→    }));
   102→    return {
   103→      type: 'invoice',
   104→      company,
   105→      nomor: parsed.nomor || entry.nomor || '',
   106→      tanggal: parsed.tanggal || entry.tanggal || '',
   107→      referensi: parsed.referensi || '',
   108→      client,
   109→      items,
   110→      ppn: parsed.ppn ?? 11,
   111→      dp: parsed.dp || 0,
   112→      catatan: parsed.catatan || '',
   113→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   114→      caraPembayaran: parsed.caraPembayaran || '',
   115→      tanggalGiro: parsed.tanggalGiro || '',
   116→      lunas: parsed.lunas === true,
   117→      tanggalPelunasan: parsed.tanggalPelunasan || '',
   118→    };
   119→  } catch {
   120→    return {
   121→      type: 'invoice',
   122→      company: { ...DEFAULT_COMPANY },
   123→      nomor: entry.nomor || '',
   124→      tanggal: entry.tanggal || '',
   125→      referensi: '',
   126→      client: { nama: '', kontak: '', alamat: '' },
   127→      items: [],
   128→      ppn: 11,
   129→      dp: 0,
   130→      catatan: '',
   131→      tanggalJatuhTempo: '',
   132→      caraPembayaran: '',
   133→      tanggalGiro: '',
   134→      lunas: false,
   135→      tanggalPelunasan: '',
   136→    };
   137→  }
   138→}

   140→export function InvoicePelunasanEditor() {
   141→  // Local invoice state (not using global store to avoid conflicts with regular editor)
   142→  const [invoiceData, setInvoiceData] = useState<InvoiceData | null>(null);
   143→  const [selectedEntry, setSelectedEntry] = useState<HistoryEntry | null>(null);
   144→  const [invoiceHistory, setInvoiceHistory] = useState<HistoryEntry[]>([]);
   145→  const [loading, setLoading] = useState(true);
   146→  const [searchQuery, setSearchQuery] = useState('');
   147→  const [dropdownOpen, setDropdownOpen] = useState(false);
   148→  const [pelunasanSaving, setPelunasanSaving] = useState(false);
   149→
   150→  // Pelunasan fields
   151→  const [lunasToggle, setLunasToggle] = useState(false);
   152→  const [tanggalPelunasan, setTanggalPelunasan] = useState('');
   153→  const [tanggalJatuhTempo, setTanggalJatuhTempo] = useState('');
   154→  const [caraPembayaran, setCaraPembayaran] = useState('');
   155→  const [tanggalGiro, setTanggalGiro] = useState('');
   156→  const [originalDpAmount, setOriginalDpAmount] = useState(0);
   157→
   158→  const fetchHistory = useCallback(async () => {
   159→    try {
   160→      setLoading(true);
   161→      const headers = getAuthHeaders();
   162→      const res = await fetch('/api/history?docType=invoice', { headers });
   163→      if (res.ok) {
   164→        const json = await res.json();
   165→        setInvoiceHistory(json.data || []);
   166→      }
   167→    } catch (err) {
   168→      console.error('Failed to fetch invoice history:', err);
   169→    } finally {
   170→      setLoading(false);
   171→    }
   172→  }, []);
   173→
   174→  useEffect(() => {
   175→    fetchHistory();
   176→  }, [fetchHistory]);
   177→
   178→  useEffect(() => {
   179→    const handler = () => fetchHistory();
   180→    window.addEventListener('dokupro:history-updated', handler);
   181→    return () => window.removeEventListener('dokupro:history-updated', handler);
   182→  }, [fetchHistory]);
   183→
   184→  // Filter pending invoices (with DP and not yet lunas)
   185→  const pendingInvoices = useMemo(() => {
   186→    return invoiceHistory.filter(entry => {
   187→      const info = parseDocInfo(entry);
   188→      return info.dpPercent > 0 && !info.lunas && info.sisa > 0;
   189→    });
   190→  }, [invoiceHistory]);
   191→
   192→  // Filter by search
   193→  const filteredPending = useMemo(() => {
   194→    if (!searchQuery.trim()) return pendingInvoices;
   195→    const q = searchQuery.toLowerCase().trim();
   196→    return pendingInvoices.filter(entry => {
   197→      const info = parseDocInfo(entry);
   198→      return entry.nomor?.toLowerCase().includes(q) || entry.pihakKedua?.toLowerCase().includes(q) || info.namaBarang?.toLowerCase().includes(q);
   199→    });
   200→  }, [pendingInvoices, searchQuery]);
   201→
   202→  // Select an invoice
   203→  const selectInvoice = (entry: HistoryEntry) => {
   204→    const parsed = parseInvoiceData(entry);
   205→    const info = parseDocInfo(entry);
   206→    setInvoiceData(parsed);
   207→    setSelectedEntry(entry);
   208→    setLunasToggle(info.lunas);
   209→    setTanggalPelunasan(info.tanggalPelunasan || getTodayStr());
   210→    setTanggalJatuhTempo(info.tanggalJatuhTempo || '');
   211→    setCaraPembayaran(parsed.caraPembayaran || '');
   212→    setTanggalGiro(parsed.tanggalGiro || '');
   213→    // Store original DP amount so it stays fixed in pelunasan view
   214→    const sub = parsed.items.reduce((s, item) => s + item.qty * item.harga, 0);
   215→    const ppn = sub * ((parsed.ppn || 0) / 100);
   216→    const tot = sub + ppn;
   217→    const dpAmt = parsed.dpAmount !== undefined ? parsed.dpAmount : tot * ((parsed.dp || 0) / 100);
   218→    setOriginalDpAmount(dpAmt);
   219→    setDropdownOpen(false);
   220→  };
   221→
   222→  // Clear selection
   223→  const clearSelection = () => {
   224→    setInvoiceData(null);
   225→    setSelectedEntry(null);
   226→    setLunasToggle(false);
   227→    setTanggalPelunasan('');
   228→    setTanggalJatuhTempo('');
   229→    setCaraPembayaran('');
   230→    setTanggalGiro('');
   231→    setOriginalDpAmount(0);
   232→  };
   233→
   234→  // Update invoice data locally
   235→  const updateInvoice = (updates: Partial<InvoiceData>) => {
   236→    setInvoiceData(prev => prev ? { ...prev, ...updates } : null);
   237→  };
   238→
   239→  const updateCompany = (company: CompanyInfo) => {
   240→    setInvoiceData(prev => prev ? { ...prev, company } : null);
   241→  };
   242→
   243→  const updateClient = (field: string, value: string) => {
   244→    setInvoiceData(prev => prev ? { ...prev, client: { ...prev.client, [field]: value } } : null);
   245→  };
   246→
   247→  // Build preview data with pelunasan overrides
   248→  const previewData: InvoiceData | null = useMemo(() => {
   249→    if (!invoiceData) {
   250→      // Default preview when no invoice is selected
   251→      return {
   252→        type: 'invoice-pelunasan',
   253→        company: { ...DEFAULT_COMPANY },
   254→        nomor: '-',
   255→        tanggal: getTodayStr(),
   256→        referensi: '',
   257→        client: { nama: '', kontak: '', alamat: '' },
   258→        items: [],
   259→        ppn: 11,
   260→        dp: 0,
   261→        catatan: '',
   262→        tanggalJatuhTempo: '',
   263→        caraPembayaran: '',
   264→        tanggalGiro: '',
   265→        lunas: false,
   266→        tanggalPelunasan: '',
   267→      };
   268→    }
   269→    return {
   270→      ...invoiceData,
   271→      tanggalJatuhTempo,
   272→      caraPembayaran: caraPembayaran as InvoiceData['caraPembayaran'],
   273→      tanggalGiro,
   274→      lunas: lunasToggle,
   275→      tanggalPelunasan: lunasToggle ? tanggalPelunasan : '',
   276→    };
   277→  }, [invoiceData, tanggalJatuhTempo, caraPembayaran, tanggalGiro, lunasToggle, tanggalPelunasan]);
   278→
   279→  // Calculate amounts — DP is FIXED at original amount
   280→  const subtotal = invoiceData?.items.reduce((sum, item) => sum + item.qty * item.harga, 0) || 0;
   281→  const ppnAmount = subtotal * ((invoiceData?.ppn || 0) / 100);
   282→  const total = subtotal + ppnAmount;
   283→  const dpPercent = invoiceData?.dp || 0;
   284→  const dpAmount = originalDpAmount;
   285→  const sisa = total - dpAmount;
   286→
   287→  // Save pelunasan — update the existing history entry
   288→  const handleSavePelunasan = async () => {
   289→    if (!selectedEntry || !invoiceData) return;
   290→    setPelunasanSaving(true);
   291→    try {
   292→      const parsed = JSON.parse(selectedEntry.dataJson);
   293→      // Update pelunasan fields
   294→      parsed.tanggalJatuhTempo = tanggalJatuhTempo;
   295→      parsed.caraPembayaran = caraPembayaran;
   296→      parsed.tanggalGiro = tanggalGiro;
   297→      if (lunasToggle) {
   298→        parsed.lunas = true;
   299→        parsed.tanggalPelunasan = tanggalPelunasan || getTodayStr();
   300→      } else {
   301→        parsed.lunas = false;
   302→        parsed.tanggalPelunasan = '';
   303→      }
   304→      // Also update any edited fields from the form
   305→      parsed.company = invoiceData.company;
   306→      parsed.client = invoiceData.client;
   307→      parsed.items = invoiceData.items;
   308→      parsed.ppn = invoiceData.ppn;
   309→      parsed.dp = invoiceData.dp;
   310→      parsed.dpAmount = originalDpAmount;
   311→      parsed.catatan = invoiceData.catatan;
   312→      parsed.nomor = invoiceData.nomor;
   313→      parsed.tanggal = invoiceData.tanggal;
   314→      parsed.referensi = invoiceData.referensi;
   315→      delete parsed.statusPembayaran;
   316→
   317→      const newDataJson = JSON.stringify(parsed);
   318→      const res = await fetcher(`/api/history/${selectedEntry.id}`, {
   319→        method: 'PUT',
   320→        headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   321→        body: JSON.stringify({ dataJson: newDataJson }),
   322→      });
   323→      if (res.ok) {
   324→        toast.success(lunasToggle ? 'Pelunasan berhasil dicatat!' : 'Invoice berhasil diperbarui');
   325→        clearSelection();
   326→        fetchHistory();
   327→        notifyDataChange('invoice');
   328→      } else {
   329→        toast.error('Gagal menyimpan perubahan');
   330→      }
   331→    } catch {
   332→      toast.error('Gagal menyimpan perubahan');
   333→    } finally {
   334→      setPelunasanSaving(false);
   335→    }
   336→  };
   337→
   338→  return (
   339→    <DocumentEditorLayout
   340→      title="Invoice Pelunasan"
   341→      previewContent={<InvoicePreview data={previewData!} showPelunasanLabel dpAmountOverride={originalDpAmount} />}
   342→      actions={
   343→        invoiceData ? (
   344→          <div className="flex items-center gap-2 flex-wrap">
   345→            <Button
   346→              variant="outline"
   347→              size="sm"
   348→              onClick={clearSelection}
   349→              className="border-gray-200 text-gray-600 hover:bg-gray-50"
   350→            >
   351→              <X className="mr-1.5 h-3.5 w-3.5" />
   352→              Batal
   353→            </Button>
   354→            <Button
   355→              size="sm"
   356→              onClick={() => {
   357→                toast.dismiss();
   358→                const origTitle = document.title;
   359→                document.title = ' ';
   360→                setTimeout(() => {
   361→                  window.print();
   362→                  document.title = origTitle;
   363→                }, 100);
   364→              }}
   365→              className="bg-emerald-600 hover:bg-emerald-700"
   366→            >
   367→              Cetak
   368→            </Button>
   369→            <Button
   370→              size="sm"
   371→              onClick={handleSavePelunasan}
   372→              disabled={pelunasanSaving}
   373→              className={cn(
   374→                'text-white',
   375→                lunasToggle
   376→                  ? 'bg-green-600 hover:bg-green-700'
   377→                  : 'bg-blue-600 hover:bg-blue-700'
   378→              )}
   379→            >
   380→              {pelunasanSaving ? (
   381→                <><Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" /> Menyimpan...</>
   382→              ) : (
   383→                <><CheckCircle2 className="mr-1.5 h-3.5 w-3.5" /> {lunasToggle ? 'Simpan Pelunasan' : 'Simpan Perubahan'}</>
   384→              )}
   385→            </Button>
   386→          </div>
   387→        ) : undefined
   388→      }
   389→    >
   390→      {/* Invoice Selector */}
   391→      <div className="rounded-lg border bg-card p-3 sm:p-4 shadow-sm">
   392→        <h3 className="mb-3 text-xs font-semibold uppercase tracking-wider text-muted-foreground flex items-center gap-1.5">
   393→          <Wallet className="w-3.5 h-3.5 text-amber-600" />
   394→          Pilih Invoice Belum Lunas
   395→        </h3>
   396→        <Popover open={dropdownOpen} onOpenChange={setDropdownOpen}>
   397→          <PopoverAnchor asChild>
   398→            <div className="relative">
   399→              <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 w-3.5 h-3.5 text-slate-400" />
   400→              <input
   401→                type="text"
   402→                placeholder={selectedEntry ? `${selectedEntry.nomor} — ${selectedEntry.pihakKedua}` : "Cari no. invoice, customer..."}
   403→                value={selectedEntry ? `${selectedEntry.nomor} — ${selectedEntry.pihakKedua}` : searchQuery}
   404→                onChange={(e) => { setSearchQuery(e.target.value); if (selectedEntry) { clearSelection(); } }}
   405→                onFocus={() => { if (selectedEntry) { clearSelection(); } setDropdownOpen(true); }}
   406→                className="flex h-9 w-full rounded-md border border-input bg-transparent pl-8 pr-8 py-2 text-sm shadow-xs transition-[color,box-shadow] outline-none focus-visible:border-ring focus-visible:ring-ring/50 focus-visible:ring-[3px]"
   407→              />
   408→              {selectedEntry && (
   409→                <button
   410→                  type="button"
   411→                  onClick={(e) => { e.preventDefault(); clearSelection(); }}
   412→                  className="absolute right-2 top-1/2 -translate-y-1/2 text-slate-400 hover:text-slate-600"
   413→                >
   414→                  <X className="w-4 h-4" />
   415→                </button>
   416→              )}
   417→            </div>
   418→          </PopoverAnchor>
   419→          <PopoverContent
   420→            align="start"
   421→            className="p-0 w-[var(--radix-popover-trigger-width)] max-h-64 overflow-y-auto"
   422→            onOpenAutoFocus={(e) => e.preventDefault()}
   423→          >
   424→            {loading ? (
   425→              <div className="px-3 py-4 text-center"><Loader2 className="w-5 h-5 mx-auto text-blue-500 animate-spin" /><p className="text-xs text-slate-400 mt-1">Memuat...</p></div>
   426→            ) : filteredPending.length > 0 ? (
   427→              <div>
   428→                <div className="px-3 py-1.5 text-[10px] font-semibold text-slate-400 uppercase bg-amber-50 border-b border-amber-100 sticky top-0">
   429→                  Menunggu Pelunasan ({filteredPending.length})
   430→                </div>
   431→                {filteredPending.map((entry) => {
   432→                  const info = parseDocInfo(entry);
   433→                  const isOverdue = info.tanggalJatuhTempo && new Date(info.tanggalJatuhTempo) < new Date(getTodayStr());
   434→                  return (
   435→                    <button
   436→                      key={entry.id}
   437→                      type="button"
   438→                      onMouseDown={(e) => { e.preventDefault(); selectInvoice(entry); }}
   439→                      className={cn(
   440→                        'w-full text-left px-3 py-2.5 text-sm transition-colors border-b border-slate-50 last:border-b-0',
   441→                        selectedEntry?.id === entry.id ? 'bg-amber-50' : 'hover:bg-amber-50/50'
   442→                      )}
   443→                    >
   444→                      <div className="flex items-center justify-between gap-2">
   445→                        <div className="min-w-0">
   446→                          <div className="flex items-center gap-1.5">
   447→                            <span className="font-semibold text-violet-700 text-xs">{entry.nomor || '-'}</span>
   448→                            {isOverdue && <AlertTriangle className="w-3 h-3 text-red-500" />}
   449→                          </div>
   450→                          <p className="text-slate-600 text-[11px] truncate">{entry.pihakKedua || '-'}</p>
   451→                        </div>
   452→                        <div className="text-right shrink-0">
   453→                          <p className="text-red-600 font-bold text-xs">{formatRupiah(info.sisa)}</p>
   454→                          <p className="text-[9px] text-violet-500">DP {info.dpPercent}%</p>
   455→                        </div>
   456→                      </div>
   457→                    </button>
   458→                  );
   459→                })}
   460→              </div>
   461→            ) : (
   462→              <div className="px-3 py-4 text-center">
   463→                <CheckCircle2 className="w-8 h-8 mx-auto text-green-300 mb-2" />
   464→                <p className="text-xs text-slate-400">Semua invoice sudah lunas</p>
   465→              </div>
   466→            )}
   467→          </PopoverContent>
   468→        </Popover>
   469→
   470→        {/* Quick list when no invoice selected */}
   471→        {!selectedEntry && !loading && pendingInvoices.length > 0 && (
   472→          <div className="mt-3 divide-y divide-slate-100 max-h-48 overflow-y-auto rounded-lg border border-slate-200">
   473→            {pendingInvoices.slice(0, 10).map((entry) => {
   474→              const info = parseDocInfo(entry);
   475→              const isOverdue = info.tanggalJatuhTempo && new Date(info.tanggalJatuhTempo) < new Date(getTodayStr());
   476→              return (
   477→                <div
   478→                  key={entry.id}
   479→                  onClick={() => selectInvoice(entry)}
   480→                  className="px-3 py-2.5 hover:bg-amber-50/50 cursor-pointer transition-colors"
   481→                >
   482→                  <div className="flex items-center justify-between gap-2">
   483→                    <div className="min-w-0">
   484→                      <div className="flex items-center gap-1.5">
   485→                        <span className="font-semibold text-violet-700 text-xs">{entry.nomor || '-'}</span>
   486→                        <CircleDot className="w-2.5 h-2.5 text-red-500" />
   487→                        {isOverdue && <AlertTriangle className="w-3 h-3 text-red-500" />}
   488→                      </div>
   489→                      <p className="text-slate-600 text-[11px] truncate">{entry.pihakKedua || '-'}</p>
