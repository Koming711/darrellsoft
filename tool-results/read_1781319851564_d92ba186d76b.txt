    65→function getTodayStr(): string {
    66→  const d = new Date()
    67→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
    68→}

    74→// --- Parse dataJson for document info ---
    75→function parseDocInfo(entry: HistoryEntry) {
    76→  try {
    77→    const parsed = JSON.parse(entry.dataJson)
    78→    const items = parsed.items || []
    79→    const firstItem = items[0]
    80→    const namaBarang = firstItem?.deskripsi || ''
    81→    const hargaSatuan = firstItem?.harga || 0
    82→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0)
    83→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0)
    84→    const ppn = parsed.ppn || 0
    85→    const dpPercent = parsed.dp || 0
    86→    const totalHarga = subtotal + (subtotal * ppn / 100)
    87→    // Use saved dpAmount if available
    88→    // If not, use originalTotal * dpPercent to get the correct original DP (not affected by pelunasan additions)
    89→    // Only fall back to totalHarga * dpPercent if originalTotal is also missing
    90→    const originalTotal = parsed.originalTotal !== undefined ? parsed.originalTotal : totalHarga
    91→    const dpAmount = parsed.dpAmount !== undefined ? parsed.dpAmount : originalTotal * (dpPercent / 100)
    92→    const sisa = totalHarga - dpAmount
    93→    const lunas = parsed.lunas === true
    94→    const tanggalJatuhTempo = parsed.tanggalJatuhTempo || ''
    95→    const tanggalPelunasan = parsed.tanggalPelunasan || ''
    96→    const referensi = parsed.referensi || ''
    97→    return { namaBarang, hargaSatuan, totalQty, totalHarga, dpPercent, dp: dpAmount, sisa, lunas, tanggalJatuhTempo, tanggalPelunasan, referensi }
    98→  } catch {
    99→    return { namaBarang: '', hargaSatuan: 0, totalQty: 0, totalHarga: 0, dpPercent: 0, dp: 0, sisa: 0, lunas: false, tanggalJatuhTempo: '', tanggalPelunasan: '', referensi: '' }
   100→  }
   101→}

   103→// --- Parse dataJson for InvoiceData ---
   104→function parseInvoiceData(entry: HistoryEntry): InvoiceData {
   105→  try {
   106→    const parsed = JSON.parse(entry.dataJson)
   107→    const company: CompanyInfo = {
   108→      nama: parsed.company?.nama || DEFAULT_COMPANY.nama,
   109→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
   110→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
   111→      email: parsed.company?.email || DEFAULT_COMPANY.email,
   112→      npwp: parsed.company?.npwp || '',
   113→      website: parsed.company?.website || '',
   114→      ppn: parsed.company?.ppn ?? parsed.ppn ?? 11,
   115→      logo: parsed.company?.logo || '',
   116→      bankName: parsed.company?.bankName || '',
   117→      bankAccount: parsed.company?.bankAccount || '',
   118→      bankHolder: parsed.company?.bankHolder || '',
   119→      bankName2: parsed.company?.bankName2 || '',
   120→      bankAccount2: parsed.company?.bankAccount2 || '',
   121→      bankHolder2: parsed.company?.bankHolder2 || '',
   122→    }
   123→    const client = parsed.client || { nama: '', kontak: '', alamat: '' }
   124→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number }, i: number) => ({
   125→      id: it.id || `item-${i}`,
   126→      deskripsi: it.deskripsi || '',
   127→      qty: it.qty || 0,
   128→      satuan: it.satuan || '',
   129→      harga: it.harga || 0,
   130→    }))
   131→    return {
   132→      type: 'invoice',
   133→      company,
   134→      nomor: parsed.nomor || entry.nomor || '',
   135→      tanggal: parsed.tanggal || entry.tanggal || '',
   136→      referensi: parsed.referensi || '',
   137→      client,
   138→      items,
   139→      ppn: parsed.ppn ?? 11,
   140→      dp: parsed.dp || 0,
   141→      dpAmount: parsed.dpAmount,
   142→      originalTotal: parsed.originalTotal,
   143→      catatan: parsed.catatan || '',
   144→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   145→      caraPembayaran: parsed.caraPembayaran || '',
   146→      tanggalGiro: parsed.tanggalGiro || '',
   147→      lunas: parsed.lunas === true,
   148→      tanggalPelunasan: parsed.tanggalPelunasan || '',
   149→    }
   150→  } catch {
   151→    return {
   152→      type: 'invoice',
   153→      company: { ...DEFAULT_COMPANY },
   154→      nomor: entry.nomor || '',
   155→      tanggal: entry.tanggal || '',
   156→      referensi: '',
   157→      client: { nama: '', kontak: '', alamat: '' },
   158→      items: [],
   159→      ppn: 11,
   160→      dp: 0,
   161→      catatan: '',
   162→      tanggalJatuhTempo: '',
   163→      caraPembayaran: '',
   164→      tanggalGiro: '',
   165→      lunas: false,
   166→      tanggalPelunasan: '',
   167→    }
   168→  }
   169→}

   633→      const headers = getAuthHeaders()
   634→      const res = await fetch('/api/history?docType=invoice', { headers })
   635→      if (res.ok) {
   636→        const json = await res.json()
   637→        setInvoiceHistory(json.data || [])

   639→    } catch (err) {
   640→      console.error('Failed to fetch invoice history:', err)

   642→      setLoading(false)
   643→    }
   644→  }, [])
   645→
   646→  useEffect(() => {
   647→    fetchHistory()
   648→  }, [fetchHistory])
   649→
   650→  useEffect(() => {
   651→    const handler = () => fetchHistory()
   652→    window.addEventListener('dokupro:history-updated', handler)
   653→    return () => window.removeEventListener('dokupro:history-updated', handler)
   654→  }, [fetchHistory])

   656→  // Filter: only invoices with DP and not yet lunas

   658→  const pendingInvoices = useMemo(() => {
   659→    return invoiceHistory.filter(entry => {
   660→      const info = parseDocInfo(entry)
