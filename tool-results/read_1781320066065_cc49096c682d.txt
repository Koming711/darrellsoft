     1→'use client';
     2→
     3→import { useEffect, useState, useCallback, useRef, useMemo } from 'react';
     4→import { useSearchParams, useRouter } from 'next/navigation';
     5→import { Input } from '@/components/ui/input';
     6→import { Label } from '@/components/ui/label';
     7→import { Textarea } from '@/components/ui/textarea';
     8→import { Popover, PopoverContent, PopoverAnchor } from '@/components/ui/popover';
     9→import { useDokuproStore } from '@/lib/store';
    10→import { CompanyFields } from './company-fields';
    11→import { ItemsFields } from './items-fields';
    12→import { InvoicePreview } from './invoice-preview';
    13→import { DocumentEditorLayout } from './document-editor-layout';
    14→import { DocumentActionButtons } from './document-action-buttons';
    15→import { formatRupiah } from '@/lib/format';
    16→import { getAuthHeaders } from '@/lib/auth';
    17→import { Truck, ImageIcon, Loader2 } from 'lucide-react';
    18→import { Button } from '@/components/ui/button';
    19→import { toast } from 'sonner';
    20→import { toJpeg } from 'html-to-image';
    21→import type { InvoiceData } from '@/lib/types';
    22→
    23→interface CustomerItem {
    24→  id: string;
    25→  name: string;
    26→  companyName?: string;
    27→  phone?: string;
    28→  email?: string;
    29→  address?: string;
    30→}
    31→
    32→interface RiwayatCetakanItem {
    33→  id: string;
    34→  nomorUrut: string;
    35→  printName: string;
    36→  customerName: string;
    37→  paperName: string;
    38→  paperGrammage: string;
    39→  cutWidth: string;
    40→  cutHeight: string;
    41→  quantity: string;
    42→  jumlahPesanan: string;
    43→  berapaMata: string;
    44→  warna: string;
    45→  machineName: string;
    46→  ongkosCetak: number;
    47→  machineName2: string;
    48→  ongkosCetak2: number;
    49→  totalPaperPrice: number;
    50→  pricePerSheet: number;
    51→  finishingNames: string;
    52→  finishingCost: number;
    53→  packingCost: number;
    54→  shippingCost: number;
    55→  otherCost: number;
    56→  otherCost2: number;
    57→  otherCostLabel: string;
    58→  otherCostLabel2: string;
    59→  glueCost: number;
    60→  glueBorongan: number;
    61→  subTotal: number;
    62→  profitPercent: number;
    63→  profitAmount: number;
    64→  grandTotal: number;
    65→  createdAt: string;
    66→}
    67→
    68→export function InvoiceEditor() {
    69→  const invoice = useDokuproStore((s) => s.invoice);
    70→  const setInvoice = useDokuproStore((s) => s.setInvoice);
    71→  const resetDocument = useDokuproStore((s) => s.resetDocument);
    72→  const loadCompanyFromAPI = useDokuproStore((s) => s.loadCompanyFromAPI);
    73→  const router = useRouter();
    74→
    75→  const searchParams = useSearchParams();
    76→  const riwayatIdFromUrl = searchParams.get('riwayatId');
    77→  const autoSelectDoneRef = useRef<string | null>(null); // track which riwayatId was auto-selected
    78→  const [savingSj, setSavingSj] = useState(false);
    79→  const [jpgGenerating, setJpgGenerating] = useState(false);
    80→
    81→  // Riwayat cetakan dropdown state
    82→  const [riwayatList, setRiwayatList] = useState<RiwayatCetakanItem[]>([]);
    83→  const [referensiInput, setReferensiInput] = useState(invoice.referensi);
    84→  const [dropdownOpen, setDropdownOpen] = useState(false);
    85→
    86→  // Customer dropdown state
    87→  const [customerList, setCustomerList] = useState<CustomerItem[]>([]);
    88→  const [clientInput, setClientInput] = useState(invoice.client.nama);
    89→  const [clientDropdownOpen, setClientDropdownOpen] = useState(false);
    90→  const [clientTyping, setClientTyping] = useState(false);
    91→
    92→  const fetchRiwayatCetakan = useCallback(async () => {
    93→    try {
    94→      const res = await fetch('/api/riwayat-cetakan', { headers: getAuthHeaders() });
    95→      if (res.ok) {
    96→        const data = await res.json();
    97→        setRiwayatList(data);
    98→        return data;
    99→      }
   100→    } catch {
   101→      // silently fail
   102→    }
   103→    return [];
   104→  }, []);
   105→
   106→  // Fetch customer list
   107→  const fetchCustomers = useCallback(async () => {
   108→    try {
   109→      const res = await fetch('/api/customers', { headers: getAuthHeaders() });
   110→      if (res.ok) {
   111→        const data = await res.json();
   112→        setCustomerList(Array.isArray(data) ? data : []);
   113→      }
   114→    } catch {
   115→      // silently fail
   116→    }
   117→  }, []);
   118→
   119→  useEffect(() => { loadCompanyFromAPI() }, [loadCompanyFromAPI]);
   120→  useEffect(() => { fetchRiwayatCetakan() }, [fetchRiwayatCetakan]);
   121→  useEffect(() => { fetchCustomers() }, [fetchCustomers]);
   122→
   123→  // Fetch next Invoice number from server
   124→  const fetchNextNumber = useCallback(() => {
   125→    fetch('/api/history?preview=next-number&docType=invoice', { headers: getAuthHeaders() })
   126→      .then(res => res.ok ? res.json() : null)
   127→      .then(data => { if (data?.nextNumber) setInvoice((prev) => ({ ...prev, nomor: data.nextNumber })) })
   128→      .catch(() => {});
   129→  }, [setInvoice]);
   130→
   131→  useEffect(() => { fetchNextNumber() }, [fetchNextNumber]);
   132→
   133→  // Re-fetch next number after a document is saved
   134→  useEffect(() => {
   135→    const handler = () => { fetchNextNumber() };
   136→    window.addEventListener('dokupro:history-updated', handler);
   137→    return () => { window.removeEventListener('dokupro:history-updated', handler) };
   138→  }, [fetchNextNumber]);
   139→
   140→  // Sync referensiInput when invoice.referensi changes externally
   141→  useEffect(() => { setReferensiInput(invoice.referensi) }, [invoice.referensi]);
   142→  // Sync clientInput when invoice.client.nama changes externally
   143→  useEffect(() => { setClientInput(invoice.client.nama) }, [invoice.client.nama]);
   144→
   145→  // Filter riwayat list by input
   146→  const filteredRiwayatList = riwayatList.filter((r) => {
   147→    const search = referensiInput.toLowerCase().trim();
   148→    if (!search) return true;
   149→    const noHc = (r.nomorUrut || '').toLowerCase();
   150→    const printName = (r.printName || '').toLowerCase();
   151→    const name = (r.customerName || '').toLowerCase();
   152→    return noHc.includes(search) || printName.includes(search) || name.includes(search);
   153→  });
   154→
   155→  const handleReferensiInputChange = (value: string) => {
   156→    setReferensiInput(value);
   157→    setInvoice((prev) => ({ ...prev, referensi: value }));
   158→    setDropdownOpen(true);
   159→  };
   160→
   161→  // Apply referensi data using functional setInvoice to avoid stale closure
   162→  const applyReferensi = (item: RiwayatCetakanItem) => {
   163→    const ref = item.nomorUrut || item.printName || '';
   164→    setReferensiInput(ref);
   165→
   166→    // Build single item with multi-line description
   167→    const descLines: string[] = [];
   168→    if (item.printName) descLines.push(item.printName);
   169→
   170→    const bahanParts: string[] = [];
   171→    if (item.paperName) bahanParts.push(item.paperName);
   172→    if (item.paperGrammage) bahanParts.push(item.paperGrammage + 'g');
   173→    if (bahanParts.length > 0) descLines.push(bahanParts.join(' '));
   174→
   175→    if (item.finishingNames) {
   176→      const finishings = item.finishingNames.split(',').map(f => f.trim()).filter(Boolean);
   177→      descLines.push('Finishing:');
   178→      finishings.forEach(f => descLines.push(f));
   179→    }
   180→
   181→    const deskripsi = descLines.join('\n');
   182→    const jumlahPesanan = parseInt(item.jumlahPesanan) || 1;
   183→    const hargaPerPcs = jumlahPesanan > 0 ? Math.round(item.grandTotal / jumlahPesanan) : 0;
   184→
   185→    const newItems: InvoiceData['items'] = [{
   186→      id: 'riwayat-0',
   187→      deskripsi: deskripsi || ref,
   188→      qty: jumlahPesanan,
   189→      satuan: 'pcs',
   190→      harga: hargaPerPcs,
   191→    }];
   192→
   193→    // Use functional form to always get the LATEST state
   194→    setInvoice((prev) => ({
   195→      ...prev,
   196→      referensi: ref,
   197→      client: {
   198→        ...prev.client,
   199→        nama: item.customerName || prev.client.nama,
   200→      },
   201→      items: newItems,
   202→      catatan: '',
   203→    }));
   204→    setDropdownOpen(false);
   205→  };
   206→
   207→  const handleReferensiSelect = (item: RiwayatCetakanItem) => {
   208→    applyReferensi(item);
   209→  };
   210→
   211→  // Auto-select referensi when coming from hitung cetakan with riwayatId
   212→  // Fetches the specific riwayat directly by ID for reliability (no race condition with list loading)
   213→  useEffect(() => {
   214→    if (!riwayatIdFromUrl) return;
   215→    if (autoSelectDoneRef.current === riwayatIdFromUrl) return;
   216→
   217→    let cancelled = false;
   218→
   219→    const fetchAndApply = async () => {
   220→      // First, try to find it in the already-loaded list (fast path)
   221→      const foundInList = riwayatList.find((r) => r.id === riwayatIdFromUrl);
   222→      if (foundInList) {
   223→        if (!cancelled) {
   224→          autoSelectDoneRef.current = riwayatIdFromUrl;
   225→          applyReferensi(foundInList);
   226→        }
   227→        return;
   228→      }
   229→
   230→      // Not in list yet — fetch directly by ID from the API (reliable path)
   231→      try {
   232→        const res = await fetch(`/api/riwayat-cetakan?id=${riwayatIdFromUrl}`, { headers: getAuthHeaders() });
   233→        if (res.ok && !cancelled) {
   234→          const data = await res.json();
   235→          if (data && !data.error) {
   236→            autoSelectDoneRef.current = riwayatIdFromUrl;
   237→            applyReferensi(data as RiwayatCetakanItem);
   238→          }
   239→        }
   240→      } catch {
   241→        // Fallback: retry once after a short delay
   242→        if (cancelled) return;
   243→        await new Promise(r => setTimeout(r, 800));
   244→        if (cancelled) return;
   245→        try {
   246→          const res = await fetch(`/api/riwayat-cetakan?id=${riwayatIdFromUrl}`, { headers: getAuthHeaders() });
   247→          if (res.ok) {
   248→            const data = await res.json();
   249→            if (data && !data.error) {
   250→              autoSelectDoneRef.current = riwayatIdFromUrl;
   251→              applyReferensi(data as RiwayatCetakanItem);
   252→            }
   253→          }
   254→        } catch {
   255→          // Give up silently
   256→        }
   257→      }
   258→    };
   259→
   260→    fetchAndApply();
   261→
   262→    return () => { cancelled = true; };
   263→  }, [riwayatIdFromUrl]);
   264→
   265→  // Filter customer list (only when user is actively typing)
   266→  const filteredCustomerList = useMemo(() =>
   267→    clientTyping
   268→      ? customerList.filter((c) => {
   269→          const search = clientInput.toLowerCase().trim();
   270→          if (!search) return true;
   271→          return c.name.toLowerCase().includes(search) ||
   272→                 (c.companyName || '').toLowerCase().includes(search) ||
   273→                 (c.phone || '').toLowerCase().includes(search) ||
   274→                 (c.address || '').toLowerCase().includes(search);
   275→        })
   276→      : customerList
   277→  , [clientTyping, customerList, clientInput]);
   278→
   279→  const handleClientInputChange = (value: string) => {
   280→    setClientInput(value);
   281→    setClientTyping(true);
   282→    updateClient('nama', value);
   283→    setClientDropdownOpen(true);
   284→  };
   285→
   286→  const handleClientSelect = (item: CustomerItem) => {
   287→    setClientInput(item.name);
   288→    setClientTyping(false);
   289→    setInvoice((prev) => ({
   290→      ...prev,
   291→      client: {
   292→        nama: item.name,
   293→        kontak: item.phone || item.email || '',
   294→        alamat: item.address || '',
   295→      },
   296→    }));
   297→    setClientDropdownOpen(false);
   298→  };
   299→
   300→  const updateCompany = (company: typeof invoice.company) => {
   301→    setInvoice((prev) => ({ ...prev, company }));
   302→  };
   303→
   304→  const updateClient = (field: string, value: string) => {
   305→    setInvoice((prev) => ({
   306→      ...prev,
   307→      client: { ...prev.client, [field]: value },
   308→    }));
   309→  };
   310→
   311→  const subtotal = invoice.items.reduce((sum, item) => sum + item.qty * item.harga, 0);
   312→  const ppnAmount = subtotal * (invoice.ppn / 100);
   313→  const total = subtotal + ppnAmount;
   314→  const dpPercent = invoice.dp || 0;
   315→  // Use saved dpAmount if available, otherwise calculate from originalTotal or current total
   316→  const originalTotalForDp = invoice.originalTotal !== undefined ? invoice.originalTotal : total;
   317→  const dpAmount = invoice.dpAmount !== undefined ? invoice.dpAmount : originalTotalForDp * (dpPercent / 100);
   318→  const sisa = total - dpAmount;
   319→
   320→  const handleSuratJalan = async () => {
   321→    // Check if data has content
   322→    const pihakKedua = invoice.client?.nama;
   323→    const hasItem = invoice.items?.some((item) => item.deskripsi.trim() !== '');
   324→    if (!pihakKedua?.trim() || !hasItem) {
   325→      toast.error('Lengkapi data invoice terlebih dahulu!');
   326→      return;
   327→    }
   328→
   329→    setSavingSj(true);
   330→    try {
   331→      // Save invoice to history first
   332→      const res = await fetch('/api/history', {
   333→        method: 'POST',
   334→        headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   335→        body: JSON.stringify({
   336→          docType: 'invoice',
   337→          nomor: invoice.nomor || '-',
   338→          tanggal: invoice.tanggal || '',
   339→          pihakKedua: invoice.client?.nama || '-',
   340→          total: '-',
   341→          dataJson: JSON.stringify({ ...invoice, dpAmount: invoice.dpAmount !== undefined ? invoice.dpAmount : dpAmount, originalTotal: invoice.originalTotal ?? total }),
   342→        }),
   343→      });
   344→      if (res.ok) {
   345→        const saved = await res.json();
   346→        window.dispatchEvent(new CustomEvent('dokupro:history-updated'));
   347→        toast.success('Invoice disimpan ke riwayat');
   348→        // Reset invoice form after saving
   349→        resetDocument('invoice');
   350→        router.push(`/surat-jalan?invoiceId=${saved.id}`);
   351→      } else if (res.status === 409) {
   352→        // Invoice already saved, get existing ID and navigate
   353→        const errData = await res.json().catch(() => ({}));
   354→        if (errData.id) {
   355→          resetDocument('invoice');
   356→          router.push(`/surat-jalan?invoiceId=${errData.id}`);
   357→        } else {
   358→          toast('Invoice sudah ada di riwayat.');
   359→        }
   360→      } else {
   361→        toast.error('Gagal menyimpan invoice');
   362→      }
   363→    } catch {
   364→      toast.error('Gagal menyimpan invoice');
   365→    }
   366→    setSavingSj(false);
   367→  };
   368→
   369→  // Generate JPG from preview (same as print output)
   370→  const handleGenerateJpg = async () => {
   371→    const previewEl = document.querySelector('[data-document-preview]') as HTMLElement;
   372→    if (!previewEl) {
   373→      toast.error('Preview tidak ditemukan');
   374→      return;
   375→    }
   376→    setJpgGenerating(true);
   377→    try {
   378→      const dataUrl = await toJpeg(previewEl, {
   379→        quality: 0.95,
   380→        pixelRatio: 2,
   381→        backgroundColor: '#ffffff',
   382→      });
   383→      const fileName = `Invoice-${invoice.nomor || 'draft'}.jpg`;
   384→      const link = document.createElement('a');
   385→      link.download = fileName;
   386→      link.href = dataUrl;
   387→      link.click();
   388→      toast.success('JPG berhasil didownload');
   389→    } catch (err) {
   390→      console.error('Failed to generate JPG:', err);
   391→      toast.error('Gagal membuat JPG');
   392→    } finally {
   393→      setJpgGenerating(false);
   394→    }
   395→  };
   396→
   397→  return (
   398→    <>
   399→      <DocumentEditorLayout
   400→        title="Invoice"
   401→        previewContent={<InvoicePreview data={invoice} />}
   402→        actions={
   403→          <div className="flex items-center gap-2 flex-wrap">
   404→            <DocumentActionButtons
   405→              docType="invoice"
   406→              documentLabel="Invoice"
   407→              currentData={invoice}
   408→              onReset={() => resetDocument('invoice')}
   409→            />
   410→            <Button
   411→              size="sm"
   412→              onClick={handleGenerateJpg}
   413→              disabled={jpgGenerating}
   414→              className="bg-violet-600 hover:bg-violet-700 text-white h-8 sm:h-9"
   415→            >
   416→              {jpgGenerating ? (
   417→                <><Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" /> Membuat...</>
   418→              ) : (
   419→                <><ImageIcon className="mr-1.5 h-3.5 w-3.5" /> JPG</>
   420→              )}
   421→            </Button>
   422→            <Button
   423→              size="sm"
   424→              onClick={handleSuratJalan}
   425→              disabled={savingSj}
   426→              className="bg-orange-600 hover:bg-orange-700 h-8 sm:h-9"
   427→            >
   428→              <Truck className="mr-1.5 h-3.5 w-3.5" />
   429→              {savingSj ? 'Menyimpan...' : 'Surat Jalan'}
   430→            </Button>
   431→          </div>
   432→        }
   433→      >
   434→        <CompanyFields company={invoice.company} onChange={updateCompany} />
   435→
   436→        <div className="rounded-lg border bg-card p-3 sm:p-4 shadow-sm">
   437→          <h3 className="mb-3 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   438→            Detail Dokumen
   439→          </h3>
   440→          <div className="grid grid-cols-2 gap-2">
   441→            <div className="space-y-1.5">
   442→              <Label className="text-xs">No. Invoice</Label>
   443→              <Input
   444→                value={invoice.nomor}
   445→                readOnly
   446→                className="bg-slate-50 text-slate-500 cursor-not-allowed"
   447→              />
   448→            </div>
   449→            <div className="space-y-1.5">
   450→              <Label className="text-xs">Tanggal</Label>
   451→              <Input
   452→                type="date"
   453→                value={invoice.tanggal}
   454→                onChange={(e) => setInvoice((prev) => ({ ...prev, tanggal: e.target.value }))}
   455→              />
   456→            </div>
   457→          </div>
   458→          <div className="space-y-2">
   459→            <Label className="text-xs">Referensi (No. HC)</Label>
   460→            <Popover open={dropdownOpen} onOpenChange={setDropdownOpen}>
   461→              <PopoverAnchor asChild>
   462→                <div className="relative">
   463→                  <input
   464→                    type="text"
   465→                    placeholder="Pilih No. HC / ketik referensi..."
   466→                    value={referensiInput}
   467→                    onChange={(e) => handleReferensiInputChange(e.target.value)}
   468→                    onFocus={() => setDropdownOpen(true)}
   469→                    className="flex h-9 w-full rounded-md border border-input bg-transparent px-3 py-2 text-sm shadow-xs transition-[color,box-shadow] outline-none focus-visible:border-ring focus-visible:ring-ring/50 focus-visible:ring-[3px] pr-9"
   470→                  />
   471→                  {/* Dropdown chevron icon */}
   472→                  <div className="absolute right-2 top-1/2 -translate-y-1/2 pointer-events-none text-slate-400">
   473→                    <svg className="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M19 9l-7 7-7-7" /></svg>
   474→                  </div>
   475→                </div>
   476→              </PopoverAnchor>
   477→              <PopoverContent
   478→                align="start"
   479→                className="p-0 w-[var(--radix-popover-trigger-width)] max-h-52 overflow-y-auto"
   480→                onOpenAutoFocus={(e) => e.preventDefault()}
   481→              >
   482→                {filteredRiwayatList.length > 0 && (
   483→                  <div>
   484→                    <div className="px-3 py-1.5 text-[10px] font-semibold text-slate-400 uppercase bg-slate-50 border-b border-slate-100 sticky top-0">Riwayat Hitung Cetakan</div>
   485→                    {filteredRiwayatList.map((r) => (
   486→                      <button
   487→                        key={r.id}
   488→                        type="button"
   489→                        onMouseDown={(e) => { e.preventDefault(); handleReferensiSelect(r) }}
   490→                        className={`w-full text-left px-3 py-2 text-sm transition-colors ${invoice.referensi === r.nomorUrut ? 'bg-blue-50 text-blue-700 font-medium' : 'text-slate-700 hover:bg-blue-50 hover:text-blue-700'}`}
   491→                      >
   492→                        <span className="font-semibold text-xs text-emerald-700">{r.nomorUrut || '-'}</span>
   493→                        <span className="ml-1.5 truncate">{r.printName || ''}</span>
   494→                        {r.customerName && <span className="text-slate-400 ml-1.5 text-[11px]">({r.customerName})</span>}
   495→                      </button>
   496→                    ))}
   497→                  </div>
   498→                )}
   499→                {filteredRiwayatList.length === 0 && (
   500→                  <div className="px-3 py-3 text-sm text-slate-400 text-center">Tidak ada riwayat cetakan</div>
   501→                )}
   502→              </PopoverContent>
   503→            </Popover>
   504→          </div>
   505→        </div>
   506→
   507→        {/* Informasi Pembayaran - moved below Detail Dokumen */}
   508→        <div className="rounded-lg border bg-card p-3 sm:p-4 shadow-sm">
   509→          <h3 className="mb-3 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   510→            Informasi Pembayaran
   511→          </h3>
   512→          <div className="grid grid-cols-2 gap-3">
   513→            <div className="space-y-1.5">
   514→              <Label className="text-xs">Tgl. Jatuh Tempo</Label>
   515→              <Input
   516→                type="date"
   517→                value={invoice.tanggalJatuhTempo}
   518→                onChange={(e) => setInvoice((prev) => ({ ...prev, tanggalJatuhTempo: e.target.value }))}
   519→              />
   520→            </div>
   521→            <div className="space-y-1.5">
   522→              <Label className="text-xs">Cara Pembayaran</Label>
   523→              <div className="flex gap-1.5">
   524→                {(['cash', 'transfer', 'giro'] as const).map((method) => {
