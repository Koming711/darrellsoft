     1→'use client';
     2→
     3→import { useEffect, useState, useCallback, useRef, useMemo } from 'react';
     4→import { useSearchParams } from 'next/navigation';
     5→import { Input } from '@/components/ui/input';
     6→import { Label } from '@/components/ui/label';
     7→import { Textarea } from '@/components/ui/textarea';
     8→import { Popover, PopoverContent, PopoverAnchor } from '@/components/ui/popover';
     9→import { useDokuproStore } from '@/lib/store';
    10→import { ItemsFields, type BarangOption } from './items-fields';
    11→import { InvoicePreview } from './invoice-preview';
    12→import { DocumentEditorLayout } from './document-editor-layout';
    13→import { DocumentActionButtons } from './document-action-buttons';
    14→import { formatRupiah } from '@/lib/format';
    15→import { getAuthHeaders } from '@/lib/auth';
    16→import { toast } from 'sonner';
    17→import type { InvoiceData } from '@/lib/types';
    18→
    19→interface CustomerItem {
    20→  id: string;
    21→  name: string;
    22→  companyName?: string;
    23→  phone?: string;
    24→  email?: string;
    25→  address?: string;
    26→}
    27→
    28→interface RiwayatCetakanItem {
    29→  id: string;
    30→  nomorUrut: string;
    31→  printName: string;
    32→  customerName: string;
    33→  paperName: string;
    34→  paperGrammage: string;
    35→  cutWidth: string;
    36→  cutHeight: string;
    37→  quantity: string;
    38→  jumlahPesanan: string;
    39→  berapaMata: string;
    40→  warna: string;
    41→  machineName: string;
    42→  ongkosCetak: number;
    43→  machineName2: string;
    44→  ongkosCetak2: number;
    45→  totalPaperPrice: number;
    46→  pricePerSheet: number;
    47→  finishingNames: string;
    48→  finishingCost: number;
    49→  packingCost: number;
    50→  shippingCost: number;
    51→  otherCost: number;
    52→  otherCost2: number;
    53→  otherCostLabel: string;
    54→  otherCostLabel2: string;
    55→  glueCost: number;
    56→  glueBorongan: number;
    57→  subTotal: number;
    58→  profitPercent: number;
    59→  profitAmount: number;
    60→  grandTotal: number;
    61→  createdAt: string;
    62→}
    63→
    64→
    65→export function InvoiceEditor({ dpDisabled = false, onSaved }: { dpDisabled?: boolean; onSaved?: (id: string) => void }) {
    66→  const invoice = useDokuproStore((s) => s.invoice);
    67→  const setInvoice = useDokuproStore((s) => s.setInvoice);
    68→  const resetDocument = useDokuproStore((s) => s.resetDocument);
    69→  const loadCompanyFromAPI = useDokuproStore((s) => s.loadCompanyFromAPI);
    70→  const invoiceEditingId = useDokuproStore((s) => s.invoiceEditingId);
    71→  const setInvoiceEditingId = useDokuproStore((s) => s.setInvoiceEditingId);
    72→  const searchParams = useSearchParams();
    73→  const riwayatIdFromUrl = searchParams.get('riwayatId');
    74→  const autoSelectDoneRef = useRef<string | null>(null); // track which riwayatId was auto-selected
    75→
    76→  // Daftar barang milik customer terpilih (Master Barang per pelanggan)
    77→  const [barangList, setBarangList] = useState<BarangOption[]>([]);
    78→  // Nama customer sebelumnya — untuk mendeteksi perubahan customer
    79→  const prevClientNameRef = useRef<string>(invoice.client.nama);
    80→  // Nama yang diisi otomatis oleh pemilihan referensi (hitung cetakan) —
    81→  // perubahan nama dari alur ini TIDAK boleh mengosongkan item yang baru diisi.
    82→  const autoFillNameRef = useRef<string>('');
    83→
    84→  // Riwayat cetakan dropdown state
    85→  const [riwayatList, setRiwayatList] = useState<RiwayatCetakanItem[]>([]);
    86→  const [referensiInput, setReferensiInput] = useState(invoice.referensi);
    87→  const [dropdownOpen, setDropdownOpen] = useState(false);
    88→
    89→  // Customer dropdown state
    90→  const [customerList, setCustomerList] = useState<CustomerItem[]>([]);
    91→  const [clientInput, setClientInput] = useState(invoice.client.nama);
    92→  const [clientDropdownOpen, setClientDropdownOpen] = useState(false);
    93→  const [clientTyping, setClientTyping] = useState(false);
    94→
    95→  const fetchRiwayatCetakan = useCallback(async () => {
    96→    try {
    97→      const res = await fetch('/api/riwayat-cetakan', { headers: getAuthHeaders() });
    98→      if (res.ok) {
    99→        const data = await res.json();
   100→        setRiwayatList(data);
   101→        return data;
   102→      }
   103→    } catch {
   104→      // silently fail
   105→    }
   106→    return [];
   107→  }, []);
   108→
   109→  // Fetch customer list
   110→  const fetchCustomers = useCallback(async () => {
   111→    try {
   112→      const res = await fetch('/api/customers', { headers: getAuthHeaders() });
   113→      if (res.ok) {
   114→        const data = await res.json();
   115→        setCustomerList(Array.isArray(data) ? data : []);
   116→      }
   117→    } catch {
   118→      // silently fail
   119→    }
   120→  }, []);
   121→
   122→  useEffect(() => { loadCompanyFromAPI() }, [loadCompanyFromAPI]);
   123→  useEffect(() => { fetchRiwayatCetakan() }, [fetchRiwayatCetakan]);
   124→  useEffect(() => { fetchCustomers() }, [fetchCustomers]);
   125→
   126→  // Muat daftar barang milik customer terpilih (Master Barang per pelanggan).
   127→  // Jika customer tidak dikenal / belum memilih, daftar barang = kosong.
   128→  useEffect(() => {
   129→    const nama = invoice.client.nama.trim().toLowerCase();
   130→    const found = nama
   131→      ? customerList.find((c) => c.name.trim().toLowerCase() === nama)
   132→      : undefined;
   133→    if (!found) {
   134→      setBarangList([]);
   135→      return;
   136→    }
   137→    let cancelled = false;
   138→    fetch(`/api/items?customerId=${encodeURIComponent(found.id)}&active=1`, { headers: getAuthHeaders() })
   139→      .then((res) => (res.ok ? res.json() : { items: [] }))
   140→      .then((data) => {
   141→        if (cancelled) return;
   142→        const rows = Array.isArray(data?.items) ? data.items : [];
   143→        setBarangList(rows.map((r: { id: string; name: string; unit: string; standardPrice: number; hpp: number | null }) => ({
   144→          id: r.id,
   145→          name: r.name,
   146→          unit: r.unit || 'pcs',
   147→          standardPrice: r.standardPrice || 0,
   148→          hpp: r.hpp ?? null,
   149→        })));
   150→      })
   151→      .catch(() => { if (!cancelled) setBarangList([]); });
   152→    return () => { cancelled = true; };
   153→  }, [invoice.client.nama, customerList]);
   154→
   155→  // Ganti customer → otomatis KOSONGKAN nama barang di kotak item
   156→  // (barang milik tiap pelanggan tidak boleh tercampur).
   157→  // Pengecualian: nama yang diisi otomatis oleh pemilihan referensi cetakan.
   158→  useEffect(() => {
   159→    const nama = invoice.client.nama.trim();
   160→    const prev = prevClientNameRef.current.trim();
   161→    if (nama !== prev) {
   162→      prevClientNameRef.current = invoice.client.nama;
   163→      if (autoFillNameRef.current && nama === autoFillNameRef.current.trim()) {
   164→        // Diisi otomatis oleh applyReferensi — jangan kosongkan
   165→        autoFillNameRef.current = '';
   166→        return;
   167→      }
   168→      autoFillNameRef.current = '';
   169→      setInvoice((prevInv) => ({
   170→        ...prevInv,
   171→        items: prevInv.items.map((it, i) => (
   172→          i === 0
   173→            ? { ...it, deskripsi: '', qty: 1, harga: 0, modal: 0 }
   174→            : null
   175→        )).filter((it): it is typeof prevInv.items[number] => it !== null),
   176→      }));
   177→    }
   178→  }, [invoice.client.nama, setInvoice]);
   179→
   180→  // Fetch next Invoice number from server (only when creating new, not editing)
   181→  const fetchNextNumber = useCallback(() => {
   182→    // Read current editingId from store to avoid stale closure
   183→    const currentEditingId = useDokuproStore.getState().invoiceEditingId;
   184→    if (currentEditingId) return; // Don't overwrite nomor when editing
   185→    fetch('/api/history?preview=next-number&docType=invoice', { headers: getAuthHeaders() })
   186→      .then(res => res.ok ? res.json() : null)
   187→      .then(data => { if (data?.nextNumber) setInvoice((prev) => ({ ...prev, nomor: data.nextNumber })) })
   188→      .catch(() => {});
   189→  }, [setInvoice]);
   190→
   191→  useEffect(() => { fetchNextNumber() }, [fetchNextNumber]);
   192→
   193→  // Re-fetch next number after a document is saved
   194→  useEffect(() => {
   195→    const handler = () => { fetchNextNumber() };
   196→    window.addEventListener('dokupro:history-updated', handler);
   197→    return () => { window.removeEventListener('dokupro:history-updated', handler) };
   198→  }, [fetchNextNumber]);
   199→
   200→  // Mode Regular (dpDisabled): paksa DP = 0 dan sembunyikan input DP
   201→  useEffect(() => {
   202→    if (dpDisabled) {
   203→      setInvoice((prev) => (prev.dp === 0 ? prev : ({ ...prev, dp: 0 })));
   204→    }
   205→  }, [dpDisabled, setInvoice]);
   206→
   207→  // Sync referensiInput when invoice.referensi changes externally
   208→  useEffect(() => { setReferensiInput(invoice.referensi) }, [invoice.referensi]);
   209→  // Sync clientInput when invoice.client.nama changes externally
   210→  useEffect(() => { setClientInput(invoice.client.nama) }, [invoice.client.nama]);
   211→
   212→  // Filter riwayat list — when dropdown is open show all, otherwise not used
   213→  const filteredRiwayatList = riwayatList;
   214→
   215→  const handleReferensiInputChange = (value: string) => {
   216→    setReferensiInput(value);
   217→    setInvoice((prev) => ({ ...prev, referensi: value }));
   218→    setDropdownOpen(true);
   219→  };
   220→
   221→  // Apply referensi data using functional setInvoice to avoid stale closure
   222→  const applyReferensi = (item: RiwayatCetakanItem) => {
   223→    const ref = item.nomorUrut || item.printName || '';
   224→    setReferensiInput(ref);
   225→
   226→    // Build single item with multi-line description
   227→    const descLines: string[] = [];
   228→    if (item.printName) descLines.push(item.printName);
   229→
   230→    const bahanParts: string[] = [];
   231→    if (item.paperName) bahanParts.push(item.paperName);
   232→    if (item.paperGrammage) bahanParts.push(item.paperGrammage + 'g');
   233→    if (bahanParts.length > 0) descLines.push(bahanParts.join(' '));
   234→
   235→    if (item.finishingNames) {
   236→      const finishings = item.finishingNames.split(',').map(f => f.trim()).filter(Boolean);
   237→      descLines.push('Finishing:');
   238→      finishings.forEach(f => descLines.push(f));
   239→    }
   240→
   241→    const deskripsi = descLines.join('\n');
   242→    const jumlahPesanan = parseInt(item.jumlahPesanan) || 1;
   243→    const hargaPerPcs = jumlahPesanan > 0 ? Math.round(item.grandTotal / jumlahPesanan) : 0;
   244→
   245→    const newItems: InvoiceData['items'] = [{
   246→      id: 'riwayat-0',
   247→      deskripsi: deskripsi || ref,
   248→      qty: jumlahPesanan,
   249→      satuan: 'pcs',
   250→      harga: hargaPerPcs,
   251→    }];
   252→
   253→    // Tandai bahwa perubahan nama customer berikutnya berasal dari alur
   254→    // referensi cetakan — jangan kosongkan item yang baru saja diisi.
   255→    if (item.customerName) autoFillNameRef.current = item.customerName;
   256→
   257→    // Use functional form to always get the LATEST state
   258→    setInvoice((prev) => ({
   259→      ...prev,
   260→      referensi: ref,
   261→      client: {
   262→        ...prev.client,
   263→        nama: item.customerName || prev.client.nama,
   264→      },
   265→      items: newItems,
   266→      catatan: '',
   267→      uangCapek: item.profitAmount || 0,
   268→    }));
   269→    setDropdownOpen(false);
   270→  };
   271→
   272→  const handleReferensiSelect = (item: RiwayatCetakanItem) => {
   273→    applyReferensi(item);
   274→  };
   275→
   276→  // Auto-select referensi when coming from hitung cetakan with riwayatId
   277→  // Fetches the specific riwayat directly by ID for reliability (no race condition with list loading)
   278→  useEffect(() => {
   279→    if (!riwayatIdFromUrl) return;
   280→    if (autoSelectDoneRef.current === riwayatIdFromUrl) return;
   281→
   282→    let cancelled = false;
   283→
   284→    const fetchAndApply = async () => {
   285→      // First, try to find it in the already-loaded list (fast path)
   286→      const foundInList = riwayatList.find((r) => r.id === riwayatIdFromUrl);
   287→      if (foundInList) {
   288→        if (!cancelled) {
   289→          autoSelectDoneRef.current = riwayatIdFromUrl;
   290→          applyReferensi(foundInList);
   291→        }
   292→        return;
   293→      }
   294→
   295→      // Not in list yet — fetch directly by ID from the API (reliable path)
   296→      try {
   297→        const res = await fetch(`/api/riwayat-cetakan?id=${riwayatIdFromUrl}`, { headers: getAuthHeaders() });
   298→        if (res.ok && !cancelled) {
   299→          const data = await res.json();
   300→          if (data && !data.error) {
   301→            autoSelectDoneRef.current = riwayatIdFromUrl;
   302→            applyReferensi(data as RiwayatCetakanItem);
   303→          }
   304→        }
   305→      } catch {
   306→        // Fallback: retry once after a short delay
   307→        if (cancelled) return;
   308→        await new Promise(r => setTimeout(r, 800));
   309→        if (cancelled) return;
   310→        try {
   311→          const res = await fetch(`/api/riwayat-cetakan?id=${riwayatIdFromUrl}`, { headers: getAuthHeaders() });
   312→          if (res.ok) {
   313→            const data = await res.json();
   314→            if (data && !data.error) {
   315→              autoSelectDoneRef.current = riwayatIdFromUrl;
   316→              applyReferensi(data as RiwayatCetakanItem);
   317→            }
   318→          }
   319→        } catch {
   320→          // Give up silently
   321→        }
   322→      }
   323→    };
   324→
   325→    fetchAndApply();
   326→
   327→    return () => { cancelled = true; };
   328→  }, [riwayatIdFromUrl]);
   329→
   330→  // Filter customer list (only when user is actively typing)
   331→  const filteredCustomerList = useMemo(() =>
   332→    clientTyping
   333→      ? customerList.filter((c) => {
   334→          const search = clientInput.toLowerCase().trim();
   335→          if (!search) return true;
   336→          return c.name.toLowerCase().includes(search) ||
   337→                 (c.companyName || '').toLowerCase().includes(search) ||
   338→                 (c.phone || '').toLowerCase().includes(search) ||
   339→                 (c.address || '').toLowerCase().includes(search);
   340→        })
   341→      : customerList
   342→  , [clientTyping, customerList, clientInput]);
   343→
   344→  const handleClientInputChange = (value: string) => {
   345→    setClientInput(value);
   346→    setClientTyping(true);
   347→    updateClient('nama', value);
   348→    setClientDropdownOpen(true);
   349→  };
   350→
   351→  const handleClientSelect = (item: CustomerItem) => {
   352→    setClientInput(item.name);
   353→    setClientTyping(false);
   354→    setInvoice((prev) => ({
   355→      ...prev,
   356→      client: {
   357→        nama: item.name,
   358→        kontak: item.phone || item.email || '',
   359→        alamat: item.address || '',
   360→      },
   361→    }));
   362→    setClientDropdownOpen(false);
   363→  };
   364→
   365→  // Pilih barang dari dropdown Master Barang customer → isi deskripsi +
   366→  // harga satuan + harga modal otomatis (semuanya tetap bisa diedit manual).
   367→  const handlePickBarang = (itemIndex: number, barang: BarangOption) => {
   368→    setInvoice((prev) => ({
   369→      ...prev,
   370→      items: prev.items.map((it, i) => (
   371→        i === itemIndex
   372→          ? {
   373→              ...it,
   374→              deskripsi: barang.name,
   375→              satuan: barang.unit || it.satuan || 'pcs',
   376→              harga: barang.standardPrice || 0,
   377→              modal: barang.hpp ?? 0,
   378→            }
   379→          : it
   380→      )),
   381→    }));
   382→  };
   383→
   384→  const updateClient = (field: string, value: string) => {
   385→    setInvoice((prev) => ({
   386→      ...prev,
   387→      client: { ...prev.client, [field]: value },
   388→    }));
   389→  };
   390→
   391→  const subtotal = invoice.items.reduce((sum, item) => sum + item.qty * item.harga, 0);
   392→  const ppnAmount = subtotal * (invoice.ppn / 100);
   393→  const total = subtotal + ppnAmount;
   394→  const dpPercent = invoice.dp || 0;
   395→  const dpAmount = total * (dpPercent / 100);
   396→  const sisa = total - dpAmount;
   397→
   398→  // Pratinjau langsung muncul begitu ada data inti (referensi dipilih /
   399→  // customer terisi / ada barang) — tanpa tombol "Lihat Pratinjau A5".
   400→  const hasPreviewData = Boolean(
   401→    referensiInput.trim() ||
   402→    invoice.client.nama.trim() ||
   403→    invoice.items.some((it) => it.deskripsi.trim() !== '')
   404→  );
   405→
   406→  return (
   407→    <>
   408→      <DocumentEditorLayout
   409→        title="Invoice"
   410→        previewMode="inline-bottom"
   411→        formColumns={2}
   412→        previewMaxWidth={560}
   413→        previewContent={
   414→          hasPreviewData ? (
   415→            <InvoicePreview data={invoice} />
   416→          ) : (
   417→            <div className="flex min-h-[320px] w-full items-center justify-center rounded-lg border-2 border-dashed border-slate-200 bg-slate-50/60 p-6 text-center">
   418→              <p className="max-w-[280px] text-sm text-slate-400">
   419→                Pratinjau invoice akan muncul di sini setelah Anda memilih referensi (No. HC) atau customer.
   420→              </p>
   421→            </div>
   422→          )
   423→        }
   424→        actions={
   425→          <DocumentActionButtons
   426→            docType="invoice"
   427→            documentLabel="Invoice"
   428→            currentData={invoice}
   429→            onReset={() => resetDocument('invoice')}
   430→            editingId={invoiceEditingId}
   431→            onUpdateSuccess={() => setInvoiceEditingId(null)}
   432→            showPrintActions={false}
   433→            onSaved={onSaved}
   434→          />
   435→        }
   436→      >
   437→        {/* FORM 2 KOLOM (desktop) — KIRI: Detail Dokumen + Informasi Pembayaran +
   438→            Kepada Yth, KANAN: Item + Informasi Tambahan. Kotak Data Perusahaan
   439→            dihapus (data perusahaan diatur di halaman Pengaturan). */}
   440→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-3 lg:gap-5 items-start">
   441→        {/* ===== KOLOM KIRI ===== */}
   442→        <div className="space-y-3 lg:space-y-5 min-w-0">
   443→        <div className="rounded-lg border bg-card p-3 sm:p-4 shadow-sm">
   444→          <h3 className="mb-3 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   445→            Detail Dokumen
   446→          </h3>
   447→          <div className="grid grid-cols-2 gap-2">
   448→            <div className="space-y-1.5">
   449→              <Label className="text-xs">No. Invoice</Label>
   450→              <Input
   451→                value={invoice.nomor}
   452→                readOnly
   453→                className="bg-slate-50 text-slate-500 cursor-not-allowed text-sm"
   454→              />
   455→            </div>
   456→            <div className="space-y-1.5">
   457→              <Label className="text-xs">Tanggal</Label>
   458→              <Input
   459→                type="date"
   460→                value={invoice.tanggal}
   461→                onChange={(e) => setInvoice((prev) => ({ ...prev, tanggal: e.target.value }))}
   462→                className="text-sm"
   463→              />
   464→            </div>
   465→          </div>
   466→          <div className="space-y-2">
   467→            <Label className="text-xs">Referensi (No. HC)</Label>
   468→            <Popover open={dropdownOpen} onOpenChange={setDropdownOpen}>
   469→              <PopoverAnchor asChild>
   470→                <button
   471→                  type="button"
   472→                  onClick={() => setDropdownOpen(true)}
   473→                  className="flex h-9 w-full items-center justify-between rounded-md border border-input bg-slate-50 px-3 py-2 text-sm shadow-xs outline-none cursor-pointer hover:bg-slate-100 transition-colors text-left"
   474→                >
   475→                  <span className={referensiInput ? 'text-slate-900' : 'text-slate-400'}>
   476→                    {referensiInput || 'Pilih No. HC...'}
   477→                  </span>
   478→                  <svg className="w-4 h-4 text-slate-400 shrink-0" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M19 9l-7 7-7-7" /></svg>
   479→                </button>
   480→              </PopoverAnchor>
   481→              <PopoverContent
   482→                align="start"
   483→                className="p-0 w-[var(--radix-popover-trigger-width)] max-h-52 overflow-y-auto"
   484→                onOpenAutoFocus={(e) => e.preventDefault()}
   485→              >
   486→                {filteredRiwayatList.length > 0 && (
   487→                  <div>
   488→                    <div className="px-3 py-1.5 text-[10px] font-semibold text-slate-400 uppercase bg-slate-50 border-b border-slate-100 sticky top-0">Riwayat Hitung Cetakan</div>
   489→                    {filteredRiwayatList.map((r) => (
   490→                      <button
   491→                        key={r.id}
   492→                        type="button"
   493→                        onMouseDown={(e) => { e.preventDefault(); handleReferensiSelect(r) }}
   494→                        className={`w-full text-left px-3 py-2 text-sm transition-colors ${invoice.referensi === r.nomorUrut ? 'bg-blue-50 text-blue-700 font-medium' : 'text-slate-700 hover:bg-blue-50 hover:text-blue-700'}`}
   495→                      >
   496→                        <span className="font-semibold text-xs text-emerald-700">{r.nomorUrut || '-'}</span>
   497→                        <span className="ml-1.5 truncate">{r.printName || ''}</span>
   498→                        {r.customerName && <span className="text-slate-400 ml-1.5 text-[11px]">({r.customerName})</span>}
   499→                      </button>
   500→                    ))}
   501→                  </div>
   502→                )}
   503→                {filteredRiwayatList.length === 0 && (
   504→                  <div className="px-3 py-3 text-sm text-slate-400 text-center">Tidak ada riwayat cetakan</div>
   505→                )}
   506→              </PopoverContent>
   507→            </Popover>
