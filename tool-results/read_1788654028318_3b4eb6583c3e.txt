     1→'use client';
     2→
     3→import { useEffect, useState, useCallback, useRef, useMemo } from 'react';
     4→import { useSearchParams, useRouter } from 'next/navigation';
     5→import { Input } from '@/components/ui/input';
     6→import { Label } from '@/components/ui/label';
     7→import { Textarea } from '@/components/ui/textarea';
     8→import { Popover, PopoverContent, PopoverAnchor } from '@/components/ui/popover';
     9→import { useDokuproStore } from '@/lib/store';
    10→import { CompanyFields } from './company-fields';
    11→import { ItemsFields } from './items-fields';
    12→import { InvoicePreview } from './invoice-preview';
    13→import { DocumentEditorLayout } from './document-editor-layout';
    14→import { DocumentActionButtons } from './document-action-buttons';
    15→import { formatRupiah } from '@/lib/format';
    16→import { getAuthHeaders } from '@/lib/auth';
    17→import { syncLinkedPelunasan } from '@/lib/sync-pelunasan';
    18→import { Truck } from 'lucide-react';
    19→import { Button } from '@/components/ui/button';
    20→import { toast } from 'sonner';
    21→import type { InvoiceData } from '@/lib/types';
    22→
    23→interface CustomerItem {
    24→  id: string;
    25→  name: string;
    26→  companyName?: string;
    27→  phone?: string;
    28→  email?: string;
    29→  address?: string;
    30→}
    31→
    32→interface RiwayatCetakanItem {
    33→  id: string;
    34→  nomorUrut: string;
    35→  printName: string;
    36→  customerName: string;
    37→  paperName: string;
    38→  paperGrammage: string;
    39→  cutWidth: string;
    40→  cutHeight: string;
    41→  quantity: string;
    42→  jumlahPesanan: string;
    43→  berapaMata: string;
    44→  warna: string;
    45→  machineName: string;
    46→  ongkosCetak: number;
    47→  machineName2: string;
    48→  ongkosCetak2: number;
    49→  totalPaperPrice: number;
    50→  pricePerSheet: number;
    51→  finishingNames: string;
    52→  finishingCost: number;
    53→  packingCost: number;
    54→  shippingCost: number;
    55→  otherCost: number;
    56→  otherCost2: number;
    57→  otherCostLabel: string;
    58→  otherCostLabel2: string;
    59→  glueCost: number;
    60→  glueBorongan: number;
    61→  subTotal: number;
    62→  profitPercent: number;
    63→  profitAmount: number;
    64→  grandTotal: number;
    65→  createdAt: string;
    66→}
    67→
    68→
    69→export function InvoiceEditor({ dpDisabled = false }: { dpDisabled?: boolean }) {
    70→  const invoice = useDokuproStore((s) => s.invoice);
    71→  const setInvoice = useDokuproStore((s) => s.setInvoice);
    72→  const resetDocument = useDokuproStore((s) => s.resetDocument);
    73→  const loadCompanyFromAPI = useDokuproStore((s) => s.loadCompanyFromAPI);
    74→  const invoiceEditingId = useDokuproStore((s) => s.invoiceEditingId);
    75→  const setInvoiceEditingId = useDokuproStore((s) => s.setInvoiceEditingId);
    76→  const router = useRouter();
    77→  const searchParams = useSearchParams();
    78→  const riwayatIdFromUrl = searchParams.get('riwayatId');
    79→  const autoSelectDoneRef = useRef<string | null>(null); // track which riwayatId was auto-selected
    80→  const [savingSj, setSavingSj] = useState(false);
    81→
    82→  // Riwayat cetakan dropdown state
    83→  const [riwayatList, setRiwayatList] = useState<RiwayatCetakanItem[]>([]);
    84→  const [referensiInput, setReferensiInput] = useState(invoice.referensi);
    85→  const [dropdownOpen, setDropdownOpen] = useState(false);
    86→
    87→  // Customer dropdown state
    88→  const [customerList, setCustomerList] = useState<CustomerItem[]>([]);
    89→  const [clientInput, setClientInput] = useState(invoice.client.nama);
    90→  const [clientDropdownOpen, setClientDropdownOpen] = useState(false);
    91→  const [clientTyping, setClientTyping] = useState(false);
    92→
    93→  const fetchRiwayatCetakan = useCallback(async () => {
    94→    try {
    95→      const res = await fetch('/api/riwayat-cetakan', { headers: getAuthHeaders() });
    96→      if (res.ok) {
    97→        const data = await res.json();
    98→        setRiwayatList(data);
    99→        return data;
   100→      }
   101→    } catch {
   102→      // silently fail
   103→    }
   104→    return [];
   105→  }, []);
   106→
   107→  // Fetch customer list
   108→  const fetchCustomers = useCallback(async () => {
   109→    try {
   110→      const res = await fetch('/api/customers', { headers: getAuthHeaders() });
   111→      if (res.ok) {
   112→        const data = await res.json();
   113→        setCustomerList(Array.isArray(data) ? data : []);
   114→      }
   115→    } catch {
   116→      // silently fail
   117→    }
   118→  }, []);
   119→
   120→  useEffect(() => { loadCompanyFromAPI() }, [loadCompanyFromAPI]);
   121→  useEffect(() => { fetchRiwayatCetakan() }, [fetchRiwayatCetakan]);
   122→  useEffect(() => { fetchCustomers() }, [fetchCustomers]);
   123→
   124→  // Fetch next Invoice number from server (only when creating new, not editing)
   125→  const fetchNextNumber = useCallback(() => {
   126→    // Read current editingId from store to avoid stale closure
   127→    const currentEditingId = useDokuproStore.getState().invoiceEditingId;
   128→    if (currentEditingId) return; // Don't overwrite nomor when editing
   129→    fetch('/api/history?preview=next-number&docType=invoice', { headers: getAuthHeaders() })
   130→      .then(res => res.ok ? res.json() : null)
   131→      .then(data => { if (data?.nextNumber) setInvoice((prev) => ({ ...prev, nomor: data.nextNumber })) })
   132→      .catch(() => {});
   133→  }, [setInvoice]);
   134→
   135→  useEffect(() => { fetchNextNumber() }, [fetchNextNumber]);
   136→
   137→  // Re-fetch next number after a document is saved
   138→  useEffect(() => {
   139→    const handler = () => { fetchNextNumber() };
   140→    window.addEventListener('dokupro:history-updated', handler);
   141→    return () => { window.removeEventListener('dokupro:history-updated', handler) };
   142→  }, [fetchNextNumber]);
   143→
   144→  // Mode Regular (dpDisabled): paksa DP = 0 dan sembunyikan input DP
   145→  useEffect(() => {
   146→    if (dpDisabled) {
   147→      setInvoice((prev) => (prev.dp === 0 ? prev : ({ ...prev, dp: 0 })));
   148→    }
   149→  }, [dpDisabled, setInvoice]);
   150→
   151→  // Sync referensiInput when invoice.referensi changes externally
   152→  useEffect(() => { setReferensiInput(invoice.referensi) }, [invoice.referensi]);
   153→  // Sync clientInput when invoice.client.nama changes externally
   154→  useEffect(() => { setClientInput(invoice.client.nama) }, [invoice.client.nama]);
   155→
   156→  // Filter riwayat list — when dropdown is open show all, otherwise not used
   157→  const filteredRiwayatList = riwayatList;
   158→
   159→  const handleReferensiInputChange = (value: string) => {
   160→    setReferensiInput(value);
   161→    setInvoice((prev) => ({ ...prev, referensi: value }));
   162→    setDropdownOpen(true);
   163→  };
   164→
   165→  // Apply referensi data using functional setInvoice to avoid stale closure
   166→  const applyReferensi = (item: RiwayatCetakanItem) => {
   167→    const ref = item.nomorUrut || item.printName || '';
   168→    setReferensiInput(ref);
   169→
   170→    // Build single item with multi-line description
   171→    const descLines: string[] = [];
   172→    if (item.printName) descLines.push(item.printName);
   173→
   174→    const bahanParts: string[] = [];
   175→    if (item.paperName) bahanParts.push(item.paperName);
   176→    if (item.paperGrammage) bahanParts.push(item.paperGrammage + 'g');
   177→    if (bahanParts.length > 0) descLines.push(bahanParts.join(' '));
   178→
   179→    if (item.finishingNames) {
   180→      const finishings = item.finishingNames.split(',').map(f => f.trim()).filter(Boolean);
   181→      descLines.push('Finishing:');
   182→      finishings.forEach(f => descLines.push(f));
   183→    }
   184→
   185→    const deskripsi = descLines.join('\n');
   186→    const jumlahPesanan = parseInt(item.jumlahPesanan) || 1;
   187→    const hargaPerPcs = jumlahPesanan > 0 ? Math.round(item.grandTotal / jumlahPesanan) : 0;
   188→
   189→    const newItems: InvoiceData['items'] = [{
   190→      id: 'riwayat-0',
   191→      deskripsi: deskripsi || ref,
   192→      qty: jumlahPesanan,
   193→      satuan: 'pcs',
   194→      harga: hargaPerPcs,
   195→    }];
   196→
   197→    // Use functional form to always get the LATEST state
   198→    setInvoice((prev) => ({
   199→      ...prev,
   200→      referensi: ref,
   201→      client: {
   202→        ...prev.client,
   203→        nama: item.customerName || prev.client.nama,
   204→      },
   205→      items: newItems,
   206→      catatan: '',
   207→      uangCapek: item.profitAmount || 0,
   208→    }));
   209→    setDropdownOpen(false);
   210→  };
   211→
   212→  const handleReferensiSelect = (item: RiwayatCetakanItem) => {
   213→    applyReferensi(item);
   214→  };
   215→
   216→  // Auto-select referensi when coming from hitung cetakan with riwayatId
   217→  // Fetches the specific riwayat directly by ID for reliability (no race condition with list loading)
   218→  useEffect(() => {
   219→    if (!riwayatIdFromUrl) return;
   220→    if (autoSelectDoneRef.current === riwayatIdFromUrl) return;
   221→
   222→    let cancelled = false;
   223→
   224→    const fetchAndApply = async () => {
   225→      // First, try to find it in the already-loaded list (fast path)
   226→      const foundInList = riwayatList.find((r) => r.id === riwayatIdFromUrl);
   227→      if (foundInList) {
   228→        if (!cancelled) {
   229→          autoSelectDoneRef.current = riwayatIdFromUrl;
   230→          applyReferensi(foundInList);
   231→        }
   232→        return;
   233→      }
   234→
   235→      // Not in list yet — fetch directly by ID from the API (reliable path)
   236→      try {
   237→        const res = await fetch(`/api/riwayat-cetakan?id=${riwayatIdFromUrl}`, { headers: getAuthHeaders() });
   238→        if (res.ok && !cancelled) {
   239→          const data = await res.json();
   240→          if (data && !data.error) {
   241→            autoSelectDoneRef.current = riwayatIdFromUrl;
   242→            applyReferensi(data as RiwayatCetakanItem);
   243→          }
   244→        }
   245→      } catch {
   246→        // Fallback: retry once after a short delay
   247→        if (cancelled) return;
   248→        await new Promise(r => setTimeout(r, 800));
   249→        if (cancelled) return;
   250→        try {
   251→          const res = await fetch(`/api/riwayat-cetakan?id=${riwayatIdFromUrl}`, { headers: getAuthHeaders() });
   252→          if (res.ok) {
   253→            const data = await res.json();
   254→            if (data && !data.error) {
   255→              autoSelectDoneRef.current = riwayatIdFromUrl;
   256→              applyReferensi(data as RiwayatCetakanItem);
   257→            }
   258→          }
   259→        } catch {
   260→          // Give up silently
   261→        }
   262→      }
   263→    };
   264→
   265→    fetchAndApply();
   266→
   267→    return () => { cancelled = true; };
   268→  }, [riwayatIdFromUrl]);
   269→
   270→  // Filter customer list (only when user is actively typing)
   271→  const filteredCustomerList = useMemo(() =>
   272→    clientTyping
   273→      ? customerList.filter((c) => {
   274→          const search = clientInput.toLowerCase().trim();
   275→          if (!search) return true;
   276→          return c.name.toLowerCase().includes(search) ||
   277→                 (c.companyName || '').toLowerCase().includes(search) ||
   278→                 (c.phone || '').toLowerCase().includes(search) ||
   279→                 (c.address || '').toLowerCase().includes(search);
   280→        })
   281→      : customerList
   282→  , [clientTyping, customerList, clientInput]);
   283→
   284→  const handleClientInputChange = (value: string) => {
   285→    setClientInput(value);
   286→    setClientTyping(true);
   287→    updateClient('nama', value);
   288→    setClientDropdownOpen(true);
   289→  };
   290→
   291→  const handleClientSelect = (item: CustomerItem) => {
   292→    setClientInput(item.name);
   293→    setClientTyping(false);
   294→    setInvoice((prev) => ({
   295→      ...prev,
   296→      client: {
   297→        nama: item.name,
   298→        kontak: item.phone || item.email || '',
   299→        alamat: item.address || '',
   300→      },
   301→    }));
   302→    setClientDropdownOpen(false);
   303→  };
   304→
   305→  const updateCompany = (company: typeof invoice.company) => {
   306→    setInvoice((prev) => ({ ...prev, company }));
   307→  };
   308→
   309→  const updateClient = (field: string, value: string) => {
   310→    setInvoice((prev) => ({
   311→      ...prev,
   312→      client: { ...prev.client, [field]: value },
   313→    }));
   314→  };
   315→
   316→  const subtotal = invoice.items.reduce((sum, item) => sum + item.qty * item.harga, 0);
   317→  const ppnAmount = subtotal * (invoice.ppn / 100);
   318→  const total = subtotal + ppnAmount;
   319→  const dpPercent = invoice.dp || 0;
   320→  const dpAmount = total * (dpPercent / 100);
   321→  const sisa = total - dpAmount;
   322→
   323→  const handleSuratJalan = async () => {
   324→    // Check if data has content
   325→    const pihakKedua = invoice.client?.nama;
   326→    const hasItem = invoice.items?.some((item) => item.deskripsi.trim() !== '');
   327→    if (!pihakKedua?.trim() || !hasItem) {
   328→      toast.error('Lengkapi data invoice terlebih dahulu!');
   329→      return;
   330→    }
   331→
   332→    setSavingSj(true);
   333→    try {
   334→      // Save invoice to history first
   335→      const dataToSave = { ...invoice, dpAmount, originalTotal: total };
   336→
   337→      if (invoiceEditingId) {
   338→        // Update existing record
   339→        const res = await fetch(`/api/history/${invoiceEditingId}`, {
   340→          method: 'PUT',
   341→          headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   342→          body: JSON.stringify({
   343→            nomor: invoice.nomor || '-',
   344→            tanggal: invoice.tanggal || '',
   345→            pihakKedua: invoice.client?.nama || '-',
   346→            total: '-',
   347→            dataJson: JSON.stringify(dataToSave),
   348→          }),
   349→        });
   350→        if (res.ok) {
   351→          // If DP was added/updated during this edit, ensure a linked
   352→          // invoice-pelunasan entry exists & is in sync so it shows up in
   353→          // the Editor Pelunasan tab. (Previously this only happened on CREATE.)
   354→          await syncLinkedPelunasan(invoiceEditingId, invoice.nomor || '-', dataToSave);
   355→          window.dispatchEvent(new CustomEvent('dokupro:history-updated'));
   356→          toast.success('Invoice diperbarui');
   357→          const invoiceId = invoiceEditingId;
   358→          setInvoiceEditingId(null);
   359→          resetDocument('invoice');
   360→          router.push(`/surat-jalan?invoiceId=${invoiceId}`);
   361→        } else {
   362→          toast.error('Gagal memperbarui invoice');
   363→        }
   364→      } else {
   365→        const res = await fetch('/api/history', {
   366→          method: 'POST',
   367→          headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   368→          body: JSON.stringify({
   369→            docType: 'invoice',
   370→            nomor: invoice.nomor || '-',
   371→            tanggal: invoice.tanggal || '',
   372→            pihakKedua: invoice.client?.nama || '-',
   373→            total: '-',
   374→            dataJson: JSON.stringify(dataToSave),
   375→          }),
   376→        });
   377→        if (res.ok) {
   378→          const saved = await res.json();
   379→          window.dispatchEvent(new CustomEvent('dokupro:history-updated'));
   380→
   381→          // If invoice has DP, ensure a linked invoice-pelunasan entry exists
   382→          // (creates if missing, updates if already present)
   383→          await syncLinkedPelunasan(saved.id, saved.nomor || invoice.nomor || '-', dataToSave);
   384→
   385→          toast.success('Invoice disimpan ke riwayat');
   386→          // Reset invoice form after saving
   387→          resetDocument('invoice');
   388→          router.push(`/surat-jalan?invoiceId=${saved.id}`);
   389→        } else if (res.status === 409) {
   390→          // Invoice already saved, get existing ID and navigate
   391→          const errData = await res.json().catch(() => ({}));
   392→          if (errData.id) {
   393→            resetDocument('invoice');
   394→            router.push(`/surat-jalan?invoiceId=${errData.id}`);
   395→          } else {
   396→            toast('Invoice sudah ada di riwayat.');
   397→          }
   398→        } else {
   399→          toast.error('Gagal menyimpan invoice');
   400→        }
   401→      }
   402→    } catch {
   403→      toast.error('Gagal menyimpan invoice');
   404→    }
   405→    setSavingSj(false);
   406→  };
   407→
   408→  return (
   409→    <>
   410→      <DocumentEditorLayout
   411→        title="Invoice"
   412→        previewMode="popup"
   413→        previewContent={<InvoicePreview data={invoice} />}
   414→        actions={
   415→          <div className="flex items-center gap-2 flex-wrap">
   416→            <DocumentActionButtons
   417→              docType="invoice"
   418→              documentLabel="Invoice"
   419→              currentData={invoice}
   420→              onReset={() => resetDocument('invoice')}
   421→              editingId={invoiceEditingId}
   422→              onUpdateSuccess={() => setInvoiceEditingId(null)}
   423→            />
   424→            <Button
   425→              size="sm"
   426→              onClick={handleSuratJalan}
   427→              disabled={savingSj}
   428→              className="bg-orange-600 hover:bg-orange-700 h-8 sm:h-9"
   429→            >
   430→              <Truck className="mr-1.5 h-3.5 w-3.5" />
   431→              {savingSj ? 'Menyimpan...' : 'Surat Jalan'}
   432→            </Button>
   433→          </div>
   434→        }
   435→      >
   436→        <CompanyFields company={invoice.company} onChange={updateCompany} />
   437→
   438→        <div className="rounded-lg border bg-card p-3 sm:p-4 shadow-sm">
   439→          <h3 className="mb-3 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   440→            Detail Dokumen
   441→          </h3>
   442→          <div className="grid grid-cols-2 gap-2">
   443→            <div className="space-y-1.5">
   444→              <Label className="text-xs">No. Invoice</Label>
   445→              <Input
   446→                value={invoice.nomor}
   447→                readOnly
   448→                className="bg-slate-50 text-slate-500 cursor-not-allowed"
   449→              />
   450→            </div>
   451→            <div className="space-y-1.5">
   452→              <Label className="text-xs">Tanggal</Label>
   453→              <Input
   454→                type="date"
   455→                value={invoice.tanggal}
   456→                onChange={(e) => setInvoice((prev) => ({ ...prev, tanggal: e.target.value }))}
   457→              />
   458→            </div>
   459→          </div>
   460→          <div className="space-y-2">
   461→            <Label className="text-xs">Referensi (No. HC)</Label>
   462→            <Popover open={dropdownOpen} onOpenChange={setDropdownOpen}>
   463→              <PopoverAnchor asChild>
   464→                <button
   465→                  type="button"
   466→                  onClick={() => setDropdownOpen(true)}
   467→                  className="flex h-9 w-full items-center justify-between rounded-md border border-input bg-slate-50 px-3 py-2 text-sm shadow-xs outline-none cursor-pointer hover:bg-slate-100 transition-colors text-left"
   468→                >
   469→                  <span className={referensiInput ? 'text-slate-900' : 'text-slate-400'}>
   470→                    {referensiInput || 'Pilih No. HC...'}
   471→                  </span>
   472→                  <svg className="w-4 h-4 text-slate-400 shrink-0" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M19 9l-7 7-7-7" /></svg>
   473→                </button>
   474→              </PopoverAnchor>
   475→              <PopoverContent
   476→                align="start"
   477→                className="p-0 w-[var(--radix-popover-trigger-width)] max-h-52 overflow-y-auto"
   478→                onOpenAutoFocus={(e) => e.preventDefault()}
   479→              >
   480→                {filteredRiwayatList.length > 0 && (
   481→                  <div>
   482→                    <div className="px-3 py-1.5 text-[10px] font-semibold text-slate-400 uppercase bg-slate-50 border-b border-slate-100 sticky top-0">Riwayat Hitung Cetakan</div>
   483→                    {filteredRiwayatList.map((r) => (
   484→                      <button
   485→                        key={r.id}
   486→                        type="button"
   487→                        onMouseDown={(e) => { e.preventDefault(); handleReferensiSelect(r) }}
   488→                        className={`w-full text-left px-3 py-2 text-sm transition-colors ${invoice.referensi === r.nomorUrut ? 'bg-blue-50 text-blue-700 font-medium' : 'text-slate-700 hover:bg-blue-50 hover:text-blue-700'}`}
   489→                      >
   490→                        <span className="font-semibold text-xs text-emerald-700">{r.nomorUrut || '-'}</span>
   491→                        <span className="ml-1.5 truncate">{r.printName || ''}</span>
   492→                        {r.customerName && <span className="text-slate-400 ml-1.5 text-[11px]">({r.customerName})</span>}
   493→                      </button>
   494→                    ))}
   495→                  </div>
   496→                )}
   497→                {filteredRiwayatList.length === 0 && (
   498→                  <div className="px-3 py-3 text-sm text-slate-400 text-center">Tidak ada riwayat cetakan</div>
   499→                )}
   500→              </PopoverContent>
   501→            </Popover>
   502→          </div>
   503→        </div>
   504→
   505→        {/* Informasi Pembayaran - moved below Detail Dokumen */}
   506→        <div className="rounded-lg border bg-card p-3 sm:p-4 shadow-sm">
   507→          <h3 className="mb-3 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   508→            Informasi Pembayaran
   509→          </h3>
   510→          <div className="grid grid-cols-2 gap-3">
   511→            <div className="space-y-1.5">
   512→              <Label className="text-xs">Tgl. Jatuh Tempo</Label>
   513→              <Input
   514→                type="date"
   515→                value={invoice.tanggalJatuhTempo}
   516→                onChange={(e) => setInvoice((prev) => ({ ...prev, tanggalJatuhTempo: e.target.value }))}
   517→              />
   518→            </div>
   519→            <div className="space-y-1.5">
   520→              <Label className="text-xs">Cara Pembayaran</Label>
   521→              <div className="flex gap-1.5">
   522→                {(['cash', 'transfer', 'giro'] as const).map((method) => {
