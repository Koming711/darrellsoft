    45→    45→// --- Helper ---
    46→    46→function getTodayStr(): string {
    47→    47→  const d = new Date();
    48→    48→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
    49→    49→}

    51→    51→// --- Parse dataJson for document info ---
    52→    52→function parseDocInfo(entry: HistoryEntry) {
    53→    53→  try {
    54→    54→    const parsed = JSON.parse(entry.dataJson);
    55→    55→    const items = parsed.items || [];
    56→    56→    const firstItem = items[0];
    57→    57→    const namaBarang = firstItem?.deskripsi || '';
    58→    58→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0);
    59→    59→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0);
    60→    60→    const ppn = parsed.ppn || 0;
    61→    61→    const dpPercent = parsed.dp || 0;
    62→    62→    const totalHarga = subtotal + (subtotal * ppn / 100);
    63→    63→    const dpAmount = totalHarga * (dpPercent / 100);
    64→    64→    const sisa = totalHarga - dpAmount;
    65→    65→    const lunas = parsed.lunas === true;
    66→    66→    const tanggalJatuhTempo = parsed.tanggalJatuhTempo || '';
    67→    67→    const tanggalPelunasan = parsed.tanggalPelunasan || '';
    68→    68→    return { namaBarang, totalQty, totalHarga, dpPercent, dp: dpAmount, sisa, lunas, tanggalJatuhTempo, tanggalPelunasan };
    69→    69→  } catch {
    70→    70→    return { namaBarang: '', totalQty: 0, totalHarga: 0, dpPercent: 0, dp: 0, sisa: 0, lunas: false, tanggalJatuhTempo: '', tanggalPelunasan: '' };
    71→    71→  }
    72→    72→}

    74→    74→// --- Parse dataJson for InvoiceData ---
    75→    75→function parseInvoiceData(entry: HistoryEntry): InvoiceData {
    76→    76→  try {
    77→    77→    const parsed = JSON.parse(entry.dataJson);
    78→    78→    const company: CompanyInfo = {
    79→    79→      nama: parsed.company?.nama || DEFAULT_COMPANY.nama,
    80→    80→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
    81→    81→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
    82→    82→      email: parsed.company?.email || DEFAULT_COMPANY.email,
    83→    83→      npwp: parsed.company?.npwp || '',
    84→    84→      website: parsed.company?.website || '',
    85→    85→      ppn: parsed.company?.ppn ?? parsed.ppn ?? 11,
    86→    86→      logo: parsed.company?.logo || '',
    87→    87→      bankName: parsed.company?.bankName || '',
    88→    88→      bankAccount: parsed.company?.bankAccount || '',
    89→    89→      bankHolder: parsed.company?.bankHolder || '',
    90→    90→      bankName2: parsed.company?.bankName2 || '',
    91→    91→      bankAccount2: parsed.company?.bankAccount2 || '',
    92→    92→      bankHolder2: parsed.company?.bankHolder2 || '',
    93→    93→    };
    94→    94→    const client = parsed.client || { nama: '', kontak: '', alamat: '' };
    95→    95→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number }, i: number) => ({
    96→    96→      id: it.id || `item-${i}`,
    97→    97→      deskripsi: it.deskripsi || '',
    98→    98→      qty: it.qty || 0,
    99→    99→      satuan: it.satuan || '',
   100→   100→      harga: it.harga || 0,
   101→   101→    }));
   102→   102→    return {
   103→   103→      type: 'invoice',
   104→   104→      company,
   105→   105→      nomor: parsed.nomor || entry.nomor || '',
   106→   106→      tanggal: parsed.tanggal || entry.tanggal || '',
   107→   107→      referensi: parsed.referensi || '',
   108→   108→      client,
   109→   109→      items,
   110→   110→      ppn: parsed.ppn ?? 11,
   111→   111→      dp: parsed.dp || 0,
   112→   112→      catatan: parsed.catatan || '',
   113→   113→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   114→   114→      caraPembayaran: parsed.caraPembayaran || '',
   115→   115→      tanggalGiro: parsed.tanggalGiro || '',
   116→   116→      lunas: parsed.lunas === true,
   117→   117→      tanggalPelunasan: parsed.tanggalPelunasan || '',
   118→   118→    };
   119→   119→  } catch {
   120→   120→    return {
   121→   121→      type: 'invoice',
   122→   122→      company: { ...DEFAULT_COMPANY },
   123→   123→      nomor: entry.nomor || '',
   124→   124→      tanggal: entry.tanggal || '',
   125→   125→      referensi: '',
   126→   126→      client: { nama: '', kontak: '', alamat: '' },
   127→   127→      items: [],
   128→   128→      ppn: 11,
   129→   129→      dp: 0,
   130→   130→      catatan: '',
   131→   131→      tanggalJatuhTempo: '',
   132→   132→      caraPembayaran: '',
   133→   133→      tanggalGiro: '',
   134→   134→      lunas: false,
   135→   135→      tanggalPelunasan: '',
   136→   136→    };
   137→   137→  }
   138→   138→}

   246→   246→

   255→   255→        tanggal: getTodayStr(),

   257→   257→        client: { nama: '', kontak: '', alamat: '' },
   258→   258→        items: [],
   259→   259→        ppn: 11,
   260→   260→        dp: 0,
   261→   261→        catatan: '',
   262→   262→        tanggalJatuhTempo: '',
   263→   263→        caraPembayaran: '',
   264→   264→        tanggalGiro: '',

   268→   268→    }
   269→   269→    return {

   389→   389→    >
   390→   390→      {/* Invoice Selector */}
   391→   391→      <div className="rounded-lg border bg-card p-3 sm:p-4 shadow-sm">
   392→   392→        <h3 className="mb-3 text-xs font-semibold uppercase tracking-wider text-muted-foreground flex items-center gap-1.5">
