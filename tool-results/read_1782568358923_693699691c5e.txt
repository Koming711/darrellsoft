     1→'use client'
     2→
     3→import { useState, useRef, useEffect, useCallback, FormEvent } from 'react'
     4→import { motion, AnimatePresence, useMotionValue } from 'framer-motion'
     5→import { Sparkles, X, Send, Trash2, Bot, User } from 'lucide-react'
     6→import { Button } from '@/components/ui/button'
     7→import { Textarea } from '@/components/ui/textarea'
     8→import { Tooltip, TooltipContent, TooltipProvider, TooltipTrigger } from '@/components/ui/tooltip'
     9→import { useLanguage } from '@/contexts/language-context'
    10→import { toast } from 'sonner'
    11→
    12→interface Message {
    13→  id: string
    14→  role: 'user' | 'assistant'
    15→  content: string
    16→}
    17→
    18→interface AIAssistantProps {
    19→  /** Bottom offset in px, to clear the mobile bottom nav when needed. Default: 84 (mobile) / 24 (desktop). */
    20→  bottomOffset?: number
    21→}
    22→
    23→// Simple id generator (avoid crypto for SSR safety)
    24→function genId() {
    25→  return Date.now().toString(36) + Math.random().toString(36).slice(2, 8)
    26→}
    27→
    28→// Very small markdown-ish renderer: supports **bold**, line breaks, and bullet lists
    29→function renderContent(text: string) {
    30→  const lines = text.split('\n')
    31→  return lines.map((line, i) => {
    32→    if (line.trim() === '') return <div key={i} className="h-2" />
    33→    // bold **text**
    34→    const parts = line.split(/(\*\*[^*]+\*\*)/g)
    35→    const rendered = parts.map((part, j) => {
    36→      if (part.startsWith('**') && part.endsWith('**')) {
    37→        return <strong key={j} className="font-semibold">{part.slice(2, -2)}</strong>
    38→      }
    39→      return <span key={j}>{part}</span>
    40→    })
    41→    return <div key={i} className="leading-relaxed">{rendered}</div>
    42→  })
    43→}
    44→
    45→interface DragState {
    46→  pointerId: number
    47→  startClientX: number
    48→  startClientY: number
    49→  startOffsetX: number
    50→  startOffsetY: number
    51→  moved: boolean
    52→  defaultLeft: number
    53→  defaultTop: number
    54→  buttonSize: number
    55→}
    56→
    57→const FAB_STORAGE_KEY = 'ai_fab_offset'
    58→const DRAG_THRESHOLD = 5 // px — pointer must move more than this to count as a drag (not a tap)
    59→const VIEWPORT_MARGIN = 8 // px — keep button at least this far from viewport edges
    60→
    61→export function AIAssistant({ bottomOffset }: AIAssistantProps) {
    62→  const { t } = useLanguage()
    63→  const [open, setOpen] = useState(false)
    64→  const [input, setInput] = useState('')
    65→  const [loading, setLoading] = useState(false) // waiting for first token / request in flight
    66→  const [streaming, setStreaming] = useState(false) // actively receiving tokens
    67→  const [messages, setMessages] = useState<Message[]>([])
    68→  const streamingMsgIdRef = useRef<string | null>(null)
    69→  const scrollRef = useRef<HTMLDivElement>(null)
    70→  const textareaRef = useRef<HTMLTextAreaElement>(null)
    71→  const fetchedWelcomeRef = useRef(false)
    72→
    73→  // FAB bottom offset — hydration-safe.
    74→  // Stable initial value (24) matches SSR + initial client render.
    75→  // Updated AFTER mount to clear mobile bottom nav (84) on small screens.
    76→  const [fabBottom, setFabBottom] = useState<number>(24)
    77→  const [isDragging, setIsDragging] = useState(false)
    78→
    79→  // Drag offset via Framer Motion motion values — hydration-safe.
    80→  // Start at 0 on both server & client; updated client-only after mount / during drag.
    81→  // Framer Motion composes `x`/`y` with `scale` (from animate) into a single transform — no conflict.
    82→  const dragX = useMotionValue(0)
    83→  const dragY = useMotionValue(0)
    84→
    85→  const dragRef = useRef<DragState | null>(null)
    86→  const wasDragRef = useRef(false)
    87→  const fabRef = useRef<HTMLDivElement>(null)
    88→
    89→  // After mount: set mobile/desktop offset + restore saved drag position
    90→  useEffect(() => {
    91→    if (bottomOffset !== undefined) {
    92→      setFabBottom(bottomOffset)
    93→    } else {
    94→      setFabBottom(window.innerWidth < 768 ? 84 : 24)
    95→    }
    96→    try {
    97→      const saved = localStorage.getItem(FAB_STORAGE_KEY)
    98→      if (saved) {
    99→        const offset = JSON.parse(saved)
   100→        if (typeof offset.x === 'number' && typeof offset.y === 'number') {
   101→          dragX.set(offset.x)
   102→          dragY.set(offset.y)
   103→        }
   104→      }
   105→    } catch {}
   106→  }, [bottomOffset, dragX, dragY])
   107→
   108→  // Re-clamp position on viewport resize (saved position might now be out of bounds)
   109→  useEffect(() => {
   110→    const handleResize = () => {
   111→      const el = fabRef.current
   112→      if (!el) return
   113→      const rect = el.getBoundingClientRect()
   114→      const currentX = dragX.get()
   115→      const currentY = dragY.get()
   116→      const defaultLeft = rect.left - currentX
   117→      const defaultTop = rect.top - currentY
   118→      const buttonSize = rect.width
   119→      const minX = VIEWPORT_MARGIN - defaultLeft
   120→      const maxX = window.innerWidth - defaultLeft - buttonSize - VIEWPORT_MARGIN
   121→      const minY = VIEWPORT_MARGIN - defaultTop
   122→      const maxY = window.innerHeight - defaultTop - buttonSize - VIEWPORT_MARGIN
   123→      dragX.set(Math.min(Math.max(currentX, minX), maxX))
   124→      dragY.set(Math.min(Math.max(currentY, minY), maxY))
   125→    }
   126→    window.addEventListener('resize', handleResize)
   127→    return () => window.removeEventListener('resize', handleResize)
   128→  }, [dragX, dragY])
   129→
   130→  // Load persisted conversation from sessionStorage
   131→  useEffect(() => {
   132→    if (fetchedWelcomeRef.current) return
   133→    fetchedWelcomeRef.current = true
   134→    try {
   135→      const saved = sessionStorage.getItem('ai_assistant_messages')
   136→      if (saved) {
   137→        const parsed = JSON.parse(saved) as Message[]
   138→        if (Array.isArray(parsed) && parsed.length > 0) {
   139→          setMessages(parsed)
   140→          return
   141→        }
   142→      }
   143→    } catch {}
   144→    // Initial welcome message
   145→    setMessages([
   146→      { id: genId(), role: 'assistant', content: t('ai_assistant_welcome') },
   147→    ])
   148→  }, [t])
   149→
   150→  // Persist messages to sessionStorage
   151→  useEffect(() => {
   152→    if (messages.length > 0) {
   153→      try {
   154→        sessionStorage.setItem('ai_assistant_messages', JSON.stringify(messages))
   155→      } catch {}
   156→    }
   157→  }, [messages])
   158→
   159→  // Auto-scroll to bottom on new message
   160→  useEffect(() => {
   161→    if (scrollRef.current) {
   162→      scrollRef.current.scrollTop = scrollRef.current.scrollHeight
   163→    }
   164→  }, [messages, loading])
   165→
   166→  // Auto-resize textarea
   167→  useEffect(() => {
   168→    if (textareaRef.current) {
   169→      textareaRef.current.style.height = 'auto'
   170→      textareaRef.current.style.height = Math.min(textareaRef.current.scrollHeight, 120) + 'px'
   171→    }
   172→  }, [input])
   173→
   174→  // Escape to close
   175→  useEffect(() => {
   176→    if (!open) return
   177→    const handler = (e: KeyboardEvent) => {
   178→      if (e.key === 'Escape') setOpen(false)
   179→    }
   180→    window.addEventListener('keydown', handler)
   181→    return () => window.removeEventListener('keydown', handler)
   182→  }, [open])
   183→
   184→  const suggestions = [
   185→    t('ai_assistant_suggestion_1'),
   186→    t('ai_assistant_suggestion_2'),
   187→    t('ai_assistant_suggestion_3'),
   188→    t('ai_assistant_suggestion_4'),
   189→    t('ai_assistant_suggestion_5'),
   190→  ]
   191→
   192→  const sendMessage = useCallback(async (text: string) => {
   193→    const trimmed = text.trim()
   194→    if (!trimmed || loading) return
   195→
   196→    const userMsg: Message = { id: genId(), role: 'user', content: trimmed }
   197→    const history = messages
   198→      .filter((m) => !(m.role === 'assistant' && m.id === messages[0]?.id)) // exclude welcome if it's first
   199→      .map((m) => ({ role: m.role, content: m.content }))
   200→
   201→    // Insert user message + an empty assistant placeholder that we'll fill as tokens stream in.
   202→    const aiMsgId = genId()
   203→    streamingMsgIdRef.current = aiMsgId
   204→    setMessages((prev) => [
   205→      ...prev,
   206→      userMsg,
   207→      { id: aiMsgId, role: 'assistant', content: '' },
   208→    ])
   209→    setInput('')
   210→    setLoading(true)
   211→    setStreaming(false)
   212→
   213→    // Safety: if no first token arrives within 25s, show a gentle nudge (but keep waiting).
   214→    let firstTokenTimer: ReturnType<typeof setTimeout> | null = null
   215→    let firstTokenReceived = false
   216→
   217→    try {
   218→      const res = await fetch('/api/ai-assistant', {
   219→        method: 'POST',
   220→        headers: { 'Content-Type': 'application/json', Accept: 'text/event-stream' },
   221→        body: JSON.stringify({ message: trimmed, history }),
   222→      })
   223→
   224→      // Non-streaming error response (JSON)
   225→      const contentType = res.headers.get('content-type') || ''
   226→      if (!res.ok || !contentType.includes('text/event-stream')) {
   227→        let errMsg = t('ai_assistant_error')
   228→        try {
   229→          const data = await res.json()
   230→          if (data?.error) errMsg = data.error
   231→        } catch {}
   232→        throw new Error(errMsg)
   233→      }
   234→
   235→      if (!res.body) {
   236→        throw new Error(t('ai_assistant_error'))
   237→      }
   238→
   239→      const reader = res.body.getReader()
   240→      const decoder = new TextDecoder()
   241→      let buffer = ''
   242→      let accumulated = ''
   243→
   244→      const flushEvents = (eventsChunk: string) => {
   245→        const events = eventsChunk.split('\n\n')
   246→        const leftover = events.pop() || ''
   247→        for (const evt of events) {
   248→          const trimmedEvt = evt.trim()
   249→          if (!trimmedEvt) continue
   250→          for (const line of trimmedEvt.split('\n')) {
   251→            if (!line.startsWith('data:')) continue
   252→            const payload = line.slice(5).trim()
   253→            if (!payload) continue
   254→            if (payload === '[DONE]') return 'done' as const
   255→            try {
   256→              const json = JSON.parse(payload)
   257→              if (json?.error) {
   258→                throw new Error(String(json.error))
   259→              }
   260→              const delta: string | undefined =
   261→                json?.delta ?? json?.choices?.[0]?.delta?.content ?? json?.content
   262→              if (typeof delta === 'string' && delta.length > 0) {
   263→                if (!firstTokenReceived) {
   264→                  firstTokenReceived = true
   265→                  setStreaming(true)
   266→                  setLoading(false)
   267→                  if (firstTokenTimer) {
   268→                    clearTimeout(firstTokenTimer)
   269→                    firstTokenTimer = null
   270→                  }
   271→                }
   272→                accumulated += delta
   273→                const snapshot = accumulated
   274→                setMessages((prev) =>
   275→                  prev.map((m) => (m.id === aiMsgId ? { ...m, content: snapshot } : m))
   276→                )
   277→              }
   278→            } catch {
   279→              // ignore unparseable keepalive lines
   280→            }
   281→          }
   282→        }
   283→        return leftover
   284→      }
   285→
   286→      firstTokenTimer = setTimeout(() => {
   287→        if (!firstTokenReceived && streamingMsgIdRef.current === aiMsgId) {
   288→          setMessages((prev) =>
   289→            prev.map((m) =>
   290→              m.id === aiMsgId
   291→                ? { ...m, content: 'Sedang berpikir… mohon tunggu sebentar.' }
   292→                : m
   293→            )
   294→          )
   295→        }
   296→      }, 25000)
   297→
   298→      while (true) {
   299→        const { done, value } = await reader.read()
   300→        if (done) break
   301→        buffer += decoder.decode(value, { stream: true })
   302→        const result = flushEvents(buffer)
   303→        if (result === 'done') {
   304→          buffer = ''
   305→          break
   306→        }
   307→        buffer = result
   308→      }
   309→      // Flush trailing
   310→      if (buffer.trim()) {
   311→        flushEvents(buffer + '\n\n')
   312→      }
   313→
   314→      if (!firstTokenReceived) {
   315→        // No tokens ever arrived — replace placeholder with an error message
   316→        setMessages((prev) =>
   317→          prev.map((m) =>
   318→            m.id === aiMsgId
   319→              ? { ...m, content: t('ai_assistant_error') }
   320→              : m
   321→          )
   322→        )
   323→        toast.error(t('ai_assistant_error'))
   324→      }
   325→    } catch (err) {
   326→      const errorMsg = err instanceof Error ? err.message : t('ai_assistant_error')
   327→      toast.error(errorMsg)
   328→      setMessages((prev) => {
   329→        // If the placeholder is still empty, replace it; otherwise append a new error msg.
   330→        const placeholder = prev.find((m) => m.id === aiMsgId)
   331→        if (placeholder && placeholder.content === '') {
   332→          return prev.map((m) =>
   333→            m.id === aiMsgId ? { ...m, content: errorMsg } : m
   334→          )
   335→        }
   336→        return [...prev, { id: genId(), role: 'assistant', content: errorMsg }]
   337→      })
   338→    } finally {
   339→      if (firstTokenTimer) clearTimeout(firstTokenTimer)
   340→      streamingMsgIdRef.current = null
   341→      setLoading(false)
   342→      setStreaming(false)
   343→    }
   344→  }, [loading, messages, t])
   345→
   346→  const handleSubmit = (e: FormEvent) => {
   347→    e.preventDefault()
   348→    sendMessage(input)
   349→  }
   350→
   351→  const handleClear = () => {
   352→    streamingMsgIdRef.current = null
   353→    setLoading(false)
   354→    setStreaming(false)
   355→    setMessages([{ id: genId(), role: 'assistant', content: t('ai_assistant_welcome') }])
   356→    try {
   357→      sessionStorage.removeItem('ai_assistant_messages')
   358→    } catch {}
   359→  }
   360→
   361→  // -------------------- Drag handlers --------------------
   362→
   363→  const handlePointerDown = (e: React.PointerEvent) => {
   364→    const el = fabRef.current
   365→    if (!el) return
   366→    // Only respond to primary mouse button; touch & pen always start drag
   367→    if (e.pointerType === 'mouse' && e.button !== 0) return
   368→    const rect = el.getBoundingClientRect()
   369→    const currentX = dragX.get()
   370→    const currentY = dragY.get()
   371→    dragRef.current = {
   372→      pointerId: e.pointerId,
   373→      startClientX: e.clientX,
   374→      startClientY: e.clientY,
   375→      startOffsetX: currentX,
   376→      startOffsetY: currentY,
   377→      moved: false,
   378→      defaultLeft: rect.left - currentX,
   379→      defaultTop: rect.top - currentY,
   380→      buttonSize: rect.width,
   381→    }
   382→    try {
   383→      ;(e.currentTarget as HTMLElement).setPointerCapture(e.pointerId)
   384→    } catch {}
   385→  }
   386→
   387→  const handlePointerMove = (e: React.PointerEvent) => {
   388→    const ds = dragRef.current
   389→    if (!ds || ds.pointerId !== e.pointerId) return
   390→    const dx = e.clientX - ds.startClientX
   391→    const dy = e.clientY - ds.startClientY
   392→    if (!ds.moved && (Math.abs(dx) > DRAG_THRESHOLD || Math.abs(dy) > DRAG_THRESHOLD)) {
   393→      ds.moved = true
   394→      setIsDragging(true)
   395→    }
   396→    if (!ds.moved) return
   397→    const minX = VIEWPORT_MARGIN - ds.defaultLeft
   398→    const maxX = window.innerWidth - ds.defaultLeft - ds.buttonSize - VIEWPORT_MARGIN
   399→    const minY = VIEWPORT_MARGIN - ds.defaultTop
   400→    const maxY = window.innerHeight - ds.defaultTop - ds.buttonSize - VIEWPORT_MARGIN
   401→    const newX = Math.min(Math.max(ds.startOffsetX + dx, minX), maxX)
   402→    const newY = Math.min(Math.max(ds.startOffsetY + dy, minY), maxY)
   403→    dragX.set(newX)
   404→    dragY.set(newY)
   405→  }
   406→
   407→  const handlePointerUp = (e: React.PointerEvent) => {
   408→    const ds = dragRef.current
   409→    if (!ds) return
   410→    wasDragRef.current = ds.moved
   411→    dragRef.current = null
   412→    try {
   413→      ;(e.currentTarget as HTMLElement).releasePointerCapture(ds.pointerId)
   414→    } catch {}
   415→    if (ds.moved) {
   416→      setIsDragging(false)
   417→      try {
   418→        localStorage.setItem(FAB_STORAGE_KEY, JSON.stringify({ x: dragX.get(), y: dragY.get() }))
   419→      } catch {}
   420→    }
   421→  }
   422→
   423→  const handleClick = () => {
   424→    // If the last pointer interaction was a drag (not a tap), don't open the chat.
   425→    if (wasDragRef.current) {
   426→      wasDragRef.current = false
   427→      return
   428→    }
   429→    setOpen(true)
   430→  }
   431→
   432→  // Reset position (double-click / context menu action)
   433→  const handleDoubleClick = () => {
   434→    dragX.set(0)
   435→    dragY.set(0)
   436→    try {
   437→      localStorage.removeItem(FAB_STORAGE_KEY)
   438→    } catch {}
   439→    toast.success('Posisi icon AI direset')
   440→  }
   441→
   442→  return (
   443→    <>
   444→      {/* Floating Action Button (FAB) — draggable */}
   445→      <AnimatePresence>
   446→        {!open && (
   447→          <motion.div
   448→            ref={fabRef}
   449→            initial={{ scale: 0, opacity: 0 }}
   450→            animate={{ scale: 1, opacity: 1 }}
   451→            exit={{ scale: 0, opacity: 0 }}
   452→            transition={{ type: 'spring', stiffness: 260, damping: 20 }}
   453→            className="fixed z-[60] right-4 md:right-6 touch-none select-none"
   454→            style={{
   455→              bottom: fabBottom,
   456→              x: dragX,
   457→              y: dragY,
   458→              cursor: isDragging ? 'grabbing' : 'grab',
   459→            }}
   460→            onPointerDown={handlePointerDown}
   461→            onPointerMove={handlePointerMove}
   462→            onPointerUp={handlePointerUp}
   463→            onPointerCancel={handlePointerUp}
   464→            onDoubleClick={handleDoubleClick}
   465→          >
   466→            <TooltipProvider delayDuration={300}>
   467→              <Tooltip>
   468→                <TooltipTrigger asChild>
   469→                  <button
   470→                    onClick={handleClick}
   471→                    aria-label={t('ai_assistant_title')}
   472→                    className={`group relative flex items-center justify-center w-14 h-14 md:w-16 md:h-16 rounded-full bg-gradient-to-br from-blue-600 to-sky-400 hover:from-blue-700 hover:to-sky-500 text-white shadow-xl shadow-blue-600/30 hover:shadow-2xl hover:shadow-blue-600/40 transition-all duration-300 hover:scale-105 active:scale-95 ${
   473→                      isDragging ? 'ring-4 ring-blue-400/50 scale-105 shadow-2xl' : ''
   474→                    }`}
   475→                  >
   476→                    {/* Pulse ring */}
   477→                    <span className="absolute inset-0 rounded-full bg-blue-500/40 animate-ping [animation-duration:2s]" />
   478→                    <Sparkles className="relative w-6 h-6 md:w-7 md:h-7" />
   479→                  </button>
   480→                </TooltipTrigger>
   481→                <TooltipContent side="left" className="mr-2">
   482→                  <p className="text-xs font-medium">{t('ai_assistant_title')}</p>
   483→                  <p className="text-[10px] text-slate-400 mt-0.5">Geser untuk memindahkan · Double-klik untuk reset</p>
   484→                </TooltipContent>
   485→              </Tooltip>
   486→            </TooltipProvider>
   487→          </motion.div>
   488→        )}
   489→      </AnimatePresence>
   490→
   491→      {/* Chat Panel */}
   492→      <AnimatePresence>
   493→        {open && (
   494→          <>
   495→            {/* Mobile backdrop */}
   496→            <motion.div
   497→              initial={{ opacity: 0 }}
   498→              animate={{ opacity: 1 }}
   499→              exit={{ opacity: 0 }}
   500→              onClick={() => setOpen(false)}
   501→              className="fixed inset-0 z-[60] bg-black/30 backdrop-blur-sm md:hidden"
   502→            />
   503→
   504→            <motion.div
   505→              initial={{ opacity: 0, y: 30, scale: 0.96 }}
   506→              animate={{ opacity: 1, y: 0, scale: 1 }}
   507→              exit={{ opacity: 0, y: 20, scale: 0.96 }}
   508→              transition={{ type: 'spring', stiffness: 280, damping: 26 }}
   509→              className="fixed z-[70] flex flex-col bg-white dark:bg-slate-900 shadow-2xl border border-slate-200 dark:border-slate-700 rounded-2xl overflow-hidden
   510→                inset-2 bottom-2 top-auto h-[85vh] max-h-[640px]
   511→                md:inset-auto md:bottom-6 md:right-6 md:w-[400px] md:h-[600px] md:max-h-[75vh]"
   512→            >
   513→              {/* Header */}
   514→              <div className="flex items-center justify-between px-4 py-3 bg-gradient-to-r from-blue-600 to-sky-500 text-white shrink-0">
   515→                <div className="flex items-center gap-3 min-w-0">
   516→                  <div className="flex items-center justify-center w-9 h-9 rounded-full bg-white/20 shrink-0">
   517→                    <Sparkles className="w-5 h-5" />
   518→                  </div>
   519→                  <div className="min-w-0">
   520→                    <h3 className="font-bold text-sm truncate">{t('ai_assistant_title')}</h3>
   521→                    <p className="text-[11px] text-white/80 truncate flex items-center gap-1">
   522→                      <span className="inline-block w-1.5 h-1.5 rounded-full bg-emerald-300 animate-pulse" />
   523→                      {t('ai_assistant_subtitle')}
   524→                    </p>
   525→                  </div>
   526→                </div>
   527→                <div className="flex items-center gap-1 shrink-0">
   528→                  <TooltipProvider delayDuration={300}>
   529→                    <Tooltip>
   530→                      <TooltipTrigger asChild>
   531→                        <button
   532→                          onClick={handleClear}
   533→                          aria-label={t('ai_assistant_clear')}
   534→                          className="p-1.5 rounded-lg hover:bg-white/20 transition-colors"
   535→                        >
   536→                          <Trash2 className="w-4 h-4" />
   537→                        </button>
   538→                      </TooltipTrigger>
   539→                      <TooltipContent side="bottom">
   540→                        <p className="text-xs">{t('ai_assistant_clear')}</p>
   541→                      </TooltipContent>
   542→                    </Tooltip>
   543→                  </TooltipProvider>
   544→                  <button
   545→                    onClick={() => setOpen(false)}
   546→                    aria-label={t('ai_assistant_close')}
   547→                    className="p-1.5 rounded-lg hover:bg-white/20 transition-colors"
   548→                  >
   549→                    <X className="w-5 h-5" />
   550→                  </button>
   551→                </div>
   552→              </div>
