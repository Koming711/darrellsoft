   458→  // Compress image to max 500KB using canvas
   459→  const compressImage = async (file: File): Promise<File> => {
   460→    return new Promise((resolve, reject) => {
   461→      const img = new Image()
   462→      const reader = new FileReader()
   463→      reader.onload = (e) => {
   464→        img.onload = () => {
   465→          const canvas = document.createElement('canvas')
   466→          const MAX_SIZE = 500 * 1024 // 500KB
   467→          let quality = 0.85
   468→          let w = img.width
   469→          let h = img.height
   470→          // Limit max dimensions to 1024px
   471→          const maxDim = 1024
   472→          if (w > maxDim || h > maxDim) {
   473→            if (w > h) { h = Math.round(h * maxDim / w); w = maxDim }
   474→            else { w = Math.round(w * maxDim / h); h = maxDim }
   475→          }
   476→          const compress = () => {
   477→            canvas.width = w
   478→            canvas.height = h
   479→            const ctx = canvas.getContext('2d')
   480→            if (!ctx) { reject(new Error('Canvas not supported')); return }
   481→            ctx.drawImage(img, 0, 0, w, h)
   482→            canvas.toBlob((blob) => {
   483→              if (!blob) { reject(new Error('Compression failed')); return }
   484→              if (blob.size <= MAX_SIZE || quality <= 0.1) {
   485→                resolve(new File([blob], file.name.replace(/\.[^.]+$/, '.jpg'), { type: 'image/jpeg' }))
   486→              } else {
   487→                quality -= 0.15
   488→                compress()
   489→              }
   490→            }, 'image/jpeg', quality)
   491→          }
   492→          compress()
   493→        }
   494→        img.onerror = () => reject(new Error('Failed to load image'))
   495→        img.src = e.target?.result as string
   496→      }
   497→      reader.onerror = () => reject(new Error('Failed to read file'))
   498→      reader.readAsDataURL(file)
   499→    })
   500→  }
