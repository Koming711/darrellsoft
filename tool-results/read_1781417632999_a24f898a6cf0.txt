     1→'use client'
     2→
     3→import { useState, useEffect, useRef } from 'react'
     4→import { History, Search, Filter, RotateCcw, Eye, Trash2, Printer, FileImage, Loader2, FileText, Calculator, Layers, Package, Truck, Percent, Scissors, Cog, Banknote } from 'lucide-react'
     5→import { useRouter } from 'next/navigation'
     6→import { MobileTable } from '@/components/mobile-table'
     7→import { useLanguage } from '@/contexts/language-context'
     8→import { Dialog, DialogContent, DialogHeader, DialogTitle } from '@/components/ui/dialog'
     9→import { toast } from 'sonner'
    10→import { getAuthHeaders } from '@/lib/auth'
    11→import { authFetch } from '@/lib/auth-fetch'
    12→import { notifyDataChange } from '@/lib/data-sync'
    13→import { useDataChange } from '@/hooks/use-data-change'
    14→
    15→interface RiwayatItem {
    16→  id: string
    17→  nomorUrut: string
    18→  type: string
    19→  printName: string
    20→  customerName: string
    21→  paperName: string
    22→  paperGrammage: string
    23→  paperLength: string
    24→  paperWidth: string
    25→  cutWidth: string
    26→  cutHeight: string
    27→  quantity: string
    28→  warna: string
    29→  warnaKhusus: string
    30→  machineName: string
    31→  hargaPlat: number
    32→  ongkosCetak: number
    33→  ongkosCetakDetail: string
    34→  machineName2: string
    35→  ongkosCetak2: number
    36→  ongkosCetak2Detail: string
    37→  totalPaperPrice: number
    38→  pricePerSheet: number
    39→  finishingNames: string
    40→  finishingBreakdown: string
    41→  finishingCost: number
    42→  packingCost: number
    43→  shippingCost: number
    44→  glueCost: number
    45→  glueBorongan: number
    46→  otherCost: number
    47→  subTotal: number
    48→  profitPercent: number
    49→  profitAmount: number
    50→  grandTotal: number
    51→  createdAt: string
    52→  updatedAt: string
    53→}
    54→
    55→interface RiwayatContentProps {
    56→  title: string
    57→  subtitle: string
    58→  defaultFilterType: 'all' | 'Hitung Cetakan' | 'Potong Kertas'
    59→}
    60→
    61→export function RiwayatContent({ title, subtitle, defaultFilterType }: RiwayatContentProps) {
    62→  const { t } = useLanguage()
    63→  const router = useRouter()
    64→  const [searchTerm, setSearchTerm] = useState('')
    65→  const [filterType, setFilterType] = useState(defaultFilterType)
    66→  const [histories, setHistories] = useState<RiwayatItem[]>([])
    67→  const [loading, setLoading] = useState(true)
    68→
    69→  // Preview
    70→  const [previewOpen, setPreviewOpen] = useState(false)
    71→  const [previewItem, setPreviewItem] = useState<RiwayatItem | null>(null)
    72→  const [isGeneratingPdf, setIsGeneratingPdf] = useState(false)
    73→  const previewRef = useRef<HTMLDivElement>(null)
    74→
    75→  useEffect(() => {
    76→    fetchRiwayat()
    77→  }, [])
    78→
    79→  useDataChange(['riwayat-cetakan', 'riwayat-potong-kertas'], () => {
    80→    fetchRiwayat()
    81→  })
    82→
    83→  const fetchRiwayat = async () => {
    84→    setLoading(true)
    85→    try {
    86→      const res = await authFetch('/api/riwayat-cetakan', { headers: getAuthHeaders() })
    87→      if (res.ok) {
    88→        const data = await res.json()
    89→        setHistories(Array.isArray(data) ? data : [])
    90→      }
    91→    } catch {
    92→      toast.error('Gagal memuat riwayat')
    93→    } finally {
    94→      setLoading(false)
    95→    }
    96→  }
    97→
    98→  const handleRestore = (item: RiwayatItem) => {
    99→    const isHitungCetak = item.type === 'hitung_cetakan'
   100→    const params = new URLSearchParams()
   101→    // Field umum
   102→    if (item.printName) params.set('printName', item.printName)
   103→    if (item.customerName) params.set('customerName', item.customerName)
   104→    if (item.paperName) params.set('paperName', item.paperName)
   105→    if (item.paperLength) params.set('paperLength', item.paperLength)
   106→    if (item.paperWidth) params.set('paperWidth', item.paperWidth)
   107→    if (item.cutWidth) params.set('cutWidth', item.cutWidth)
   108→    if (item.cutHeight) params.set('cutHeight', item.cutHeight)
   109→    if (item.quantity) params.set('quantity', item.quantity)
   110→    if (item.totalPaperPrice) params.set('totalPaperPrice', item.totalPaperPrice.toString())
   111→    if (item.profitPercent) params.set('profitPercent', item.profitPercent.toString())
   112→    params.set('restoredFromRiwayat', '1')
   113→
   114→    if (isHitungCetak) {
   115→      // Field khusus hitung cetakan
   116→      if (item.paperGrammage && item.paperGrammage !== '0') params.set('paperGrammage', item.paperGrammage)
   117→      if (item.pricePerSheet) params.set('pricePerSheet', item.pricePerSheet.toString())
   118→      if (item.warna && item.warna !== '-') params.set('warna', item.warna)
   119→      if (item.warnaKhusus && item.warnaKhusus !== '-' && parseInt(item.warnaKhusus) > 0) params.set('warnaKhusus', item.warnaKhusus)
   120→      if (item.hargaPlat) params.set('hargaPlat', item.hargaPlat.toString())
   121→      if (item.machineName && item.machineName !== '-') params.set('machineName', item.machineName)
   122→      if (item.machineName2 && item.machineName2 !== '-') params.set('machineName2', item.machineName2)
   123→      if (item.finishingNames && item.finishingNames !== '-') params.set('finishingNames', item.finishingNames)
   124→      if (item.packingCost) params.set('packingCost', item.packingCost.toString())
   125→      if (item.shippingCost) params.set('shippingCost', item.shippingCost.toString())
   126→      if (item.glueCost) params.set('glueCost', item.glueCost.toString())
   127→      if (item.glueBorongan) params.set('glueBorongan', item.glueBorongan.toString())
   128→      if (item.otherCost) params.set('otherCost', item.otherCost.toString())
   129→      window.location.href = `/hitung-cetakan?${params.toString()}`
   130→    } else {
   131→      window.location.href = `/potong-kertas?${params.toString()}`
   132→    }
   133→  }
   134→
   135→  const handlePreview = (item: RiwayatItem) => {
   136→    setPreviewItem(item)
   137→    setPreviewOpen(true)
   138→  }
   139→
   140→  const handleDelete = async (item: RiwayatItem) => {
   141→    if (!confirm('Beneran mau dihapus nih?')) return
   142→    try {
   143→      const res = await authFetch(`/api/riwayat-cetakan/${item.id}`, {
   144→        method: 'DELETE',
   145→        headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   146→      })
   147→      if (res.ok) {
   148→        toast.success('Riwayat berhasil dihapus')
   149→        // Optimistic update: remove from state immediately
   150→        setHistories(prev => prev.filter(h => h.id !== item.id))
   151→        notifyDataChange('riwayat-cetakan')
   152→      } else {
   153→        toast.error('Gagal menghapus riwayat')
   154→      }
   155→    } catch {
   156→      toast.error('Gagal menghapus riwayat')
   157→    }
   158→  }
   159→
   160→  const handlePrint = () => {
   161→    const el = previewRef.current
   162→    if (!el || !previewItem) return
   163→    const pw = window.open('', '_blank')
   164→    if (!pw) { toast.error('Popup diblokir'); return }
   165→    pw.document.write(`<!DOCTYPE html><html><head><meta charset="utf-8"><title>Preview - ${previewItem.printName}</title>
   166→      <style>
   167→        * { margin: 0; padding: 0; box-sizing: border-box; }
   168→        @page { size: A4; margin: 10mm; }
   169→        body { font-family: 'Segoe UI', Arial, sans-serif; color: #1e293b; }
   170→        .header { text-align: center; margin-bottom: 12px; padding-bottom: 8px; border-bottom: 2px solid #e2e8f0; }
   171→        .header h1 { font-size: 16px; font-weight: 700; color: #0f172a; }
   172→        .header p { font-size: 10px; color: #64748b; margin-top: 2px; }
   173→        .grid { display: grid; grid-template-columns: 1fr 1fr; gap: 6px; margin-bottom: 10px; }
   174→        .cell { background: #f8fafc; border: 1px solid #e2e8f0; border-radius: 5px; padding: 6px 8px; }
   175→        .cell .lbl { font-size: 8px; color: #64748b; font-weight: 500; }
   176→        .cell .val { font-size: 12px; font-weight: 700; color: #0f172a; }
   177→        .cell .val.grn { color: #059669; }
   178→        .cell .val.red { color: #e11d48; }
   179→        .row { display: flex; gap: 6px; margin-bottom: 6px; }
   180→        .row .full { flex: 1; }
   181→        .breakdown { background: #fef2f2; border: 1px solid #fecaca; border-radius: 5px; padding: 6px 8px; margin-bottom: 8px; }
   182→        .breakdown-title { font-size: 9px; font-weight: 700; color: #991b1b; margin-bottom: 3px; }
   183→        .breakdown-text { font-size: 9px; color: #7f1d1d; white-space: pre-line; }
   184→        .total-bar { background: #0f172a; color: white; border-radius: 6px; padding: 8px 12px; display: flex; justify-content: space-between; align-items: center; }
   185→        .total-bar .lbl { font-size: 9px; color: #94a3b8; }
   186→        .total-bar .val { font-size: 16px; font-weight: 800; color: #22c55e; }
   187→      </style>
   188→    </head><body>${el.innerHTML}</body></html>`)
   189→    pw.document.close()
   190→    pw.onload = () => pw.print()
   191→  }
   192→
   193→  const handlePdf = async () => {
   194→    const el = previewRef.current
   195→    if (!el) return
   196→    setIsGeneratingPdf(true)
   197→    try {
   198→      const { toCanvas } = await import('html-to-image')
   199→      const canvas = await toCanvas(el, { backgroundColor: '#ffffff', pixelRatio: 2 })
   200→      const imgData = canvas.toDataURL('image/jpeg', 0.95)
   201→      const { jsPDF } = await import('jspdf')
   202→      const pdf = new jsPDF('p', 'mm', 'a4')
   203→      const pdfW = pdf.internal.pageSize.getWidth()
   204→      const pdfH = pdf.internal.pageSize.getHeight()
   205→      const margin = 8
   206→      const cw = pdfW - margin * 2
   207→      const ih = (canvas.height * cw) / canvas.width
   208→      const maxH = pdfH - margin * 2
   209→      let fw = cw, fh = ih
   210→      if (fh > maxH) { fw = (maxH * cw) / ih; fh = maxH }
   211→      pdf.addImage(imgData, 'JPEG', margin + (cw - fw) / 2, margin, fw, fh)
   212→      pdf.save(`riwayat-${Date.now()}.pdf`)
   213→      toast.success('PDF berhasil diunduh!')
   214→    } catch { toast.error('Gagal menghasilkan PDF') }
   215→    finally { setIsGeneratingPdf(false) }
   216→  }
   217→
   218→  const filteredHistories = histories.filter(h => {
   219→    const term = searchTerm.toLowerCase()
   220→    const matchesSearch = h.printName.toLowerCase().includes(term) || h.customerName.toLowerCase().includes(term) || h.paperName.toLowerCase().includes(term) || h.machineName.toLowerCase().includes(term)
   221→    const isHitungCetak = h.type === 'hitung_cetakan'
   222→    const matchesFilter = filterType === 'all' || (filterType === 'Hitung Cetakan' && isHitungCetak) || (filterType === 'Potong Kertas' && !isHitungCetak)
   223→    return matchesSearch && matchesFilter
   224→  })
   225→
   226→  const formatDate = (d: string) => {
   227→    try {
   228→      return new Date(d).toLocaleDateString('id-ID', { day: 'numeric', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit' })
   229→    } catch { return d }
   230→  }
   231→
   232→  const formatRp = (n: number) => `Rp ${n.toLocaleString('id-ID')}`
   233→
   234→  const isItemHitungCetak = (h: RiwayatItem) => h.type === 'hitung_cetakan'
   235→
   236→  const columns = [
   237→    {
   238→      key: 'jenis',
   239→      title: 'Jenis',
   240→      render: (h: RiwayatItem) => {
   241→        const jenis = isItemHitungCetak(h) ? 'Hitung Cetakan' : 'Potong Kertas'
   242→        const isPotong = jenis === 'Potong Kertas'
   243→        return (
   244→          <span className={`inline-flex items-center px-2 py-0.5 rounded-full text-[10px] font-bold ${isPotong ? 'bg-violet-100 text-violet-700' : 'bg-blue-100 text-blue-700'}`}>
   245→            {isPotong ? '✂️' : '🖨️'} {jenis}
   246→          </span>
   247→        )
   248→      }
   249→    },
   250→    {
   251→      key: 'nomorUrut',
   252→      title: 'Nomor',
   253→      render: (h: RiwayatItem) => {
   254→        if (!h.nomorUrut) return <span className="text-slate-400">-</span>
   255→        const isHC = isItemHitungCetak(h)
   256→        return (
   257→          <span className={`font-semibold text-xs ${isHC ? 'text-blue-700' : 'text-teal-700'}`}>{h.nomorUrut}</span>
   258→        )
   259→      }
   260→    },
   261→    {
   262→      key: 'customerName',
   263→      title: 'Nama Customer',
   264→      render: (h: RiwayatItem) => (
   265→        <span className="text-slate-700 truncate">{h.customerName || '-'}</span>
   266→      )
   267→    },
   268→    {
   269→      key: 'printName',
   270→      title: 'Nama Cetakan',
   271→      render: (h: RiwayatItem) => (
   272→        <div className="flex items-center gap-2">
   273→          {isItemHitungCetak(h) ? (
   274→            <Calculator className="w-4 h-4 text-blue-600 flex-shrink-0" />
   275→          ) : (
   276→            <Scissors className="w-4 h-4 text-teal-600 flex-shrink-0" />
   277→          )}
   278→          <span className="font-medium text-slate-800 truncate">{h.printName}</span>
   279→        </div>
   280→      )
   281→    },
   282→    {
   283→      key: 'profitAmount',
   284→      title: 'Uang Capek',
   285→      render: (h: RiwayatItem) => (
   286→        <span className={`font-semibold ${h.profitAmount > 0 ? 'text-violet-700' : 'text-slate-400'}`}>
   287→          {h.profitAmount > 0 ? formatRp(h.profitAmount) : '-'}
   288→        </span>
   289→      )
   290→    },
   291→    {
   292→      key: 'quantity',
   293→      title: 'Jumlah',
   294→      render: (h: RiwayatItem) => `${parseInt(h.quantity || '0').toLocaleString()} lbr`
   295→    },
   296→    {
   297→      key: 'pricePerSheet',
   298→      title: 'Harga/Lembar',
   299→      render: (h: RiwayatItem) => {
   300→        const qty = parseInt(h.quantity || '0')
   301→        if (qty <= 0) return <span className="text-slate-400">-</span>
   302→        return <span className="font-semibold text-slate-700">{formatRp(Math.round(h.grandTotal / qty))}</span>
   303→      }
   304→    },
   305→    {
   306→      key: 'grandTotal',
   307→      title: 'Total Harga',
   308→      render: (h: RiwayatItem) => (
   309→        <span className="font-bold text-emerald-700">{formatRp(h.grandTotal)}</span>
   310→      )
   311→    },
   312→    {
   313→      key: 'createdAt',
   314→      title: 'Tanggal',
   315→      render: (h: RiwayatItem) => (
   316→        <span className="text-xs text-slate-500">{formatDate(h.createdAt)}</span>
   317→      )
   318→    }
   319→  ]
   320→
   321→  return (
   322→    <div className="space-y-4 lg:space-y-6">
   323→      {/* Filters */}
   324→      <div className="bg-white rounded-xl shadow-sm border border-slate-200 p-4 lg:p-6">
   325→        <div className="flex flex-col lg:flex-row gap-4 items-start lg:items-center justify-between">
   326→          <div className="flex flex-col lg:flex-row gap-4 w-full lg:flex-1">
   327→            <div className="relative flex-1">
   328→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
   329→              <input type="text" placeholder="Cari riwayat..." value={searchTerm} onChange={(e) => setSearchTerm(e.target.value)}
   330→                className="w-full pl-9 pr-4 py-2.5 border border-slate-300 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-blue-500" />
   331→            </div>
   332→            <div className="relative">
   333→              <Filter className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
   334→              <select value={filterType} onChange={(e) => setFilterType(e.target.value)}
   335→                className="pl-9 pr-4 py-2.5 border border-slate-300 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-blue-500 appearance-none bg-white w-full lg:w-auto">
   336→                <option value="all">Semua Tipe</option>
   337→                <option value="Hitung Cetakan">Hitung Cetakan</option>
   338→                <option value="Potong Kertas">Potong Kertas</option>
   339→              </select>
   340→            </div>
   341→          </div>
   342→          <div className="text-xs text-slate-400">{filteredHistories.length} data</div>
   343→        </div>
   344→      </div>
   345→
   346→      {/* Table */}
   347→      <div className="bg-white rounded-xl shadow-sm border border-slate-200 p-4 lg:p-6">
   348→        {loading ? (
   349→          <div className="flex items-center justify-center py-12">
   350→            <Loader2 className="w-6 h-6 animate-spin text-blue-600 mr-2" />
   351→            <span className="text-sm text-slate-500">Memuat riwayat...</span>
   352→          </div>
   353→        ) : (
   354→          <MobileTable
   355→            data={filteredHistories}
   356→            columns={columns}
   357→            keyField="id"
   358→            onDelete={handleDelete}
   359→            showAsButtons={true}
   360→            emptyMessage="Belum ada riwayat perhitungan"
   361→            emptyIcon={<History className="w-12 h-12 mx-auto text-slate-400" />}
   362→            extraActions={(item: RiwayatItem) => (
   363→              <div className="flex items-center gap-1">
   364→                <button onClick={() => handlePreview(item)} title={t('preview')}
   365→                  className="p-1.5 rounded-lg bg-violet-100 hover:bg-violet-200 text-violet-700 transition-colors">
   366→                  <Eye className="w-3.5 h-3.5" />
   367→                </button>
   368→                <button onClick={() => handleRestore(item)} title={t('restore_ke_hitung')}
   369→                  className="p-1.5 rounded-lg bg-emerald-100 hover:bg-emerald-200 text-emerald-700 transition-colors">
   370→                  <RotateCcw className="w-3.5 h-3.5" />
   371→                </button>
   372→              </div>
   373→            )}
   374→            mobileCardActions={(item: RiwayatItem) => (
   375→              <div className="flex items-center gap-2 mt-3 pt-3 border-t border-slate-100">
   376→                <button onClick={() => handlePreview(item)}
   377→                  className="flex-1 flex items-center justify-center gap-1.5 py-2 rounded-lg bg-violet-600 hover:bg-violet-700 text-white text-xs font-medium transition-colors">
   378→                  <Eye className="w-3.5 h-3.5" /> Preview
   379→                </button>
   380→                <button onClick={() => handleRestore(item)}
   381→                  className="flex-1 flex items-center justify-center gap-1.5 py-2 rounded-lg bg-emerald-600 hover:bg-emerald-700 text-white text-xs font-medium transition-colors">
   382→                  <RotateCcw className="w-3.5 h-3.5" /> Restore
   383→                </button>
   384→                <button onClick={() => handleDelete(item)}
   385→                  className="py-2 px-3 rounded-lg bg-red-100 hover:bg-red-200 text-red-600 transition-colors">
   386→                  <Trash2 className="w-3.5 h-3.5" />
   387→                </button>
   388→              </div>
   389→            )}
   390→          />
   391→        )}
   392→      </div>
   393→
   394→      {/* ===== PREVIEW DIALOG ===== */}
   395→      <Dialog open={previewOpen} onOpenChange={setPreviewOpen}>
   396→        <DialogContent className="max-w-2xl max-h-[92vh] overflow-y-auto p-0">
   397→          <DialogHeader className="p-4 pb-0">
   398→            <DialogTitle className="flex items-center gap-2">
   399→              <Eye className="w-5 h-5 text-violet-600" />
   400→              Detail Riwayat Cetakan
   401→            </DialogTitle>
   402→          </DialogHeader>
   403→
   404→          {previewItem && (
   405→            <>
   406→              <div ref={previewRef} className="p-4 bg-white space-y-3">
   407→                {/* Header */}
   408→                <div className="text-center pb-3 border-b-2 border-slate-200">
   409→                  <div className="flex items-center justify-center gap-2 mb-1">
   410→                    {isItemHitungCetak(previewItem) ? (
   411→                      <Calculator className="w-5 h-5 text-blue-600" />
   412→                    ) : (
   413→                      <Scissors className="w-5 h-5 text-teal-600" />
   414→                    )}
   415→                    <h1 className="text-lg font-bold text-slate-900">
   416→                      {isItemHitungCetak(previewItem) ? 'Rincian Harga Cetakan' : 'Rincian Potong Kertas'}
   417→                    </h1>
   418→                  </div>
   419→                  <p className="text-xs text-slate-500 mt-1">
   420→                    {previewItem.printName} · {formatDate(previewItem.createdAt)}
   421→                  </p>
   422→                </div>
   423→
   424→                {/* === INFORMASI CETAKAN === */}
   425→                <div>
   426→                  <div className="flex items-center gap-1.5 mb-2">
   427→                    <div className="w-5 h-5 rounded bg-blue-100 flex items-center justify-center">
   428→                      <FileText className="w-3 h-3 text-blue-600" />
   429→                    </div>
   430→                    <p className="text-xs font-bold text-slate-700 uppercase tracking-wide">Informasi Cetakan</p>
   431→                  </div>
   432→                  <div className="grid grid-cols-2 gap-2">
   433→                    <div className="bg-blue-50 border border-blue-100 rounded-lg p-2.5">
   434→                      <p className="text-[10px] text-blue-500 font-medium">Nama Customer</p>
   435→                      <p className="text-sm font-bold text-blue-800">{previewItem.customerName || '-'}</p>
   436→                    </div>
   437→                    <div className="bg-indigo-50 border border-indigo-100 rounded-lg p-2.5">
   438→                      <p className="text-[10px] text-indigo-500 font-medium">Nama Cetakan</p>
   439→                      <p className="text-sm font-bold text-indigo-800">{previewItem.printName || '-'}</p>
