    30→/** Tanggal hari ini pada pukul 00:00 (untuk deteksi jatuh tempo). */
    31→function startOfToday(): number {
    32→  const n = new Date()
    33→  return new Date(n.getFullYear(), n.getMonth(), n.getDate()).getTime()
    34→}
    35→
    36→/**
    37→ * Badge status invoice. Warna (spesifikasi user):
    38→ * Lunas → hijau, Belum Lunas → oranye, Jatuh Tempo → merah, Batal → abu.
    39→ * `dueDate` opsional: BELUM_BAYAR + jatuh tempo terlewat → tampil "Jatuh Tempo" (merah).
    40→ */
    41→function StatusBadge({ status, dueDate }: { status: 'BELUM_BAYAR' | 'LUNAS' | 'BATAL'; dueDate?: string | null }) {
    42→  if (status === 'BELUM_BAYAR' && dueDate && new Date(dueDate).getTime() < startOfToday()) {
    43→    return (
    44→      <Badge variant="outline" className="bg-red-50 text-red-700 border-red-200 text-[11px] shrink-0">
    45→        Jatuh Tempo
    46→      </Badge>
    47→    )
    48→  }
    49→  const map = {
    50→    BELUM_BAYAR: 'bg-amber-50 text-amber-700 border-amber-200',
    51→    LUNAS: 'bg-emerald-50 text-emerald-700 border-emerald-200',
    52→    BATAL: 'bg-stone-100 text-stone-500 border-stone-200',
    53→  } as const
    54→  return <Badge variant="outline" className={`${map[status]} text-[11px] shrink-0`}>{STATUS_LABEL[status]}</Badge>
    55→}
    56→
    57→export { StatusBadge }

    73→/** Format ringkas utk sumbu grafik: 850rb / 1,2jt / 2M */
    74→function compactIDR(v: number): string {
    75→  const trim = (n: number) => {
    76→    const s = n.toFixed(1)
    77→    return s.endsWith('.0') ? s.slice(0, -2) : s
    78→  }
    79→  if (v >= 1_000_000_000) return `${trim(v / 1_000_000_000)}M`
    80→  if (v >= 1_000_000) return `${trim(v / 1_000_000)}jt`
    81→  if (v >= 1_000) return `${Math.round(v / 1_000)}rb`
    82→  return `${Math.round(v)}`
    83→}

    99→      <p className="text-sm font-semibold text-stone-900">{formatIDR(Number(payload[0]?.value ?? 0))}</p>

   101→  )
   102→}
