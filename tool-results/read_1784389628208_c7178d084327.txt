    64→// ===== Date filter helpers (same scheme as rekap-penjualan) =====
    65→type FilterType = 'all' | 'today' | 'week' | 'month' | 'custom'
    66→
    67→function getFilterDates(
    68→  filter: FilterType,
    69→  customStart?: Date,
    70→  customEnd?: Date
    71→): { startDate: string; endDate: string } {
    72→  const today = new Date()
    73→  const fmt = (d: Date) => {
    74→    const y = d.getFullYear()
    75→    const m = String(d.getMonth() + 1).padStart(2, '0')
    76→    const day = String(d.getDate()).padStart(2, '0')
    77→    return `${y}-${m}-${day}`
    78→  }
    79→  switch (filter) {
    80→    case 'all':
    81→      return { startDate: '2000-01-01', endDate: '2099-12-31' }
    82→    case 'today':
    83→      return { startDate: fmt(today), endDate: fmt(today) }
    84→    case 'week': {
    85→      const s = new Date(today)
    86→      const day = s.getDay()
    87→      s.setDate(s.getDate() + (day === 0 ? -6 : 1 - day))
    88→      return { startDate: fmt(s), endDate: fmt(today) }
    89→    }
    90→    case 'month': {
    91→      const s = new Date(today.getFullYear(), today.getMonth(), 1)
    92→      return { startDate: fmt(s), endDate: fmt(today) }
    93→    }
    94→    case 'custom':
    95→      return {
    96→        startDate: customStart ? fmt(customStart) : fmt(today),
    97→        endDate: customEnd ? fmt(customEnd) : fmt(today),
    98→      }
    99→  }
   100→}
   101→
   102→function formatDateShort(iso: string): string {
   103→  if (!iso) return '-'
   104→  try {
   105→    return new Date(iso).toLocaleDateString('id-ID', {
   106→      day: '2-digit',
   107→      month: 'short',
   108→      year: 'numeric',
   109→    })
   110→  } catch {
   111→    return iso
   112→  }
   113→}
   114→
   115→function formatDateShortId(dateStr: string): string {
   116→  if (!dateStr) return '-'
   117→  try {
   118→    return new Date(dateStr + 'T00:00:00').toLocaleDateString('id-ID', {
   119→      day: '2-digit',
   120→      month: 'short',
   121→      year: 'numeric',
   122→    })
   123→  } catch {
   124→    return dateStr
   125→  }
   126→}
   127→
   128→function formatRupiahShort(n: number): string {
   129→  return new Intl.NumberFormat('id-ID', {
   130→    style: 'currency',
   131→    currency: 'IDR',
   132→    minimumFractionDigits: 0,
   133→  }).format(n)
   134→}

   142→  const [data, setData] = useState<DpResponse | null>(null)

   152→  const [customEndStr, setCustomEndStr] = useState('')
   153→

   156→  const [searchTerm, setSearchTerm] = useState('')
   157→  const [sortKey, setSortKey] = useState<SortKey>('terbaru')
   158→
   159→  const fetchData = useCallback(async () => {
   160→    setLoading(true)
   161→    setError(null)
   162→    try {
   163→      const { startDate, endDate } = getFilterDates(
   164→        filterType,
   165→        customStartDate,
   166→        customEndDate
   167→      )
   168→      const res = await authFetch(
   169→        `/api/laporan-invoice-dp?startDate=${startDate}&endDate=${endDate}&status=${statusFilter}`
   170→      )
   171→      if (res.ok) {
   172→        const json = (await res.json()) as DpResponse
   173→        setData(json)
   174→      } else {
   175→        setError('Gagal memuat data laporan invoice DP')
   176→      }
   177→    } catch (err) {
   178→      console.error('[LaporanInvoiceDP] fetchData error:', err)
   179→      setError('Gagal memuat data laporan invoice DP')
   180→    } finally {
   181→      setLoading(false)
   182→    }
   183→  }, [filterType, customStartDate, customEndDate, statusFilter])
   184→
   185→  useEffect(() => {
   186→    fetchData()
   187→  }, [fetchData])
   188→
   189→  const handleFilterChange = (type: FilterType) => {
   190→    if (type === 'custom') {
   191→      setShowCustomDialog(true)
   192→    } else {
   193→      setFilterType(type)
   194→    }
   195→  }
   196→
   197→  const applyCustomFilter = () => {
   198→    if (customStartStr && customEndStr) {
   199→      setCustomStartDate(new Date(customStartStr))
   200→      setCustomEndDate(new Date(customEndStr))
   201→      setFilterType('custom')
   202→      setShowCustomDialog(false)
   203→    }
   204→  }
   205→
   206→  // Apply search + sort on the invoice list
   207→  const processedInvoices = useMemo<DpInvoice[]>(() => {
   208→    if (!data) return []
   209→    const term = searchTerm.trim().toLowerCase()
   210→    let list = data.invoices
   211→    if (term) {
   212→      list = list.filter(
   213→        (i) =>
   214→          i.customerName.toLowerCase().includes(term) ||
   215→          i.nomor.toLowerCase().includes(term)
   216→      )
   217→    }
   218→    const sorted = [...list]
   219→    switch (sortKey) {
   220→      case 'terbaru':
   221→        sorted.sort(
   222→          (a, b) =>
   223→            new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()
   224→        )
   225→        break
   226→      case 'jatuh_tempo':
   227→        // overdue first, then nearest due date, then no due date
   228→        sorted.sort((a, b) => {
   229→          if (a.isOverdue !== b.isOverdue) return a.isOverdue ? -1 : 1
   230→          if (!a.tanggalJatuhTempo) return 1
   231→          if (!b.tanggalJatuhTempo) return -1
   232→          return (
   233→            new Date(a.tanggalJatuhTempo).getTime() -
   234→            new Date(b.tanggalJatuhTempo).getTime()
   235→          )
   236→        })
   237→        break
   238→      case 'sisa_desc':
   239→        sorted.sort((a, b) => b.sisa - a.sisa)
   240→        break
   241→      case 'dp_desc':
   242→        sorted.sort((a, b) => b.dpAmount - a.dpAmount)
   243→        break
   244→      case 'nama_asc':
   245→        sorted.sort((a, b) => a.customerName.localeCompare(b.customerName, 'id'))
   246→        break
   247→    }
   248→    return sorted

   252→

   255→    { type: 'today', label: t('today') },
   256→    { type: 'week', label: t('this_week') },
   257→    { type: 'month', label: t('this_month') },
   258→    { type: 'custom', label: t('custom') },
   259→  ]
   260→
   261→  const statusButtons: { type: StatusFilter; label: string }[] = [
   262→    { type: 'all', label: t('laporan_dp_status_all') },
   263→    { type: 'belum_lunas', label: t('laporan_dp_status_belum_lunas') },
   264→    { type: 'lunas', label: t('laporan_dp_status_lunas') },
   265→    { type: 'jatuh_tempo', label: t('laporan_dp_status_jatuh_tempo') },
   266→  ]
   267→

   269→  const handleExportExcel = () => {
   270→    if (!data || processedInvoices.length === 0) return

   272→    rows.push([
   273→      'No',
   274→      'No. Invoice',
   275→      'Tanggal',
   276→      'Customer',
   277→      'Total Invoice',
   278→      'DP (%)',
   279→      'Nilai DP',
   280→      'Sisa Piutang',
   281→      'Status',
   282→      'Jatuh Tempo',
   283→      'Tanggal Pelunasan',
   284→      'Cara Pembayaran',
   285→    ])

   287→      const statusLabel = inv.lunas
   288→        ? 'Lunas'
   289→        : inv.isOverdue
   290→          ? 'Jatuh Tempo'
   291→          : 'Belum Lunas'
   292→      rows.push([
   293→        String(i + 1),
   294→        inv.nomor,
   295→        inv.tanggal || formatDateShort(inv.createdAt),
   296→        inv.customerName,
   297→        String(Math.round(inv.totalHarga)),
   298→        String(inv.dpPercent) + '%',
   299→        String(Math.round(inv.dpAmount)),
   300→        String(Math.round(inv.sisa)),
   301→        statusLabel,
   302→        inv.tanggalJatuhTempo || '-',
   303→        inv.tanggalPelunasan || '-',
   304→        inv.caraPembayaran || '-',
   305→      ])
   306→    })
   307→    // Summary row
   308→    if (summary) {
   309→      rows.push([])
   310→      rows.push([
   311→        '',
   312→        'TOTAL',
   313→        '',
   314→        '',
   315→        String(Math.round(summary.totalNilai)),
   316→        '',
   317→        String(Math.round(summary.totalDPMasuk)),
   318→        String(Math.round(summary.totalSisaPiutang)),
   319→        `${summary.totalLunas} Lunas / ${summary.totalBelumLunas} Belum`,
   320→        '',
   321→        '',
   322→        '',
   323→      ])
   324→    }
   325→    // Convert to CSV (Excel-friendly: semicolon separator for ID locale)
   326→    const csv = rows
   327→      .map((r) =>
   328→        r.map((cell) => `"${(cell || '').replace(/"/g, '""')}"`).join(';')
   329→      )
   330→      .join('\r\n')
   331→    // Prepend BOM so Excel detects UTF-8
   332→    const blob = new Blob(['\uFEFF' + csv], { type: 'text/csv;charset=utf-8;' })
   333→    const url = URL.createObjectURL(blob)
   334→    const a = document.createElement('a')
   335→    a.href = url
   336→    const todayStr = new Date().toISOString().slice(0, 10)
   337→    a.download = `Laporan_Invoice_DP_${todayStr}.csv`
   338→    document.body.appendChild(a)
   339→    a.click()
   340→    document.body.removeChild(a)
   341→    URL.revokeObjectURL(url)
   342→  }
   343→
   344→  // ===== Print =====
   345→  const handlePrint = () => {
   346→    window.print()
   347→  }
   348→
   349→  return (
   350→    <DashboardLayout
   351→      title={t('laporan_invoice_dp')}
   352→      subtitle={t('subtitle_laporan_invoice_dp')}
   353→    >
   354→      <div className="space-y-4 sm:space-y-6 pb-6">
   355→        {/* ===== Summary Cards ===== */}
   356→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-3 print:hidden">
   357→          {/* Total Invoice DP */}
   358→          <div className="bg-white dark:bg-zinc-900 border border-slate-200 dark:border-zinc-700 rounded-xl p-3 sm:p-4">
   359→            <div className="w-8 h-8 rounded-lg bg-blue-100 text-blue-600 dark:bg-blue-950 dark:text-blue-400 flex items-center justify-center mb-2">
   360→              <Receipt className="w-5 h-5" />
   361→            </div>
   362→            <p className="text-xs text-slate-500 dark:text-slate-400 mb-0.5">
   363→              {t('laporan_dp_total_invoice')}
   364→            </p>
   365→            {loading ? (
   366→              <div className="h-6 w-16 bg-slate-200 dark:bg-zinc-700 rounded animate-pulse" />
   367→            ) : (
   368→              <p className="text-base sm:text-lg font-bold text-blue-700 dark:text-blue-400 leading-tight">
   369→                {summary?.totalInvoiceDP ?? 0}
   370→              </p>
   371→            )}
   372→          </div>
   373→
   374→          {/* Total DP Masuk */}
   375→          <div className="bg-white dark:bg-zinc-900 border border-slate-200 dark:border-zinc-700 rounded-xl p-3 sm:p-4">
   376→            <div className="w-8 h-8 rounded-lg bg-emerald-100 text-emerald-600 dark:bg-emerald-950 dark:text-emerald-400 flex items-center justify-center mb-2">
   377→              <TrendingUp className="w-5 h-5" />
   378→            </div>
   379→            <p className="text-xs text-slate-500 dark:text-slate-400 mb-0.5">
   380→              {t('laporan_dp_total_dp')}
   381→            </p>
   382→            {loading ? (
   383→              <div className="h-6 w-24 bg-slate-200 dark:bg-zinc-700 rounded animate-pulse" />
   384→            ) : (
   385→              <p className="text-base sm:text-lg font-bold text-emerald-700 dark:text-emerald-400 leading-tight">
   386→                {formatRupiahShort(summary?.totalDPMasuk ?? 0)}
   387→              </p>
   388→            )}
   389→          </div>
   390→
   391→          {/* Sisa Piutang */}
   392→          <div className="bg-white dark:bg-zinc-900 border border-slate-200 dark:border-zinc-700 rounded-xl p-3 sm:p-4">
   393→            <div className="w-8 h-8 rounded-lg bg-amber-100 text-amber-600 dark:bg-amber-950 dark:text-amber-400 flex items-center justify-center mb-2">
   394→              <Wallet className="w-5 h-5" />
   395→            </div>
   396→            <p className="text-xs text-slate-500 dark:text-slate-400 mb-0.5">
   397→              {t('laporan_dp_total_sisa')}
   398→            </p>
   399→            {loading ? (
   400→              <div className="h-6 w-24 bg-slate-200 dark:bg-zinc-700 rounded animate-pulse" />
   401→            ) : (
   402→              <p className="text-base sm:text-lg font-bold text-amber-700 dark:text-amber-400 leading-tight">
