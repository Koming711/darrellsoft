    52→function errText(e: unknown): string {
    53→  return e instanceof Error ? e.message : 'Terjadi kesalahan'
    54→}
    55→
    56→interface CustomerOption {
    57→  id: string
    58→  name: string
    59→  companyName: string | null
    60→}
    61→
    62→interface ItemFormState {
    63→  name: string
    64→  unit: string
    65→  standardPrice: string
    66→  hpp: string
    67→  /** Bidang praktis: saat diisi → Harga Jual otomatis = Harga Modal + Profit */
    68→  profit: string
    69→  keterangan: string
    70→  isActive: boolean
    71→}
    72→
    73→const EMPTY_FORM: ItemFormState = {
    74→  name: '', unit: 'pcs', standardPrice: '', hpp: '', profit: '', keterangan: '', isActive: true,
    75→}
    76→
    77→function toNum(v: string): number | null {
    78→  if (v.trim() === '') return null
    79→  const n = Number(v)
    80→  return Number.isFinite(n) ? n : null
    81→}
    82→
    83→function profitOf(jual: string, modal: string): string {
    84→  const j = toNum(jual)
    85→  const m = toNum(modal)
    86→  return j !== null && m !== null ? String(j - m) : ''
    87→}
    88→
    89→function ActiveBadge({ active }: { active: boolean }) {
    90→  return active
    91→    ? <Badge variant="outline" className="bg-emerald-50 text-emerald-700 border-emerald-200 text-[11px] shrink-0">Aktif</Badge>
    92→    : <Badge variant="outline" className="bg-stone-100 text-stone-500 border-stone-200 text-[11px] shrink-0">Nonaktif</Badge>
    93→}
    94→
    95→interface ItemsViewProps {
    96→  user: SessionUser
    97→  /** Izin granular Matriks Hak Akses (master-barang-tambah/edit/hapus); undefined = fallback role lama */
    98→  canAdd?: boolean
    99→  canEdit?: boolean
   100→  canDelete?: boolean
   101→}
   102→
   103→export default function ItemsView({ user, canAdd: canAddProp, canEdit: canEditProp, canDelete: canDeleteProp }: ItemsViewProps) {
   104→  const canManage = user.role !== 'KASIR'
   105→  const canAdd = canAddProp ?? canManage
   106→  const canEditItem = canEditProp ?? canManage
   107→  const canDelete = canDeleteProp ?? canManage
   108→  // Harga Modal & Profit tampil untuk SEMUA role (permintaan owner — sebelumnya disembunyikan untuk KASIR)
   109→  const showHpp = true
   110→
   111→  const [items, setItems] = useState<Item[]>([])
   112→  const [loading, setLoading] = useState(true)
   113→  const [searchInput, setSearchInput] = useState('')
   114→  const [query, setQuery] = useState('')
   115→
   116→  // Pilih Customer — filter daftar barang per pelanggan (barang terdaftar)
   117→  const [customers, setCustomers] = useState<CustomerOption[]>([])
   118→  const [customerId, setCustomerId] = useState<string>('all')
   119→
   120→  const [dialogOpen, setDialogOpen] = useState(false)
   121→  const [editing, setEditing] = useState<Item | null>(null)
   122→  const [form, setForm] = useState<ItemFormState>(EMPTY_FORM)
   123→  const [saving, setSaving] = useState(false)
   124→
   125→  const [deleteTarget, setDeleteTarget] = useState<Item | null>(null)
   126→  const [deleting, setDeleting] = useState(false)
   127→
   128→  useEffect(() => {
   129→    const t = setTimeout(() => setQuery(searchInput.trim()), 300)
   130→    return () => clearTimeout(t)
   131→  }, [searchInput])
   132→
   133→  useEffect(() => {
   134→    apiFetch<CustomerOption[]>('/api/customers')
   135→      .then((data) => setCustomers(Array.isArray(data) ? data : []))
   136→      .catch(() => setCustomers([]))
   137→  }, [])
   138→
   139→  const load = useCallback(async () => {
   140→    setLoading(true)
   141→    try {
   142→      // active=all agar item nonaktif tetap terlihat (bisa diaktifkan kembali)
   143→      const cs = customerId !== 'all' ? `&customerId=${encodeURIComponent(customerId)}` : ''
   144→      const data = await apiFetch<{ items: Item[] }>(
   145→        `/api/items?q=${encodeURIComponent(query)}&active=all${cs}`
   146→      )
   147→      setItems(data.items)
   148→    } catch (e) {
   149→      toast.error(errText(e))
   150→    } finally {
   151→      setLoading(false)
   152→    }
   153→  }, [query, customerId])
   154→
   155→  useEffect(() => {
   156→    void load()
   157→  }, [load])
   158→
   159→  const openCreate = () => {
   160→    setEditing(null)
   161→    setForm(EMPTY_FORM)
   162→    setDialogOpen(true)
   163→  }
   164→
   165→  const openEdit = (it: Item) => {
   166→    setEditing(it)
   167→    const modalStr = it.hpp != null ? String(it.hpp) : ''
   168→    setForm({
   169→      name: it.name,
   170→      unit: it.unit,
   171→      standardPrice: String(it.standardPrice),
   172→      hpp: modalStr,
   173→      profit: profitOf(String(it.standardPrice), modalStr),
   174→      keterangan: it.keterangan ?? '',
   175→      isActive: it.isActive,
   176→    })
   177→    setDialogOpen(true)
   178→  }
   179→
   180→  // Profit = harga jual − harga modal; Margin = profit / harga jual × 100
   181→  const marginInfo = (() => {
   182→    const std = Number(form.standardPrice)
   183→    const hpp = Number(form.hpp)
   184→    if (form.standardPrice === '' || form.hpp === '' || !Number.isFinite(std) || !Number.isFinite(hpp) || std <= 0) {
   185→      return null
   186→    }
   187→    const profit = std - hpp
   188→    const margin = (profit / std) * 100
   189→    return { profit, margin, negative: margin < 0 }
   190→  })()
   191→
   192→  /* Sinkronisasi dua arah: Jual ↔ Profit dengan Modal sebagai dasar.
   193→   - Ubah Harga Jual → Profit = Jual − Modal
   194→   - Ubah Harga Modal → Profit mengikuti Jual; bila Jual kosong tapi Profit terisi → Jual = Modal + Profit
   195→   - Ubah Profit → Jual = Modal + Profit */
   196→  const setJual = (v: string) => {
   197→    setForm((f) => ({ ...f, standardPrice: v, profit: profitOf(v, f.hpp) }))
   198→  }
   199→  const setModal = (v: string) => {
   200→    setForm((f) => {
   201→      const jual = toNum(f.standardPrice)
   202→      const modal = toNum(v)
   203→      if (jual !== null && modal !== null) {
   204→        return { ...f, hpp: v, profit: String(jual - modal) }
   205→      }
   206→      const profit = toNum(f.profit)
   207→      if (profit !== null && modal !== null) {
   208→        return { ...f, hpp: v, standardPrice: String(modal + profit) }
   209→      }
   210→      return { ...f, hpp: v, profit: f.standardPrice !== '' || f.profit === '' ? '' : f.profit }
   211→    })
   212→  }
   213→  const setProfit = (v: string) => {
   214→    setForm((f) => {
   215→      const modal = toNum(f.hpp)
   216→      const profit = toNum(v)
   217→      if (modal !== null && profit !== null) {
   218→        return { ...f, profit: v, standardPrice: String(modal + profit) }
   219→      }
   220→      return { ...f, profit: v }
   221→    })
   222→  }
   223→
   224→  const handleSave = async () => {
   225→    const std = Number(form.standardPrice)
   226→    if (!form.name.trim()) {
   227→      toast.error('Nama barang wajib diisi')
   228→      return
   229→    }
   230→    if (form.standardPrice === '' || !Number.isFinite(std) || std < 0) {
   231→      toast.error('Harga jual wajib diisi (min 0)')
   232→      return
   233→    }
   234→    const hppNum = form.hpp === '' ? null : Number(form.hpp)
   235→    if (hppNum !== null && (!Number.isFinite(hppNum) || hppNum < 0)) {
   236→      toast.error('HPP tidak boleh negatif')
   237→      return
   238→    }
   239→    setSaving(true)
   240→    try {
   241→      const body = {
   242→        name: form.name.trim(),
   243→        unit: form.unit,
   244→        standardPrice: std,
   245→        hpp: hppNum,
   246→        keterangan: form.keterangan.trim(),
   247→        ...(editing ? { isActive: form.isActive } : { customerId: customerId !== 'all' ? customerId : undefined }),
   248→      }
   249→      const selectedCustomer = customers.find((c) => c.id === customerId)
   250→      if (editing) {
   251→        await apiFetch<{ item: Item }>(`/api/items/${editing.id}`, {
   252→          method: 'PUT',
   253→          body: JSON.stringify(body),
   254→        })
   255→        toast.success('Barang berhasil diperbarui')
   256→      } else {
   257→        await apiFetch<{ item: Item }>('/api/items', {
   258→          method: 'POST',
   259→          body: JSON.stringify(body),
   260→        })
   261→        toast.success(
   262→          selectedCustomer
   263→            ? `Barang berhasil ditambahkan untuk ${selectedCustomer.name}`
   264→            : 'Barang berhasil ditambahkan'
   265→        )
   266→      }
   267→      setDialogOpen(false)
   268→      void load()
   269→    } catch (e) {
   270→      toast.error(errText(e))
   271→    } finally {
   272→      setSaving(false)
   273→    }
   274→  }
   275→
   276→  const handleDelete = async () => {
   277→    if (!deleteTarget) return
   278→    setDeleting(true)
   279→    try {
   280→      await apiFetch<{ ok: boolean }>(`/api/items/${deleteTarget.id}`, { method: 'DELETE' })
   281→      toast.success('Barang berhasil dihapus')
   282→      setDeleteTarget(null)
   283→      void load()
   284→    } catch (e) {
   285→      // 409 → item dipakai di invoice
   286→      toast.error(errText(e))
   287→    } finally {
   288→      setDeleting(false)
   289→    }
   290→  }
   291→
   292→  const hasResults = items.length > 0
   293→  const selectedCustomerName = customers.find((c) => c.id === customerId)?.name ?? null
   294→  // Tambah/Edit hanya tampil saat pelanggan spesifik dipilih — bukan "Semua Barang"
   295→  const showTambah = canAdd && customerId !== 'all'
   296→  const showEdit = canEditItem && customerId !== 'all'
   297→  // Hapus SELALU tampil di tabel (selaras Master Pelanggan), termasuk mode "Semua Barang"
   298→  const showHapus = canDelete
   299→  const countLabel = loading
   300→    ? 'Memuat data…'
   301→    : selectedCustomerName
   302→      ? `${items.length} barang untuk ${selectedCustomerName}`
   303→      : `${items.length} barang`
   304→
   305→  return (
   306→    <div className="space-y-5">
   307→      {/* Header */}
   308→      <div className="flex flex-col gap-3 md:flex-row md:items-end md:justify-between">
   309→        <div>
   310→          <h1 className="text-xl md:text-2xl font-bold tracking-tight">Master Barang</h1>
   311→          <p className="text-sm text-muted-foreground mt-1">{countLabel}</p>
   312→        </div>
   313→        <div className="flex flex-col sm:flex-row sm:items-center gap-2">
   314→          <div className="relative flex-1 sm:w-64">
   315→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-stone-400 pointer-events-none" />
   316→            <Input
   317→              value={searchInput}
   318→              onChange={(e) => setSearchInput(e.target.value)}
   319→              placeholder="Cari nama / kode barang…"
   320→              aria-label="Cari barang"
   321→              className="pl-9 min-h-[44px]"
   322→            />
   323→          </div>
   324→          {showTambah && (
   325→            <Button
   326→              onClick={openCreate}
   327→              title="Tambah barang untuk pelanggan ini"
   328→              className="bg-emerald-600 hover:bg-emerald-700 min-h-[44px] w-full sm:w-auto"
   329→            >
   330→              <Plus className="h-4 w-4" /> Tambah
   331→            </Button>
   332→          )}
   333→        </div>
   334→      </div>
   335→
   336→      {/* Pilih Pelanggan — kotak filter tepat di bawah judul Master Barang */}
   337→      <section
   338→        aria-label="Filter pelanggan"
   339→        className="rounded-xl border border-stone-200 bg-white p-3 md:px-4 md:py-3"
   340→      >
   341→        <div className="flex flex-col gap-2.5 lg:flex-row lg:items-center lg:gap-4">
   342→          <div className="flex items-center gap-2.5 lg:shrink-0">
   343→            <span className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg border border-stone-200 bg-stone-100">
   344→              <Users className="h-4 w-4 text-stone-500" aria-hidden="true" />
   345→            </span>
   346→            <div className="min-w-0">
   347→              <Label
   348→                htmlFor="pilih-pelanggan"
   349→                className="block text-sm font-semibold text-stone-800 leading-tight"
   350→              >
   351→                Pilih Pelanggan
   352→              </Label>
   353→              <p className="text-[11px] text-muted-foreground leading-tight mt-0.5">
   354→                Tampilkan barang per pelanggan
   355→              </p>
   356→            </div>
   357→          </div>
   358→          <div className="lg:w-80">
   359→            <Select value={customerId} onValueChange={(v) => setCustomerId(v || 'all')}>
   360→              <SelectTrigger
   361→                id="pilih-pelanggan"
   362→                className="w-full min-h-[44px] bg-white"
   363→                aria-label="Pilih Pelanggan"
   364→              >
   365→                <SelectValue placeholder="Pilih pelanggan" />
   366→              </SelectTrigger>
   367→              <SelectContent>
   368→                <SelectItem value="all">Semua Barang</SelectItem>
   369→                {customers.map((c) => (
   370→                  <SelectItem key={c.id} value={c.id}>
   371→                    {c.name}{c.companyName ? ` — ${c.companyName}` : ''}
   372→                  </SelectItem>
   373→                ))}
   374→              </SelectContent>
   375→            </Select>
   376→          </div>
   377→          {canAdd && customerId === 'all' && (
   378→            <p className="text-xs text-muted-foreground lg:ml-auto self-center">
   379→              Pilih nama pelanggan untuk menambah / mengelola barang
   380→            </p>
   381→          )}
   382→          {customerId !== 'all' && selectedCustomerName && (
   383→            <div className="flex items-center gap-2 lg:ml-auto">
   384→              <Badge
   385→                variant="outline"
   386→                className="bg-emerald-50 text-emerald-700 border-emerald-200 text-xs max-w-[240px]"
   387→              >
   388→                <span className="truncate font-medium">{selectedCustomerName}</span>
   389→              </Badge>
   390→              <Button
   391→                variant="ghost"
   392→                size="sm"
   393→                className="h-8 px-2 text-xs text-muted-foreground hover:text-stone-800"
   394→                onClick={() => setCustomerId('all')}
   395→                aria-label="Reset filter pelanggan"
   396→              >
   397→                <X className="h-3.5 w-3.5" /> Reset
   398→              </Button>
   399→            </div>
   400→          )}
   401→        </div>
   402→      </section>
   403→
   404→      {/* Desktop table */}
   405→      <div className="hidden md:block rounded-xl border border-stone-200 bg-white overflow-hidden">
   406→        {loading ? (
   407→          <div className="p-4 space-y-3">
   408→            {[1, 2, 3, 4].map((i) => <Skeleton key={i} className="h-10 w-full" />)}
   409→          </div>
   410→        ) : !hasResults ? (
   411→          <EmptyState
   412→            filtered={query !== ''}
   413→            customerName={selectedCustomerName}
   414→            action={
   415→              showTambah && !query ? (
   416→                <Button
   417→                  onClick={openCreate}
   418→                  className="bg-emerald-600 hover:bg-emerald-700 min-h-[44px]"
   419→                >
   420→                  <Plus className="h-4 w-4" /> Tambah Barang
   421→                </Button>
   422→              ) : undefined
   423→            }
   424→          />
   425→        ) : (
   426→          <div className="max-h-96 overflow-y-auto scrollbar-thin">
   427→            <Table>
   428→              <TableHeader className="sticky top-0 z-10 bg-stone-50">
   429→                <TableRow className="bg-stone-50 hover:bg-stone-50">
   430→                  <TableHead>Kode</TableHead>
   431→                  <TableHead>Nama</TableHead>
   432→                  <TableHead className="text-center">Satuan</TableHead>
   433→                  <TableHead className="text-right">Harga Jual</TableHead>
   434→                  {showHpp && <TableHead className="text-right">HPP</TableHead>}
   435→                  <TableHead>Keterangan</TableHead>
   436→                  <TableHead>Status</TableHead>
   437→                  {(showEdit || showHapus) && <TableHead className="text-right">Aksi</TableHead>}
   438→                </TableRow>
   439→              </TableHeader>
   440→              <TableBody>
   441→                {items.map((it) => (
   442→                  <TableRow key={it.id}>
   443→                    <TableCell className="font-mono text-xs text-muted-foreground whitespace-nowrap">{it.code}</TableCell>
   444→                    <TableCell className="font-medium">{it.name}</TableCell>
   445→                    <TableCell className="text-center">{it.unit}</TableCell>
   446→                    <TableCell className="text-right whitespace-nowrap">{formatIDR(it.standardPrice)}</TableCell>
   447→                    {showHpp && (
   448→                      <TableCell className="text-right whitespace-nowrap">
   449→                        {it.hpp != null ? formatIDR(it.hpp) : '-'}
   450→                      </TableCell>
   451→                    )}
   452→                    <TableCell className="max-w-[200px]">
   453→                      <span
   454→                        className="block truncate text-sm text-muted-foreground"
   455→                        title={it.keterangan || undefined}
   456→                      >
   457→                        {it.keterangan || '-'}
   458→                      </span>
   459→                    </TableCell>
   460→                    <TableCell><ActiveBadge active={it.isActive} /></TableCell>
   461→                    {(showEdit || showHapus) && (
   462→                      <TableCell className="text-right">
   463→                        <div className="flex justify-end gap-1">
   464→                          {showEdit && (
   465→                            <Button
   466→                              variant="ghost"
   467→                              size="icon"
   468→                              className="h-9 w-9"
   469→                              onClick={() => openEdit(it)}
   470→                              aria-label={`Edit ${it.name}`}
   471→                              title="Edit"
   472→                            >
   473→                              <Pencil className="h-4 w-4" />
   474→                            </Button>
   475→                          )}
   476→                          {showHapus && (
   477→                            <Button
   478→                              variant="ghost"
   479→                              size="icon"
   480→                              className="h-9 w-9 text-destructive hover:text-destructive"
   481→                              onClick={() => setDeleteTarget(it)}
   482→                              aria-label={`Hapus ${it.name}`}
   483→                              title="Hapus"
   484→                            >
   485→                              <Trash2 className="h-4 w-4" />
   486→                            </Button>
   487→                          )}
   488→                        </div>
   489→                      </TableCell>
   490→                    )}
   491→                  </TableRow>
   492→                ))}
   493→              </TableBody>
   494→            </Table>
   495→          </div>
   496→        )}
   497→      </div>
   498→
   499→      {/* Mobile cards */}
   500→      <div className="md:hidden space-y-3">
   501→        {loading ? (
   502→          [1, 2, 3].map((i) => <Skeleton key={i} className="h-36 w-full rounded-xl" />)
   503→        ) : !hasResults ? (
   504→          <EmptyState
   505→            filtered={query !== ''}
   506→            customerName={selectedCustomerName}
   507→            action={
   508→              showTambah && !query ? (
   509→                <Button
   510→                  onClick={openCreate}
   511→                  className="bg-emerald-600 hover:bg-emerald-700 min-h-[44px] w-full sm:w-auto"
   512→                >
   513→                  <Plus className="h-4 w-4" /> Tambah Barang
   514→                </Button>
   515→              ) : undefined
   516→            }
   517→          />
   518→        ) : (
   519→          items.map((it) => (
   520→            <Card key={it.id} className="p-0 gap-0">
   521→              <CardContent className="p-4 space-y-2">
   522→                <div className="flex items-start justify-between gap-2">
   523→                  <div className="min-w-0">
   524→                    <p className="font-medium truncate">{it.name}</p>
   525→                    <p className="text-xs text-muted-foreground font-mono">{it.code}</p>
   526→                  </div>
   527→                  <ActiveBadge active={it.isActive} />
   528→                </div>
   529→                <div className="flex items-center gap-4 text-sm">
   530→                  <p className="text-muted-foreground">Satuan: <span className="font-medium text-stone-700">{it.unit}</span></p>
   531→                  <p className="text-muted-foreground">Harga Jual: <span className="font-semibold text-stone-700">{formatIDR(it.standardPrice)}</span></p>
   532→                </div>
   533→                {showHpp && (
   534→                  <p className="text-sm text-muted-foreground">
   535→                    HPP: <span className="font-medium text-stone-700">{it.hpp != null ? formatIDR(it.hpp) : '-'}</span>
   536→                  </p>
   537→                )}
   538→                {it.keterangan && (
   539→                  <p className="text-sm text-muted-foreground">
   540→                    Keterangan: <span className="text-stone-700">{it.keterangan}</span>
   541→                  </p>
   542→                )}
   543→                {(showEdit || showHapus) && (
   544→                  <div className="flex gap-2 pt-1">
   545→                    {showEdit && (
   546→                      <Button
   547→                        variant="outline"
   548→                        size="sm"
   549→                        className="flex-1 min-h-[44px]"
   550→                        onClick={() => openEdit(it)}
   551→                      >
   552→                        <Pencil className="h-4 w-4" /> Edit
   553→                      </Button>
   554→                    )}
   555→                    {showHapus && (
   556→                      <Button
   557→                        variant="outline"
   558→                        size="sm"
   559→                        className="flex-1 min-h-[44px] text-destructive border-stone-200 hover:bg-destructive/10 hover:text-destructive"
   560→                        onClick={() => setDeleteTarget(it)}
   561→                      >
   562→                        <Trash2 className="h-4 w-4" /> Hapus

   564→                    )}
   565→                  </div>
   566→                )}
   567→              </CardContent>
   568→            </Card>
   569→          ))
   570→        )}
   571→      </div>
   572→
   573→      {/* Dialog create / edit */}
   574→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   575→        <DialogContent className="sm:max-w-lg">
   576→          <DialogHeader>
   577→            <DialogTitle>{editing ? 'Edit Barang' : 'Tambah Barang'}</DialogTitle>
   578→            <DialogDescription>
   579→              {editing
   580→                ? `Kode ${editing.code} — perbarui data barang.`
   581→                : selectedCustomerName
   582→                  ? `Kode barang dibuat otomatis. Barang akan didaftarkan untuk ${selectedCustomerName}.`
   583→                  : 'Kode barang dibuat otomatis oleh sistem.'}
   584→            </DialogDescription>
   585→          </DialogHeader>
   586→          <div className="grid gap-4">
   587→            <div className="grid gap-1.5">
   588→              <Label htmlFor="item-name">Nama Barang <span className="text-destructive">*</span></Label>
   589→              <Input
   590→                id="item-name"
   591→                value={form.name}
   592→                onChange={(e) => setForm((f) => ({ ...f, name: e.target.value }))}
   593→                placeholder="Nama barang / layanan"
   594→                autoComplete="off"
   595→              />
   596→            </div>
   597→            <div className="grid sm:grid-cols-2 gap-4">
   598→              <div className="grid gap-1.5">
   599→                <Label htmlFor="item-unit">Satuan</Label>
   600→                <Select value={form.unit} onValueChange={(v) => setForm((f) => ({ ...f, unit: v }))}>
   601→                  <SelectTrigger id="item-unit" className="w-full min-h-[44px]" aria-label="Satuan barang">
   602→                    <SelectValue placeholder="Pilih satuan" />
   603→                  </SelectTrigger>
   604→                  <SelectContent>
   605→                    {UNIT_OPTIONS.map((u) => (
   606→                      <SelectItem key={u} value={u}>{u}</SelectItem>
   607→                    ))}
   608→                  </SelectContent>
   609→                </Select>
   610→              </div>
   611→              <div className="grid gap-1.5">
   612→                <Label htmlFor="item-price">Harga Jual (Rp) <span className="text-destructive">*</span></Label>
   613→                <Input
   614→                  id="item-price"
   615→                  type="number"
   616→                  inputMode="numeric"
   617→                  min={0}
   618→                  step="any"
   619→                  value={form.standardPrice}
   620→                  onChange={(e) => setJual(e.target.value)}
   621→                  placeholder="0"
   622→                />
   623→              </div>
   624→            </div>
   625→            {showHpp && (
   626→              <div className="grid gap-1.5">
   627→                <div className="grid sm:grid-cols-2 gap-4">
   628→                  <div className="grid gap-1.5">
   629→                    <Label htmlFor="item-modal">Harga Modal (Rp)</Label>
   630→                    <Input
   631→                      id="item-modal"
   632→                      type="number"
   633→                      inputMode="numeric"
   634→                      min={0}
   635→                      step="any"
   636→                      value={form.hpp}
   637→                      onChange={(e) => setModal(e.target.value)}
   638→                      placeholder="0"
   639→                    />
   640→                  </div>
   641→                  <div className="grid gap-1.5">
   642→                    <Label htmlFor="item-profit">Profit (Rp)</Label>
   643→                    <Input
   644→                      id="item-profit"
   645→                      type="number"
   646→                      inputMode="numeric"
   647→                      step="any"
   648→                      value={form.profit}
   649→                      onChange={(e) => setProfit(e.target.value)}
   650→                      placeholder="Otomatis dari Harga Jual − Modal"
   651→                      className={marginInfo?.negative ? 'text-red-600' : ''}
   652→                    />
   653→                  </div>
   654→                </div>
   655→                <p className="text-[11px] text-muted-foreground">
   656→                  Isi Modal + Profit → Harga Jual terisi otomatis. Isi Harga Jual → Profit terhitung sendiri.
   657→                </p>
   658→                {marginInfo && (
   659→                  <div className={`text-xs rounded-md px-2 py-1.5 ${marginInfo.negative ? 'bg-red-50 text-red-600' : 'bg-emerald-50 text-emerald-700'}`}>

   661→                    {' '}(Margin: {formatNum(marginInfo.margin, 1)}%)
   662→                    {marginInfo.negative ? ' — rugi! Harga Jual lebih kecil dari Modal.' : ''}
   663→                  </div>
   664→                )}
   665→              </div>
   666→            )}
   667→            <div className="grid gap-1.5">
   668→              <Label htmlFor="item-keterangan">Keterangan</Label>
   669→              <Input
   670→                id="item-keterangan"
   671→                value={form.keterangan}
   672→                onChange={(e) => setForm((f) => ({ ...f, keterangan: e.target.value }))}
   673→                placeholder="Keterangan tambahan (opsional)"
   674→                maxLength={500}
   675→                autoComplete="off"
   676→              />
   677→            </div>
   678→            {editing && (
   679→              <div className="flex items-center justify-between gap-3 rounded-lg border border-stone-200 p-3">
   680→                <div>
   681→                  <Label htmlFor="item-active">Status Aktif</Label>
   682→                  <p className="text-xs text-muted-foreground">Nonaktif = tidak muncul saat buat invoice.</p>
   683→                </div>
   684→                <Switch
   685→                  id="item-active"
   686→                  checked={form.isActive}
   687→                  onCheckedChange={(v) => setForm((f) => ({ ...f, isActive: v }))}
   688→                  className="data-[state=checked]:bg-emerald-600"
   689→                />
   690→              </div>
   691→            )}
   692→          </div>
   693→          <DialogFooter className="gap-2">
   694→            <Button variant="outline" onClick={() => setDialogOpen(false)} disabled={saving} className="min-h-[44px]">
   695→              Batal
   696→            </Button>
   697→            <Button
   698→              onClick={() => void handleSave()}
   699→              disabled={saving}
   700→              className="bg-emerald-600 hover:bg-emerald-700 min-h-[44px]"
   701→            >
   702→              {saving ? 'Menyimpan…' : 'Simpan'}
   703→            </Button>
   704→          </DialogFooter>
   705→        </DialogContent>
   706→      </Dialog>
   707→
   708→      {/* AlertDialog hapus */}
   709→      <AlertDialog open={!!deleteTarget} onOpenChange={(o) => { if (!o) setDeleteTarget(null) }}>
   710→        <AlertDialogContent>
   711→          <AlertDialogHeader>
   712→            <AlertDialogTitle>Hapus barang?</AlertDialogTitle>
   713→            <AlertDialogDescription>
   714→              {deleteTarget?.name} akan dihapus permanen dari daftar beserta daftar harga khususnya.
   715→              Invoice lama tetap tersimpan. Jika tidak ingin dihapus, gunakan opsi Nonaktifkan sebagai gantinya.
   716→            </AlertDialogDescription>
   717→          </AlertDialogHeader>
   718→          <AlertDialogFooter>
   719→            <AlertDialogCancel disabled={deleting}>Batal</AlertDialogCancel>
   720→            <AlertDialogAction
   721→              disabled={deleting}
   722→              className="bg-destructive text-white hover:bg-destructive/90"
   723→              onClick={(e) => { e.preventDefault(); void handleDelete() }}
   724→            >
   725→              {deleting ? 'Menghapus…' : 'Hapus'}
   726→            </AlertDialogAction>
   727→          </AlertDialogFooter>
   728→        </AlertDialogContent>
   729→      </AlertDialog>
   730→    </div>
   731→  )
   732→}
