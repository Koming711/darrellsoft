    42→function FadeIn({
    43→  children,
    44→  delay = 0,
    45→  className = '',
    46→  direction = 'up',
    47→}: {
    48→  children: React.ReactNode;
    49→  delay?: number;
    50→  className?: string;
    51→  direction?: 'up' | 'down' | 'left' | 'right';
    52→}) {
    53→  const ref = useRef<HTMLDivElement>(null);
    54→  const isInView = useInView(ref, { once: true });
    55→
    56→  const dirMap = {
    57→    up: { y: 40, x: 0 },
    58→    down: { y: -40, x: 0 },
    59→    left: { x: 40, y: 0 },
    60→    right: { x: -40, y: 0 },
    61→  };
    62→  const { x, y } = dirMap[direction];
    63→
    64→  return (
    65→    <motion.div
    66→      ref={ref}
    67→      className={className}
    68→      initial={isInView ? { opacity: 0, x, y } : false}
    69→      animate={{ opacity: 1, x: 0, y: 0 }}
    70→      transition={{ duration: 0.6, delay, ease: 'easeOut' }}
    71→    >
    72→      {children}
    73→    </motion.div>
    74→  );
    75→}

    77→function CountUp({ end, suffix = '', prefix = '', duration = 2000 }: { end: number; suffix?: string; prefix?: string; duration?: number }) {
    78→  const ref = useRef<HTMLSpanElement>(null);
    79→  const isInView = useInView(ref, { once: true });
    80→
    81→  return (
    82→    <motion.span
    83→      ref={ref}
    84→      initial={false}
    85→      animate={{ opacity: 1 }}
    86→      transition={{ duration: 0.4 }}
    87→    >
    88→      {isInView ? (
    89→        <Counter end={end} suffix={suffix} prefix={prefix} duration={duration} />
    90→      ) : (
    91→        <span>{prefix}{end.toLocaleString('id-ID')}{suffix}</span>
    92→      )}
    93→    </motion.span>
    94→  );
    95→}
    96→
    97→function Counter({ end, suffix, prefix, duration }: { end: number; suffix: string; prefix: string; duration: number }) {
    98→  const [count, setCount] = useState(0);
    99→  const ref = useRef<HTMLSpanElement>(null);
   100→  const started = useRef(false);
   101→
   102→  useInView(ref, { once: true });
   103→
   104→  if (!started.current) {
   105→    started.current = true;
   106→    const startTime = Date.now();
   107→    const step = () => {
   108→      const elapsed = Date.now() - startTime;
   109→      const progress = Math.min(elapsed / duration, 1);
   110→      const eased = 1 - Math.pow(1 - progress, 3);
   111→      setCount(Math.round(eased * end));
   112→      if (progress < 1) requestAnimationFrame(step);
   113→    };
   114→    requestAnimationFrame(step);
   115→  }
   116→
   117→  return <span ref={ref}>{prefix}{count.toLocaleString('id-ID')}{suffix}</span>;
   118→}
