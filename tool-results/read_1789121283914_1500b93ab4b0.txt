    89→    89→export default function ItemsView({ user, canAdd: canAddProp, canEdit: canEditProp, canDelete: canDeleteProp }: ItemsViewProps) {
    90→    90→  const canManage = user.role !== 'KASIR'
    91→    91→  const canAdd = canAddProp ?? canManage
    92→    92→  const canEditItem = canEditProp ?? canManage
    93→    93→  const canDelete = canDeleteProp ?? canManage
    94→    94→  const showHpp = user.role !== 'KASIR'
    95→    95→
    96→    96→  const [items, setItems] = useState<Item[]>([])
    97→    97→  const [loading, setLoading] = useState(true)
    98→    98→  const [searchInput, setSearchInput] = useState('')
    99→    99→  const [query, setQuery] = useState('')
   100→   100→
   101→   101→  // Pilih Customer — filter daftar barang per pelanggan (barang terdaftar)
   102→   102→  const [customers, setCustomers] = useState<CustomerOption[]>([])
   103→   103→  const [customerId, setCustomerId] = useState<string>('all')
   104→   104→
   105→   105→  const [dialogOpen, setDialogOpen] = useState(false)
   106→   106→  const [editing, setEditing] = useState<Item | null>(null)
   107→   107→  const [form, setForm] = useState<ItemFormState>(EMPTY_FORM)
   108→   108→  const [saving, setSaving] = useState(false)
   109→   109→
   110→   110→  const [deleteTarget, setDeleteTarget] = useState<Item | null>(null)
   111→   111→  const [deleting, setDeleting] = useState(false)
   112→   112→
   113→   113→  useEffect(() => {
   114→   114→    const t = setTimeout(() => setQuery(searchInput.trim()), 300)
   115→   115→    return () => clearTimeout(t)
   116→   116→  }, [searchInput])
   117→   117→
   118→   118→  useEffect(() => {
   119→   119→    apiFetch<CustomerOption[]>('/api/customers')
   120→   120→      .then((data) => setCustomers(Array.isArray(data) ? data : []))
   121→   121→      .catch(() => setCustomers([]))
   122→   122→  }, [])
   123→   123→
   124→   124→  const load = useCallback(async () => {
   125→   125→    setLoading(true)

   127→   127→      // active=all agar item nonaktif tetap terlihat (bisa diaktifkan kembali)
   128→   128→      const cs = customerId !== 'all' ? `&customerId=${encodeURIComponent(customerId)}` : ''
   129→   129→      const data = await apiFetch<{ items: Item[] }>(
   130→   130→        `/api/items?q=${encodeURIComponent(query)}&active=all${cs}`
   131→   131→      )
   132→   132→      setItems(data.items)
   133→   133→    } catch (e) {
   134→   134→      toast.error(errText(e))
   135→   135→    } finally {
   136→   136→      setLoading(false)
   137→   137→    }
   138→   138→  }, [query, customerId])
   139→   139→
   140→   140→  useEffect(() => {
   141→   141→    void load()
   142→   142→  }, [load])
   143→   143→
   144→   144→  const openCreate = () => {
   145→   145→    setEditing(null)
   146→   146→    setForm(EMPTY_FORM)
   147→   147→    setDialogOpen(true)
   148→   148→  }
   149→   149→
   150→   150→  const openEdit = (it: Item) => {
   151→   151→    setEditing(it)
   152→   152→    setForm({
   153→   153→      name: it.name,
   154→   154→      unit: it.unit,
   155→   155→      standardPrice: String(it.standardPrice),
   156→   156→      hpp: it.hpp != null ? String(it.hpp) : '',
   157→   157→      keterangan: it.keterangan ?? '',
   158→   158→      isActive: it.isActive,
   159→   159→    })
   160→   160→    setDialogOpen(true)
   161→   161→  }
   162→   162→
   163→   163→  // Profit = harga jual − HPP; Margin = profit / harga jual × 100
   164→   164→  const marginInfo = (() => {
   165→   165→    const std = Number(form.standardPrice)
   166→   166→    const hpp = Number(form.hpp)

   168→   168→      return null
   169→   169→    }
   170→   170→    const profit = std - hpp
   171→   171→    const margin = (profit / std) * 100
   172→   172→    return { profit, margin, negative: margin < 0 }
   173→   173→  })()

   175→   175→  const handleSave = async () => {

   178→   178→      toast.error('Nama barang wajib diisi')
   179→   179→      return
   180→   180→    }

   182→   182→      toast.error('Harga jual wajib diisi (min 0)')
   183→   183→      return
   184→   184→    }

   186→   186→    if (hppNum !== null && (!Number.isFinite(hppNum) || hppNum < 0)) {
   187→   187→      toast.error('HPP tidak boleh negatif')

   189→   189→    }

   191→   191→    try {
   192→   192→      const body = {
   193→   193→        name: form.name.trim(),
   194→   194→        unit: form.unit,
   195→   195→        standardPrice: std,
   196→   196→        hpp: showHpp ? hppNum : null,
   197→   197→        keterangan: form.keterangan.trim(),
   198→   198→        ...(editing ? { isActive: form.isActive } : { customerId: customerId !== 'all' ? customerId : undefined }),
   199→   199→      }
   200→   200→      const selectedCustomer = customers.find((c) => c.id === customerId)
   201→   201→      if (editing) {
   202→   202→        await apiFetch<{ item: Item }>(`/api/items/${editing.id}`, {
   203→   203→          method: 'PUT',
   204→   204→          body: JSON.stringify(body),
   205→   205→        })
   206→   206→        toast.success('Barang berhasil diperbarui')
   207→   207→      } else {
   208→   208→        await apiFetch<{ item: Item }>('/api/items', {
   209→   209→          method: 'POST',
   210→   210→          body: JSON.stringify(body),
   211→   211→        })
   212→   212→        toast.success(
   213→   213→          selectedCustomer
   214→   214→            ? `Barang berhasil ditambahkan untuk ${selectedCustomer.name}`
   215→   215→            : 'Barang berhasil ditambahkan'
   216→   216→        )
   217→   217→      }
   218→   218→      setDialogOpen(false)
   219→   219→      void load()
   220→   220→    } catch (e) {
   221→   221→      toast.error(errText(e))
   222→   222→    } finally {
   223→   223→      setSaving(false)
   224→   224→    }
   225→   225→  }
   226→   226→
   227→   227→  const handleDelete = async () => {
   228→   228→    if (!deleteTarget) return
   229→   229→    setDeleting(true)

   231→   231→      await apiFetch<{ ok: boolean }>(`/api/items/${deleteTarget.id}`, { method: 'DELETE' })
   232→   232→      toast.success('Barang berhasil dihapus')
   233→   233→      setDeleteTarget(null)
   234→   234→      void load()
   235→   235→    } catch (e) {
   236→   236→      // 409 → item dipakai di invoice
   237→   237→      toast.error(errText(e))
   238→   238→    } finally {
   239→   239→      setDeleting(false)
   240→   240→    }
   241→   241→  }
   242→   242→
   243→   243→  const hasResults = items.length > 0
   244→   244→  const selectedCustomerName = customers.find((c) => c.id === customerId)?.name ?? null
   245→   245→  // Tambah/Edit hanya tampil saat pelanggan spesifik dipilih — bukan "Semua Barang"
   246→   246→  const showTambah = canAdd && customerId !== 'all'
   247→   247→  const showEdit = canEditItem && customerId !== 'all'
   248→   248→  // Hapus SELALU tampil di tabel (selaras Master Pelanggan), termasuk mode "Semua Barang"
   249→   249→  const showHapus = canDelete
   250→   250→  const countLabel = loading
   251→   251→    ? 'Memuat data…'
   252→   252→    : selectedCustomerName
   253→   253→      ? `${items.length} barang untuk ${selectedCustomerName}`
   254→   254→      : `${items.length} barang`
   255→   255→
   256→   256→  return (
   257→   257→    <div className="space-y-5">
   258→   258→      {/* Header */}
   259→   259→      <div className="flex flex-col gap-3 md:flex-row md:items-end md:justify-between">
   260→   260→        <div>
   261→   261→          <h1 className="text-xl md:text-2xl font-bold tracking-tight">Master Barang</h1>
   262→   262→          <p className="text-sm text-muted-foreground mt-1">{countLabel}</p>
   263→   263→        </div>
   264→   264→        <div className="flex flex-col sm:flex-row sm:items-center gap-2">
   265→   265→          <div className="relative flex-1 sm:w-64">
   266→   266→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-stone-400 pointer-events-none" />
   267→   267→            <Input
   268→   268→              value={searchInput}
   269→   269→              onChange={(e) => setSearchInput(e.target.value)}
   270→   270→              placeholder="Cari nama / kode barang…"
   271→   271→              aria-label="Cari barang"
   272→   272→              className="pl-9 min-h-[44px]"
   273→   273→            />
   274→   274→          </div>
   275→   275→          {showTambah && (
   276→   276→            <Button
   277→   277→              onClick={openCreate}
   278→   278→              title="Tambah barang untuk pelanggan ini"
   279→   279→              className="bg-emerald-600 hover:bg-emerald-700 min-h-[44px] w-full sm:w-auto"
   280→   280→            >
   281→   281→              <Plus className="h-4 w-4" /> Tambah
   282→   282→            </Button>
   283→   283→          )}
   284→   284→        </div>
   285→   285→      </div>
   286→   286→
   287→   287→      {/* Pilih Pelanggan — kotak filter tepat di bawah judul Master Barang */}
   288→   288→      <section
   289→   289→        aria-label="Filter pelanggan"
   290→   290→        className="rounded-xl border border-stone-200 bg-white p-3 md:px-4 md:py-3"
   291→   291→      >
   292→   292→        <div className="flex flex-col gap-2.5 lg:flex-row lg:items-center lg:gap-4">
   293→   293→          <div className="flex items-center gap-2.5 lg:shrink-0">
   294→   294→            <span className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg border border-stone-200 bg-stone-100">
   295→   295→              <Users className="h-4 w-4 text-stone-500" aria-hidden="true" />
   296→   296→            </span>
   297→   297→            <div className="min-w-0">
   298→   298→              <Label
   299→   299→                htmlFor="pilih-pelanggan"
   300→   300→                className="block text-sm font-semibold text-stone-800 leading-tight"
   301→   301→              >
   302→   302→                Pilih Pelanggan
   303→   303→              </Label>
   304→   304→              <p className="text-[11px] text-muted-foreground leading-tight mt-0.5">
   305→   305→                Tampilkan barang per pelanggan
   306→   306→              </p>
   307→   307→            </div>
   308→   308→          </div>
   309→   309→          <div className="lg:w-80">
   310→   310→            <Select value={customerId} onValueChange={(v) => setCustomerId(v || 'all')}>
   311→   311→              <SelectTrigger
   312→   312→                id="pilih-pelanggan"
   313→   313→                className="w-full min-h-[44px] bg-white"
   314→   314→                aria-label="Pilih Pelanggan"

   316→   316→                <SelectValue placeholder="Pilih pelanggan" />
   317→   317→              </SelectTrigger>
   318→   318→              <SelectContent>
   319→   319→                <SelectItem value="all">Semua Barang</SelectItem>
   320→   320→                {customers.map((c) => (
   321→   321→                  <SelectItem key={c.id} value={c.id}>
   322→   322→                    {c.name}{c.companyName ? ` — ${c.companyName}` : ''}
   323→   323→                  </SelectItem>
   324→   324→                ))}
   325→   325→              </SelectContent>
   326→   326→            </Select>
   327→   327→          </div>
   328→   328→          {canAdd && customerId === 'all' && (
   329→   329→            <p className="text-xs text-muted-foreground lg:ml-auto self-center">
   330→   330→              Pilih nama pelanggan untuk menambah / mengelola barang
   331→   331→            </p>
   332→   332→          )}
   333→   333→          {customerId !== 'all' && selectedCustomerName && (
   334→   334→            <div className="flex items-center gap-2 lg:ml-auto">
   335→   335→              <Badge
   336→   336→                variant="outline"
   337→   337→                className="bg-emerald-50 text-emerald-700 border-emerald-200 text-xs max-w-[240px]"
   338→   338→              >
   339→   339→                <span className="truncate font-medium">{selectedCustomerName}</span>
   340→   340→              </Badge>
   341→   341→              <Button
   342→   342→                variant="ghost"
   343→   343→                size="sm"
   344→   344→                className="h-8 px-2 text-xs text-muted-foreground hover:text-stone-800"
   345→   345→                onClick={() => setCustomerId('all')}
   346→   346→                aria-label="Reset filter pelanggan"
   347→   347→              >
   348→   348→                <X className="h-3.5 w-3.5" /> Reset
   349→   349→              </Button>
   350→   350→            </div>
   351→   351→          )}
   352→   352→        </div>
   353→   353→      </section>
   354→   354→
   355→   355→      {/* Desktop table */}
   356→   356→      <div className="hidden md:block rounded-xl border border-stone-200 bg-white overflow-hidden">
   357→   357→        {loading ? (
   358→   358→          <div className="p-4 space-y-3">
   359→   359→            {[1, 2, 3, 4].map((i) => <Skeleton key={i} className="h-10 w-full" />)}
   360→   360→          </div>
   361→   361→        ) : !hasResults ? (
   362→   362→          <EmptyState
   363→   363→            filtered={query !== ''}
   364→   364→            customerName={selectedCustomerName}
   365→   365→            action={
   366→   366→              showTambah && !query ? (
   367→   367→                <Button
   368→   368→                  onClick={openCreate}
   369→   369→                  className="bg-emerald-600 hover:bg-emerald-700 min-h-[44px]"
   370→   370→                >
   371→   371→                  <Plus className="h-4 w-4" /> Tambah Barang
   372→   372→                </Button>
   373→   373→              ) : undefined
   374→   374→            }
   375→   375→          />
   376→   376→        ) : (
   377→   377→          <div className="max-h-96 overflow-y-auto scrollbar-thin">
   378→   378→            <Table>
   379→   379→              <TableHeader className="sticky top-0 z-10 bg-stone-50">
   380→   380→                <TableRow className="bg-stone-50 hover:bg-stone-50">
   381→   381→                  <TableHead>Kode</TableHead>
   382→   382→                  <TableHead>Nama</TableHead>
   383→   383→                  <TableHead className="text-center">Satuan</TableHead>
   384→   384→                  <TableHead className="text-right">Harga Jual</TableHead>
   385→   385→                  {showHpp && <TableHead className="text-right">HPP</TableHead>}
   386→   386→                  <TableHead>Keterangan</TableHead>

   389→   389→                </TableRow>

   391→   391→              <TableBody>

   395→   395→                    <TableCell className="font-medium">{it.name}</TableCell>
   396→   396→                    <TableCell className="text-center">{it.unit}</TableCell>
   397→   397→                    <TableCell className="text-right whitespace-nowrap">{formatIDR(it.standardPrice)}</TableCell>
   398→   398→                    {showHpp && (
   399→   399→                      <TableCell className="text-right whitespace-nowrap">
   400→   400→                        {it.hpp != null ? formatIDR(it.hpp) : '-'}
   401→   401→                      </TableCell>
   402→   402→                    )}
   403→   403→                    <TableCell className="max-w-[200px]">
   404→   404→                      <span
   405→   405→                        className="block truncate text-sm text-muted-foreground"
   406→   406→                        title={it.keterangan || undefined}
   407→   407→                      >

   425→   425→                            </Button>
   426→   426→                          )}
   427→   427→                          {showHapus && (

   431→   431→                              className="h-9 w-9 text-destructive hover:text-destructive"

   437→   437→                            </Button>
   438→   438→                          )}
   439→   439→                        </div>
   440→   440→                      </TableCell>
   441→   441→                    )}

   448→   448→      </div>
   449→   449→
   450→   450→      {/* Mobile cards */}
   451→   451→      <div className="md:hidden space-y-3">
   452→   452→        {loading ? (
   453→   453→          [1, 2, 3].map((i) => <Skeleton key={i} className="h-36 w-full rounded-xl" />)
   454→   454→        ) : !hasResults ? (
   455→   455→          <EmptyState
   456→   456→            filtered={query !== ''}
   457→   457→            customerName={selectedCustomerName}
   458→   458→            action={
   459→   459→              showTambah && !query ? (
