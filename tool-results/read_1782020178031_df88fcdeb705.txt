    15→// === MODULE-LEVEL SESSION CACHE ===
    16→// Persists across navigations so DashboardLayout doesn't need to re-verify on every mount
    17→interface CachedSession {
    18→  user: any
    19→  userProfile: { createdAt: string | null; validUntil: string | null } | null
    20→  autoLogoutMin: number
    21→  logoutWarningSec: number
    22→  sessionWarning: string | null
    23→  forceLogoutAvailable: boolean
    24→  accountExpired: boolean
    25→  cachedAt: number
    26→}
    27→let _sessionCache: CachedSession | null = null
    28→const SESSION_CACHE_TTL = 120_000 // 2 minutes
    29→
    30→function getSessionCache(): CachedSession | null {
    31→  // Check module-level cache first
    32→  if (_sessionCache && Date.now() - _sessionCache.cachedAt <= SESSION_CACHE_TTL) {
    33→    return _sessionCache
    34→  }
    35→  // Fallback to sessionStorage
    36→  if (typeof window !== 'undefined') {
    37→    try {
    38→      const stored = sessionStorage.getItem('__dashboard_session_cache')
    39→      if (stored) {
    40→        const parsed = JSON.parse(stored) as CachedSession
    41→        if (Date.now() - parsed.cachedAt <= SESSION_CACHE_TTL) {
    42→          _sessionCache = parsed
    43→          return parsed
    44→        }
    45→        sessionStorage.removeItem('__dashboard_session_cache')
    46→      }
    47→    } catch {}
    48→  }
    49→  _sessionCache = null
    50→  return null
    51→}
    52→
    53→function setSessionCache(data: Omit<CachedSession, 'cachedAt'>) {
    54→  const entry = { ...data, cachedAt: Date.now() }
    55→  _sessionCache = entry
    56→  if (typeof window !== 'undefined') {
    57→    try {
    58→      sessionStorage.setItem('__dashboard_session_cache', JSON.stringify(entry))
    59→    } catch {}
    60→  }
    61→}
    62→
    63→function invalidateSessionCache() {
    64→  _sessionCache = null
    65→  if (typeof window !== 'undefined') {
    66→    try {
    67→      sessionStorage.removeItem('__dashboard_session_cache')
    68→    } catch {}
    69→  }
    70→}

   540→        onLogout={handleLogout}
   541→        permVersion={permVersion}
   542→      />
   543→
   544→      <div className={`${desktopMargin} transition-all duration-300`}>
   545→        <MobileHeader
   546→          username={user?.username}
   547→          title={title}
   548→          subtitle={subtitle}
   549→          userProfile={userProfile}
   550→        />

   552→        {/* Main Content — extra bottom padding on mobile for bottom nav + safe area */}

   554→          {children}
   555→        </main>
   556→      </div>
   557→

   560→
   561→      {/* ===== MODAL: ACCOUNT EXPIRED ===== */}
   562→      {accountExpired && sessionWarning && (
   563→        <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/50 backdrop-blur-sm px-4">
   564→          <div className="rounded-2xl shadow-2xl border border-red-200 dark:border-red-800 max-w-sm w-full p-6 animate-in fade-in zoom-in duration-200" style={{ backgroundColor: 'var(--card)' }}>
   565→            <div className="flex items-start gap-3 mb-4">

   567→                <TimerOff className="w-5 h-5 text-red-600 dark:text-red-400" />

   578→              onClick={handleLogout}
   579→              className="w-full bg-red-600 hover:bg-red-700 text-white font-semibold py-2.5 rounded-xl transition-colors"

   581→            >

   594→            <div className="flex items-start gap-3 mb-4">

   649→                  className="w-full flex items-center justify-center gap-2 border-2 border-red-200 dark:border-red-800 text-red-600 dark:text-red-400 hover:bg-red-50 dark:hover:bg-red-950 font-semibold py-2.5 rounded-xl transition-colors"
   650→                >
   651→                  <LogOut className="w-4 h-4" /> Logout dari Sini
   652→                </button>
   653→              </div>
   654→            ) : (
   655→              <button
   656→                onClick={handleLogout}
   657→                className="w-full bg-red-600 hover:bg-red-700 text-white font-semibold py-2.5 rounded-xl transition-colors"
   658→                autoFocus
   659→              >
   660→                OK, Logout
   661→              </button>
   662→            )}
   663→          </div>
   664→        </div>
   665→      )}
   666→
   667→      {/* ===== MODAL: AUTO-LOGOUT COUNTDOWN ===== */}
   668→      {showCountdown && (
   669→        <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/50 backdrop-blur-sm px-4">
   670→          <div className="rounded-2xl shadow-2xl border border-amber-200 dark:border-amber-800 max-w-sm w-full p-6 animate-in fade-in zoom-in duration-200" style={{ backgroundColor: 'var(--card)' }}>
   671→            <div className="flex items-start gap-3 mb-4">
   672→              <div className="w-10 h-10 rounded-xl bg-amber-100 dark:bg-amber-900/30 flex items-center justify-center flex-shrink-0">
   673→                <AlertTriangle className="w-5 h-5 text-amber-600 dark:text-amber-400" />
   674→              </div>
   675→              <div>
   676→                <h3 className="text-lg font-bold text-foreground">Sesi Akan Berakhir</h3>

   678→              </div>
   679→            </div>
   680→            <div className="text-center py-5">
   681→              <div className="text-6xl font-bold text-amber-600 dark:text-amber-400 tabular-nums">{countdown}</div>
   682→              <p className="text-sm text-muted-foreground mt-2">detik tersisa</p>
   683→            </div>
   684→            <button
   685→              onClick={handleStayLoggedIn}
   686→              className="w-full bg-amber-500 hover:bg-amber-600 text-white font-semibold py-2.5 rounded-xl transition-colors"
   687→              autoFocus
   688→            >
   689→              Tetap Login
   690→            </button>
   691→          </div>
   692→        </div>
   693→      )}
   694→
   695→      {/* ===== MODAL: COMPANY DATA POPUP (untuk demo user baru dari checkout) ===== */}
