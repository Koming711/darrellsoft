    40→function FadeIn({
    41→  children,
    42→  delay = 0,
    43→  className = '',
    44→  direction = 'up',
    45→}: {
    46→  children: React.ReactNode;
    47→  delay?: number;
    48→  className?: string;
    49→  direction?: 'up' | 'down' | 'left' | 'right';
    50→}) {
    51→  const ref = useRef<HTMLDivElement>(null);
    52→  const isInView = useInView(ref, { once: true });
    53→
    54→  const dirMap = {
    55→    up: { y: 40, x: 0 },
    56→    down: { y: -40, x: 0 },
    57→    left: { x: 40, y: 0 },
    58→    right: { x: -40, y: 0 },
    59→  };
    60→  const { x, y } = dirMap[direction];
    61→
    62→  return (
    63→    <motion.div
    64→      ref={ref}
    65→      className={className}
    66→      initial={isInView ? { opacity: 0, x, y } : false}
    67→      animate={{ opacity: 1, x: 0, y: 0 }}
    68→      transition={{ duration: 0.6, delay, ease: 'easeOut' }}
    69→    >
    70→      {children}
    71→    </motion.div>
    72→  );
    73→}

    75→function CountUp({ end, suffix = '', prefix = '', duration = 2000 }: { end: number; suffix?: string; prefix?: string; duration?: number }) {
    76→  const ref = useRef<HTMLSpanElement>(null);
    77→  const isInView = useInView(ref, { once: true });
    78→
    79→  return (
    80→    <motion.span
    81→      ref={ref}
    82→      initial={false}
    83→      animate={{ opacity: 1 }}
    84→      transition={{ duration: 0.4 }}
    85→    >
    86→      {isInView ? (
    87→        <Counter end={end} suffix={suffix} prefix={prefix} duration={duration} />
    88→      ) : (
    89→        <span>{prefix}{end.toLocaleString('id-ID')}{suffix}</span>
    90→      )}
    91→    </motion.span>
    92→  );
    93→}
    94→
    95→function Counter({ end, suffix, prefix, duration }: { end: number; suffix: string; prefix: string; duration: number }) {
    96→  const [count, setCount] = useState(0);
    97→  const ref = useRef<HTMLSpanElement>(null);
    98→  const started = useRef(false);
    99→
   100→  useInView(ref, { once: true });
   101→
   102→  if (!started.current) {
   103→    started.current = true;
   104→    const startTime = Date.now();
   105→    const step = () => {
   106→      const elapsed = Date.now() - startTime;
   107→      const progress = Math.min(elapsed / duration, 1);
   108→      const eased = 1 - Math.pow(1 - progress, 3);
   109→      setCount(Math.round(eased * end));
   110→      if (progress < 1) requestAnimationFrame(step);
   111→    };
   112→    requestAnimationFrame(step);
   113→  }
   114→
   115→  return <span ref={ref}>{prefix}{count.toLocaleString('id-ID')}{suffix}</span>;
   116→}
