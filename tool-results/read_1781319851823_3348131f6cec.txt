    47→// --- Helper ---
    48→function getTodayStr(): string {
    49→  const d = new Date();
    50→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
    51→}

    53→// --- Parse dataJson for document info ---
    54→function parseDocInfo(entry: HistoryEntry) {
    55→  try {
    56→    const parsed = JSON.parse(entry.dataJson);
    57→    const items = parsed.items || [];
    58→    const firstItem = items[0];
    59→    const namaBarang = firstItem?.deskripsi || '';
    60→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0);
    61→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0);
    62→    const ppn = parsed.ppn || 0;
    63→    const dpPercent = parsed.dp || 0;
    64→    const totalHarga = subtotal + (subtotal * ppn / 100);
    65→    // Use saved dpAmount if available
    66→    // If not, use originalTotal * dpPercent to get the correct original DP (not affected by pelunasan additions)
    67→    // Only fall back to totalHarga * dpPercent if originalTotal is also missing
    68→    const originalTotal = parsed.originalTotal !== undefined ? parsed.originalTotal : totalHarga;
    69→    const dpAmount = parsed.dpAmount !== undefined ? parsed.dpAmount : originalTotal * (dpPercent / 100);
    70→    const sisa = totalHarga - dpAmount;
    71→    const lunas = parsed.lunas === true;
    72→    const tanggalJatuhTempo = parsed.tanggalJatuhTempo || '';
    73→    const tanggalPelunasan = parsed.tanggalPelunasan || '';
    74→    return { namaBarang, totalQty, totalHarga, dpPercent, dp: dpAmount, sisa, lunas, tanggalJatuhTempo, tanggalPelunasan };
    75→  } catch {
    76→    return { namaBarang: '', totalQty: 0, totalHarga: 0, dpPercent: 0, dp: 0, sisa: 0, lunas: false, tanggalJatuhTempo: '', tanggalPelunasan: '' };
    77→  }
    78→}
    79→
    80→// --- Parse dataJson for InvoiceData ---
    81→function parseInvoiceData(entry: HistoryEntry): InvoiceData {
    82→  try {
    83→    const parsed = JSON.parse(entry.dataJson);
    84→    const company: CompanyInfo = {
    85→      nama: parsed.company?.nama || DEFAULT_COMPANY.nama,
    86→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
    87→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
    88→      email: parsed.company?.email || DEFAULT_COMPANY.email,
    89→      npwp: parsed.company?.npwp || '',
    90→      website: parsed.company?.website || '',
    91→      ppn: parsed.company?.ppn ?? parsed.ppn ?? 11,
    92→      logo: parsed.company?.logo || '',
    93→      bankName: parsed.company?.bankName || '',
    94→      bankAccount: parsed.company?.bankAccount || '',
    95→      bankHolder: parsed.company?.bankHolder || '',
    96→      bankName2: parsed.company?.bankName2 || '',
    97→      bankAccount2: parsed.company?.bankAccount2 || '',
    98→      bankHolder2: parsed.company?.bankHolder2 || '',
    99→    };
   100→    const client = parsed.client || { nama: '', kontak: '', alamat: '' };
   101→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number }, i: number) => ({
   102→      id: it.id || `item-${i}`,
   103→      deskripsi: it.deskripsi || '',
   104→      qty: it.qty || 0,
   105→      satuan: it.satuan || '',
   106→      harga: it.harga || 0,
   107→    }));
   108→    return {
   109→      type: 'invoice',
   110→      company,
   111→      nomor: parsed.nomor || entry.nomor || '',
   112→      tanggal: parsed.tanggal || entry.tanggal || '',
   113→      referensi: parsed.referensi || '',
   114→      client,
   115→      items,
   116→      ppn: parsed.ppn ?? 11,
   117→      dp: parsed.dp || 0,
   118→      dpAmount: parsed.dpAmount,
   119→      originalTotal: parsed.originalTotal,
   120→      catatan: parsed.catatan || '',
   121→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   122→      caraPembayaran: parsed.caraPembayaran || '',
   123→      tanggalGiro: parsed.tanggalGiro || '',
   124→      lunas: parsed.lunas === true,
   125→      tanggalPelunasan: parsed.tanggalPelunasan || '',
   126→    };
   127→  } catch {
   128→    return {
   129→      type: 'invoice',
   130→      company: { ...DEFAULT_COMPANY },
   131→      nomor: entry.nomor || '',
   132→      tanggal: entry.tanggal || '',
   133→      referensi: '',
   134→      client: { nama: '', kontak: '', alamat: '' },
   135→      items: [],
   136→      ppn: 11,
   137→      dp: 0,
   138→      catatan: '',
   139→      tanggalJatuhTempo: '',
   140→      caraPembayaran: '',
   141→      tanggalGiro: '',
   142→      lunas: false,
   143→      tanggalPelunasan: '',
   144→    };
   145→  }
   146→}

   148→export function InvoicePelunasanEditor() {
   149→  // Local invoice state (not using global store to avoid conflicts with regular editor)
   150→  const [invoiceData, setInvoiceData] = useState<InvoiceData | null>(null);
   151→  const [selectedEntry, setSelectedEntry] = useState<HistoryEntry | null>(null);
   152→  const [invoiceHistory, setInvoiceHistory] = useState<HistoryEntry[]>([]);
   153→  const [loading, setLoading] = useState(true);
   154→  const [searchQuery, setSearchQuery] = useState('');
   155→  const [dropdownOpen, setDropdownOpen] = useState(false);
   156→  const [pelunasanSaving, setPelunasanSaving] = useState(false);
   157→  const [jpgGenerating, setJpgGenerating] = useState(false);
   158→
   159→  // Pelunasan fields
   160→  const [lunasToggle, setLunasToggle] = useState(false);
   161→  const [tanggalPelunasan, setTanggalPelunasan] = useState('');
   162→  const [tanggalJatuhTempo, setTanggalJatuhTempo] = useState('');
   163→  const [caraPembayaran, setCaraPembayaran] = useState('');
   164→  const [tanggalGiro, setTanggalGiro] = useState('');
   165→
   166→  // Original DP amount — fixed when invoice is first selected, does NOT change when items are added
   167→  const [originalDpAmount, setOriginalDpAmount] = useState(0);
   168→
   169→  const fetchHistory = useCallback(async () => {
   170→    try {
   171→      setLoading(true);
   172→      const headers = getAuthHeaders();
   173→      const res = await fetch('/api/history?docType=invoice', { headers });
   174→      if (res.ok) {
   175→        const json = await res.json();
   176→        setInvoiceHistory(json.data || []);
   177→      }
   178→    } catch (err) {
   179→      console.error('Failed to fetch invoice history:', err);
   180→    } finally {
   181→      setLoading(false);
   182→    }
   183→  }, []);
   184→
   185→  useEffect(() => {
   186→    fetchHistory();
   187→  }, [fetchHistory]);
   188→
   189→  useEffect(() => {
   190→    const handler = () => fetchHistory();
   191→    window.addEventListener('dokupro:history-updated', handler);
   192→    return () => window.removeEventListener('dokupro:history-updated', handler);
   193→  }, [fetchHistory]);
   194→
   195→  // Filter pending invoices (with DP and not yet lunas)
   196→  // Business rule: having DP means belum lunas, only explicit lunas flag marks it as paid
   197→  const pendingInvoices = useMemo(() => {
   198→    return invoiceHistory.filter(entry => {
   199→      const info = parseDocInfo(entry);
   200→      return (info.dpPercent > 0 || info.dp > 0) && !info.lunas;
   201→    });
   202→  }, [invoiceHistory]);
   203→
   204→  // Filter by search
   205→  const filteredPending = useMemo(() => {
   206→    if (!searchQuery.trim()) return pendingInvoices;
   207→    const q = searchQuery.toLowerCase().trim();
   208→    return pendingInvoices.filter(entry => {
   209→      const info = parseDocInfo(entry);
   210→      return entry.nomor?.toLowerCase().includes(q) || entry.pihakKedua?.toLowerCase().includes(q) || info.namaBarang?.toLowerCase().includes(q);
   211→    });
   212→  }, [pendingInvoices, searchQuery]);
   213→
   214→  // Select an invoice
   215→  const selectInvoice = (entry: HistoryEntry) => {
   216→    const parsed = parseInvoiceData(entry);
   217→    const info = parseDocInfo(entry);
   218→    setInvoiceData(parsed);
   219→    setSelectedEntry(entry);
   220→    setLunasToggle(info.lunas);
   221→    setTanggalPelunasan(info.tanggalPelunasan || getTodayStr());
   222→    setTanggalJatuhTempo(info.tanggalJatuhTempo || '');
   223→    setCaraPembayaran(parsed.caraPembayaran || '');
   224→    setTanggalGiro(parsed.tanggalGiro || '');
   225→    // Calculate original DP amount from originalTotal (not from saved dpAmount which may be wrong)
   226→    // This ensures DP stays at the original amount even when pelunasan items are added
   227→    const rawParsed = JSON.parse(entry.dataJson);
   228→    const dpPercent = rawParsed.dp || 0;
   229→    const items = rawParsed.items || [];
   230→    const sub = items.reduce((s: number, it: { qty: number; harga: number }) => s + it.qty * it.harga, 0);
   231→    const ppn = rawParsed.ppn || 0;
   232→    const currentTotal = sub + (sub * ppn / 100);
   233→    const originalTotal = rawParsed.originalTotal !== undefined ? rawParsed.originalTotal : currentTotal;
   234→    // Priority: saved dpAmount → originalTotal * dpPercent → currentTotal * dpPercent
   235→    // But if dpAmount equals currentTotal * dpPercent, it was likely recalculated incorrectly
   236→    const recalculatedDp = currentTotal * (dpPercent / 100);
   237→    let fixedDpAmount: number;
   238→    if (rawParsed.dpAmount !== undefined && rawParsed.dpAmount !== recalculatedDp) {
   239→      // dpAmount was explicitly saved and is NOT a simple recalculation from current total — trust it
   240→      fixedDpAmount = rawParsed.dpAmount;
   241→    } else {
   242→      // dpAmount is either missing or was recalculated from current total — use originalTotal instead
   243→      fixedDpAmount = originalTotal * (dpPercent / 100);
   244→    }
   245→    setOriginalDpAmount(fixedDpAmount);
   246→    setDropdownOpen(false);
   247→  };
   248→
   249→  // Clear selection
   250→  const clearSelection = () => {
   251→    setInvoiceData(null);
   252→    setSelectedEntry(null);
   253→    setLunasToggle(false);
   254→    setTanggalPelunasan('');
   255→    setTanggalJatuhTempo('');
   256→    setCaraPembayaran('');
   257→    setTanggalGiro('');
   258→    setOriginalDpAmount(0);
   259→  };
   260→
   261→  // Update invoice data locally
   262→  const updateInvoice = (updates: Partial<InvoiceData>) => {
   263→    setInvoiceData(prev => prev ? { ...prev, ...updates } : null);
   264→  };
   265→
   266→  const updateCompany = (company: CompanyInfo) => {
   267→    setInvoiceData(prev => prev ? { ...prev, company } : null);
   268→  };
   269→
   270→  const updateClient = (field: string, value: string) => {
   271→    setInvoiceData(prev => prev ? { ...prev, client: { ...prev.client, [field]: value } } : null);
   272→  };
   273→
   274→  // Build preview data with pelunasan overrides
   275→  const previewData: InvoiceData | null = useMemo(() => {
   276→    if (!invoiceData) {
   277→      // Default preview when no invoice is selected
   278→      return {
   279→        type: 'invoice',
   280→        company: { ...DEFAULT_COMPANY },
   281→        nomor: '-',
   282→        tanggal: getTodayStr(),
   283→        referensi: '',
   284→        client: { nama: '', kontak: '', alamat: '' },
   285→        items: [],
   286→        ppn: 11,
   287→        dp: 0,
   288→        catatan: '',
   289→        tanggalJatuhTempo: '',
   290→        caraPembayaran: '',
   291→        tanggalGiro: '',
   292→        lunas: false,
   293→        tanggalPelunasan: '',
   294→      };
   295→    }
   296→    return {
   297→      ...invoiceData,
   298→      tanggalJatuhTempo,
   299→      caraPembayaran: caraPembayaran as InvoiceData['caraPembayaran'],
   300→      tanggalGiro,
   301→      lunas: lunasToggle,
   302→      tanggalPelunasan: lunasToggle ? tanggalPelunasan : '',
   303→    };
   304→  }, [invoiceData, tanggalJatuhTempo, caraPembayaran, tanggalGiro, lunasToggle, tanggalPelunasan]);
   305→
   306→  // Calculate amounts — DP is FIXED at original amount, only subtotal and sisa change
   307→  const subtotal = invoiceData?.items.reduce((sum, item) => sum + item.qty * item.harga, 0) || 0;
   308→  const ppnAmount = subtotal * ((invoiceData?.ppn || 0) / 100);
   309→  const total = subtotal + ppnAmount;
   310→  const dpPercent = invoiceData?.dp || 0;
   311→  // DP amount stays fixed at original value, doesn't recalculate when items change
   312→  const dpAmount = originalDpAmount;
   313→  const sisa = total - dpAmount;
   314→
   315→  // Generate JPG from preview and send to WhatsApp
   316→  const handleGenerateJpg = async () => {
   317→    const previewEl = document.querySelector('[data-document-preview]') as HTMLElement;
   318→    if (!previewEl) {
   319→      toast.error('Preview tidak ditemukan');
   320→      return;
   321→    }
   322→    setJpgGenerating(true);
   323→    try {
   324→      const dataUrl = await toJpeg(previewEl, {
   325→        quality: 0.95,
   326→        pixelRatio: 2,
   327→        backgroundColor: '#ffffff',
   328→      });
   329→
   330→      // Convert data URL to Blob for sharing
   331→      const res = await fetch(dataUrl);
   332→      const blob = await res.blob();
   333→      const fileName = `Invoice-PEL-${invoiceData?.nomor || 'draft'}.jpg`;
   334→      const file = new File([blob], fileName, { type: 'image/jpeg' });
   335→
   336→      // Try Web Share API (mobile) — can share file directly to WhatsApp
   337→      if (navigator.share && navigator.canShare?.({ files: [file] })) {
   338→        await navigator.share({
   339→          files: [file],
   340→          title: `Invoice ${invoiceData?.nomor || ''}`,
   341→          text: `Berikut invoice ${invoiceData?.nomor || ''} dari ${invoiceData?.company?.nama || ''}`,
   342→        });
   343→        toast.success('JPG berhasil dikirim ke WhatsApp');
   344→      } else {
   345→        // Fallback (desktop): download JPG + open WhatsApp link
   346→        const link = document.createElement('a');
   347→        link.download = fileName;
   348→        link.href = dataUrl;
   349→        link.click();
   350→
   351→        const phone = invoiceData?.client?.kontak?.replace(/\D/g, '') || '';
   352→        const message = encodeURIComponent(
   353→          `Berikut invoice ${invoiceData?.nomor || ''} dari ${invoiceData?.company?.nama || ''}\n\nJPG invoice sudah didownload, silakan lampirkan ke chat ini.`
   354→        );
   355→        const waUrl = phone
   356→          ? `https://wa.me/${phone}?text=${message}`
   357→          : `https://wa.me/?text=${message}`;
   358→        window.open(waUrl, '_blank');
   359→        toast.success('JPG didownload & WhatsApp terbuka');
   360→      }
   361→    } catch (err) {
   362→      console.error('Failed to generate JPG:', err);
   363→      toast.error('Gagal membuat JPG');
   364→    } finally {
   365→      setJpgGenerating(false);
   366→    }
   367→  };
   368→
   369→  // Save pelunasan — update the existing history entry
   370→  const handleSavePelunasan = async () => {
   371→    if (!selectedEntry || !invoiceData) return;
   372→    setPelunasanSaving(true);
   373→    try {
   374→      const parsed = JSON.parse(selectedEntry.dataJson);
   375→      // Save originalTotal if not already saved — must be done BEFORE updating items
   376→      // so we capture the total from the original items only
   377→      if (parsed.originalTotal === undefined) {
   378→        const origItems = parsed.items || [];
   379→        const origSub = origItems.reduce((s: number, it: { qty: number; harga: number }) => s + it.qty * it.harga, 0);
   380→        const origTot = origSub + (origSub * (parsed.ppn || 0) / 100);
   381→        parsed.originalTotal = origTot;
   382→      }
   383→      // Update pelunasan fields
   384→      parsed.tanggalJatuhTempo = tanggalJatuhTempo;
   385→      parsed.caraPembayaran = caraPembayaran;
   386→      parsed.tanggalGiro = tanggalGiro;
   387→      if (lunasToggle) {
   388→        parsed.lunas = true;
   389→        parsed.tanggalPelunasan = tanggalPelunasan || getTodayStr();
   390→      } else {
   391→        parsed.lunas = false;
   392→        parsed.tanggalPelunasan = '';
   393→      }
   394→      // Also update any edited fields from the form
   395→      parsed.company = invoiceData.company;
   396→      parsed.client = invoiceData.client;
   397→      parsed.items = invoiceData.items;
   398→      parsed.ppn = invoiceData.ppn;
   399→      parsed.dp = invoiceData.dp;
   400→      // Save fixed DP amount so it doesn't change on restore
   401→      parsed.dpAmount = originalDpAmount;
   402→      parsed.catatan = invoiceData.catatan;
   403→      parsed.nomor = invoiceData.nomor;
   404→      parsed.tanggal = invoiceData.tanggal;
   405→      parsed.referensi = invoiceData.referensi;
   406→      delete parsed.statusPembayaran;
   407→
   408→      const newDataJson = JSON.stringify(parsed);
   409→      const res = await fetcher(`/api/history/${selectedEntry.id}`, {
   410→        method: 'PUT',
   411→        headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   412→        body: JSON.stringify({ dataJson: newDataJson }),
   413→      });
   414→      if (res.ok) {
   415→        toast.success(lunasToggle ? 'Pelunasan berhasil dicatat!' : 'Invoice berhasil diperbarui');
   416→        clearSelection();
   417→        fetchHistory();
   418→        notifyDataChange('invoice');
   419→      } else {
   420→        toast.error('Gagal menyimpan perubahan');
   421→      }
   422→    } catch {
   423→      toast.error('Gagal menyimpan perubahan');
   424→    } finally {
   425→      setPelunasanSaving(false);
   426→    }
   427→  };
   428→
   429→  return (
   430→    <DocumentEditorLayout
   431→      title="Invoice Pelunasan"
   432→      previewContent={<InvoicePreview data={previewData!} showPelunasanLabel dpAmountOverride={originalDpAmount} />}
   433→      actions={
   434→        invoiceData ? (
   435→          <div className="flex items-center gap-2 flex-wrap">
   436→            <Button
   437→              variant="outline"
   438→              size="sm"
   439→              onClick={clearSelection}
   440→              className="border-gray-200 text-gray-600 hover:bg-gray-50"
   441→            >
   442→              <X className="mr-1.5 h-3.5 w-3.5" />
   443→              Batal
   444→            </Button>
   445→            <Button
   446→              size="sm"
   447→              onClick={() => {
   448→                toast.dismiss();
   449→                const origTitle = document.title;
   450→                document.title = ' ';
   451→                setTimeout(() => {
   452→                  window.print();
   453→                  document.title = origTitle;
   454→                }, 100);
   455→              }}
   456→              className="bg-emerald-600 hover:bg-emerald-700"
   457→            >
   458→              Cetak
   459→            </Button>
   460→            <Button
   461→              size="sm"
   462→              onClick={handleGenerateJpg}
   463→              disabled={jpgGenerating}
   464→              className="bg-violet-600 hover:bg-violet-700 text-white"
   465→            >
   466→              {jpgGenerating ? (
   467→                <><Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" /> Membuat...</>
   468→              ) : (
   469→                <><ImageIcon className="mr-1.5 h-3.5 w-3.5" /> JPG</>
   470→              )}
   471→            </Button>
   472→            <Button
   473→              size="sm"
   474→              onClick={handleSavePelunasan}
   475→              disabled={pelunasanSaving}
   476→              className={cn(
   477→                'text-white',
   478→                lunasToggle
   479→                  ? 'bg-green-600 hover:bg-green-700'
   480→                  : 'bg-blue-600 hover:bg-blue-700'
   481→              )}
   482→            >
   483→              {pelunasanSaving ? (
   484→                <><Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" /> Menyimpan...</>
   485→              ) : (
   486→                <><CheckCircle2 className="mr-1.5 h-3.5 w-3.5" /> {lunasToggle ? 'Simpan Pelunasan' : 'Simpan Perubahan'}</>
   487→              )}
   488→            </Button>
   489→          </div>
   490→        ) : undefined
   491→      }
   492→    >
   493→      {/* Invoice Selector */}
   494→      <div className="rounded-lg border bg-card p-3 sm:p-4 shadow-sm">
   495→        <h3 className="mb-3 text-xs font-semibold uppercase tracking-wider text-muted-foreground flex items-center gap-1.5">
   496→          <Wallet className="w-3.5 h-3.5 text-amber-600" />
   497→          Pilih Invoice Belum Lunas
   498→        </h3>
   499→        <Popover open={dropdownOpen} onOpenChange={setDropdownOpen}>
   500→          <PopoverAnchor asChild>
   501→            <div className="relative">
   502→              <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 w-3.5 h-3.5 text-slate-400" />
   503→              <input
