    18→function rp(amount: number): string {
    19→  if (isNaN(amount)) return 'Rp0'
    20→  return 'Rp' + Math.round(amount).toLocaleString('id-ID')
    21→}

    23→function fmtDate(dateStr: string): string {
    24→  if (!dateStr) return '-'
    25→  try {
    26→    const date = new Date(dateStr + 'T00:00:00')
    27→    const bulan = [
    28→      'Januari', 'Februari', 'Maret', 'April', 'Mei', 'Juni',
    29→      'Juli', 'Agustus', 'September', 'Oktober', 'November', 'Desember'
    30→    ]
    31→    return `${date.getDate()} ${bulan[date.getMonth()]} ${date.getFullYear()}`
    32→  } catch {
    33→    return dateStr
    34→  }
    35→}

    37→/** Format date short: 01 Jan 25 */
    38→function fmtDateShort(dateStr: string): string {
    39→  if (!dateStr) return '-'
    40→  try {
    41→    const date = new Date(dateStr + 'T00:00:00')
    42→    return date.toLocaleDateString('id-ID', { day: '2-digit', month: 'short', year: '2-digit' })
    43→  } catch {
    44→    return dateStr
    45→  }
    46→}

    48→/** Try to add a base64 logo image to the PDF. Returns true if successful. */
    49→function tryAddLogoImage(pdf: jsPDF, logoDataUrl: string | undefined, x: number, y: number, size: number): boolean {
    50→  if (!logoDataUrl) return false
    51→  try {
    52→    // Detect format from data URL
    53→    const pngMatch = logoDataUrl.match(/^data:image\/png;base64,/)
    54→    const jpegMatch = logoDataUrl.match(/^data:image\/jpeg;base64,/)
    55→    const jpgMatch = logoDataUrl.match(/^data:image\/jpg;base64,/)
    56→    const webpMatch = logoDataUrl.match(/^data:image\/webp;base64,/)
    57→
    58→    let format: string | null = null
    59→    if (pngMatch) format = 'PNG'
    60→    else if (jpegMatch || jpgMatch) format = 'JPEG'
    61→    else if (webpMatch) format = 'PNG' // jsPDF may handle webp as PNG
    62→
    63→    if (!format) return false
    64→
    65→    pdf.addImage(logoDataUrl, format, x, y, size, size)
    66→    return true
    67→  } catch {
    68→    return false
    69→  }
    70→}

    72→/** Draw a standard document header (logo + company info + title). Returns new Y. */
    73→function drawDocHeader(
    74→  pdf: jsPDF,
    75→  opts: {
    76→    pageW: number; m: number; y: number
    77→    company: InvoiceData['company']
    78→    title: string
    79→    subtitle?: string
    80→    jatuhTempo?: string  // optional due date display in header
    81→  }
    82→): number {
    83→  const { pageW, m, y, company, title, subtitle, jatuhTempo } = opts
    84→
    85→  const logoSize = 12
    86→  const logoX = m
    87→  const logoY = y
    88→
    89→  // Logo — try base64 image first, fallback to bordered box with 2-char initials (matches print)
    90→  const logoAdded = tryAddLogoImage(pdf, company.logo, logoX, logoY, logoSize)
    91→
    92→  if (!logoAdded) {
    93→    // Matches print: black border, transparent bg, black text, 2-char initials
    94→    const companyInitials = (company.nama || 'C').split(/\s+/).map(w => w.charAt(0)).join('').toUpperCase().slice(0, 2)
    95→    pdf.setDrawColor(0, 0, 0)
    96→    pdf.setLineWidth(0.8)
    97→    pdf.roundedRect(logoX, logoY, logoSize, logoSize, 1.2, 1.2, 'S') // stroke only
    98→    pdf.setFontSize(8)
    99→    pdf.setFont('helvetica', 'bold')
   100→    pdf.setTextColor(0, 0, 0)
   101→    pdf.text(companyInitials, logoX + logoSize / 2, logoY + logoSize / 2 + 1.5, { align: 'center' })
   102→  }
   103→
   104→  const infoX = m + 15
   105→
   106→  // Company name — matches print: 12pt bold black
   107→  pdf.setTextColor(0, 0, 0)
   108→  pdf.setFontSize(11)
   109→  pdf.setFont('helvetica', 'bold')
   110→  pdf.text(company.nama || '', infoX, y + 4)
   111→
   112→  // Address — matches print: all text black (CSS forces #000 in print)
   113→  pdf.setFontSize(8)
   114→  pdf.setFont('helvetica', 'normal')
   115→  pdf.setTextColor(0, 0, 0)
   116→  let nextY = y + 8.5
   117→  if (company.alamat) {
   118→    pdf.text(company.alamat, infoX, nextY)
   119→    nextY += 3.5
   120→  }
   121→
   122→  // Telepon & Email
   123→  const contactLine = [company.telepon, company.email].filter(Boolean).join('    ')
   124→  if (contactLine) {
   125→    pdf.text(contactLine, infoX, nextY)
   126→    nextY += 3.5
   127→  }
   128→
   129→  // Bank info — matches print: inline on same line, 9pt
   130→  let bankY = nextY
   131→  if (company.bankName || company.bankName2) {
   132→    pdf.setFontSize(6.5)
   133→    pdf.setTextColor(0, 0, 0)
   134→    const bankParts: string[] = []
   135→    if (company.bankName) {
   136→      bankParts.push(`${company.bankName} ${company.bankAccount} a.n. ${company.bankHolder}`)
   137→    }
   138→    if (company.bankName2) {
   139→      bankParts.push(`${company.bankName2} ${company.bankAccount2} a.n. ${company.bankHolder2}`)
   140→    }
   141→    pdf.text(bankParts.join('   '), infoX, bankY)
   142→    bankY += 3.5
   143→  }
   144→
   145→  // Title (right side) — matches print: 12pt bold black
   146→  const rightX = pageW - m
   147→  pdf.setFontSize(11)
   148→  pdf.setFont('helvetica', 'bold')
   149→  pdf.setTextColor(0, 0, 0)
   150→  pdf.text(title, rightX, y + 5, { align: 'right' })
   151→
   152→  if (subtitle) {
   153→    pdf.setFontSize(6.5)
   154→    pdf.setFont('helvetica', 'normal')
   155→    pdf.setTextColor(0, 0, 0) // print forces black
   156→    pdf.text(subtitle, rightX, y + 9.5, { align: 'right' })
   157→  }
   158→
   159→  // Jatuh Tempo — matches print: shown in header area (right side, below title), black text
   160→  if (jatuhTempo) {
   161→    const jtY = subtitle ? y + 14 : y + 10
   162→    pdf.setFontSize(6.5)
   163→    pdf.setFont('helvetica', 'bold')
   164→    pdf.setTextColor(0, 0, 0) // print forces black
   165→    pdf.text(`Jatuh Tempo: ${fmtDateShort(jatuhTempo)}`, rightX, jtY, { align: 'right' })
   166→  }
   167→
   168→  return Math.max(bankY, y + 20) + 2
   169→}

   271→/** Draw items table with prices — matches preview with border-top/bottom header, no zebra, no outer border. Returns new Y. */
   272→function drawItemsTableWithPrice(
   273→  pdf: jsPDF,
   274→  opts: {
   275→    m: number; cw: number; y: number
   276→    items: { deskripsi: string; qty: number; harga: number }[]
   277→    maxRows: number
   278→  }
   279→): number {
   280→  const { m, cw, y, items, maxRows } = opts
   281→  const tableTop = y
   282→
   283→  // Column widths — adjusted for A5
   284→  const colQty = 14
   285→  const colPrice = 24
   286→  const colTotal = 30
   287→  const colDesc = cw - colQty - colPrice - colTotal
   288→  const minRowH = 6.5
   289→  const lineH = 3.2
   290→  const cellPad = 1.5
   291→
   292→  // Pre-calculate each row's height based on description length
   293→  const rows = [...items]
   294→  while (rows.length < maxRows) {
   295→    rows.push({ deskripsi: '', qty: 0, harga: 0 })
   296→  }
   297→
   298→  const rowHeights: number[] = rows.map((item, i) => {
   299→    if (i >= items.length || !item.deskripsi) return minRowH
   300→    const descLines = pdf.splitTextToSize(item.deskripsi, colDesc - cellPad * 2)
   301→    const neededH = Math.max(minRowH, 3 + descLines.length * lineH)
   302→    return neededH
   303→  })
   304→
   305→  // Header — matches preview: border top/bottom, black text on white bg
   306→  const headerH = minRowH
   307→  pdf.setDrawColor(0, 0, 0)
   308→  pdf.setLineWidth(0.6)
   309→  pdf.line(m, y, m + cw, y) // top border
   310→  pdf.line(m, y + headerH, m + cw, y + headerH) // bottom border
   311→
   312→  pdf.setFontSize(6.5)
   313→  pdf.setFont('helvetica', 'bold')
   314→  pdf.setTextColor(0, 0, 0)
   315→  pdf.text('Qty', m + colQty - cellPad, y + headerH - 2, { align: 'right' })
   316→  pdf.text('Nama Barang', m + colQty + cellPad, y + headerH - 2)
   317→  pdf.text('Harga Satuan', m + colQty + colDesc + colPrice - cellPad, y + headerH - 2, { align: 'right' })
   318→  pdf.text('Jumlah', m + cw - cellPad, y + headerH - 2, { align: 'right' })
   319→
   320→  let curY = y + headerH
   321→
   322→  // Draw rows — no zebra striping, all white bg (matches preview)
   323→  rows.forEach((item, i) => {
   324→    const hasData = i < items.length
   325→    const rowH = rowHeights[i]
   326→
   327→    // No background fill — matches preview (white bg, no zebra)
   328→
   329→    if (hasData) {
   330→      pdf.setTextColor(0, 0, 0)
   331→      pdf.setFontSize(6.5)
   332→      pdf.setFont('helvetica', 'normal')
   333→
   334→      const textY = curY + 4
   335→
   336→      // Qty — right aligned
   337→      pdf.text(String(item.qty), m + colQty - cellPad, textY, { align: 'right' })
   338→
   339→      // Description — multi-line support, left aligned
   340→      const descLines = pdf.splitTextToSize(item.deskripsi || '', colDesc - cellPad * 2)
   341→      pdf.text(descLines[0] || '', m + colQty + cellPad, textY)
   342→      if (descLines.length > 1) {
   343→        for (let li = 1; li < descLines.length; li++) {
   344→          pdf.text(descLines[li], m + colQty + cellPad, textY + li * lineH)
   345→        }
   346→      }
   347→
   348→      // Harga satuan — right aligned
   349→      pdf.setFontSize(6.5)
   350→      pdf.setFont('helvetica', 'normal')
   351→      pdf.text(rp(item.harga), m + colQty + colDesc + colPrice - cellPad, textY, { align: 'right' })
   352→
   353→      // Jumlah — right aligned, bold
   354→      pdf.setFont('helvetica', 'bold')
   355→      pdf.text(rp(item.qty * item.harga), m + cw - cellPad, textY, { align: 'right' })
   356→    }
   357→    curY += rowH
   358→  })
   359→
   360→  // No outer border — matches preview (no border around table)
   361→
   362→  return curY + 3
   363→}

   441→/** Draw totals block (Subtotal, PPN, Total, DP, Sisa). Returns new Y. */
   442→function drawTotals(
   443→  pdf: jsPDF,
   444→  opts: {
   445→    m: number; pageW: number; y: number
   446→    subtotal: number; ppnPercent: number; ppnAmount: number; total: number
   447→    dp?: number  // down payment amount (calculated from percentage)
   448→    dpPercent?: number  // down payment percentage
   449→  }
   450→): number {
   451→  const { m, pageW, y, subtotal, ppnPercent, ppnAmount, total, dp = 0, dpPercent = 0 } = opts
   452→  const rightX = pageW - m
   453→  const totalsW = 58
   454→  const labelX = rightX - totalsW
   455→  const sisa = total - dp
   456→
   457→  // Subtotal — print forces black
   458→  pdf.setFontSize(7)
   459→  pdf.setFont('helvetica', 'normal')
   460→  pdf.setTextColor(0, 0, 0)
   461→  pdf.text('Subtotal', labelX, y)
   462→  pdf.text(rp(subtotal), rightX, y, { align: 'right' })
   463→  let curY = y + 4.5
   464→
   465→  // PPN
   466→  if (ppnPercent > 0) {
   467→    pdf.text(`PPN (${ppnPercent}%)`, labelX, curY)
   468→    pdf.text(rp(ppnAmount), rightX, curY, { align: 'right' })
   469→    curY += 4.5
   470→  }
   471→
   472→  // Total line — matches print: 2px solid #000 border, then TOTAL + amount
   473→  pdf.setDrawColor(0, 0, 0)
   474→  pdf.setLineWidth(0.5)
   475→  pdf.line(labelX - 2, curY, rightX, curY)
   476→  curY += 3.5
   477→
   478→  pdf.setFontSize(8.5)
   479→  pdf.setFont('helvetica', 'bold')
   480→  pdf.setTextColor(0, 0, 0)
   481→  pdf.text('TOTAL', labelX, curY)
   482→  pdf.setFontSize(9)
   483→  pdf.text(rp(total), rightX, curY, { align: 'right' })
   484→  curY += 5
   485→
   486→  // DP (%) + Sisa Pembayaran — matches print exactly
   487→  if (dp > 0) {
   488→    // DP line
   489→    pdf.setFontSize(7)
   490→    pdf.setFont('helvetica', 'normal')
   491→    pdf.setTextColor(0, 0, 0)
   492→    pdf.text(`DP (${dpPercent}%)`, labelX, curY)
   493→    pdf.text(rp(dp), rightX, curY, { align: 'right' })
   494→    curY += 4.5
   495→
   496→    // Sisa Pembayaran line — matches print: 1px solid #000 border, bold
   497→    pdf.setDrawColor(0, 0, 0)
   498→    pdf.setLineWidth(0.3)
   499→    pdf.line(labelX - 2, curY, rightX, curY)
   500→    curY += 3.5
   501→
   502→    pdf.setFontSize(8.5)
   503→    pdf.setFont('helvetica', 'bold')
   504→    pdf.setTextColor(0, 0, 0)
   505→    pdf.text('SISA PEMBAYARAN', labelX, curY)
   506→    pdf.setFontSize(9)
   507→    pdf.text(rp(sisa), rightX, curY, { align: 'right' })
   508→    curY += 5
   509→  }
   510→
   511→  return curY
   512→}

   523→/** Draw catatan box. Returns new Y. */
   524→function drawCatatan(pdf: jsPDF, m: number, cw: number, y: number, catatan: string): number {
   525→  if (!catatan) return y
   526→
   527→  // Split by explicit newlines first (whitespace-pre-wrap behavior)
   528→  const paragraphs = catatan.split('\n')
   529→  const allLines: string[] = []
   530→  for (const para of paragraphs) {
   531→    if (para === '') {
   532→      allLines.push('')
   533→    } else {
   534→      const lines = pdf.splitTextToSize(para, cw - 4)
   535→      allLines.push(...lines)
   536→    }
   537→  }
   538→
   539→  const boxH = Math.max(10, 5 + allLines.length * 3.5 + 2)
   540→
   541→  pdf.setFillColor(245, 245, 245)
   542→  pdf.roundedRect(m, y, cw, boxH, 1, 1, 'F')
   543→
   544→  pdf.setFontSize(6.5)
   545→  pdf.setFont('helvetica', 'bold')
   546→  pdf.setTextColor(0, 0, 0)
   547→  pdf.text('Catatan:', m + 2, y + 4)
   548→
   549→  pdf.setFont('helvetica', 'normal')
   550→  pdf.setFontSize(6.5)
   551→  pdf.text(allLines, m + 2, y + 7.5)
   552→
   553→  return y + boxH + 3
   554→}

   594→function drawSignatures2Col(
   595→  pdf: jsPDF,
   596→  opts: {
   597→    m: number; cw: number; y: number
   598→    leftLabel: string; rightLabel: string
   599→    gap?: number  // mm between label and line (default 20)
   600→  }
   601→): number {
   602→  const { m, cw, y, leftLabel, rightLabel, gap = 20 } = opts
   603→  const sigGap = 12
   604→  const sigW = (cw - sigGap) / 2
   605→  const sigL = m
   606→  const sigR = m + sigW + sigGap

   608→  pdf.setFontSize(8)
   609→  pdf.setFont('helvetica', 'bold')
   610→  pdf.setTextColor(0, 0, 0)
   611→  pdf.text(leftLabel, sigL + sigW / 2, y, { align: 'center' })
   612→  pdf.text(rightLabel, sigR + sigW / 2, y, { align: 'center' })
   613→
   614→  const lineY = y + gap
   615→  pdf.setDrawColor(0, 0, 0)
   616→  pdf.setLineWidth(0.3)
   617→  pdf.line(sigL + sigW * 0.15, lineY, sigL + sigW * 0.85, lineY)
   618→  pdf.line(sigR + sigW * 0.15, lineY, sigR + sigW * 0.85, lineY)
   619→
   620→  return lineY + 3
   621→}

  1037→export async function sharePdfViaWhatsApp(
  1038→  blob: Blob,
  1039→  fileName: string,
  1040→  documentLabel: string,
  1041→  waWindowRef?: React.MutableRefObject<Window | null>
  1042→): Promise<void> {
  1043→  if (!blob || !(blob instanceof Blob)) {
  1044→    throw new Error('Invalid blob: PDF generation may have failed')

  1046→
  1047→  const isMobile = /Android|iPhone|iPad|iPod/i.test(navigator.userAgent)
  1048→
  1049→  // ===== MOBILE: Share PDF file directly to WhatsApp via Web Share API =====
  1050→  if (isMobile && navigator.share && navigator.canShare) {
  1051→    const file = new File([blob], fileName, { type: 'application/pdf' })
  1052→    if (navigator.canShare({ files: [file] })) {
  1053→      try {
  1054→        await navigator.share({
  1055→          files: [file],
  1056→          text: `Dokumen ${documentLabel} - www.darrellsoft.com`,
  1057→        })
  1058→        return
  1059→      } catch (err: unknown) {
  1060→        if (err instanceof Error && err.name === 'AbortError') return
  1061→        // Fall through to download + open WhatsApp
  1062→      }
  1063→    }
  1064→  }
  1065→
  1066→  // ===== DESKTOP / FALLBACK: Download PDF + open WhatsApp app =====
  1067→  const url = URL.createObjectURL(blob)
  1068→  const a = document.createElement('a')
  1069→  a.href = url
  1070→  a.download = fileName
  1071→  a.style.display = 'none'
  1072→  document.body.appendChild(a)
  1073→  a.click()
  1074→  setTimeout(() => { document.body.removeChild(a); URL.revokeObjectURL(url) }, 5000)
  1075→
  1076→  await new Promise(resolve => setTimeout(resolve, 500))
  1077→
  1078→  const msg = `Dokumen ${documentLabel} dalam format PDF sudah diunduh. Silakan lampirkan file PDF tersebut.`
  1079→  const encoded = encodeURIComponent(msg)
  1080→
  1081→  const { openWhatsApp } = await import('@/lib/whatsapp-business')
  1082→  openWhatsApp(encoded, waWindowRef ? { waWindowRef } : undefined)
  1083→}

  1095→// ============================================================
  1096→// Generate JPG from PDF (hasil cetak) — A5 size
  1097→// ============================================================
  1098→
  1099→/**
  1100→ * Generate a JPG image from a PDF blob at A5 resolution.
  1101→ * Renders the first page of the PDF to a canvas using pdfjs-dist,
  1102→ * then converts to JPG. This produces an image that matches
  1103→ * the print/cetak output exactly.
  1104→ *
  1105→ * A5 = 148mm × 210mm. At 2×150 DPI → ~1750 × 2480 px.
  1106→ */
  1107→export async function generateJpgFromPdf(pdfBlob: Blob): Promise<Blob> {
  1108→  const pdfjsLib = await import('pdfjs-dist')
  1109→
  1110→  // Use the worker file copied to /public/pdf.worker.min.mjs
  1111→  pdfjsLib.GlobalWorkerOptions.workerSrc = '/pdf.worker.min.mjs'
  1112→
  1113→  const arrayBuffer = await pdfBlob.arrayBuffer()
  1114→  const pdf = await pdfjsLib.getDocument({ data: new Uint8Array(arrayBuffer) }).promise
  1115→  const page = await pdf.getPage(1)
  1116→
  1117→  // A5 at 2×150 DPI for high quality
  1118→  const A5_W_MM = 148
  1119→  const A5_H_MM = 210
  1120→  const DPI = 150
  1121→  const SCALE = 2
  1122→  const targetW = Math.round(A5_W_MM * DPI * SCALE / 25.4) // ~1750px
  1123→  const targetH = Math.round(A5_H_MM * DPI * SCALE / 25.4) // ~2480px
  1124→
  1125→  // Calculate scale to fit A5 target size
  1126→  const viewport = page.getViewport({ scale: 1 })
  1127→  const scaleToTarget = Math.min(targetW / viewport.width, targetH / viewport.height)
  1128→  const scaledViewport = page.getViewport({ scale: scaleToTarget })
  1129→
  1130→  // Create canvas and render
  1131→  const canvas = document.createElement('canvas')
  1132→  canvas.width = Math.round(scaledViewport.width)
  1133→  canvas.height = Math.round(scaledViewport.height)
  1134→  const ctx = canvas.getContext('2d')!
  1135→
  1136→  // Fill white background first
  1137→  ctx.fillStyle = '#ffffff'
  1138→  ctx.fillRect(0, 0, canvas.width, canvas.height)
  1139→
  1140→  await page.render({
  1141→    canvasContext: ctx,
  1142→    viewport: scaledViewport,
  1143→  }).promise
  1144→
  1145→  return new Promise<Blob>((resolve, reject) => {
  1146→    canvas.toBlob(
  1147→      (blob) => {
  1148→        if (blob) resolve(blob)
  1149→        else reject(new Error('Failed to generate JPG from PDF'))
  1150→      },
  1151→      'image/jpeg',
  1152→      0.92
  1153→    )
  1154→  })
  1155→}

  1473→    const origScaleParentStyle = scaleParent?.getAttribute('style') || ''
  1474→    const origScaleParentClass = scaleParent?.className || ''
  1475→    const origCenteringParentStyle = centeringParent?.getAttribute('style') || ''
  1476→    const origCenteringParentClass = centeringParent?.className || ''
  1477→
  1478→    try {
  1479→      // Apply print-mode to the element
  1480→      element.classList.add('print-mode')
  1481→      element.style.cssText = `
  1482→        box-shadow: none !important;
  1483→        border: none !important;
  1484→        border-radius: 0 !important;
  1485→        margin: 0 !important;
  1486→        background-color: #fff !important;
  1487→        width: 100% !important;
  1488→        overflow: hidden !important;
  1489→        padding: ${marginPx}px !important;
  1490→        box-sizing: border-box !important;
  1491→      `
  1492→
  1493→      // Set scale parent to A5 size, remove transform
  1494→      if (scaleParent) {
  1495→        scaleParent.classList.add('print-mode')
  1496→        scaleParent.style.cssText = `
  1497→          position: fixed !important;
  1498→          left: 0 !important;
  1499→          top: 0 !important;
  1500→          width: ${a5WidthPx}px !important;
  1501→          height: ${a5HeightPx}px !important;
  1502→          max-width: ${a5WidthPx}px !important;
  1503→          max-height: ${a5HeightPx}px !important;
  1504→          border: none !important;
  1505→          border-radius: 0 !important;
  1506→          box-shadow: none !important;
  1507→          overflow: hidden !important;
  1508→          background: white !important;
  1509→          z-index: 99999 !important;
  1510→          transform: none !important;
  1511→          font-size: 10pt !important;
  1512→        `
  1513→      }
  1514→
  1515→      // Make centering parent not interfere
  1516→      if (centeringParent) {
  1517→        centeringParent.style.cssText = `
  1518→          position: static !important;
  1519→          padding: 0 !important;
  1520→          margin: 0 !important;
  1521→        `
  1522→      }
  1523→
  1524→      await new Promise(resolve => setTimeout(resolve, 200))
  1525→
  1526→      const CAPTURE_SCALE = 2
  1527→      const captureTarget = scaleParent || element
  1528→      canvas = await toCanvas(captureTarget, {
  1529→        width: a5WidthPx,
  1530→        height: a5HeightPx,
  1531→        canvasWidth: a5WidthPx * CAPTURE_SCALE,
  1532→        canvasHeight: a5HeightPx * CAPTURE_SCALE,
  1533→        backgroundColor: '#ffffff',
  1534→        pixelRatio: 1,
  1535→      })
  1536→    } finally {
  1537→      element.setAttribute('style', origElementStyle)
  1538→      element.className = origElementClass
  1539→      if (scaleParent) {
  1540→        scaleParent.setAttribute('style', origScaleParentStyle)
  1541→        scaleParent.className = origScaleParentClass
  1542→      }
  1543→      if (centeringParent) {
  1544→        centeringParent.setAttribute('style', origCenteringParentStyle)
  1545→        centeringParent.className = origCenteringParentClass
  1546→      }
  1547→    }
  1548→  }
  1549→
  1550→  // Create A5 PDF and embed image
  1551→  const pdf = new jsPDF({ orientation: 'p', unit: 'mm', format: 'a5' })
  1552→  const pageW = pdf.internal.pageSize.getWidth()  // 148
  1553→  const pageH = pdf.internal.pageSize.getHeight() // 210
  1554→
  1555→  const imgData = canvas.toDataURL('image/jpeg', 0.92)
  1556→  pdf.addImage(imgData, 'JPEG', 0, 0, pageW, pageH)
  1557→
  1558→  return pdf.output('blob')
  1559→}

  1561→/**
  1562→ * Share a JPG image via WhatsApp.
  1563→ * On mobile: uses Web Share API to share directly to WhatsApp.
  1564→ * On desktop: downloads the JPG and opens WhatsApp with a message.
  1565→ */
  1566→export async function shareJpgViaWhatsApp(
  1567→  blob: Blob,
  1568→  fileName: string,
  1569→  documentLabel: string,
  1570→  waWindowRef?: React.MutableRefObject<Window | null>
  1571→): Promise<void> {
  1572→  if (!blob || !(blob instanceof Blob)) {
  1573→    throw new Error('Invalid blob: JPG generation may have failed')
  1574→  }
  1575→
  1576→  const isMobile = /Android|iPhone|iPad|iPod/i.test(navigator.userAgent)
  1577→
  1578→  // ===== MOBILE: Share JPG file directly to WhatsApp via Web Share API =====
  1579→  if (isMobile && navigator.share && navigator.canShare) {
  1580→    const file = new File([blob], fileName, { type: 'image/jpeg' })
  1581→    if (navigator.canShare({ files: [file] })) {
  1582→      try {
  1583→        await navigator.share({
  1584→          files: [file],
  1585→          text: `${documentLabel} - www.darrellsoft.com`,
  1586→        })
  1587→        return
  1588→      } catch (err: unknown) {
  1589→        if (err instanceof Error && err.name === 'AbortError') return
  1590→        // Fall through to download + open WhatsApp
  1591→      }
  1592→    }
  1593→  }
  1594→
  1595→  // ===== DESKTOP / FALLBACK: Download JPG + open WhatsApp app =====
  1596→  const url = URL.createObjectURL(blob)
  1597→  const a = document.createElement('a')
  1598→  a.href = url
  1599→  a.download = fileName
  1600→  a.style.display = 'none'
  1601→  document.body.appendChild(a)
  1602→  a.click()
  1603→  setTimeout(() => { document.body.removeChild(a); URL.revokeObjectURL(url) }, 5000)
  1604→
  1605→  await new Promise(resolve => setTimeout(resolve, 500))
  1606→
  1607→  const msg = `Dokumen ${documentLabel} dalam format gambar sudah diunduh. Silakan lampirkan file gambar tersebut.`
  1608→  const encoded = encodeURIComponent(msg)
  1609→
  1610→  const { openWhatsApp } = await import('@/lib/whatsapp-business')
  1611→  openWhatsApp(encoded, waWindowRef ? { waWindowRef } : undefined)
  1612→}
