    53→// --- Date filter ---
    54→type FilterType = 'today' | 'week' | 'month' | 'custom'

    56→function getFilterDates(filter: FilterType, customStart?: Date, customEnd?: Date): { startDate: string; endDate: string } {
    57→  const today = new Date()
    58→  const fmt = (d: Date) => {
    59→    const y = d.getFullYear()
    60→    const m = String(d.getMonth() + 1).padStart(2, '0')
    61→    const day = String(d.getDate()).padStart(2, '0')
    62→    return `${y}-${m}-${day}`
    63→  }
    64→  switch (filter) {
    65→    case 'today': return { startDate: fmt(today), endDate: fmt(today) }
    66→    case 'week': {
    67→      const s = new Date(today)
    68→      const day = s.getDay()
    69→      s.setDate(s.getDate() + (day === 0 ? -6 : 1 - day))
    70→      return { startDate: fmt(s), endDate: fmt(today) }
    71→    }
    72→    case 'month': {
    73→      const s = new Date(today.getFullYear(), today.getMonth(), 1)
    74→      return { startDate: fmt(s), endDate: fmt(today) }
    75→    }
    76→    case 'custom': {
    77→      return {
    78→        startDate: customStart ? fmt(customStart) : fmt(today),
    79→        endDate: customEnd ? fmt(customEnd) : fmt(today),
    80→      }
    81→    }
    82→  }
    83→}

    85→function formatDateShort(iso: string): string {
    86→  return new Date(iso).toLocaleDateString('id-ID', { day: '2-digit', month: 'short', year: 'numeric' })
    87→}

    89→function formatRupiahShort(n: number): string {
    90→  return new Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR', minimumFractionDigits: 0 }).format(n)
    91→}

    93→function getTodayStr(): string {
    94→  const d = new Date()
    95→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
    96→}

    98→interface DocInfo {
    99→  namaCustomer: string;
   100→  namaBarang: string;
   101→  totalQty: number;
   102→  totalHarga: number;
   103→  ppn: number;
   104→  dpPercent: number;
   105→  dp: number;
   106→  sisa: number;
   107→  tanggalJatuhTempo: string;
   108→  catatan: string;
   109→  referensi: string;
   110→  caraPembayaran: string;
   111→  lunas: boolean;
   112→  tanggalPelunasan: string;
   113→}

   115→function parseDocInfo(entry: HistoryEntry): DocInfo {
   116→  try {
   117→    const parsed = JSON.parse(entry.dataJson)
   118→    const items = parsed.items || []
   119→    const client = parsed.client || {}
   120→    const namaCustomer = client.nama || entry.pihakKedua || ''
   121→    const ppn = parsed.ppn || 0
   122→    const dpPercent = parsed.dp || 0
   123→    const tanggalJatuhTempo = parsed.tanggalJatuhTempo || ''
   124→    const catatan = parsed.catatan || ''
   125→    const referensi = parsed.referensi || ''
   126→    const caraPembayaran = parsed.caraPembayaran || ''
   127→    const lunas = parsed.lunas === true
   128→    const tanggalPelunasan = parsed.tanggalPelunasan || ''
   129→
   130→    const firstItem = items[0]
   131→    const namaBarang = firstItem?.deskripsi || ''
   132→
   133→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0)
   134→    const totalHarga = subtotal + (subtotal * ppn / 100)
   135→    const dpAmount = totalHarga * (dpPercent / 100)
   136→    const sisa = totalHarga - dpAmount
   137→
   138→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0)
   139→
   140→    return { namaCustomer, namaBarang, totalQty, totalHarga, ppn, dpPercent, dp: dpAmount, sisa, tanggalJatuhTempo, catatan, referensi, caraPembayaran, lunas, tanggalPelunasan }
   141→  } catch {
   142→    return { namaCustomer: '', namaBarang: '', totalQty: 0, totalHarga: 0, ppn: 0, dpPercent: 0, dp: 0, sisa: 0, tanggalJatuhTempo: '', catatan: '', referensi: '', caraPembayaran: '', lunas: false, tanggalPelunasan: '' }
   143→  }
   144→}

   146→// --- Parse dataJson for InvoiceData ---
   147→function parseInvoiceData(entry: HistoryEntry): InvoiceData {
   148→  try {
   149→    const parsed = JSON.parse(entry.dataJson)
   150→    const company: CompanyInfo = {
   151→      nama: parsed.company?.nama || DEFAULT_COMPANY.nama,
   152→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
   153→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
   154→      email: parsed.company?.email || DEFAULT_COMPANY.email,
   155→      npwp: parsed.company?.npwp || '',
   156→      website: parsed.company?.website || '',
   157→      ppn: parsed.company?.ppn ?? parsed.ppn ?? 11,
   158→      logo: parsed.company?.logo || '',
   159→      bankName: parsed.company?.bankName || '',
   160→      bankAccount: parsed.company?.bankAccount || '',
   161→      bankHolder: parsed.company?.bankHolder || '',
   162→      bankName2: parsed.company?.bankName2 || '',
   163→      bankAccount2: parsed.company?.bankAccount2 || '',
   164→      bankHolder2: parsed.company?.bankHolder2 || '',
   165→    }
   166→    const client = parsed.client || { nama: '', kontak: '', alamat: '' }
   167→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number }, i: number) => ({
   168→      id: it.id || `item-${i}`,
   169→      deskripsi: it.deskripsi || '',
   170→      qty: it.qty || 0,
   171→      satuan: it.satuan || '',
   172→      harga: it.harga || 0,
   173→    }))
   174→    return {
   175→      type: 'invoice',
   176→      company,
   177→      nomor: parsed.nomor || entry.nomor || '',
   178→      tanggal: parsed.tanggal || entry.tanggal || '',
   179→      referensi: parsed.referensi || '',
   180→      client,
   181→      items,
   182→      ppn: parsed.ppn ?? 11,
   183→      dp: parsed.dp || 0,
   184→      catatan: parsed.catatan || '',
   185→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   186→      caraPembayaran: parsed.caraPembayaran || '',
   187→      tanggalGiro: parsed.tanggalGiro || '',
   188→      lunas: parsed.lunas === true,
   189→      tanggalPelunasan: parsed.tanggalPelunasan || '',
   190→    }
   191→  } catch {
   192→    return {
   193→      type: 'invoice',
   194→      company: { ...DEFAULT_COMPANY },
   195→      nomor: entry.nomor || '',
   196→      tanggal: entry.tanggal || '',
   197→      referensi: '',
   198→      client: { nama: '', kontak: '', alamat: '' },
   199→      items: [],
   200→      ppn: 11,
   201→      dp: 0,
   202→      catatan: '',
   203→      tanggalJatuhTempo: '',
   204→      caraPembayaran: '',
   205→      tanggalGiro: '',
   206→      lunas: false,
   207→      tanggalPelunasan: '',
   208→    }
   209→  }
   210→}
