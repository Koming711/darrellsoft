    96→export default function ReportSalesView({ user, onNavigate }: ReportSalesViewProps) {
    97→  const [now] = useState(() => new Date())
    98→  const [period, setPeriod] = useState<ReportPeriod>('month')
    99→  const [month, setMonth] = useState<number | null>(now.getMonth() + 1)
   100→  const [year, setYear] = useState<number | null>(now.getFullYear())
   101→  const [from, setFrom] = useState<string>(() => toInputDate(new Date(now.getFullYear(), now.getMonth(), 1)))
   102→  const [to, setTo] = useState<string>(() => toInputDate(now))
   103→  const [customerId, setCustomerId] = useState<string>('all')
   104→  const [payStatus, setPayStatus] = useState<'all' | 'LUNAS' | 'BELUM'>('all')
   105→  const [searchInput, setSearchInput] = useState('')
   106→  const [q, setQ] = useState('')
   107→
   108→  const [data, setData] = useState<SalesReportResponse | null>(null)
   109→  const [loading, setLoading] = useState(true)
   110→  const [customers, setCustomers] = useState<Customer[]>([])
   111→
   112→  // Dialog edit
   113→  const [editRow, setEditRow] = useState<SalesReportRow | null>(null)
   114→  const [editDate, setEditDate] = useState('')
   115→  const [editStatus, setEditStatus] = useState<InvoiceStatus>('BELUM_BAYAR')
   116→  const [editDueDate, setEditDueDate] = useState('')
   117→  const [editNotes, setEditNotes] = useState('')
   118→  const [saving, setSaving] = useState(false)
   119→
   120→  // Dialog hapus
   121→  const [deleteRow, setDeleteRow] = useState<SalesReportRow | null>(null)
   122→  const [deleting, setDeleting] = useState(false)
   123→
   124→  // Dialog cetak invoice per baris
   125→  const [printId, setPrintId] = useState<string | null>(null)
   126→  const [printInv, setPrintInv] = useState<Invoice | null>(null)
   127→
   128→  const { print, savePdf } = useReportPrint()
   129→
   130→  // Debounce pencarian 400ms
   131→  useEffect(() => {

   134→  }, [searchInput])
   135→

   138→    let cancelled = false
   139→    apiFetch<{ customers: Customer[] }>('/api/customers')
   140→      .then((d) => {
   141→        if (!cancelled) setCustomers(d.customers.filter((c) => c.isActive))
   142→      })
   143→      .catch((e) => {
   144→        if (!cancelled) toast.error(errText(e))
   145→      })

   147→      cancelled = true
   148→    }
   149→  }, [])
   150→

   156→      if (to) p.set('to', to)

   159→      p.set('month', String(month ?? now.getMonth() + 1))
   160→      p.set('year', String(year ?? now.getFullYear()))
   161→    }
   162→    if (period === 'year') {
   163→      p.set('year', String(year ?? now.getFullYear()))
   164→    }
   165→    if (customerId !== 'all') p.set('customerId', customerId)
   166→    if (payStatus !== 'all') p.set('status', payStatus)
   167→    if (q) p.set('q', q)
   168→    return p.toString()

   170→
   171→  const load = useCallback(async () => {
   172→    setLoading(true)
   173→    try {
   174→      const d = await apiFetch<SalesReportResponse>(`/api/reports/sales?${query}`)
   175→      setData(d)
   176→    } catch (e) {
   177→      toast.error(errText(e))
   178→    } finally {
   179→      setLoading(false)
   180→    }
   181→  }, [query])
   182→
   183→  useEffect(() => {
   184→    void load()
   185→  }, [load])
   186→
   187→  // Inisialisasi form edit saat dialog dibuka

   191→    setEditStatus(editRow.status)
   192→    setEditDueDate(editRow.dueDate ? toInputDate(editRow.dueDate) : '')
   193→    setEditNotes(editRow.notes ?? '')
   194→  }, [editRow])
   195→
   196→  // Fetch invoice bentuk penuh saat dialog cetak dibuka
   197→  useEffect(() => {
   198→    if (!printId) return
   199→    let cancelled = false
   200→    setPrintInv(null)

   202→      .then((d) => {
   203→        if (!cancelled) setPrintInv(d.invoice)
   204→      })
   205→      .catch((e) => {
   206→        if (cancelled) return
   207→        toast.error(errText(e))
   208→        setPrintId(null)
   209→      })
   210→    return () => {
   211→      cancelled = true
   212→    }
   213→  }, [printId])
   214→
   215→  const closePrintDialog = useCallback(() => {
   216→    setPrintId(null)
   217→    setPrintInv(null)
   218→  }, [])
   219→
   220→  const saveEdit = async () => {
   221→    if (!editRow) return
   222→    setSaving(true)
   223→    try {
   224→      await apiFetch<{ invoice: Invoice }>(`/api/invoices/${encodeURIComponent(editRow.id)}`, {
   225→        method: 'PUT',
   226→        body: JSON.stringify({
   227→          date: editDate,
   228→          status: editStatus,
   229→          dueDate: editDueDate || '',
   230→          notes: editNotes,
   231→        }),
   232→      })

   235→      await load()
   236→    } catch (e) {
   237→      toast.error(errText(e))
   238→    } finally {
   239→      setSaving(false)
   240→    }

   242→

   248→      toast.success('Invoice dihapus')
   249→      setDeleteRow(null)
   250→      await load()
   251→    } catch (e) {
   252→      // mis. 409: invoice DP masih memiliki pelunasan → pesan dari server
   253→      toast.error(errText(e))
   254→    } finally {
   255→      setDeleting(false)
   256→    }
   257→  }
   258→
   259→  const rows = data?.rows ?? []
   260→  const s = data?.summary
   261→  const periodLabel = periodText(period, from, to, month, year)

   263→
   264→  return (

   279→            <Printer className="h-4 w-4" /> Print
   280→          </Button>

   292→      {/* Filter */}

   294→        <CardContent className="p-4 space-y-4">
   295→          <PeriodFilter
   296→            period={period}
   297→            onChangePeriod={setPeriod}
   298→            from={from}
   299→            to={to}
   300→            onFromChange={setFrom}
   301→            onToChange={setTo}
   302→            month={month}
   303→            onMonthChange={setMonth}
   304→            year={year}
   305→            onYearChange={setYear}
   306→          />
   307→          <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-3">
   308→            <div className="grid gap-1.5">
   309→              <Label htmlFor="report-customer">Pelanggan</Label>

   322→            <div className="grid gap-1.5">
   323→              <Label htmlFor="report-pay-status">Status Pembayaran</Label>

   326→                  <SelectValue placeholder="Semua Status" />
   327→                </SelectTrigger>
   328→                <SelectContent>
   329→                  {PAYMENT_STATUS_OPTIONS.map((o) => (

   332→                </SelectContent>
   333→              </Select>

   335→            <div className="grid gap-1.5 sm:col-span-2 lg:col-span-1">

   339→                <Input
   340→                  id="report-search"
   341→                  type="search"

   346→                  className="pl-9 min-h-[44px]"
   347→                />
   348→              </div>
   349→            </div>
   350→          </div>
   351→        </CardContent>
   352→      </Card>
   353→
   354→      {/* Ringkasan */}
   355→      {loading && !data ? (
   356→        <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-4">
   357→          {[1, 2, 3, 4].map((i) => <Skeleton key={i} className="h-24 w-full rounded-xl" />)}
   358→        </div>
   359→      ) : s ? (
   360→        <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-4">
   361→          <Card className="p-0 gap-0">
   362→            <CardContent className="p-4">
   363→              <p className="text-xs font-medium text-muted-foreground">Total Penjualan</p>
   364→              <p className="text-lg md:text-xl font-bold mt-1">{formatIDR(s.totalPenjualan)}</p>
   365→              <p className="text-[11px] text-muted-foreground mt-1">{s.salesCount} invoice reguler &amp; DP</p>
   366→            </CardContent>
   367→          </Card>
   368→          <Card className="p-0 gap-0">
   369→            <CardContent className="p-4">
   370→              <p className="text-xs font-medium text-muted-foreground">Pembayaran Masuk</p>
   371→              <p className="text-lg md:text-xl font-bold mt-1 text-emerald-700">{formatIDR(s.pembayaranMasuk)}</p>
   372→              <p className="text-[11px] text-muted-foreground mt-1">
   373→                DP {formatIDR(s.totalDp)} · Pelunasan {formatIDR(s.totalPelunasan)} · Reguler lunas {formatIDR(s.regulerLunasTotal)}
   374→              </p>
   375→            </CardContent>
   376→          </Card>
   377→          <Card className="p-0 gap-0">
   378→            <CardContent className="p-4">
   379→              <p className="text-xs font-medium text-muted-foreground">Total Piutang</p>
   380→              <p className="text-lg md:text-xl font-bold mt-1 text-amber-700">{formatIDR(s.totalPiutang)}</p>
   381→              <p className="text-[11px] text-muted-foreground mt-1">Belum dibayar dari invoice reguler &amp; DP</p>
   382→            </CardContent>
   383→          </Card>
   384→          <Card className="p-0 gap-0">
   385→            <CardContent className="p-4">
   386→              <p className="text-xs font-medium text-muted-foreground">Jumlah Invoice</p>
   387→              <p className="text-lg md:text-xl font-bold mt-1">{s.invoiceCount}</p>
   388→              <p className="text-[11px] text-muted-foreground mt-1">{s.salesCount} transaksi penjualan</p>
   389→            </CardContent>
   390→          </Card>
   391→        </div>
   392→      ) : null}
   393→
   394→      {/* Peringatan data terpotong */}
   395→      {data?.truncated && (
   396→        <div className="rounded-lg border border-amber-200 bg-amber-50 px-3 py-2 text-xs text-amber-800" role="alert">
   397→          Menampilkan 500 invoice terbaru dari {data.summary.invoiceCount} — persempit periode untuk melihat semua.
   398→        </div>
   399→      )}
   400→
   401→      {/* Daftar invoice */}
   402→      {loading && !data ? (
   403→        <Skeleton className="h-72 w-full rounded-xl" />
   404→      ) : rows.length === 0 ? (
   405→        <EmptyState />
   406→      ) : (
   407→        <>
   408→          {/* Tabel desktop */}
   409→          <div className="hidden md:block rounded-xl border border-stone-200 bg-white overflow-hidden">
   410→            <div className="max-h-96 overflow-y-auto scrollbar-thin">
   411→              <Table>
   412→                <TableHeader className="sticky top-0 z-10 bg-stone-50">
   413→                  <TableRow className="bg-stone-50 hover:bg-stone-50">
   414→                    <TableHead className="w-10">No.</TableHead>
   415→                    <TableHead>Nomor Invoice</TableHead>
   416→                    <TableHead>Tanggal</TableHead>
   417→                    <TableHead>Customer</TableHead>
   418→                    <TableHead>Jenis Invoice</TableHead>
   419→                    <TableHead className="text-right">Total Penjualan</TableHead>
   420→                    <TableHead className="text-right">DP</TableHead>
   421→                    <TableHead className="text-right">Pelunasan</TableHead>
   422→                    <TableHead className="text-right">Sisa Tagihan</TableHead>
   423→                    <TableHead>Status</TableHead>
   424→                    <TableHead>Aksi</TableHead>
   425→                  </TableRow>
   426→                </TableHeader>
   427→                <TableBody>
   428→                  {rows.map((r, i) => (
   429→                    <TableRow key={r.id} className={cn(r.status === 'BATAL' && 'opacity-60')}>

   431→                      <TableCell>
   432→                        <div className="flex items-center gap-1.5 flex-wrap">
   433→                          <span className="font-mono text-xs">{r.number}</span>
   434→                          {r.type !== 'REGULER' && <TypeBadge type={r.type} />}
   435→                        </div>
   436→                        {r.type === 'PELUNASAN' && r.parentNumber && (
   437→                          <p className="text-[11px] text-muted-foreground mt-0.5">untuk {r.parentNumber}</p>

   439→                      </TableCell>
   440→                      <TableCell className="whitespace-nowrap">{formatDate(r.date)}</TableCell>

   451→                          isAdmin={isAdmin}

   453→                          onEdit={() => setEditRow(r)}
   454→                          onPrint={() => setPrintId(r.id)}
   455→                          onDelete={() => setDeleteRow(r)}
   456→                        />
   457→                      </TableCell>
   458→                    </TableRow>
   459→                  ))}
   460→                </TableBody>
   461→              </Table>
   462→            </div>
   463→          </div>
   464→
   465→          {/* Kartu mobile */}
   466→          <div className="md:hidden space-y-3">
   467→            {rows.map((r) => (
   468→              <Card key={r.id} className={cn('p-0 gap-0', r.status === 'BATAL' && 'opacity-60')}>
   469→                <CardContent className="p-4 space-y-2">
   470→                  <div className="flex items-start justify-between gap-2">
   471→                    <div className="min-w-0">
   472→                      <div className="flex items-center gap-1.5 flex-wrap">
   473→                        <p className="font-mono text-xs font-semibold break-all">{r.number}</p>
   474→                        {r.type !== 'REGULER' && <TypeBadge type={r.type} />}
   475→                      </div>
   476→                      {r.type === 'PELUNASAN' && r.parentNumber && (
   477→                        <p className="text-[11px] text-muted-foreground mt-0.5">untuk {r.parentNumber}</p>
   478→                      )}
   479→                    </div>
   480→                    <StatusBadge status={r.status} dueDate={r.dueDate} />
   481→                  </div>
   482→                  <p className="text-sm">

   485→                  </p>
   486→                  <div className="grid grid-cols-2 gap-x-4 gap-y-1 text-sm border-t border-stone-100 pt-2">
