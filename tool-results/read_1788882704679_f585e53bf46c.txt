    67→    67→// --- Parse dataJson for document info ---
    68→    68→function parseDocInfo(entry: HistoryEntry) {
    69→    69→  try {
    70→    70→    const parsed = JSON.parse(entry.dataJson)
    71→    71→    const items = parsed.items || []
    72→    72→    const firstItem = items[0]
    73→    73→    const namaBarang = firstItem?.deskripsi || ''
    74→    74→    const hargaSatuan = firstItem?.harga || 0
    75→    75→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0)
    76→    76→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0)
    77→    77→    const ppn = parsed.ppn || 0
    78→    78→    const totalHarga = subtotal + (subtotal * ppn / 100)
    79→    79→    const referensi = parsed.referensi || ''
    80→    80→    const tanggalJatuhTempo = parsed.tanggalJatuhTempo || ''
    81→    81→    return { namaBarang, hargaSatuan, totalQty, totalHarga, referensi, tanggalJatuhTempo }
    82→    82→  } catch {
    83→    83→    return { namaBarang: '', hargaSatuan: 0, totalQty: 0, totalHarga: 0, referensi: '', tanggalJatuhTempo: '' }
    84→    84→  }
    85→    85→}

    87→    87→// --- Parse dataJson for PurchaseOrderData ---
    88→    88→function parsePurchaseOrderData(entry: HistoryEntry): PurchaseOrderData {
    89→    89→  try {
    90→    90→    const parsed = JSON.parse(entry.dataJson)
    91→    91→    const company: CompanyInfo = {
    92→    92→      nama: parsed.company?.nama || DEFAULT_COMPANY.nama,
    93→    93→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
    94→    94→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
    95→    95→      email: parsed.company?.email || DEFAULT_COMPANY.email,
    96→    96→      npwp: parsed.company?.npwp || '',
    97→    97→      website: parsed.company?.website || '',
    98→    98→      ppn: parsed.company?.ppn ?? parsed.ppn ?? 0,
    99→    99→      logo: parsed.company?.logo || '',
   100→   100→      bankName: parsed.company?.bankName || '',
   101→   101→      bankAccount: parsed.company?.bankAccount || '',
   102→   102→      bankHolder: parsed.company?.bankHolder || '',
   103→   103→      bankName2: parsed.company?.bankName2 || '',
   104→   104→      bankAccount2: parsed.company?.bankAccount2 || '',
   105→   105→      bankHolder2: parsed.company?.bankHolder2 || '',
   106→   106→    }
   107→   107→    const pemasok = parsed.pemasok || { nama: '', jenisBarang: '', kontak: '', alamat: '' }
   108→   108→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number }, i: number) => ({
   109→   109→      id: it.id || `item-${i}`,
   110→   110→      deskripsi: it.deskripsi || '',
   111→   111→      qty: it.qty || 0,
   112→   112→      satuan: it.satuan || '',
   113→   113→      harga: it.harga || 0,
   114→   114→    }))
   115→   115→    return {
   116→   116→      type: 'purchase-order',
   117→   117→      company,
   118→   118→      nomor: parsed.nomor || entry.nomor || '',
   119→   119→      tanggal: parsed.tanggal || entry.tanggal || '',
   120→   120→      referensi: parsed.referensi || '',
   121→   121→      pemasok,
   122→   122→      items,
   123→   123→      ppn: parsed.ppn ?? 0,
   124→   124→      catatan: parsed.catatan || '',
   125→   125→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   126→   126→      riwayatPotongKertasId: parsed.riwayatPotongKertasId || '',
   127→   127→    }
   128→   128→  } catch {
   129→   129→    return {
   130→   130→      type: 'purchase-order',
   131→   131→      company: { ...DEFAULT_COMPANY },
   132→   132→      nomor: entry.nomor || '',
   133→   133→      tanggal: entry.tanggal || '',
   134→   134→      referensi: '',
   135→   135→      pemasok: { nama: '', jenisBarang: '', kontak: '', alamat: '' },
   136→   136→      items: [],
   137→   137→      ppn: 0,
   138→   138→      catatan: '',
   139→   139→      tanggalJatuhTempo: '',
   140→   140→    }
   141→   141→  }
   142→   142→}

   146→   146→function AutoOpenEditor({ param, onOpen }: { param: string; onOpen: () => void }) {
   147→   147→  const searchParams = useSearchParams()
   148→   148→  const doneRef = useRef(false)
   149→   149→  useEffect(() => {
   150→   150→    if (doneRef.current) return
   151→   151→    if (searchParams.get(param)) {
   152→   152→      doneRef.current = true
   153→   153→      onOpen()
   154→   154→    }
   155→   155→  }, [searchParams, param, onOpen])
   156→   156→  return null
   157→   157→}
