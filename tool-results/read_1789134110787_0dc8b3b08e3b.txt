    77→    77→function toNum(v: string): number | null {
    78→    78→  if (v.trim() === '') return null
    79→    79→  const n = Number(v)
    80→    80→  return Number.isFinite(n) ? n : null
    81→    81→}
    82→    82→
    83→    83→function profitOf(jual: string, modal: string): string {
    84→    84→  const j = toNum(jual)
    85→    85→  const m = toNum(modal)
    86→    86→  return j !== null && m !== null ? String(j - m) : ''
    87→    87→}

    93→    93→}

   208→   208→        return { ...f, hpp: v, standardPrice: String(modal + profit) }

   258→   258→          method: 'POST',

   262→   262→          selectedCustomer

   266→   266→      }
   267→   267→      setDialogOpen(false)
   268→   268→      void load()
   269→   269→    } catch (e) {
   270→   270→      toast.error(errText(e))

   286→   286→      toast.error(errText(e))
   287→   287→    } finally {
   288→   288→      setDeleting(false)
   289→   289→    }
   290→   290→  }

   292→   292→  const hasResults = items.length > 0
   293→   293→  const selectedCustomerName = customers.find((c) => c.id === customerId)?.name ?? null

   296→   296→  const showEdit = canEditItem && customerId !== 'all'

   298→   298→  const showHapus = canDelete
