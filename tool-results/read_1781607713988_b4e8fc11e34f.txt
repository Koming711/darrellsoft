   146→// --- Date filter types ---
   147→type FilterType = 'today' | 'week' | 'month' | 'custom'
   148→
   149→function getFilterDates(filter: FilterType, customStart?: Date, customEnd?: Date): { startDate: string; endDate: string } {
   150→  const today = new Date()
   151→  const formatDate = (d: Date) => {
   152→    const y = d.getFullYear()
   153→    const m = String(d.getMonth() + 1).padStart(2, '0')
   154→    const day = String(d.getDate()).padStart(2, '0')
   155→    return `${y}-${m}-${day}`
   156→  }
   157→
   158→  switch (filter) {
   159→    case 'today': {
   160→      return { startDate: formatDate(today), endDate: formatDate(today) }
   161→    }
   162→    case 'week': {
   163→      const startOfWeek = new Date(today)
   164→      // Get Monday of current week
   165→      const day = startOfWeek.getDay()
   166→      const diff = day === 0 ? -6 : 1 - day
   167→      startOfWeek.setDate(startOfWeek.getDate() + diff)
   168→      return { startDate: formatDate(startOfWeek), endDate: formatDate(today) }
   169→    }
   170→    case 'month': {
   171→      const startOfMonth = new Date(today.getFullYear(), today.getMonth(), 1)
   172→      return { startDate: formatDate(startOfMonth), endDate: formatDate(today) }
   173→    }
   174→    case 'custom': {
   175→      return {
   176→        startDate: customStart ? formatDate(customStart) : formatDate(today),
   177→        endDate: customEnd ? formatDate(customEnd) : formatDate(today),
   178→      }
   179→    }
   180→  }
   181→}

   234→// --- Parse dataJson for document info ---
   235→function parseDocInfo(entry: HistoryEntry) {
   236→  try {
   237→    const parsed = JSON.parse(entry.dataJson);
   238→    const items = parsed.items || [];
   239→    const firstItem = items[0];
   240→    const namaBarang = firstItem?.deskripsi || '';
   241→    const hargaSatuan = firstItem?.harga || 0;
   242→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0);
   243→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0);
   244→    const ppn = parsed.ppn || 0;
   245→    const totalHarga = subtotal + (subtotal * ppn / 100);
   246→    const dp = parsed.dp || 0;
   247→    const originalTotal = parsed.originalTotal !== undefined ? parsed.originalTotal : totalHarga;
   248→    const grandTotal = dp > 0 ? originalTotal : totalHarga;
   249→    const referensi = parsed.referensi || '';
   250→    return { namaBarang, hargaSatuan, totalQty, totalHarga, dp, originalTotal, grandTotal, referensi };
   251→  } catch {
   252→    return { namaBarang: '', hargaSatuan: 0, totalQty: 0, totalHarga: 0, dp: 0, originalTotal: 0, grandTotal: 0, referensi: '' };
   253→  }
   254→}

   372→// Calculate uang capek (profit) for each invoice by matching referensi with riwayat cetakan profitAmount
   373→// Referensi can be either a nomorUrut (e.g. "HC/05/2026/0007") or a printName (e.g. "Brosur")
   374→function calculateUangCapek(invoices: HistoryEntry[], cetakanRecords: { nomorUrut: string; printName: string; profitAmount: number }[]): Map<string, number> {
   375→  const result = new Map<string, number>()
   376→  
   377→  // Build cetakan lookup by both nomorUrut and printName
   378→  const cetakanByRef = new Map<string, number>()
   379→  for (const c of cetakanRecords) {
   380→    // Index by nomorUrut (preferred match)
   381→    if (c.nomorUrut) {
   382→      cetakanByRef.set(c.nomorUrut, (cetakanByRef.get(c.nomorUrut) || 0) + c.profitAmount)
   383→    }
   384→    // Also index by printName (fallback for older data)
   385→    if (c.printName && !cetakanByRef.has(c.printName)) {
   386→      cetakanByRef.set(c.printName, (cetakanByRef.get(c.printName) || 0) + c.profitAmount)
   387→    }
   388→  }
   389→
   390→  // Calculate uang capek per invoice
   391→  for (const inv of invoices) {
   392→    const info = parseDocInfo(inv)
   393→    const uangCapek = info.referensi ? (cetakanByRef.get(info.referensi) || 0) : 0
   394→    result.set(inv.id, uangCapek)
   395→  }
   396→
   397→  return result
   398→}
