    66→function formatRupiah(n: number): string {
    67→  return new Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR', minimumFractionDigits: 0 }).format(n || 0)
    68→}
    69→
    70→function formatDateID(iso: string): string {
    71→  if (!iso) return '-'
    72→  const d = new Date(iso + (iso.length === 10 ? 'T00:00:00' : ''))
    73→  if (isNaN(d.getTime())) return iso
    74→  return d.toLocaleDateString('id-ID', { day: '2-digit', month: 'short', year: 'numeric' })
    75→}
    76→
    77→function todayISO(): string {
    78→  const d = new Date()
    79→  const tz = d.getTimezoneOffset() * 60000
    80→  return new Date(d.getTime() - tz).toISOString().slice(0, 10)
    81→}

   667→/**
   668→ * Panel Selaras Laba Rugi — menampilkan ringkasan Laporan Rugi Laba untuk
   669→ * periode yang aktif difilter (atau bulan berjalan), sehingga pengguna melihat
   670→ * langsung bagaimana biaya di halaman ini mengurangi laba. Sumber data & filter
   671→ * periode identik dengan halaman /laporan/rugi-laba.
   672→ */
   673→function LabaRugiPanel({ filterMonth }: { filterMonth: string }) {
   674→  const [data, setData] = useState<{ labaKotor: number; biayaOperasional: number; labaBersih: number; count: number } | null>(null)
   675→  const [loading, setLoading] = useState(true)
   676→
   677→  const range = useMemo(() => {
   678→    if (filterMonth) {
   679→      const [y, m] = filterMonth.split('-').map(Number)
   680→      const lastDay = new Date(y, m, 0).getDate()
   681→      return { start: `${filterMonth}-01`, end: `${filterMonth}-${String(lastDay).padStart(2, '0')}` }
   682→    }
   683→    const now = new Date()
   684→    const y = now.getFullYear()
   685→    const m = now.getMonth()
   686→    const ym = `${y}-${String(m + 1).padStart(2, '0')}`
   687→    const lastDay = new Date(y, m + 1, 0).getDate()
   688→    return { start: `${ym}-01`, end: `${ym}-${String(lastDay).padStart(2, '0')}` }
   689→  }, [filterMonth])
   690→
   691→  const periodeLabel = useMemo(() => {
   692→    if (filterMonth) {
   693→      const [y, m] = filterMonth.split('-').map(Number)
   694→      return new Date(y, m - 1, 1).toLocaleDateString('id-ID', { month: 'long', year: 'numeric' })
   695→    }
   696→    return 'Bulan Ini'
   697→  }, [filterMonth])
   698→
   699→  const load = useCallback(async () => {
   700→    setLoading(true)
   701→    try {
   702→      const res = await authFetch(`/api/laporan/rugi-laba?start=${range.start}&end=${range.end}`)
   703→      if (!res.ok) throw new Error('gagal memuat')
   704→      const json = await res.json()
   705→      setData({
   706→        labaKotor: Number(json?.summary?.labaKotor) || 0,
   707→        biayaOperasional: Number(json?.summary?.biayaOperasional) || 0,
   708→        labaBersih: Number(json?.summary?.labaBersih) || 0,
   709→        count: Number(json?.biayaCount) || 0,
   710→      })
   711→    } catch {
   712→      setData(null)
   713→    } finally {
   714→      setLoading(false)
   715→    }
   716→  }, [range])
   717→
   718→  useEffect(() => { void load() }, [load])
   719→
   720→  // Selaras: refresh saat biaya / invoice berubah di mana pun
   721→  useDataChange(['biaya', 'invoice'], () => { void load() })
   722→
   723→  const miniStats = [
   724→    { label: 'Laba Kotor', value: data?.labaKotor ?? 0, tone: 'text-slate-800' },
   725→    { label: 'Biaya Operasional', value: -(data?.biayaOperasional ?? 0), tone: 'text-rose-600' },
   726→    { label: 'Laba Bersih', value: data?.labaBersih ?? 0, tone: (data?.labaBersih ?? 0) < 0 ? 'text-red-600' : 'text-emerald-600' },
   727→  ]
   728→
   729→  return (
   730→    <div
   731→      className="bg-card rounded-xl shadow-sm border border-slate-200 mb-4"
   732→      aria-label="Panel selaras dengan laporan rugi laba"
   733→    >
   734→      <div className="p-3 sm:p-4 flex flex-col gap-3 lg:flex-row lg:items-center lg:gap-4">
   735→        <div className="flex items-start gap-2.5 flex-1 min-w-0">
   736→          <div className="w-9 h-9 rounded-lg bg-emerald-50 flex items-center justify-center flex-shrink-0">
   737→            {loading ? (
   738→              <Loader2 className="w-4.5 h-4.5 text-emerald-600 animate-spin" />
   739→            ) : data ? (
   740→              <CheckCircle2 className="w-4.5 h-4.5 text-emerald-600" />
   741→            ) : (
   742→              <FileBarChart className="w-4.5 h-4.5 text-slate-400" />
   743→            )}
   744→          </div>
   745→          <div className="min-w-0">
   746→            <p className="text-sm font-semibold text-slate-800">Selaras dengan Laporan Rugi Laba</p>
   747→            <p className="text-xs text-slate-500 mt-0.5">
   748→              {data
   749→                ? `Periode ${periodeLabel} · ${data.count} transaksi biaya otomatis mengurangi laba — sumber data & periode sama dengan laporan.`
   750→                : 'Ringkasan laba rugi untuk periode ini belum tersedia.'}
   751→            </p>
   752→          </div>
   753→        </div>
   754→
   755→        {loading ? (
   756→          <div className="grid grid-cols-3 gap-2 w-full lg:max-w-md">
   757→            {[1, 2, 3].map((i) => <Skeleton key={i} className="h-14 w-full rounded-lg" />)}
   758→          </div>
   759→        ) : data ? (
   760→          <div className="grid grid-cols-3 gap-2 w-full lg:max-w-md">
   761→            {miniStats.map((s) => (
   762→              <div key={s.label} className="rounded-lg border border-slate-200 bg-slate-50/60 px-2.5 py-2 min-w-0">
   763→                <p className="text-[10px] text-slate-500 truncate">{s.label}</p>
   764→                <p className={`text-xs sm:text-sm font-bold truncate mt-0.5 ${s.tone}`}>{formatRupiah(s.value)}</p>
   765→              </div>
   766→            ))}
   767→          </div>
   768→        ) : null}
   769→
   770→        <Button
   771→          asChild
   772→          variant="outline"
   773→          size="sm"
   774→          className="shrink-0 border-emerald-200 text-emerald-700 hover:bg-emerald-50 hover:text-emerald-800 min-h-[36px]"
   775→        >
   776→          <Link href="/laporan/rugi-laba">
   777→            Buka Rugi Laba <ArrowRight className="w-3.5 h-3.5" aria-hidden="true" />
   778→          </Link>
   779→        </Button>
   780→      </div>
   781→    </div>
   782→  )
   783→}
