   150→function getFilterDates(filter: FilterType, customStart?: Date, customEnd?: Date): { startDate: string; endDate: string } {
   151→  const today = new Date()
   152→  const formatDate = (d: Date) => {
   153→    const y = d.getFullYear()
   154→    const m = String(d.getMonth() + 1).padStart(2, '0')
   155→    const day = String(d.getDate()).padStart(2, '0')
   156→    return `${y}-${m}-${day}`
   157→  }
   158→
   159→  switch (filter) {
   160→    case 'all': {
   161→      // Wide range so both dashboard & history APIs return all records
   162→      return { startDate: '2000-01-01', endDate: '2099-12-31' }
   163→    }
   164→    case 'today': {
   165→      return { startDate: formatDate(today), endDate: formatDate(today) }
   166→    }
   167→    case 'week': {
   168→      const startOfWeek = new Date(today)
   169→      // Get Monday of current week
   170→      const day = startOfWeek.getDay()
   171→      const diff = day === 0 ? -6 : 1 - day
   172→      startOfWeek.setDate(startOfWeek.getDate() + diff)
   173→      return { startDate: formatDate(startOfWeek), endDate: formatDate(today) }
   174→    }
   175→    case 'month': {
   176→      const startOfMonth = new Date(today.getFullYear(), today.getMonth(), 1)
   177→      return { startDate: formatDate(startOfMonth), endDate: formatDate(today) }
   178→    }
   179→    case 'custom': {
   180→      return {
   181→        startDate: customStart ? formatDate(customStart) : formatDate(today),
   182→        endDate: customEnd ? formatDate(customEnd) : formatDate(today),
   183→      }
   184→    }
   185→  }
   186→}

   200→// --- Greeting ---
   201→function getGreetingKey(): 'greeting_morning' | 'greeting_afternoon' | 'greeting_evening' | 'greeting_night' {
   202→  const hour = new Date().getHours()
   203→  if (hour >= 0 && hour < 11) return 'greeting_morning'
   204→  if (hour >= 11 && hour < 15) return 'greeting_afternoon'
   205→  if (hour >= 15 && hour < 18) return 'greeting_evening'
   206→  return 'greeting_night'
   207→}

   242→// --- Parse dataJson for document info ---
   243→function parseDocInfo(entry: HistoryEntry) {
   244→  try {
   245→    const parsed = JSON.parse(entry.dataJson);
   246→    const items = parsed.items || [];
   247→    const firstItem = items[0];
   248→    const namaBarang = firstItem?.deskripsi || '';
   249→    const hargaSatuan = firstItem?.harga || 0;
   250→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0);
   251→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0);
   252→    const ppn = parsed.ppn || 0;
   253→    const totalHarga = subtotal + (subtotal * ppn / 100);
   254→    const dp = parsed.dp || 0;
   255→    const originalTotal = parsed.originalTotal !== undefined ? parsed.originalTotal : totalHarga;
   256→    const grandTotal = dp > 0 ? originalTotal : totalHarga;
   257→    const referensi = parsed.referensi || '';
   258→    return { namaBarang, hargaSatuan, totalQty, totalHarga, dp, originalTotal, grandTotal, referensi };
   259→  } catch {
   260→    return { namaBarang: '', hargaSatuan: 0, totalQty: 0, totalHarga: 0, dp: 0, originalTotal: 0, grandTotal: 0, referensi: '' };
   261→  }
   262→}

   380→// Calculate profit for each invoice by matching referensi with riwayat cetakan profitAmount
   381→// Referensi can be either a nomorUrut (e.g. "HC/05/2026/0007") or a printName (e.g. "Brosur")
   382→function calculateUangCapek(invoices: HistoryEntry[], cetakanRecords: { nomorUrut: string; printName: string; profitAmount: number }[]): Map<string, number> {
   383→  const result = new Map<string, number>()
   384→  
   385→  // Build cetakan lookup by both nomorUrut and printName
   386→  const cetakanByRef = new Map<string, number>()
   387→  for (const c of cetakanRecords) {
   388→    // Index by nomorUrut (preferred match)
   389→    if (c.nomorUrut) {
   390→      cetakanByRef.set(c.nomorUrut, (cetakanByRef.get(c.nomorUrut) || 0) + c.profitAmount)
   391→    }
   392→    // Also index by printName (fallback for older data)
   393→    if (c.printName && !cetakanByRef.has(c.printName)) {
   394→      cetakanByRef.set(c.printName, (cetakanByRef.get(c.printName) || 0) + c.profitAmount)
   395→    }
   396→  }
   397→
   398→  // Calculate profit per invoice
   399→  for (const inv of invoices) {
   400→    const info = parseDocInfo(inv)
   401→    const uangCapek = info.referensi ? (cetakanByRef.get(info.referensi) || 0) : 0
   402→    result.set(inv.id, uangCapek)
   403→  }
   404→
   405→  return result
   406→}
