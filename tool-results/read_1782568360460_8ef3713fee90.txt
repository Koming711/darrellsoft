    23→    23→// Simple id generator (avoid crypto for SSR safety)
    24→    24→function genId() {
    25→    25→  return Date.now().toString(36) + Math.random().toString(36).slice(2, 8)
    26→    26→}

    28→    28→// Very small markdown-ish renderer: supports **bold**, line breaks, and bullet lists
    29→    29→function renderContent(text: string) {
    30→    30→  const lines = text.split('\n')
    31→    31→  return lines.map((line, i) => {
    32→    32→    if (line.trim() === '') return <div key={i} className="h-2" />
    33→    33→    // bold **text**
    34→    34→    const parts = line.split(/(\*\*[^*]+\*\*)/g)
    35→    35→    const rendered = parts.map((part, j) => {
    36→    36→      if (part.startsWith('**') && part.endsWith('**')) {
    37→    37→        return <strong key={j} className="font-semibold">{part.slice(2, -2)}</strong>
    38→    38→      }
    39→    39→      return <span key={j}>{part}</span>
    40→    40→    })
    41→    41→    return <div key={i} className="leading-relaxed">{rendered}</div>
    42→    42→  })
    43→    43→}

    47→    47→  startClientX: number
    48→    48→  startClientY: number

    57→    57→const FAB_STORAGE_KEY = 'ai_fab_offset'
    58→    58→const DRAG_THRESHOLD = 5 // px — pointer must move more than this to count as a drag (not a tap)
    59→    59→const VIEWPORT_MARGIN = 8 // px — keep button at least this far from viewport edges

    61→    61→export function AIAssistant({ bottomOffset }: AIAssistantProps) {
    62→    62→  const { t } = useLanguage()
    63→    63→  const [open, setOpen] = useState(false)
    64→    64→  const [input, setInput] = useState('')
    65→    65→  const [loading, setLoading] = useState(false) // waiting for first token / request in flight
    66→    66→  const [streaming, setStreaming] = useState(false) // actively receiving tokens
    67→    67→  const [messages, setMessages] = useState<Message[]>([])
    68→    68→  const streamingMsgIdRef = useRef<string | null>(null)
    69→    69→  const scrollRef = useRef<HTMLDivElement>(null)
    70→    70→  const textareaRef = useRef<HTMLTextAreaElement>(null)
    71→    71→  const fetchedWelcomeRef = useRef(false)
    72→    72→
    73→    73→  // FAB bottom offset — hydration-safe.
    74→    74→  // Stable initial value (24) matches SSR + initial client render.
    75→    75→  // Updated AFTER mount to clear mobile bottom nav (84) on small screens.
    76→    76→  const [fabBottom, setFabBottom] = useState<number>(24)
    77→    77→  const [isDragging, setIsDragging] = useState(false)
    78→    78→
    79→    79→  // Drag offset via Framer Motion motion values — hydration-safe.
    80→    80→  // Start at 0 on both server & client; updated client-only after mount / during drag.
    81→    81→  // Framer Motion composes `x`/`y` with `scale` (from animate) into a single transform — no conflict.
    82→    82→  const dragX = useMotionValue(0)
    83→    83→  const dragY = useMotionValue(0)
    84→    84→
    85→    85→  const dragRef = useRef<DragState | null>(null)
    86→    86→  const wasDragRef = useRef(false)
    87→    87→  const fabRef = useRef<HTMLDivElement>(null)
    88→    88→
    89→    89→  // After mount: set mobile/desktop offset + restore saved drag position
    90→    90→  useEffect(() => {
    91→    91→    if (bottomOffset !== undefined) {
    92→    92→      setFabBottom(bottomOffset)
    93→    93→    } else {
    94→    94→      setFabBottom(window.innerWidth < 768 ? 84 : 24)
    95→    95→    }
    96→    96→    try {
    97→    97→      const saved = localStorage.getItem(FAB_STORAGE_KEY)
    98→    98→      if (saved) {
    99→    99→        const offset = JSON.parse(saved)
   100→   100→        if (typeof offset.x === 'number' && typeof offset.y === 'number') {
   101→   101→          dragX.set(offset.x)
   102→   102→          dragY.set(offset.y)
   103→   103→        }
   104→   104→      }
   105→   105→    } catch {}
   106→   106→  }, [bottomOffset, dragX, dragY])
   107→   107→
   108→   108→  // Re-clamp position on viewport resize (saved position might now be out of bounds)
   109→   109→  useEffect(() => {
   110→   110→    const handleResize = () => {
   111→   111→      const el = fabRef.current
   112→   112→      if (!el) return
   113→   113→      const rect = el.getBoundingClientRect()
   114→   114→      const currentX = dragX.get()
   115→   115→      const currentY = dragY.get()
   116→   116→      const defaultLeft = rect.left - currentX
   117→   117→      const defaultTop = rect.top - currentY
   118→   118→      const buttonSize = rect.width
   119→   119→      const minX = VIEWPORT_MARGIN - defaultLeft
   120→   120→      const maxX = window.innerWidth - defaultLeft - buttonSize - VIEWPORT_MARGIN
   121→   121→      const minY = VIEWPORT_MARGIN - defaultTop
   122→   122→      const maxY = window.innerHeight - defaultTop - buttonSize - VIEWPORT_MARGIN
   123→   123→      dragX.set(Math.min(Math.max(currentX, minX), maxX))
   124→   124→      dragY.set(Math.min(Math.max(currentY, minY), maxY))
   125→   125→    }
   126→   126→    window.addEventListener('resize', handleResize)
   127→   127→    return () => window.removeEventListener('resize', handleResize)
   128→   128→  }, [dragX, dragY])
   129→   129→
   130→   130→  // Load persisted conversation from sessionStorage
   131→   131→  useEffect(() => {
   132→   132→    if (fetchedWelcomeRef.current) return
   133→   133→    fetchedWelcomeRef.current = true
   134→   134→    try {
   135→   135→      const saved = sessionStorage.getItem('ai_assistant_messages')
   136→   136→      if (saved) {
   137→   137→        const parsed = JSON.parse(saved) as Message[]
   138→   138→        if (Array.isArray(parsed) && parsed.length > 0) {
   139→   139→          setMessages(parsed)
   140→   140→          return
   141→   141→        }
   142→   142→      }
   143→   143→    } catch {}
   144→   144→    // Initial welcome message
   145→   145→    setMessages([
   146→   146→      { id: genId(), role: 'assistant', content: t('ai_assistant_welcome') },
   147→   147→    ])
   148→   148→  }, [t])
   149→   149→
   150→   150→  // Persist messages to sessionStorage
   151→   151→  useEffect(() => {
   152→   152→    if (messages.length > 0) {
   153→   153→      try {
   154→   154→        sessionStorage.setItem('ai_assistant_messages', JSON.stringify(messages))
   155→   155→      } catch {}
   156→   156→    }
   157→   157→  }, [messages])
   158→   158→
   159→   159→  // Auto-scroll to bottom on new message
   160→   160→  useEffect(() => {
   161→   161→    if (scrollRef.current) {
   162→   162→      scrollRef.current.scrollTop = scrollRef.current.scrollHeight
   163→   163→    }
   164→   164→  }, [messages, loading])
   165→   165→
   166→   166→  // Auto-resize textarea
   167→   167→  useEffect(() => {
   168→   168→    if (textareaRef.current) {
   169→   169→      textareaRef.current.style.height = 'auto'
   170→   170→      textareaRef.current.style.height = Math.min(textareaRef.current.scrollHeight, 120) + 'px'
   171→   171→    }
   172→   172→  }, [input])
   173→   173→
   174→   174→  // Escape to close
   175→   175→  useEffect(() => {
   176→   176→    if (!open) return
   177→   177→    const handler = (e: KeyboardEvent) => {
   178→   178→      if (e.key === 'Escape') setOpen(false)
   179→   179→    }
   180→   180→    window.addEventListener('keydown', handler)
   181→   181→    return () => window.removeEventListener('keydown', handler)
   182→   182→  }, [open])
   183→   183→
   184→   184→  const suggestions = [
   185→   185→    t('ai_assistant_suggestion_1'),
   186→   186→    t('ai_assistant_suggestion_2'),
   187→   187→    t('ai_assistant_suggestion_3'),
   188→   188→    t('ai_assistant_suggestion_4'),
   189→   189→    t('ai_assistant_suggestion_5'),
   190→   190→  ]
   191→   191→
   192→   192→  const sendMessage = useCallback(async (text: string) => {
   193→   193→    const trimmed = text.trim()
   194→   194→    if (!trimmed || loading) return
   195→   195→
   196→   196→    const userMsg: Message = { id: genId(), role: 'user', content: trimmed }
   197→   197→    const history = messages
   198→   198→      .filter((m) => !(m.role === 'assistant' && m.id === messages[0]?.id)) // exclude welcome if it's first
   199→   199→      .map((m) => ({ role: m.role, content: m.content }))
   200→   200→
   201→   201→    // Insert user message + an empty assistant placeholder that we'll fill as tokens stream in.
   202→   202→    const aiMsgId = genId()
   203→   203→    streamingMsgIdRef.current = aiMsgId
   204→   204→    setMessages((prev) => [
   205→   205→      ...prev,
   206→   206→      userMsg,
   207→   207→      { id: aiMsgId, role: 'assistant', content: '' },
   208→   208→    ])
   209→   209→    setInput('')
   210→   210→    setLoading(true)
   211→   211→    setStreaming(false)
   212→   212→
   213→   213→    // Safety: if no first token arrives within 25s, show a gentle nudge (but keep waiting).
   214→   214→    let firstTokenTimer: ReturnType<typeof setTimeout> | null = null
   215→   215→    let firstTokenReceived = false
   216→   216→
   217→   217→    try {
   218→   218→      const res = await fetch('/api/ai-assistant', {
   219→   219→        method: 'POST',
   220→   220→        headers: { 'Content-Type': 'application/json', Accept: 'text/event-stream' },
   221→   221→        body: JSON.stringify({ message: trimmed, history }),
   222→   222→      })
   223→   223→
   224→   224→      // Non-streaming error response (JSON)
   225→   225→      const contentType = res.headers.get('content-type') || ''
   226→   226→      if (!res.ok || !contentType.includes('text/event-stream')) {
   227→   227→        let errMsg = t('ai_assistant_error')
   228→   228→        try {
   229→   229→          const data = await res.json()
   230→   230→          if (data?.error) errMsg = data.error
   231→   231→        } catch {}
   232→   232→        throw new Error(errMsg)
   233→   233→      }
   234→   234→
   235→   235→      if (!res.body) {
   236→   236→        throw new Error(t('ai_assistant_error'))
   237→   237→      }
   238→   238→
   239→   239→      const reader = res.body.getReader()
   240→   240→      const decoder = new TextDecoder()
   241→   241→      let buffer = ''
   242→   242→      let accumulated = ''
   243→   243→
   244→   244→      const flushEvents = (eventsChunk: string) => {
   245→   245→        const events = eventsChunk.split('\n\n')
   246→   246→        const leftover = events.pop() || ''
   247→   247→        for (const evt of events) {
   248→   248→          const trimmedEvt = evt.trim()
   249→   249→          if (!trimmedEvt) continue
   250→   250→          for (const line of trimmedEvt.split('\n')) {
   251→   251→            if (!line.startsWith('data:')) continue
   252→   252→            const payload = line.slice(5).trim()
   253→   253→            if (!payload) continue
   254→   254→            if (payload === '[DONE]') return 'done' as const
   255→   255→            try {
   256→   256→              const json = JSON.parse(payload)
   257→   257→              if (json?.error) {
   258→   258→                throw new Error(String(json.error))
   259→   259→              }
   260→   260→              const delta: string | undefined =
   261→   261→                json?.delta ?? json?.choices?.[0]?.delta?.content ?? json?.content
   262→   262→              if (typeof delta === 'string' && delta.length > 0) {
   263→   263→                if (!firstTokenReceived) {
   264→   264→                  firstTokenReceived = true
   265→   265→                  setStreaming(true)
   266→   266→                  setLoading(false)
   267→   267→                  if (firstTokenTimer) {
   268→   268→                    clearTimeout(firstTokenTimer)
   269→   269→                    firstTokenTimer = null
   270→   270→                  }
   271→   271→                }
   272→   272→                accumulated += delta
   273→   273→                const snapshot = accumulated
   274→   274→                setMessages((prev) =>
   275→   275→                  prev.map((m) => (m.id === aiMsgId ? { ...m, content: snapshot } : m))
   276→   276→                )
   277→   277→              }
   278→   278→            } catch {
   279→   279→              // ignore unparseable keepalive lines
   280→   280→            }
   281→   281→          }
   282→   282→        }
   283→   283→        return leftover
   284→   284→      }
   285→   285→
   286→   286→      firstTokenTimer = setTimeout(() => {
   287→   287→        if (!firstTokenReceived && streamingMsgIdRef.current === aiMsgId) {
   288→   288→          setMessages((prev) =>
   289→   289→            prev.map((m) =>
   290→   290→              m.id === aiMsgId
   291→   291→                ? { ...m, content: 'Sedang berpikir… mohon tunggu sebentar.' }
   292→   292→                : m
   293→   293→            )
   294→   294→          )
   295→   295→        }
   296→   296→      }, 25000)
   297→   297→
   298→   298→      while (true) {
   299→   299→        const { done, value } = await reader.read()
   300→   300→        if (done) break
   301→   301→        buffer += decoder.decode(value, { stream: true })
   302→   302→        const result = flushEvents(buffer)
   303→   303→        if (result === 'done') {
   304→   304→          buffer = ''
   305→   305→          break
   306→   306→        }
   307→   307→        buffer = result
   308→   308→      }
   309→   309→      // Flush trailing
   310→   310→      if (buffer.trim()) {
   311→   311→        flushEvents(buffer + '\n\n')
   312→   312→      }
   313→   313→
   314→   314→      if (!firstTokenReceived) {
   315→   315→        // No tokens ever arrived — replace placeholder with an error message
   316→   316→        setMessages((prev) =>
   317→   317→          prev.map((m) =>
   318→   318→            m.id === aiMsgId
   319→   319→              ? { ...m, content: t('ai_assistant_error') }
   320→   320→              : m
   321→   321→          )
   322→   322→        )
   323→   323→        toast.error(t('ai_assistant_error'))
   324→   324→      }
   325→   325→    } catch (err) {
   326→   326→      const errorMsg = err instanceof Error ? err.message : t('ai_assistant_error')
   327→   327→      toast.error(errorMsg)
   328→   328→      setMessages((prev) => {
   329→   329→        // If the placeholder is still empty, replace it; otherwise append a new error msg.
   330→   330→        const placeholder = prev.find((m) => m.id === aiMsgId)
   331→   331→        if (placeholder && placeholder.content === '') {
   332→   332→          return prev.map((m) =>
   333→   333→            m.id === aiMsgId ? { ...m, content: errorMsg } : m
   334→   334→          )
   335→   335→        }
   336→   336→        return [...prev, { id: genId(), role: 'assistant', content: errorMsg }]
   337→   337→      })
   338→   338→    } finally {
   339→   339→      if (firstTokenTimer) clearTimeout(firstTokenTimer)
   340→   340→      streamingMsgIdRef.current = null
   341→   341→      setLoading(false)
   342→   342→      setStreaming(false)
   343→   343→    }
   344→   344→  }, [loading, messages, t])
   345→   345→
   346→   346→  const handleSubmit = (e: FormEvent) => {
   347→   347→    e.preventDefault()
   348→   348→    sendMessage(input)
   349→   349→  }
   350→   350→
   351→   351→  const handleClear = () => {
   352→   352→    streamingMsgIdRef.current = null
   353→   353→    setLoading(false)
   354→   354→    setStreaming(false)
   355→   355→    setMessages([{ id: genId(), role: 'assistant', content: t('ai_assistant_welcome') }])
   356→   356→    try {
   357→   357→      sessionStorage.removeItem('ai_assistant_messages')
   358→   358→    } catch {}
   359→   359→  }
   360→   360→
   361→   361→  // -------------------- Drag handlers --------------------
   362→   362→
   363→   363→  const handlePointerDown = (e: React.PointerEvent) => {
   364→   364→    const el = fabRef.current
   365→   365→    if (!el) return
   366→   366→    // Only respond to primary mouse button; touch & pen always start drag
   367→   367→    if (e.pointerType === 'mouse' && e.button !== 0) return
   368→   368→    const rect = el.getBoundingClientRect()
   369→   369→    const currentX = dragX.get()
   370→   370→    const currentY = dragY.get()
   371→   371→    dragRef.current = {
   372→   372→      pointerId: e.pointerId,
   373→   373→      startClientX: e.clientX,
   374→   374→      startClientY: e.clientY,
   375→   375→      startOffsetX: currentX,
   376→   376→      startOffsetY: currentY,
   377→   377→      moved: false,
   378→   378→      defaultLeft: rect.left - currentX,
   379→   379→      defaultTop: rect.top - currentY,
   380→   380→      buttonSize: rect.width,
   381→   381→    }
   382→   382→    try {
   383→   383→      ;(e.currentTarget as HTMLElement).setPointerCapture(e.pointerId)
   384→   384→    } catch {}
   385→   385→  }
   386→   386→
   387→   387→  const handlePointerMove = (e: React.PointerEvent) => {
   388→   388→    const ds = dragRef.current
   389→   389→    if (!ds || ds.pointerId !== e.pointerId) return
   390→   390→    const dx = e.clientX - ds.startClientX
   391→   391→    const dy = e.clientY - ds.startClientY
   392→   392→    if (!ds.moved && (Math.abs(dx) > DRAG_THRESHOLD || Math.abs(dy) > DRAG_THRESHOLD)) {
   393→   393→      ds.moved = true
   394→   394→      setIsDragging(true)
   395→   395→    }
   396→   396→    if (!ds.moved) return
   397→   397→    const minX = VIEWPORT_MARGIN - ds.defaultLeft
   398→   398→    const maxX = window.innerWidth - ds.defaultLeft - ds.buttonSize - VIEWPORT_MARGIN
   399→   399→    const minY = VIEWPORT_MARGIN - ds.defaultTop
   400→   400→    const maxY = window.innerHeight - ds.defaultTop - ds.buttonSize - VIEWPORT_MARGIN
   401→   401→    const newX = Math.min(Math.max(ds.startOffsetX + dx, minX), maxX)
   402→   402→    const newY = Math.min(Math.max(ds.startOffsetY + dy, minY), maxY)
   403→   403→    dragX.set(newX)
   404→   404→    dragY.set(newY)
   405→   405→  }
   406→   406→
   407→   407→  const handlePointerUp = (e: React.PointerEvent) => {
   408→   408→    const ds = dragRef.current
   409→   409→    if (!ds) return
   410→   410→    wasDragRef.current = ds.moved
   411→   411→    dragRef.current = null
   412→   412→    try {
   413→   413→      ;(e.currentTarget as HTMLElement).releasePointerCapture(ds.pointerId)
   414→   414→    } catch {}
   415→   415→    if (ds.moved) {
   416→   416→      setIsDragging(false)
   417→   417→      try {
   418→   418→        localStorage.setItem(FAB_STORAGE_KEY, JSON.stringify({ x: dragX.get(), y: dragY.get() }))
   419→   419→      } catch {}
   420→   420→    }
   421→   421→  }
   422→   422→
   423→   423→  const handleClick = () => {
   424→   424→    // If the last pointer interaction was a drag (not a tap), don't open the chat.
   425→   425→    if (wasDragRef.current) {
   426→   426→      wasDragRef.current = false
   427→   427→      return
   428→   428→    }
   429→   429→    setOpen(true)
   430→   430→  }
   431→   431→
   432→   432→  // Reset position (double-click / context menu action)
   433→   433→  const handleDoubleClick = () => {
   434→   434→    dragX.set(0)
   435→   435→    dragY.set(0)
   436→   436→    try {
   437→   437→      localStorage.removeItem(FAB_STORAGE_KEY)
   438→   438→    } catch {}
   439→   439→    toast.success('Posisi icon AI direset')
   440→   440→  }
   441→   441→
   442→   442→  return (
   443→   443→    <>
   444→   444→      {/* Floating Action Button (FAB) — draggable */}
   445→   445→      <AnimatePresence>
   446→   446→        {!open && (
   447→   447→          <motion.div
   448→   448→            ref={fabRef}
   449→   449→            initial={{ scale: 0, opacity: 0 }}
   450→   450→            animate={{ scale: 1, opacity: 1 }}
   451→   451→            exit={{ scale: 0, opacity: 0 }}
   452→   452→            transition={{ type: 'spring', stiffness: 260, damping: 20 }}
   453→   453→            className="fixed z-[60] right-4 md:right-6 touch-none select-none"
   454→   454→            style={{
   455→   455→              bottom: fabBottom,
   456→   456→              x: dragX,
   457→   457→              y: dragY,
   458→   458→              cursor: isDragging ? 'grabbing' : 'grab',
   459→   459→            }}
   460→   460→            onPointerDown={handlePointerDown}
   461→   461→            onPointerMove={handlePointerMove}
   462→   462→            onPointerUp={handlePointerUp}
   463→   463→            onPointerCancel={handlePointerUp}
   464→   464→            onDoubleClick={handleDoubleClick}
   465→   465→          >
   466→   466→            <TooltipProvider delayDuration={300}>
   467→   467→              <Tooltip>
   468→   468→                <TooltipTrigger asChild>
   469→   469→                  <button
