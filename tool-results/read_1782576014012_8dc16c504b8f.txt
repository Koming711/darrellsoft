    23→    23→// Simple id generator (avoid crypto for SSR safety)
    24→    24→function genId() {
    25→    25→  return Date.now().toString(36) + Math.random().toString(36).slice(2, 8)

    28→    28→// Very small markdown-ish renderer: supports **bold**, line breaks, and bullet lists
    29→    29→function renderContent(text: string) {
    30→    30→  const lines = text.split('\n')
    31→    31→  return lines.map((line, i) => {
    32→    32→    if (line.trim() === '') return <div key={i} className="h-2" />
    33→    33→    // bold **text**
    34→    34→    const parts = line.split(/(\*\*[^*]+\*\*)/g)
    35→    35→    const rendered = parts.map((part, j) => {
    36→    36→      if (part.startsWith('**') && part.endsWith('**')) {
    37→    37→        return <strong key={j} className="font-semibold">{part.slice(2, -2)}</strong>
    38→    38→      }
    39→    39→      return <span key={j}>{part}</span>
    40→    40→    })
    41→    41→    return <div key={i} className="leading-relaxed">{rendered}</div>
    42→    42→  })
    43→    43→}

    57→    57→const FAB_STORAGE_KEY = 'ai_fab_offset'
    58→    58→// Pointer must move MORE than this (net displacement) to count as a drag, not a tap.
    59→    59→// Higher threshold for touch because fingers naturally jitter more than a mouse.
    60→    60→const DRAG_THRESHOLD_MOUSE = 6
    61→    61→const DRAG_THRESHOLD_TOUCH = 14
    62→    62→const VIEWPORT_MARGIN = 8 // px — keep button at least this far from viewport edges
    63→    63→
    64→    64→function dragThresholdFor(pointerType: string) {
    65→    65→  return pointerType === 'mouse' ? DRAG_THRESHOLD_MOUSE : DRAG_THRESHOLD_TOUCH
    66→    66→}

   216→   216→  const sendMessage = useCallback(async (text: string) => {
   217→   217→    const trimmed = text.trim()
   218→   218→    if (!trimmed || loading) return
   219→   219→
   220→   220→    const userMsg: Message = { id: genId(), role: 'user', content: trimmed }
   221→   221→    const history = messages
   222→   222→      .filter((m) => !(m.role === 'assistant' && m.id === messages[0]?.id)) // exclude welcome if it's first
   223→   223→      .map((m) => ({ role: m.role, content: m.content }))
   224→   224→

   226→   226→    const aiMsgId = genId()
   227→   227→    streamingMsgIdRef.current = aiMsgId
   228→   228→    setMessages((prev) => [
   229→   229→      ...prev,
   230→   230→      userMsg,
   231→   231→      { id: aiMsgId, role: 'assistant', content: '' },
   232→   232→    ])
   233→   233→    setInput('')
   234→   234→    setLoading(true)
   235→   235→    setStreaming(false)
   236→   236→
   237→   237→    // Safety: if no first token arrives within 25s, show a gentle nudge (but keep waiting).
   238→   238→    let firstTokenTimer: ReturnType<typeof setTimeout> | null = null
   239→   239→    let firstTokenReceived = false
   240→   240→
   241→   241→    try {
   242→   242→      const res = await fetch('/api/ai-assistant', {
   243→   243→        method: 'POST',
   244→   244→        headers: { 'Content-Type': 'application/json', Accept: 'text/event-stream' },
   245→   245→        body: JSON.stringify({ message: trimmed, history }),
   246→   246→      })
   247→   247→
   248→   248→      // Non-streaming error response (JSON)
   249→   249→      const contentType = res.headers.get('content-type') || ''
   250→   250→      if (!res.ok || !contentType.includes('text/event-stream')) {
   251→   251→        let errMsg = t('ai_assistant_error')
   252→   252→        try {
   253→   253→          const data = await res.json()
   254→   254→          if (data?.error) errMsg = data.error
   255→   255→        } catch {}
   256→   256→        throw new Error(errMsg)
   257→   257→      }
   258→   258→
   259→   259→      if (!res.body) {
   260→   260→        throw new Error(t('ai_assistant_error'))
   261→   261→      }
   262→   262→
   263→   263→      const reader = res.body.getReader()
   264→   264→      const decoder = new TextDecoder()
   265→   265→      let buffer = ''
   266→   266→      let accumulated = ''
   267→   267→
   268→   268→      let backendError: string | null = null
   269→   269→
   270→   270→      const flushEvents = (eventsChunk: string) => {
   271→   271→        const events = eventsChunk.split('\n\n')
   272→   272→        const leftover = events.pop() || ''
   273→   273→        for (const evt of events) {
   274→   274→          const trimmedEvt = evt.trim()
   275→   275→          if (!trimmedEvt) continue
   276→   276→          for (const line of trimmedEvt.split('\n')) {
   277→   277→            if (!line.startsWith('data:')) continue
   278→   278→            const payload = line.slice(5).trim()
   279→   279→            if (!payload) continue
   280→   280→            if (payload === '[DONE]') return 'done' as const
   281→   281→            // Try JSON parse first; if it fails, treat payload as raw text delta
   282→   282→            let json: any = null
   283→   283→            try {
   284→   284→              json = JSON.parse(payload)
   285→   285→            } catch {
   286→   286→              // Plain-text delta (not JSON) — stream as raw text
   287→   287→              if (payload.length > 0) {
   288→   288→                if (!firstTokenReceived) {
   289→   289→                  firstTokenReceived = true
   290→   290→                  setStreaming(true)
   291→   291→                  setLoading(false)
   292→   292→                  if (firstTokenTimer) {
   293→   293→                    clearTimeout(firstTokenTimer)
   294→   294→                    firstTokenTimer = null
   295→   295→                  }
   296→   296→                }
   297→   297→                accumulated += payload
   298→   298→                const snapshot = accumulated
   299→   299→                setMessages((prev) =>
   300→   300→                  prev.map((m) => (m.id === aiMsgId ? { ...m, content: snapshot } : m))
   301→   301→                )
   302→   302→              }
   303→   303→              continue
   304→   304→            }
   305→   305→            // Backend sent an error event — capture it (don't throw inside try/catch)
   306→   306→            if (json?.error) {
   307→   307→              backendError = typeof json.error === 'string' ? json.error : JSON.stringify(json.error)
   308→   308→              return 'error' as const
   309→   309→            }
   310→   310→            const delta: string | undefined =
   311→   311→              json?.delta ??
   312→   312→              json?.choices?.[0]?.delta?.content ??
   313→   313→              json?.choices?.[0]?.message?.content ??
   314→   314→              json?.content
   315→   315→            if (typeof delta === 'string' && delta.length > 0) {
   316→   316→              if (!firstTokenReceived) {
   317→   317→                firstTokenReceived = true
   318→   318→                setStreaming(true)
   319→   319→                setLoading(false)
   320→   320→                if (firstTokenTimer) {
   321→   321→                  clearTimeout(firstTokenTimer)
   322→   322→                  firstTokenTimer = null
   323→   323→                }
   324→   324→              }
   325→   325→              accumulated += delta
   326→   326→              const snapshot = accumulated
   327→   327→              setMessages((prev) =>
   328→   328→                prev.map((m) => (m.id === aiMsgId ? { ...m, content: snapshot } : m))
   329→   329→              )
   330→   330→            }
   331→   331→          }
   332→   332→        }
   333→   333→        return leftover
   334→   334→      }
   335→   335→
   336→   336→      firstTokenTimer = setTimeout(() => {
   337→   337→        if (!firstTokenReceived && streamingMsgIdRef.current === aiMsgId) {
   338→   338→          setMessages((prev) =>
   339→   339→            prev.map((m) =>
   340→   340→              m.id === aiMsgId
   341→   341→                ? { ...m, content: 'Sedang berpikir… mohon tunggu sebentar.' }
   342→   342→                : m
   343→   343→            )
   344→   344→          )
   345→   345→        }
   346→   346→      }, 25000)
   347→   347→
   348→   348→      while (true) {
   349→   349→        const { done, value } = await reader.read()
   350→   350→        if (done) break
   351→   351→        buffer += decoder.decode(value, { stream: true })
   352→   352→        const result = flushEvents(buffer)
   353→   353→        if (result === 'done' || result === 'error') {
   354→   354→          buffer = ''
   355→   355→          break
   356→   356→        }
   357→   357→        buffer = result
   358→   358→      }
   359→   359→      // Flush trailing
   360→   360→      if (buffer.trim() && backendError === null) {
   361→   361→        flushEvents(buffer + '\n\n')
   362→   362→      }
   363→   363→
   364→   364→      // Surface backend error to the user
   365→   365→      if (backendError) {
   366→   366→        throw new Error(backendError)
   367→   367→      }
   368→   368→
   369→   369→      if (!firstTokenReceived) {
   370→   370→        // No tokens ever arrived — replace placeholder with an error message
   371→   371→        setMessages((prev) =>
   372→   372→          prev.map((m) =>
   373→   373→            m.id === aiMsgId
   374→   374→              ? { ...m, content: t('ai_assistant_error') }
   375→   375→              : m
   376→   376→          )
   377→   377→        )
   378→   378→        toast.error(t('ai_assistant_error'))
   379→   379→      }
   380→   380→    } catch (err) {
   381→   381→      const errorMsg = err instanceof Error ? err.message : t('ai_assistant_error')
   382→   382→      toast.error(errorMsg)
   383→   383→      setMessages((prev) => {
   384→   384→        // If the placeholder is still empty, replace it; otherwise append a new error msg.
   385→   385→        const placeholder = prev.find((m) => m.id === aiMsgId)
   386→   386→        if (placeholder && placeholder.content === '') {
   387→   387→          return prev.map((m) =>
   388→   388→            m.id === aiMsgId ? { ...m, content: errorMsg } : m
   389→   389→          )
   390→   390→        }
   391→   391→        return [...prev, { id: genId(), role: 'assistant', content: errorMsg }]
   392→   392→      })
   393→   393→    } finally {
   394→   394→      if (firstTokenTimer) clearTimeout(firstTokenTimer)
   395→   395→      streamingMsgIdRef.current = null
   396→   396→      setLoading(false)
   397→   397→      setStreaming(false)
   398→   398→    }
   399→   399→  }, [loading, messages, t])
   400→   400→
   401→   401→  const handleSubmit = (e: FormEvent) => {
   402→   402→    e.preventDefault()
   403→   403→    sendMessage(input)
   404→   404→  }
   405→   405→
   406→   406→  const handleClear = () => {
   407→   407→    streamingMsgIdRef.current = null
   408→   408→    setLoading(false)
   409→   409→    setStreaming(false)
   410→   410→    setMessages([{ id: genId(), role: 'assistant', content: t('ai_assistant_welcome') }])
   411→   411→    try {
   412→   412→      sessionStorage.removeItem('ai_assistant_messages')
   413→   413→    } catch {}
   414→   414→  }

   416→   416→  // -------------------- Drag handlers --------------------
   417→   417→
   418→   418→  const handlePointerDown = (e: React.PointerEvent) => {
   419→   419→    const el = fabRef.current
   420→   420→    if (!el) return
   421→   421→    // Only respond to primary mouse button; touch & pen always start drag
   422→   422→    if (e.pointerType === 'mouse' && e.button !== 0) return
   423→   423→    const rect = el.getBoundingClientRect()
   424→   424→    const currentX = dragX.get()
   425→   425→    const currentY = dragY.get()
   426→   426→    dragRef.current = {
   427→   427→      pointerId: e.pointerId,
   428→   428→      startClientX: e.clientX,
   429→   429→      startClientY: e.clientY,
   430→   430→      startOffsetX: currentX,
   431→   431→      startOffsetY: currentY,
   432→   432→      moved: false,
   433→   433→      defaultLeft: rect.left - currentX,
   434→   434→      defaultTop: rect.top - currentY,
   435→   435→      buttonSize: rect.width,
   436→   436→    }
   437→   437→    try {
   438→   438→      ;(e.currentTarget as HTMLElement).setPointerCapture(e.pointerId)
   439→   439→    } catch {}
   440→   440→  }
   441→   441→
   442→   442→  const handlePointerMove = (e: React.PointerEvent) => {
   443→   443→    const ds = dragRef.current
   444→   444→    if (!ds || ds.pointerId !== e.pointerId) return
   445→   445→    const dx = e.clientX - ds.startClientX
   446→   446→    const dy = e.clientY - ds.startClientY
   447→   447→    const threshold = dragThresholdFor(e.pointerType)
   448→   448→    if (!ds.moved && (Math.abs(dx) > threshold || Math.abs(dy) > threshold)) {
   449→   449→      ds.moved = true
   450→   450→      setIsDragging(true)
   451→   451→    }
   452→   452→    if (!ds.moved) return
   453→   453→    const minX = VIEWPORT_MARGIN - ds.defaultLeft
   454→   454→    const maxX = window.innerWidth - ds.defaultLeft - ds.buttonSize - VIEWPORT_MARGIN
   455→   455→    const minY = VIEWPORT_MARGIN - ds.defaultTop
   456→   456→    const maxY = window.innerHeight - ds.defaultTop - ds.buttonSize - VIEWPORT_MARGIN
   457→   457→    const newX = Math.min(Math.max(ds.startOffsetX + dx, minX), maxX)
   458→   458→    const newY = Math.min(Math.max(ds.startOffsetY + dy, minY), maxY)
   459→   459→    dragX.set(newX)
   460→   460→    dragY.set(newY)
   461→   461→  }
   462→   462→
   463→   463→  const handlePointerUp = (e: React.PointerEvent) => {
   464→   464→    const ds = dragRef.current
   465→   465→    if (!ds) return
   466→   466→    // Compute NET displacement from pointerdown to pointerup.
