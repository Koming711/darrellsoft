     1→'use client';
     2→
     3→import { useEffect, useState, useCallback, useRef, useMemo } from 'react';
     4→import { useSearchParams, useRouter } from 'next/navigation';
     5→import { Input } from '@/components/ui/input';
     6→import { Label } from '@/components/ui/label';
     7→import { Textarea } from '@/components/ui/textarea';
     8→import { Popover, PopoverContent, PopoverAnchor } from '@/components/ui/popover';
     9→import { useDokuproStore } from '@/lib/store';
    10→import { CompanyFields } from './company-fields';
    11→import { ItemsFields } from './items-fields';
    12→import { InvoicePreview } from './invoice-preview';
    13→import { DocumentEditorLayout } from './document-editor-layout';
    14→import { DocumentActionButtons } from './document-action-buttons';
    15→import { formatRupiah } from '@/lib/format';
    16→import { getAuthHeaders } from '@/lib/auth';
    17→import { Truck } from 'lucide-react';
    18→import { Button } from '@/components/ui/button';
    19→import { toast } from 'sonner';
    20→import type { InvoiceData } from '@/lib/types';
    21→
    22→interface CustomerItem {
    23→  id: string;
    24→  name: string;
    25→  companyName?: string;
    26→  phone?: string;
    27→  email?: string;
    28→  address?: string;
    29→}
    30→
    31→interface RiwayatCetakanItem {
    32→  id: string;
    33→  nomorUrut: string;
    34→  printName: string;
    35→  customerName: string;
    36→  paperName: string;
    37→  paperGrammage: string;
    38→  cutWidth: string;
    39→  cutHeight: string;
    40→  quantity: string;
    41→  jumlahPesanan: string;
    42→  berapaMata: string;
    43→  warna: string;
    44→  machineName: string;
    45→  ongkosCetak: number;
    46→  machineName2: string;
    47→  ongkosCetak2: number;
    48→  totalPaperPrice: number;
    49→  pricePerSheet: number;
    50→  finishingNames: string;
    51→  finishingCost: number;
    52→  packingCost: number;
    53→  shippingCost: number;
    54→  otherCost: number;
    55→  otherCost2: number;
    56→  otherCostLabel: string;
    57→  otherCostLabel2: string;
    58→  glueCost: number;
    59→  glueBorongan: number;
    60→  subTotal: number;
    61→  profitPercent: number;
    62→  profitAmount: number;
    63→  grandTotal: number;
    64→  createdAt: string;
    65→}
    66→
    67→export function InvoiceEditor() {
    68→  const invoice = useDokuproStore((s) => s.invoice);
    69→  const setInvoice = useDokuproStore((s) => s.setInvoice);
    70→  const resetDocument = useDokuproStore((s) => s.resetDocument);
    71→  const loadCompanyFromAPI = useDokuproStore((s) => s.loadCompanyFromAPI);
    72→  const router = useRouter();
    73→
    74→  const searchParams = useSearchParams();
    75→  const riwayatIdFromUrl = searchParams.get('riwayatId');
    76→  const autoSelectDoneRef = useRef<string | null>(null); // track which riwayatId was auto-selected
    77→  const [savingSj, setSavingSj] = useState(false);
    78→
    79→  // Riwayat cetakan dropdown state
    80→  const [riwayatList, setRiwayatList] = useState<RiwayatCetakanItem[]>([]);
    81→  const [referensiInput, setReferensiInput] = useState(invoice.referensi);
    82→  const [dropdownOpen, setDropdownOpen] = useState(false);
    83→
    84→  // Customer dropdown state
    85→  const [customerList, setCustomerList] = useState<CustomerItem[]>([]);
    86→  const [clientInput, setClientInput] = useState(invoice.client.nama);
    87→  const [clientDropdownOpen, setClientDropdownOpen] = useState(false);
    88→  const [clientTyping, setClientTyping] = useState(false);
    89→
    90→  const fetchRiwayatCetakan = useCallback(async () => {
    91→    try {
    92→      const res = await fetch('/api/riwayat-cetakan', { headers: getAuthHeaders() });
    93→      if (res.ok) {
    94→        const data = await res.json();
    95→        setRiwayatList(data);
    96→        return data;
    97→      }
    98→    } catch {
    99→      // silently fail
   100→    }
   101→    return [];
   102→  }, []);
   103→
   104→  // Fetch customer list
   105→  const fetchCustomers = useCallback(async () => {
   106→    try {
   107→      const res = await fetch('/api/customers', { headers: getAuthHeaders() });
   108→      if (res.ok) {
   109→        const data = await res.json();
   110→        setCustomerList(Array.isArray(data) ? data : []);
   111→      }
   112→    } catch {
   113→      // silently fail
   114→    }
   115→  }, []);
   116→
   117→  useEffect(() => { loadCompanyFromAPI() }, [loadCompanyFromAPI]);
   118→  useEffect(() => { fetchRiwayatCetakan() }, [fetchRiwayatCetakan]);
   119→  useEffect(() => { fetchCustomers() }, [fetchCustomers]);
   120→
   121→  // Fetch next Invoice number from server
   122→  const fetchNextNumber = useCallback(() => {
   123→    fetch('/api/history?preview=next-number&docType=invoice', { headers: getAuthHeaders() })
   124→      .then(res => res.ok ? res.json() : null)
   125→      .then(data => { if (data?.nextNumber) setInvoice((prev) => ({ ...prev, nomor: data.nextNumber })) })
   126→      .catch(() => {});
   127→  }, [setInvoice]);
   128→
   129→  useEffect(() => { fetchNextNumber() }, [fetchNextNumber]);
   130→
   131→  // Re-fetch next number after a document is saved
   132→  useEffect(() => {
   133→    const handler = () => { fetchNextNumber() };
   134→    window.addEventListener('dokupro:history-updated', handler);
   135→    return () => { window.removeEventListener('dokupro:history-updated', handler) };
   136→  }, [fetchNextNumber]);
   137→
   138→  // Sync referensiInput when invoice.referensi changes externally
   139→  useEffect(() => { setReferensiInput(invoice.referensi) }, [invoice.referensi]);
   140→  // Sync clientInput when invoice.client.nama changes externally
   141→  useEffect(() => { setClientInput(invoice.client.nama) }, [invoice.client.nama]);
   142→
   143→  // Filter riwayat list by input
   144→  const filteredRiwayatList = riwayatList.filter((r) => {
   145→    const search = referensiInput.toLowerCase().trim();
   146→    if (!search) return true;
   147→    const noHc = (r.nomorUrut || '').toLowerCase();
   148→    const printName = (r.printName || '').toLowerCase();
   149→    const name = (r.customerName || '').toLowerCase();
   150→    return noHc.includes(search) || printName.includes(search) || name.includes(search);
   151→  });
   152→
   153→  const handleReferensiInputChange = (value: string) => {
   154→    setReferensiInput(value);
   155→    setInvoice((prev) => ({ ...prev, referensi: value }));
   156→    setDropdownOpen(true);
   157→  };
   158→
   159→  // Apply referensi data using functional setInvoice to avoid stale closure
   160→  const applyReferensi = (item: RiwayatCetakanItem) => {
   161→    const ref = item.nomorUrut || item.printName || '';
   162→    setReferensiInput(ref);
   163→
   164→    // Build single item with multi-line description
   165→    const descLines: string[] = [];
   166→    if (item.printName) descLines.push(item.printName);
   167→
   168→    const bahanParts: string[] = [];
   169→    if (item.paperName) bahanParts.push(item.paperName);
   170→    if (item.paperGrammage) bahanParts.push(item.paperGrammage + 'g');
   171→    if (bahanParts.length > 0) descLines.push(bahanParts.join(' '));
   172→
   173→    if (item.finishingNames) {
   174→      const finishings = item.finishingNames.split(',').map(f => f.trim()).filter(Boolean);
   175→      descLines.push('Finishing:');
   176→      finishings.forEach(f => descLines.push(f));
   177→    }
   178→
   179→    const deskripsi = descLines.join('\n');
   180→    const jumlahPesanan = parseInt(item.jumlahPesanan) || 1;
   181→    const hargaPerPcs = jumlahPesanan > 0 ? Math.round(item.grandTotal / jumlahPesanan) : 0;
   182→
   183→    const newItems: InvoiceData['items'] = [{
   184→      id: 'riwayat-0',
   185→      deskripsi: deskripsi || ref,
   186→      qty: jumlahPesanan,
   187→      satuan: 'pcs',
   188→      harga: hargaPerPcs,
   189→    }];
   190→
   191→    // Use functional form to always get the LATEST state
   192→    setInvoice((prev) => ({
   193→      ...prev,
   194→      referensi: ref,
   195→      client: {
   196→        ...prev.client,
   197→        nama: item.customerName || prev.client.nama,
   198→      },
   199→      items: newItems,
   200→      catatan: '',
   201→    }));
   202→    setDropdownOpen(false);
   203→  };
   204→
   205→  const handleReferensiSelect = (item: RiwayatCetakanItem) => {
   206→    applyReferensi(item);
   207→  };
   208→
   209→  // Auto-select referensi when coming from hitung cetakan with riwayatId
   210→  // Fetches the specific riwayat directly by ID for reliability (no race condition with list loading)
   211→  useEffect(() => {
   212→    if (!riwayatIdFromUrl) return;
   213→    if (autoSelectDoneRef.current === riwayatIdFromUrl) return;
   214→
   215→    let cancelled = false;
   216→
   217→    const fetchAndApply = async () => {
   218→      // First, try to find it in the already-loaded list (fast path)
   219→      const foundInList = riwayatList.find((r) => r.id === riwayatIdFromUrl);
   220→      if (foundInList) {
   221→        if (!cancelled) {
   222→          autoSelectDoneRef.current = riwayatIdFromUrl;
   223→          applyReferensi(foundInList);
   224→        }
   225→        return;
   226→      }
   227→
   228→      // Not in list yet — fetch directly by ID from the API (reliable path)
   229→      try {
   230→        const res = await fetch(`/api/riwayat-cetakan?id=${riwayatIdFromUrl}`, { headers: getAuthHeaders() });
   231→        if (res.ok && !cancelled) {
   232→          const data = await res.json();
   233→          if (data && !data.error) {
   234→            autoSelectDoneRef.current = riwayatIdFromUrl;
   235→            applyReferensi(data as RiwayatCetakanItem);
   236→          }
   237→        }
   238→      } catch {
   239→        // Fallback: retry once after a short delay
   240→        if (cancelled) return;
   241→        await new Promise(r => setTimeout(r, 800));
   242→        if (cancelled) return;
   243→        try {
   244→          const res = await fetch(`/api/riwayat-cetakan?id=${riwayatIdFromUrl}`, { headers: getAuthHeaders() });
   245→          if (res.ok) {
   246→            const data = await res.json();
   247→            if (data && !data.error) {
   248→              autoSelectDoneRef.current = riwayatIdFromUrl;
   249→              applyReferensi(data as RiwayatCetakanItem);
   250→            }
   251→          }
   252→        } catch {
   253→          // Give up silently
   254→        }
   255→      }
   256→    };
   257→
   258→    fetchAndApply();
   259→
   260→    return () => { cancelled = true; };
   261→  }, [riwayatIdFromUrl]);
   262→
   263→  // Filter customer list (only when user is actively typing)
   264→  const filteredCustomerList = useMemo(() =>
   265→    clientTyping
   266→      ? customerList.filter((c) => {
   267→          const search = clientInput.toLowerCase().trim();
   268→          if (!search) return true;
   269→          return c.name.toLowerCase().includes(search) ||
   270→                 (c.companyName || '').toLowerCase().includes(search) ||
   271→                 (c.phone || '').toLowerCase().includes(search) ||
   272→                 (c.address || '').toLowerCase().includes(search);
   273→        })
   274→      : customerList
   275→  , [clientTyping, customerList, clientInput]);
   276→
   277→  const handleClientInputChange = (value: string) => {
   278→    setClientInput(value);
   279→    setClientTyping(true);
   280→    updateClient('nama', value);
   281→    setClientDropdownOpen(true);
   282→  };
   283→
   284→  const handleClientSelect = (item: CustomerItem) => {
   285→    setClientInput(item.name);
   286→    setClientTyping(false);
   287→    setInvoice((prev) => ({
   288→      ...prev,
   289→      client: {
   290→        nama: item.name,
   291→        kontak: item.phone || item.email || '',
   292→        alamat: item.address || '',
   293→      },
   294→    }));
   295→    setClientDropdownOpen(false);
   296→  };
   297→
   298→  const updateCompany = (company: typeof invoice.company) => {
   299→    setInvoice((prev) => ({ ...prev, company }));
   300→  };
   301→
   302→  const updateClient = (field: string, value: string) => {
   303→    setInvoice((prev) => ({
   304→      ...prev,
   305→      client: { ...prev.client, [field]: value },
   306→    }));
   307→  };
   308→
   309→  const subtotal = invoice.items.reduce((sum, item) => sum + item.qty * item.harga, 0);
   310→  const ppnAmount = subtotal * (invoice.ppn / 100);
   311→  const total = subtotal + ppnAmount;
   312→  const dpPercent = invoice.dp || 0;
   313→  const dpAmount = total * (dpPercent / 100);
   314→  const sisa = total - dpAmount;
   315→
   316→  const handleSuratJalan = async () => {
   317→    // Check if data has content
   318→    const pihakKedua = invoice.client?.nama;
   319→    const hasItem = invoice.items?.some((item) => item.deskripsi.trim() !== '');
   320→    if (!pihakKedua?.trim() || !hasItem) {
   321→      toast.error('Lengkapi data invoice terlebih dahulu!');
   322→      return;
   323→    }
   324→
   325→    setSavingSj(true);
   326→    try {
   327→      // Save invoice to history first
   328→      const dataToSave = { ...invoice, dpAmount, originalTotal: total };
   329→      const res = await fetch('/api/history', {
   330→        method: 'POST',
   331→        headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   332→        body: JSON.stringify({
   333→          docType: 'invoice',
   334→          nomor: invoice.nomor || '-',
   335→          tanggal: invoice.tanggal || '',
   336→          pihakKedua: invoice.client?.nama || '-',
   337→          total: '-',
   338→          dataJson: JSON.stringify(dataToSave),
   339→        }),
   340→      });
   341→      if (res.ok) {
   342→        const saved = await res.json();
   343→        window.dispatchEvent(new CustomEvent('dokupro:history-updated'));
   344→
   345→        // If invoice has DP, also create an invoice-pelunasan entry
   346→        if (invoice.dp > 0) {
   347→          try {
   348→            const pelunasanData = {
   349→              ...dataToSave,
   350→              type: 'invoice-pelunasan',
   351→              referensiInvoiceId: saved.id,
   352→              referensiInvoiceNomor: saved.nomor || invoice.nomor,
   353→              lunas: false,
   354→              tanggalPelunasan: '',
   355→            };
   356→            await fetch('/api/history', {
   357→              method: 'POST',
   358→              headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   359→              body: JSON.stringify({
   360→                docType: 'invoice-pelunasan',
   361→                tanggal: invoice.tanggal || '',
   362→                pihakKedua: invoice.client?.nama || '-',
   363→                total: '-',
   364→                dataJson: JSON.stringify(pelunasanData),
   365→              }),
   366→            });
   367→          } catch {
   368→            console.error('Failed to create pelunasan invoice entry');
   369→          }
   370→        }
   371→
   372→        toast.success('Invoice disimpan ke riwayat');
   373→        // Reset invoice form after saving
   374→        resetDocument('invoice');
   375→        router.push(`/surat-jalan?invoiceId=${saved.id}`);
   376→      } else if (res.status === 409) {
   377→        // Invoice already saved, get existing ID and navigate
   378→        const errData = await res.json().catch(() => ({}));
   379→        if (errData.id) {
   380→          resetDocument('invoice');
   381→          router.push(`/surat-jalan?invoiceId=${errData.id}`);
   382→        } else {
   383→          toast('Invoice sudah ada di riwayat.');
   384→        }
   385→      } else {
   386→        toast.error('Gagal menyimpan invoice');
   387→      }
   388→    } catch {
   389→      toast.error('Gagal menyimpan invoice');
   390→    }
   391→    setSavingSj(false);
   392→  };
   393→
   394→  return (
   395→    <>
   396→      <DocumentEditorLayout
   397→        title="Invoice"
   398→        previewContent={<InvoicePreview data={invoice} />}
   399→        actions={
   400→          <div className="flex items-center gap-2 flex-wrap">
   401→            <DocumentActionButtons
   402→              docType="invoice"
   403→              documentLabel="Invoice"
   404→              currentData={invoice}
   405→              onReset={() => resetDocument('invoice')}
   406→            />
   407→            <Button
   408→              size="sm"
   409→              onClick={handleSuratJalan}
   410→              disabled={savingSj}
   411→              className="bg-orange-600 hover:bg-orange-700 h-8 sm:h-9"
   412→            >
   413→              <Truck className="mr-1.5 h-3.5 w-3.5" />
   414→              {savingSj ? 'Menyimpan...' : 'Surat Jalan'}
   415→            </Button>
   416→          </div>
   417→        }
   418→      >
   419→        <CompanyFields company={invoice.company} onChange={updateCompany} />
   420→
   421→        <div className="rounded-lg border bg-card p-3 sm:p-4 shadow-sm">
   422→          <h3 className="mb-3 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   423→            Detail Dokumen
   424→          </h3>
   425→          <div className="grid grid-cols-2 gap-2">
   426→            <div className="space-y-1.5">
   427→              <Label className="text-xs">No. Invoice</Label>
   428→              <Input
   429→                value={invoice.nomor}
   430→                readOnly
   431→                className="bg-slate-50 text-slate-500 cursor-not-allowed"
   432→              />
   433→            </div>
   434→            <div className="space-y-1.5">
   435→              <Label className="text-xs">Tanggal</Label>
   436→              <Input
   437→                type="date"
   438→                value={invoice.tanggal}
   439→                onChange={(e) => setInvoice((prev) => ({ ...prev, tanggal: e.target.value }))}
   440→              />
   441→            </div>
   442→          </div>
   443→          <div className="space-y-2">
   444→            <Label className="text-xs">Referensi (No. HC)</Label>
   445→            <Popover open={dropdownOpen} onOpenChange={setDropdownOpen}>
   446→              <PopoverAnchor asChild>
   447→                <div className="relative">
   448→                  <input
   449→                    type="text"
   450→                    placeholder="Pilih No. HC / ketik referensi..."
   451→                    value={referensiInput}
   452→                    onChange={(e) => handleReferensiInputChange(e.target.value)}
   453→                    onFocus={() => setDropdownOpen(true)}
   454→                    className="flex h-9 w-full rounded-md border border-input bg-transparent px-3 py-2 text-sm shadow-xs transition-[color,box-shadow] outline-none focus-visible:border-ring focus-visible:ring-ring/50 focus-visible:ring-[3px] pr-9"
   455→                  />
   456→                  {/* Dropdown chevron icon */}
   457→                  <div className="absolute right-2 top-1/2 -translate-y-1/2 pointer-events-none text-slate-400">
   458→                    <svg className="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M19 9l-7 7-7-7" /></svg>
   459→                  </div>
   460→                </div>
   461→              </PopoverAnchor>
   462→              <PopoverContent
   463→                align="start"
   464→                className="p-0 w-[var(--radix-popover-trigger-width)] max-h-52 overflow-y-auto"
   465→                onOpenAutoFocus={(e) => e.preventDefault()}
   466→              >
   467→                {filteredRiwayatList.length > 0 && (
   468→                  <div>
   469→                    <div className="px-3 py-1.5 text-[10px] font-semibold text-slate-400 uppercase bg-slate-50 border-b border-slate-100 sticky top-0">Riwayat Hitung Cetakan</div>
   470→                    {filteredRiwayatList.map((r) => (
   471→                      <button
   472→                        key={r.id}
   473→                        type="button"
   474→                        onMouseDown={(e) => { e.preventDefault(); handleReferensiSelect(r) }}
   475→                        className={`w-full text-left px-3 py-2 text-sm transition-colors ${invoice.referensi === r.nomorUrut ? 'bg-blue-50 text-blue-700 font-medium' : 'text-slate-700 hover:bg-blue-50 hover:text-blue-700'}`}
   476→                      >
   477→                        <span className="font-semibold text-xs text-emerald-700">{r.nomorUrut || '-'}</span>
   478→                        <span className="ml-1.5 truncate">{r.printName || ''}</span>
   479→                        {r.customerName && <span className="text-slate-400 ml-1.5 text-[11px]">({r.customerName})</span>}
   480→                      </button>
   481→                    ))}
   482→                  </div>
   483→                )}
   484→                {filteredRiwayatList.length === 0 && (
   485→                  <div className="px-3 py-3 text-sm text-slate-400 text-center">Tidak ada riwayat cetakan</div>
   486→                )}
   487→              </PopoverContent>
   488→            </Popover>
   489→          </div>
   490→        </div>
   491→
   492→        {/* Informasi Pembayaran - moved below Detail Dokumen */}
   493→        <div className="rounded-lg border bg-card p-3 sm:p-4 shadow-sm">
   494→          <h3 className="mb-3 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   495→            Informasi Pembayaran
   496→          </h3>
   497→          <div className="grid grid-cols-2 gap-3">
   498→            <div className="space-y-1.5">
   499→              <Label className="text-xs">Tgl. Jatuh Tempo</Label>
   500→              <Input
   501→                type="date"
   502→                value={invoice.tanggalJatuhTempo}
   503→                onChange={(e) => setInvoice((prev) => ({ ...prev, tanggalJatuhTempo: e.target.value }))}
   504→              />
   505→            </div>
   506→            <div className="space-y-1.5">
   507→              <Label className="text-xs">Cara Pembayaran</Label>
   508→              <div className="flex gap-1.5">
   509→                {(['cash', 'transfer', 'giro'] as const).map((method) => {
   510→                  const labels: Record<string, string> = { cash: 'Cash', transfer: 'Transfer', giro: 'Giro' };
   511→                  const isSelected = invoice.caraPembayaran === method;
   512→                  return (
   513→                    <button
   514→                      key={method}
   515→                      type="button"
   516→                      onClick={() => setInvoice((prev) => ({
   517→                        ...prev,
   518→                        caraPembayaran: prev.caraPembayaran === method ? '' : method,
   519→                        tanggalGiro: method === 'giro' && prev.caraPembayaran !== method ? prev.tanggalGiro : (method !== 'giro' ? '' : prev.tanggalGiro),
   520→                      }))}
   521→                      className={`flex-1 py-2 rounded-lg border text-[11px] font-semibold transition-colors ${
   522→                        isSelected
   523→                          ? 'bg-emerald-100 text-emerald-700 border-emerald-300 ring-2 ring-offset-1 ring-emerald-400 shadow-sm'
   524→                          : 'bg-slate-50 text-slate-500 border-slate-200 hover:bg-slate-100'
   525→                      }`}
   526→                    >
