   286→      const vh = window.innerHeight

   293→      const scale = Math.min(availW / naturalW, availH / naturalH, 1.4)

   295→      setPreviewDims({ w: naturalW * scale, h: naturalH * scale })

   314→    }

   317→

   323→      if (updates.tanggalJatuhTempo !== undefined) parsed.tanggalJatuhTempo = updates.tanggalJatuhTempo

   328→        const originalTotal = parsed.originalTotal !== undefined ? parsed.originalTotal : (() => {
   329→          const sub = (parsed.items || []).reduce((s: number, it: { qty: number; harga: number }) => s + it.qty * it.harga, 0)
   330→          return sub + (sub * (parsed.ppn || 0) / 100)

   332→        parsed.dpAmount = originalTotal * (parsed.dp / 100)
   333→        parsed.originalTotal = originalTotal

   337→
   338→      const res = await fetcher(`/api/history/${pelunasanDialogItem.id}`, {
   339→        method: 'PUT',
   340→        headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   341→        body: JSON.stringify({ dataJson: newDataJson }),
   342→      })
   343→      if (res.ok) {
   344→        toast.success(updates.lunas ? 'Pelunasan berhasil dicatat' : 'Berhasil diperbarui')
   345→        setPelunasanDialogOpen(false)
   346→        fetchHistory()
   347→        notifyDataChange('invoice')
   348→      } else {
   349→        toast.error('Gagal menyimpan perubahan')
   350→      }
   351→    } catch {
   352→      toast.error('Gagal menyimpan perubahan')
   353→    } finally {
   354→      setPelunasanUpdating(false)
   355→    }
   356→  }
   357→
   358→  const openPelunasanDialog = (item: HistoryEntry, e?: React.MouseEvent) => {
   359→    if (e) e.stopPropagation()
   360→    const info = parseDocInfo(item)
   361→    setPelunasanToggle(info.lunas)
   362→    setPelunasanDate(info.tanggalPelunasan || getTodayStr())
   363→    setJatuhTempoDate(info.tanggalJatuhTempo || '')
   364→    setPelunasanDialogItem(item)
   365→    setPelunasanDialogOpen(true)
   366→  }
   367→
   368→  const handleSendJpg = useCallback(async () => {

   370→    setSendingPdf(true)
   371→    try {
   372→      const previewEl = document.querySelector('[data-document-preview]') as HTMLElement
   373→      if (previewEl) {
   374→        const blob = await captureElementAsJpg(previewEl)
   375→        const fileName = `${(invData.nomor || 'draft').replace(/\//g, '-')}.jpg`
   376→        const phone = invData.client?.kontak || ''
   377→
   378→        // No-API sharing: Web Share API first (auto-attaches file),
   379→        // then fallback to download + WhatsApp Web.
   380→        const result = await shareJpgToWhatsApp({
   381→          blob,
   382→          fileName,
   383→          documentLabel: `Invoice ${invData.nomor}`,
   384→          phone,

   387→        if (result.status === 'shared') {
   388→          toast.success('Gambar dibagikan ke WhatsApp')

   390→          // silent
   391→        } else if (result.status === 'downloaded') {
   392→          toast.success(`${fileName} tersimpan ke perangkat`, {
   393→            description: 'File JPG telah diunduh ke folder Downloads.',
   394→          })
   395→        } else {
   396→          toast.error(result.error || 'Gagal memproses JPG')
   397→        }
   398→      } else {
   399→        toast.error('Preview tidak ditemukan')
   400→      }
   401→    } catch (err) {
   402→      console.error(err)
   403→      toast.error('Gagal mengirim gambar')

   405→      setSendingPdf(false)
   406→    }
   407→  }, [invData])
   408→
   409→  const handleBackup = async () => {
   410→    setBackupLoading('backup')
   411→    try {
   412→      const res = await authFetch(`/api/database/backup-master?table=invoice_history`)
   413→      if (!res.ok) {
   414→        let errMsg = 'Gagal backup data riwayat invoice'
   415→        try { const errData = await res.json(); errMsg = errData?.error || errMsg } catch {}
   416→        toast.error(errMsg)
   417→        return
   418→      }
   419→      const blob = await res.blob()
   420→      if (blob.size === 0) { toast.error('Backup kosong'); return }
   421→      const url = URL.createObjectURL(blob)
   422→      const a = document.createElement('a')
   423→      a.href = url
   424→      const disposition = res.headers.get('Content-Disposition')
   425→      const match = disposition?.match(/filename\*?=(?:UTF-8'')?"?([^";]+)"?/i)
   426→      a.download = match ? match[1] : `backup-invoice-history-${Date.now()}.xlsx`
   427→      document.body.appendChild(a); a.click(); document.body.removeChild(a)
   428→      URL.revokeObjectURL(url)
   429→      toast.success('Backup berhasil diunduh')
   430→    } catch (e) { console.error('Backup error:', e); toast.error('Gagal backup data') }
   431→    setBackupLoading(null)
   432→  }
   433→
   434→  const handleRestore = async () => {
   435→    const input = document.createElement('input')

   448→        } else { toast.error(data.error || 'Gagal restore') }
   449→      } catch { toast.error('File backup tidak valid') }
   450→      setBackupLoading(null)
   451→    }
   452→    input.click()
   453→  }
   454→
   455→  const filteredHistory = useMemo(() => {

   457→    const q = searchQuery.toLowerCase().trim()
   458→    return allHistory.filter(entry => {
   459→      const info = parseDocInfo(entry)
   460→      return entry.nomor?.toLowerCase().includes(q) || entry.pihakKedua?.toLowerCase().includes(q) || info.namaBarang?.toLowerCase().includes(q) || entry.tanggal?.toLowerCase().includes(q)
   461→    })
   462→  }, [allHistory, searchQuery])
   463→
   464→  // Separate into DP invoices and Pelunasan invoices
   465→  const dpInvoices = useMemo(() => filteredHistory.filter(e => e.docType === 'invoice'), [filteredHistory])
   466→  const pelunasanInvoices = useMemo(() => filteredHistory.filter(e => e.docType === 'invoice-pelunasan'), [filteredHistory])
   467→
   468→  // Calculate uang capek per invoice — use saved value from dataJson, fallback to cetakan lookup
   469→  const invoiceUangCapek = useMemo(() => {
   470→    const result = new Map<string, number>()

   473→    for (const c of cetakanList) {
   474→      if (c.nomorUrut) {
   475→        cetakanByRef.set(c.nomorUrut, (cetakanByRef.get(c.nomorUrut) || 0) + c.profitAmount)
   476→      }
   477→      if (c.printName && !cetakanByRef.has(c.printName)) {
   478→        cetakanByRef.set(c.printName, (cetakanByRef.get(c.printName) || 0) + c.profitAmount)
   479→      }
   480→    }
   481→    for (const inv of dpInvoices) {
   482→      const info = parseDocInfo(inv)
   483→      // Prefer saved uangCapek from dataJson, fallback to cetakan lookup
   484→      const uangCapek = info.uangCapek > 0 ? info.uangCapek : (info.referensi ? (cetakanByRef.get(info.referensi) || 0) : 0)
   485→      result.set(inv.id, uangCapek)
   486→    }
   487→    return result
   488→  }, [dpInvoices, cetakanList])
   489→
   490→  return (
   491→    <>
   492→      <div className="bg-white dark:bg-zinc-900 rounded-2xl shadow-sm border border-slate-200 dark:border-zinc-700 overflow-hidden">
   493→        {/* Header */}
   494→        <div className="flex items-center justify-between gap-2 px-4 py-3 border-b border-slate-200 dark:border-zinc-700 bg-white dark:bg-zinc-900">
   495→          <div className="flex items-center gap-2">
   496→            <History className="w-4 h-4 text-violet-600" />
   497→            <h2 className="text-sm font-bold text-slate-700 uppercase tracking-wide">Riwayat Invoice</h2>
   498→            <span className="text-[10px] font-medium text-slate-500 bg-slate-100 px-2 py-0.5 rounded-full">{allHistory.length} data</span>
   499→          </div>
   500→          <div className="flex items-center gap-1.5">
   501→            <Button onClick={handleBackup} variant="outline" size="sm" disabled={backupLoading === 'backup'} className="h-7 gap-1.5 text-xs">
   502→              {backupLoading === 'backup' ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <DatabaseBackup className="w-3.5 h-3.5" />} Backup
   503→            </Button>
   504→            <Button onClick={handleRestore} variant="outline" size="sm" disabled={backupLoading === 'restore'} className="h-7 gap-1.5 text-xs">

   506→            </Button>

   508→        </div>
   509→
   510→        {/* Search */}

   513→            <div className="relative">
   514→              <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 w-3.5 h-3.5 text-slate-400" />
   515→              <input type="text" value={searchQuery} onChange={(e) => setSearchQuery(e.target.value)} placeholder="Cari no. invoice, customer, barang..." className="w-full h-8 pl-8 pr-8 text-xs bg-white border border-slate-200 rounded-lg focus:outline-none focus:ring-1 focus:ring-blue-400 focus:border-blue-400 placeholder:text-slate-400" />
   516→              {searchQuery && (<button onClick={() => setSearchQuery('')} className="absolute right-2 top-1/2 -translate-y-1/2"><X className="w-3.5 h-3.5 text-slate-400 hover:text-slate-600" /></button>)}
   517→            </div>
   518→          </div>
   519→        )}
   520→
   521→        {/* Table / Cards */}
   522→        {loading ? (
   523→          <div className="px-4 py-6 text-center"><Loader2 className="w-6 h-6 mx-auto text-blue-500 animate-spin" /><p className="text-xs text-slate-400 mt-2">Memuat riwayat...</p></div>
   524→        ) : filteredHistory.length > 0 ? (
   525→          <>
   526→            {/* === INVOICE DP Section === */}
   527→            {dpInvoices.length > 0 && (
   528→              <div>
   529→                <div className="px-4 py-2 bg-white dark:bg-zinc-900 border-b border-slate-200 dark:border-zinc-700">
   530→                  <p className="text-xs font-bold text-violet-700 uppercase tracking-wide flex items-center gap-1.5">
   531→                    <FileText className="w-3.5 h-3.5" />

   536→                <div className="sm:hidden divide-y divide-slate-100">
   537→                  {dpInvoices.slice(0, 100).map((entry) => {
   538→                    const info = parseDocInfo(entry)
   539→                    const uc = invoiceUangCapek.get(entry.id) ?? 0
   540→                    return (
   541→                      <div key={entry.id} className="px-4 py-3 hover:bg-violet-50/30 active:bg-violet-100/40 transition-colors cursor-pointer" onClick={() => { setPreviewItem(entry); setPreviewOpen(true) }}>
   542→                        <div className="flex items-start justify-between gap-2 mb-1">
   543→                          <div className="min-w-0 flex items-center gap-2">
   544→                            <p className="text-violet-700 font-semibold text-[13px] truncate">{entry.nomor || '-'}</p>

   577→                        <th className="text-right py-3 px-3 text-slate-500 font-semibold whitespace-nowrap">Total DP</th>

   582→                    <tbody>
   583→                      {dpInvoices.slice(0, 100).map((entry, idx) => {
