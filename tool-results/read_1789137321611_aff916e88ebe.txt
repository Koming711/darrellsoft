   100→function getTodayStr(): string {
   101→  const d = new Date()
   102→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
   103→}
   104→
   105→function formatRupiahShort(n: number): string {
   106→  return new Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR', minimumFractionDigits: 0 }).format(n)
   107→}

   109→// --- Parse dataJson for document info ---
   110→function parseDocInfo(entry: HistoryEntry) {
   111→  try {
   112→    const parsed = JSON.parse(entry.dataJson)
   113→    const items = parsed.items || []
   114→    const itemCount = items.length
   115→    const firstItem = items[0]
   116→    const namaBarang = firstItem?.deskripsi || ''
   117→    const hargaSatuan = firstItem?.harga || 0
   118→    const totalQty = items.reduce((sum: number, it: { qty: number }) => sum + (it.qty || 0), 0)
   119→    const subtotal = items.reduce((sum: number, it: { qty: number; harga: number }) => sum + it.qty * it.harga, 0)
   120→    const ppn = parsed.ppn || 0
   121→    const dpPercent = parsed.dp || 0
   122→    const totalHarga = subtotal + (subtotal * ppn / 100)
   123→    // Always derive dpAmount from originalTotal
   124→    const originalTotal = parsed.originalTotal !== undefined ? parsed.originalTotal : totalHarga
   125→    const dpAmount = originalTotal * (dpPercent / 100)
   126→    const sisa = totalHarga - dpAmount
   127→    const lunas = parsed.lunas === true
   128→    const batal = parsed.batal === true
   129→    const tanggalJatuhTempo = parsed.tanggalJatuhTempo || ''
   130→    const tanggalPelunasan = parsed.tanggalPelunasan || ''
   131→    const referensi = parsed.referensi || ''
   132→    const referensiInvoiceNomor = parsed.referensiInvoiceNomor || ''
   133→    const isPelunasan = entry.docType === 'invoice-pelunasan'
   134→    const uangCapek = parsed.uangCapek || 0
   135→    return { namaBarang, hargaSatuan, totalQty, totalHarga, dpPercent, dp: dpAmount, sisa, lunas, batal, tanggalJatuhTempo, tanggalPelunasan, referensi, referensiInvoiceNomor, isPelunasan, originalTotal, uangCapek, itemCount }
   136→  } catch {
   137→    return { namaBarang: '', hargaSatuan: 0, totalQty: 0, totalHarga: 0, dpPercent: 0, dp: 0, sisa: 0, lunas: false, batal: false, tanggalJatuhTempo: '', tanggalPelunasan: '', referensi: '', referensiInvoiceNomor: '', isPelunasan: false, originalTotal: 0, uangCapek: 0, itemCount: 0 }
   138→  }
   139→}

   141→// --- Status invoice (batal > lunas > belum) ---
   142→function getInvoiceStatus(info: ReturnType<typeof parseDocInfo>): InvoiceStatus {
   143→  if (info.batal) return 'batal'
   144→  if (info.isPelunasan) return info.lunas ? 'lunas' : 'belum'
   145→  const hasDP = info.dpPercent > 0 || info.dp > 0
   146→  return (hasDP ? info.lunas : (info.lunas || info.sisa <= 0)) ? 'lunas' : 'belum'
   147→}

   149→// --- Parse dataJson for InvoiceData ---
   150→function parseInvoiceData(entry: HistoryEntry): InvoiceData {
   151→  try {
   152→    const parsed = JSON.parse(entry.dataJson)
   153→    const company: CompanyInfo = {
   154→      nama: parsed.company?.nama || DEFAULT_COMPANY.nama,
   155→      telepon: parsed.company?.telepon || DEFAULT_COMPANY.telepon,
   156→      alamat: parsed.company?.alamat || DEFAULT_COMPANY.alamat,
   157→      email: parsed.company?.email || DEFAULT_COMPANY.email,
   158→      npwp: parsed.company?.npwp || '',
   159→      website: parsed.company?.website || '',
   160→      ppn: parsed.company?.ppn ?? parsed.ppn ?? 11,
   161→      logo: parsed.company?.logo || '',
   162→      bankName: parsed.company?.bankName || '',
   163→      bankAccount: parsed.company?.bankAccount || '',
   164→      bankHolder: parsed.company?.bankHolder || '',
   165→      bankName2: parsed.company?.bankName2 || '',
   166→      bankAccount2: parsed.company?.bankAccount2 || '',
   167→      bankHolder2: parsed.company?.bankHolder2 || '',
   168→    }
   169→    const client = parsed.client || { nama: '', kontak: '', alamat: '' }
   170→    const items = (parsed.items || []).map((it: { id?: string; deskripsi?: string; qty?: number; satuan?: string; harga?: number; modal?: number }, i: number) => ({
   171→      id: it.id || `item-${i}`,
   172→      deskripsi: it.deskripsi || '',
   173→      qty: it.qty || 0,
   174→      satuan: it.satuan || '',
   175→      harga: it.harga || 0,
   176→      ...(it.modal !== undefined ? { modal: it.modal } : {}),
   177→    }))
   178→    const docType = (entry.docType === 'invoice-pelunasan' ? 'invoice-pelunasan' : 'invoice') as 'invoice' | 'invoice-pelunasan'
   179→    return {
   180→      type: docType,
   181→      company,
   182→      nomor: parsed.nomor || entry.nomor || '',
   183→      tanggal: parsed.tanggal || entry.tanggal || '',
   184→      referensi: parsed.referensi || '',
   185→      client,
   186→      items,
   187→      ppn: parsed.ppn ?? 11,
   188→      dp: parsed.dp || 0,
   189→      dpAmount: parsed.dpAmount,
   190→      catatan: parsed.catatan || '',
   191→      tanggalJatuhTempo: parsed.tanggalJatuhTempo || '',
   192→      caraPembayaran: parsed.caraPembayaran || '',
   193→      tanggalGiro: parsed.tanggalGiro || '',
   194→      lunas: parsed.lunas === true,
   195→      tanggalPelunasan: parsed.tanggalPelunasan || '',
   196→      referensiInvoiceId: parsed.referensiInvoiceId || '',
   197→      referensiInvoiceNomor: parsed.referensiInvoiceNomor || '',
   198→      originalTotal: parsed.originalTotal,
   199→      uangCapek: parsed.uangCapek || 0,
   200→    }
   201→  } catch {
   202→    const docType = (entry.docType === 'invoice-pelunasan' ? 'invoice-pelunasan' : 'invoice') as 'invoice' | 'invoice-pelunasan'
   203→    return {
   204→      type: docType,
   205→      company: { ...DEFAULT_COMPANY },
   206→      nomor: entry.nomor || '',
   207→      tanggal: entry.tanggal || '',
   208→      referensi: '',
   209→      client: { nama: '', kontak: '', alamat: '' },
   210→      items: [],
   211→      ppn: 11,
   212→      dp: 0,
   213→      catatan: '',
   214→      tanggalJatuhTempo: '',
   215→      caraPembayaran: '',
   216→      tanggalGiro: '',
   217→      lunas: false,
   218→      tanggalPelunasan: '',
   219→      referensiInvoiceId: '',
   220→      referensiInvoiceNomor: '',
   221→      uangCapek: 0,
   222→    }
   223→  }
   224→}

   249→// ============================================================
   250→// DetailInvoiceView — "halaman pratinjau" setelah Simpan di Buat
   251→// Invoice & halaman detail saat baris riwayat diklik.
   252→// Tombol Cetak / JPG / Surat Jalan / Tandai Lunas / Hapus / Batal
   253→// ada DI ATAS pratinjau. Pratinjau A5 ber-outline, +20% (desktop),
   254→// full-width di mobile.
   255→// ============================================================
   256→function DetailInvoiceView({ id, onBack }: { id: string; onBack: () => void }) {
   257→  const router = useRouter()
   258→  const resetDocument = useDokuproStore((s) => s.resetDocument)
   259→  const [entry, setEntry] = useState<HistoryEntry | null>(null)
   260→  const [loading, setLoading] = useState(true)
   261→  const [error, setError] = useState('')
   262→  const [jpgGenerating, setJpgGenerating] = useState(false)
   263→  const [updating, setUpdating] = useState(false)
   264→
   265→  // Dialog konfirmasi
   266→  const [lunasOpen, setLunasOpen] = useState(false)
   267→  const [batalOpen, setBatalOpen] = useState(false)
   268→  const [hapusOpen, setHapusOpen] = useState(false)
   269→
   270→  // Pratinjau scaler
   271→  const scalerRef = useRef<HTMLDivElement>(null)
   272→
   273→  const loadEntry = useCallback(async () => {
   274→    setLoading(true)
   275→    setError('')
   276→    try {
   277→      const res = await fetcher(`/api/history/${id}`, { headers: getAuthHeaders() })
   278→      if (res.ok) {
   279→        const json = await res.json()
   280→        setEntry(json.data || null)
   281→      } else {
   282→        setError('Invoice tidak ditemukan')
   283→      }
   284→    } catch {
   285→      setError('Gagal memuat invoice')
   286→    } finally {
   287→      setLoading(false)
   288→    }
   289→  }, [id])
   290→
   291→  useEffect(() => { loadEntry() }, [loadEntry])
   292→
   293→  const data = useMemo(() => (entry ? parseInvoiceData(entry) : null), [entry])
   294→  const info = useMemo(() => (entry ? parseDocInfo(entry) : null), [entry])
   295→  const status: InvoiceStatus = info ? getInvoiceStatus(info) : 'belum'
   296→
   297→  // Scale pratinjau A5 agar pas dengan container (desktop ~+20% dari ukuran
   298→  // asli 148mm, mobile full-width). offsetWidth/Height tidak terpengaruh transform.
   299→  useLayoutEffect(() => {
   300→    if (!data) return
   301→    const fit = () => {
   302→      const wrapper = scalerRef.current
   303→      if (!wrapper) return
   304→      const a5 = wrapper.querySelector('.a5-page') as HTMLElement | null
   305→      if (!a5) return
   306→      const naturalW = a5.offsetWidth
   307→      const naturalH = a5.offsetHeight
   308→      if (naturalW === 0 || naturalH === 0) {
   309→        requestAnimationFrame(fit)
   310→        return
   311→      }
   312→      const availW = wrapper.parentElement?.clientWidth || wrapper.clientWidth
   313→      if (availW === 0) {
   314→        requestAnimationFrame(fit)
   315→        return
   316→      }
   317→      const scale = availW / naturalW
   318→      a5.style.transform = `scale(${scale})`
   319→      a5.style.transformOrigin = 'top left'
   320→      wrapper.style.width = `${naturalW * scale}px`
   321→      wrapper.style.height = `${naturalH * scale}px`
   322→    }
   323→    fit()
   324→    const raf = requestAnimationFrame(fit)
   325→    const timer = setTimeout(fit, 250)
   326→    window.addEventListener('resize', fit)
   327→    return () => { cancelAnimationFrame(raf); clearTimeout(timer); window.removeEventListener('resize', fit) }
   328→  }, [data])
   329→
   330→  const handleJpg = async () => {
   331→    if (!data) return
   332→    setJpgGenerating(true)
   333→    try {
   334→      const previewEl = document.querySelector('[data-document-preview]') as HTMLElement
   335→      if (!previewEl) { toast.error('Pratinjau tidak ditemukan'); return }
   336→      const blob = await captureElementAsJpg(previewEl)
   337→      const fileName = `${(data.nomor || 'draft').replace(/\//g, '-')}.jpg`
   338→      const phone = data.client?.kontak || ''
   339→      const result = await shareJpgToWhatsApp({
   340→        blob,
   341→        fileName,
   342→        documentLabel: `Invoice ${data.nomor}`,
   343→        phone,
   344→      })
   345→      if (result.status === 'shared') toast.success('Gambar dibagikan ke WhatsApp')
   346→      else if (result.status === 'cancelled') { /* silent */ }
   347→      else if (result.status === 'downloaded') toast.success(`${fileName} tersimpan ke perangkat`, { description: 'File JPG telah diunduh ke folder Downloads.' })
   348→      else toast.error(result.error || 'Gagal memproses JPG')
   349→    } catch (err) {
   350→      console.error(err)
   351→      toast.error('Gagal membuat JPG')
   352→    } finally {
   353→      setJpgGenerating(false)
   354→    }
   355→  }
   356→
   357→  // Tandai Lunas — invoice dinyatakan lunas penuh hari ini (+ sinkron PEL)
   358→  const handleTandaiLunas = async () => {
   359→    if (!entry || !data) return
   360→    setUpdating(true)
   361→    try {
   362→      const parsed = JSON.parse(entry.dataJson)
   363→      parsed.lunas = true
   364→      parsed.tanggalPelunasan = getTodayStr()
   365→      delete parsed.statusPembayaran
   366→      if ((parsed.dp || 0) > 0) {
   367→        const sub = (parsed.items || []).reduce((s: number, it: { qty: number; harga: number }) => s + it.qty * it.harga, 0)
   368→        const totalHarga = sub + (sub * (parsed.ppn || 0) / 100)
   369→        parsed.originalTotal = parsed.originalTotal !== undefined ? parsed.originalTotal : totalHarga
   370→        parsed.dpAmount = parsed.originalTotal * (parsed.dp / 100)
   371→      }
   372→      const res = await fetcher(`/api/history/${entry.id}`, {
   373→        method: 'PUT',
   374→        headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   375→        body: JSON.stringify({ dataJson: JSON.stringify(parsed) }),
   376→      })
   377→      if (res.ok) {
   378→        await syncLinkedPelunasan(entry.id, parsed.nomor || entry.nomor, parsed)
   379→        toast.success('Invoice ditandai LUNAS')
   380→        setLunasOpen(false)
   381→        window.dispatchEvent(new CustomEvent('dokupro:history-updated'))
   382→        notifyDataChange('invoice')
   383→        loadEntry()
   384→      } else {
   385→        toast.error('Gagal menandai lunas')
   386→      }
   387→    } catch {
   388→      toast.error('Gagal menandai lunas')
   389→    } finally {
   390→      setUpdating(false)
   391→    }
   392→  }
   393→
   394→  // Batal — invoice dibatalkan (status Batal di riwayat & detail)
   395→  const handleBatal = async () => {
   396→    if (!entry) return
   397→    setUpdating(true)
   398→    try {
   399→      const parsed = JSON.parse(entry.dataJson)
   400→      parsed.batal = true
   401→      const res = await fetcher(`/api/history/${entry.id}`, {
   402→        method: 'PUT',
   403→        headers: { 'Content-Type': 'application/json', ...getAuthHeaders() },
   404→        body: JSON.stringify({ dataJson: JSON.stringify(parsed) }),
   405→      })
   406→      if (res.ok) {
   407→        toast.success('Invoice dibatalkan')
   408→        setBatalOpen(false)
   409→        window.dispatchEvent(new CustomEvent('dokupro:history-updated'))
   410→        notifyDataChange('invoice')
   411→        loadEntry()
   412→      } else {
   413→        toast.error('Gagal membatalkan invoice')
   414→      }
   415→    } catch {
   416→      toast.error('Gagal membatalkan invoice')
   417→    } finally {
   418→      setUpdating(false)
   419→    }
   420→  }
   421→
   422→  // Hapus — soft delete → masuk tab Sampah (bisa dipulihkan)
   423→  const handleHapus = async () => {
   424→    if (!entry) return
   425→    setUpdating(true)
   426→    try {
   427→      const res = await fetcher(`/api/history/${entry.id}`, { method: 'DELETE', headers: getAuthHeaders() })
   428→      if (res.ok) {
   429→        toast.success('Invoice dipindahkan ke Sampah')
   430→        setHapusOpen(false)
   431→        window.dispatchEvent(new CustomEvent('dokupro:history-updated'))
   432→        notifyDataChange('invoice')
   433→        resetDocument('invoice')
   434→        onBack()
   435→      } else {
   436→        toast.error('Gagal menghapus invoice')
   437→      }
   438→    } catch {
   439→      toast.error('Gagal menghapus invoice')
   440→    } finally {
   441→      setUpdating(false)
   442→    }
   443→  }
   444→
   445→  const actionButtons = (
   446→    <div className="flex flex-wrap gap-2 print:hidden">
   447→      <Button size="sm" onClick={() => { toast.dismiss(); const origTitle = document.title; document.title = ' '; setTimeout(() => { window.print(); document.title = origTitle }, 100) }} className="bg-emerald-600 hover:bg-emerald-700 min-h-[36px]">
   448→        <Printer className="mr-1.5 h-3.5 w-3.5" /> Cetak
   449→      </Button>
   450→      <Button size="sm" onClick={handleJpg} disabled={jpgGenerating} className="bg-green-600 hover:bg-green-700 min-h-[36px]">
   451→        {jpgGenerating ? <><Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" /> JPG...</> : <><ImageIcon className="mr-1.5 h-3.5 w-3.5" /> JPG</>}
   452→      </Button>
   453→      <Button size="sm" onClick={() => router.push(`/surat-jalan?invoiceId=${id}`)} className="bg-orange-600 hover:bg-orange-700 min-h-[36px]">
   454→        <Truck className="mr-1.5 h-3.5 w-3.5" /> Surat Jalan
   455→      </Button>
   456→      {status !== 'lunas' && status !== 'batal' && (
   457→        <Button size="sm" onClick={() => setLunasOpen(true)} className="bg-violet-600 hover:bg-violet-700 min-h-[36px]">
   458→          <CheckCircle2 className="mr-1.5 h-3.5 w-3.5" /> Tandai Lunas
   459→        </Button>
   460→      )}
   461→      {status !== 'batal' && (
   462→        <Button size="sm" variant="outline" onClick={() => setBatalOpen(true)} className="border-amber-300 text-amber-700 hover:bg-amber-50 min-h-[36px]">
   463→          <Ban className="mr-1.5 h-3.5 w-3.5" /> Batal
   464→        </Button>
   465→      )}
   466→      <Button size="sm" variant="outline" onClick={() => setHapusOpen(true)} className="border-red-200 text-destructive hover:bg-red-50 hover:text-destructive min-h-[36px]">
   467→        <Trash2 className="mr-1.5 h-3.5 w-3.5" /> Hapus
   468→      </Button>
   469→    </div>
   470→  )
   471→
   472→  return (
   473→    <div>
   474→      {/* Header — Kembali + judul "Detail Invoice" + status */}
   475→      <div className="flex items-center gap-2 mb-3 print:hidden">
   476→        <Button onClick={onBack} variant="outline" size="sm" className="h-9 gap-1.5 text-xs">
   477→          <ArrowLeft className="w-3.5 h-3.5" /> Kembali
   478→        </Button>
